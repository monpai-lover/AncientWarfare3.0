using AncientWarfare3.core.lineage;
using System;
using System.Collections.Generic;
using System.IO;

internal static class CityReservePoolRulesTests
{
    internal static void Run()
    {
        Equal(20, CityReservePoolRules.ResolveAvailableManpower(
            authenticResidents: 30, authenticMobilized: 10,
            activeCitySourcedMilitary: 0),
            "city manpower is half of all authentic residents");
        Equal(5, CityReservePoolRules.ResolveAvailableManpower(
            authenticResidents: 30, authenticMobilized: 10,
            activeCitySourcedMilitary: 15),
            "active source-city soldiers consume manpower headroom");
        Equal(false, CityReservePoolRules.CanConfirmManpowerExhausted(
            ledgerReady: false, availableManpower: 0),
            "a missing ledger cannot masquerade as zero manpower");
        Equal(true, CityReservePoolRules.CanConfirmManpowerExhausted(
            ledgerReady: true, availableManpower: 0),
            "a ready empty ledger confirms exhaustion");
        Equal(15, CityReservePoolRules.Capacity(50, 30),
            "limited law registers thirty percent of eligible civilians");
        Equal(25, CityReservePoolRules.Capacity(50, 50),
            "standard law registers half of eligible civilians");
        Equal(35, CityReservePoolRules.Capacity(50, 70),
            "expanded law registers seventy percent of eligible civilians");
        Equal(50, CityReservePoolRules.Capacity(50, 100),
            "full law registers every eligible civilian");
        Equal(1, CityReservePoolRules.Capacity(1, 30),
            "a non-zero limited law keeps one eligible civilian available");
        Equal(1, CityReservePoolRules.Capacity(3, 50),
            "small standard-law pools do not collapse to zero");
        Equal(2, CityReservePoolRules.Capacity(3, 70),
            "small expanded-law pools retain their exact rounded-down share");
        Equal(0, CityReservePoolRules.Capacity(0, 30),
            "an empty city still has no reserve capacity");
        Equal(true, CityReservePoolRules.ShouldAddForLawChange(
            frozen: true, oldPercent: 50, newPercent: 70),
            "raising the law is the only frozen-pool addition path");
        Equal(false, CityReservePoolRules.ShouldAddForLawChange(
            frozen: true, oldPercent: 70, newPercent: 30),
            "lowering a wartime law cannot add candidates");
        Equal(20, CityReservePoolRules.RequiredRemovalCount(
            memberCount: 35, capacity: 15),
            "lowering removes only excess registrations");

        Equal(true, CityReservePoolRules.CanEnroll(
            alive: true, adult: true, localResident: true,
            baseEligible: true, frozen: false, memberCount: 4,
            capacity: 5), "eligible adult enters a non-full peace pool");
        Equal(false, CityReservePoolRules.CanEnroll(
            alive: true, adult: true, localResident: true,
            baseEligible: true, frozen: false, memberCount: 5,
            capacity: 5), "a full pool rejects without a waiting list");
        Equal(false, CityReservePoolRules.CanEnroll(
            alive: true, adult: true, localResident: true,
            baseEligible: true, frozen: true, memberCount: 0,
            capacity: 5), "an adult event cannot add during war");

        Equal(1, CityReservePoolRules.CityBudget(preparation: false),
            "peace maintenance visits one city per world day");
        Equal(4, CityReservePoolRules.CityBudget(preparation: true),
            "preparation accelerates city maintenance");
        Equal(8, CityReservePoolRules.ActorBudget(preparation: false),
            "peace candidate validation is bounded");
        Equal(int.MaxValue, CityReservePoolRules.ActorBudget(preparation: true),
            "preparation scans every resident without using it as an allocation size");
        Equal(18, CityReservePoolRules.CapacityForPreparation(
            eligibleCivilians: 61, lawPercent: 30, preparation: true),
            "preparation reserve capacity still follows the conscription law");
        Equal(18, CityReservePoolRules.CapacityForPreparation(
            eligibleCivilians: 61, lawPercent: 30, preparation: false),
            "peacetime reserve capacity still follows the conscription law");
        Equal(73, CityReservePoolRules.FullReconciliationBudget(
            residentCount: 61, indexedMemberCount: 12),
            "war start budgets one complete resident and member pass");
        Equal(1, CityReservePoolRules.FullReconciliationBudget(
            residentCount: 0, indexedMemberCount: 0),
            "an empty uncached city still receives a reconciliation visit");

        Equal(true, CityReservePoolRules.CanRestoreRejectedCandidate(
            sameKingdom: true, sameCity: true, alive: true,
            reserveEligible: true, enlistedIntoTargetArmy: false),
            "a rejected enlistment candidate returns to its source city pool");
        Equal(false, CityReservePoolRules.CanRestoreRejectedCandidate(
            sameKingdom: true, sameCity: true, alive: true,
            reserveEligible: true, enlistedIntoTargetArmy: true),
            "a successfully enlisted candidate cannot remain in reserve");
        Equal(false, CityReservePoolRules.CanRestoreRejectedCandidate(
            sameKingdom: true, sameCity: false, alive: true,
            reserveEligible: true, enlistedIntoTargetArmy: false),
            "a migrant cannot be returned to the former city's pool");
        Equal(true, CityReservePoolRules.MatchesSourceCity(
            sourceCityId: 10L, candidateCityId: 10L),
            "an army may consume reserves raised by its source city");
        Equal(false, CityReservePoolRules.MatchesSourceCity(
            sourceCityId: 10L, candidateCityId: 20L),
            "an army cannot consume reserves raised by another city");
        Equal(true, CityReservePoolRules.ShouldReconcileJoiningKingdom(
            warActive: true, liveKingdom: true),
            "a realm joining an active war reconciles every city before freezing");
        Equal(11L, CityReservePoolRules.ResolveWarEmergencyId(
            frozen: true, currentEmergencyId: 11L,
            requestedEmergencyId: 22L),
            "a parallel war cannot reset the first wartime reserve snapshot");
        Equal(22L, CityReservePoolRules.ResolveWarEmergencyId(
            frozen: false, currentEmergencyId: -1L,
            requestedEmergencyId: 22L),
            "the first formal war opens a new reserve snapshot");
        Equal(false, CityReservePoolRules.ShouldReconcileJoiningKingdom(
            warActive: false, liveKingdom: true),
            "an ended war cannot trigger a joining-realm reconciliation");
        Equal(false, CityReservePoolRules.ShouldReconcileJoiningKingdom(
            warActive: true, liveKingdom: false),
            "an invalid realm cannot enter reserve reconciliation");

        Equal(true, CityReservePoolRules.ShouldApplyReserveExhaustion(
            attackAssignment: true, reinforcementShortage: 1,
            kingdomFrozen: true, exhaustionConfirmed: true,
            alreadyApplied: false),
            "an attacking army with no reserve receives the penalty");
        Equal(false, CityReservePoolRules.ShouldApplyReserveExhaustion(
            attackAssignment: false, reinforcementShortage: 10,
            kingdomFrozen: true, exhaustionConfirmed: true,
            alreadyApplied: false),
            "defense-only shortage does not receive the penalty");
        Equal(false, CityReservePoolRules.ShouldApplyReserveExhaustion(
            attackAssignment: true, reinforcementShortage: 10,
            kingdomFrozen: true, exhaustionConfirmed: false,
            alreadyApplied: false),
            "an incomplete reserve check defers the penalty");
        Equal(false, CityReservePoolRules.ShouldApplyReserveExhaustion(
            attackAssignment: true, reinforcementShortage: 10,
            kingdomFrozen: true, exhaustionConfirmed: true,
            alreadyApplied: true),
            "the same side and war cannot receive the penalty twice");
        Equal(20, CityReservePoolRules.ReserveExhaustionContribution,
            "reserve exhaustion adds exactly twenty points");
        Equal(100, CityReservePoolRules.ComposeExhaustion(90, 20),
            "total exhaustion remains clamped to one hundred");
        Equal(55, CityReservePoolRules.ComposeExhaustion(35, 20),
            "reserve exhaustion composes with casualty and duration exhaustion");
        int attackerReserve = CityReservePoolRules.
            ApplyReserveExhaustionContribution(0);
        attackerReserve = CityReservePoolRules.
            ApplyReserveExhaustionContribution(attackerReserve);
        Equal(20, attackerReserve,
            "repeated reserve exhaustion checks remain idempotent");
        int defenderReserve = 0;
        defenderReserve = CityReservePoolRules.
            ApplyReserveExhaustionContribution(defenderReserve);
        Equal(20, attackerReserve,
            "applying defender exhaustion does not alter attackers");
        Equal(20, defenderReserve,
            "defenders maintain an independent reserve contribution");

        Equal(true, TemporaryLevyRules.CanRegisterReserve(
            originalEligible: true, protectedIdentity: false, age: 18f),
            "a newly adult eligible civilian can enter the reserve");
        Equal(false, TemporaryLevyRules.CanRegisterReserve(
            originalEligible: true, protectedIdentity: false, age: 65f),
            "a person at the enlistment age ceiling cannot enter");
        Equal(false, TemporaryLevyRules.CanRegisterReserve(
            originalEligible: true, protectedIdentity: true, age: 20f),
            "protected identities cannot enter the reserve");

        Equal(true, CityReservePoolRules.CanMaintain(
            frozen: false, worldDayChanged: true),
            "peace maintenance runs once on a new world day");
        Equal(false, CityReservePoolRules.CanMaintain(
            frozen: true, worldDayChanged: true),
            "war freeze blocks all additions");
        Equal(false, CityReservePoolRules.ShouldUnfreeze(
            activeWarCount: 1),
            "ending one concurrent war does not unfreeze the pool");
        Equal(true, CityReservePoolRules.ShouldUnfreeze(
            activeWarCount: 0),
            "leaving the final war reopens peace maintenance");

        var actorIds = new SortedSet<long> { 9L, 3L, 7L };
        Equal(true, CityReservePoolRules.TryTakeNextActorId(actorIds,
            out long firstActorId),
            "a populated city reserve yields an actor");
        Equal(3L, firstActorId,
            "city reserve consumption uses ascending actor id");
        Equal(true, CityReservePoolRules.TryTakeNextActorId(actorIds,
            out long secondActorId),
            "a second request consumes the next actor");
        Equal(7L, secondActorId,
            "two requests cannot return the same actor id");
        Equal(false, actorIds.Contains(firstActorId),
            "consumed actor ids are removed atomically");

        Equal(false, CityReservePoolRules.CanConsumeFromCity(
            kingdomFrozen: true, realmControlled: true, population: 20),
            "a city at the twenty-person floor cannot donate");
        Equal(false, CityReservePoolRules.CanConsumeFromCity(
            kingdomFrozen: true, realmControlled: false, population: 200),
            "an enemy-controlled city cannot donate");
        Equal(true, CityReservePoolRules.CanConsumeFromCity(
            kingdomFrozen: true, realmControlled: true, population: 21),
            "a frozen controlled city above the floor can donate");
        Equal(false, CityReservePoolRules.CanConsumeDuringPreparation(
            activeNotice: true, realmControlled: true, population: 20),
            "preparation preserves the twenty-person city floor");
        Equal(false, CityReservePoolRules.CanConsumeDuringPreparation(
            activeNotice: false, realmControlled: true, population: 200),
            "preparation cannot consume after its notice ends");
        Equal(true, CityReservePoolRules.CanConsumeDuringPreparation(
            activeNotice: true, realmControlled: true, population: 21),
            "a prepared controlled city above the floor can donate");

        Equal(true, CityReservePoolRules.CanConsumeForMobilization(
            ArmyMobilizationPhase.Notice, realmControlled: true,
            population: 21),
            "notice preparation consumes the shared city ledger");
        Equal(true, CityReservePoolRules.CanConsumeForMobilization(
            ArmyMobilizationPhase.War, realmControlled: true,
            population: 21),
            "formal war consumes the same city ledger");
        Equal(false, CityReservePoolRules.CanConsumeForMobilization(
            ArmyMobilizationPhase.Peace, realmControlled: true,
            population: 200),
            "peace maintains but never consumes the reserve ledger");
        Equal(false, CityReservePoolRules.CanConsumeForMobilization(
            ArmyMobilizationPhase.Notice, realmControlled: false,
            population: 200),
            "a captured source city cannot supply its army");

        Equal(false, CityReservePoolRules.CanConfirmExhausted(
            reconciliationComplete: false, availableActorCount: 0),
            "an uninitialized city ledger cannot masquerade as exhausted");
        Equal(false, CityReservePoolRules.CanConfirmExhausted(
            reconciliationComplete: true, availableActorCount: 1),
            "a reconciled city with one reserve remains available");
        Equal(true, CityReservePoolRules.CanConfirmExhausted(
            reconciliationComplete: true, availableActorCount: 0),
            "a complete empty city reconciliation confirms exhaustion");

        Equal(true, CityReservePoolRules.CanConfirmExhausted(
            kingdomFrozen: true, allIndexedCitiesChecked: true,
            remainingIndexedActors: 0),
            "a fully validated empty frozen pool confirms exhaustion");
        Equal(false, CityReservePoolRules.CanConfirmExhausted(
            kingdomFrozen: true, allIndexedCitiesChecked: false,
            remainingIndexedActors: 0),
            "an incomplete validation pass cannot confirm exhaustion");
        Equal(false, CityReservePoolRules.CanConfirmExhausted(
            kingdomFrozen: true, allIndexedCitiesChecked: true,
            remainingIndexedActors: 1),
            "an indexed reserve actor prevents exhaustion");
        Equal(false, CityReservePoolRules.HasRemainingUsableActors(0),
            "eligible civilians without registrations do not block exhaustion");
        Equal(true, CityReservePoolRules.HasRemainingUsableActors(1),
            "one registered reserve actor keeps the pool available");

        Equal(true, CityReservePoolRules.ShouldDeferPersistedMemberValidation(
            hasPersistedMember: true, restoreValidationReady: false),
            "save restore defers member validation until actor state is ready");
        Equal(false, CityReservePoolRules.ShouldDeferPersistedMemberValidation(
            hasPersistedMember: true, restoreValidationReady: true),
            "ready restore may validate persisted reserve membership");
        Equal(false, CityReservePoolRules.ShouldDeferPersistedMemberValidation(
            hasPersistedMember: false, restoreValidationReady: false),
            "unmarked actors do not enter deferred member validation");

        Equal(true, CityReservePoolRules.ShouldDeferCallbackInvalidation(
            hasPersistedMember: true, restoreInFlight: true),
            "load-time profession callbacks preserve persisted reserve members");
        Equal(false, CityReservePoolRules.ShouldDeferCallbackInvalidation(
            hasPersistedMember: false, restoreInFlight: true),
            "load-time callbacks do not defer unmarked actors");
        Equal(false, CityReservePoolRules.ShouldDeferCallbackInvalidation(
            hasPersistedMember: true, restoreInFlight: false),
            "normal profession changes still validate reserve members");

        PreparationUsesMaintainedActorPool();
    }

    private static void PreparationUsesMaintainedActorPool()
    {
        string source = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "CityReservePoolService.cs"));
        Equal(true, source.Contains(
                "int actorBudget = CityReservePoolRules.ActorBudget(preparation);",
                StringComparison.Ordinal),
            "daily and notice maintenance must budget actor-index work");
        Equal(true, source.Contains(
                "bool complete = MaintainCity(kingdom, city, state,",
                StringComparison.Ordinal),
            "daily and notice maintenance must populate the city actor pool");
        Equal(true, source.Contains(
                "if (!IndexEligibleActor(actor, kingdom, city, pool)) return;",
                StringComparison.Ordinal),
            "an eligible actor enters the reserve when becoming adult");
        Equal(true, source.Contains(
                "CityReservePoolRules.TryTakeNextActorId(pool.ActorIds,",
                StringComparison.Ordinal),
            "notice mobilization must consume the law-limited actor pool");
        Equal(true, source.Contains(
                "CityReservePoolRules.CanRestoreRejectedCandidate(",
                StringComparison.Ordinal),
            "a failed enlistment must restore its actor to the source pool");
        Equal(true, source.Contains(
                "foreach (Actor actor in World.world.units)",
                StringComparison.Ordinal),
            "load rebuild must reconstruct actor pools from living residents");
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(name + ": expected " +
                expected + ", got " + actual);
    }
}
