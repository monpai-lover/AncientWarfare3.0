using System.Reflection;
using AncientWarfare3.core.lineage;

internal static class TemporaryLevyRulesTests
{
    public static void Run()
    {
        AnnualRecruitmentStopsAfterWarStarts();
        RecoversEligibleOrphanWarriorsDuringWar();
        PreservesGarrisonAndAssignedSoldiers();
        NotifiesRtsForEveryRosterMembershipChange();
        PreparationScanRunsOnlyOnMonthTickOrInvalidation();
        DirectedReplenishmentNeverLeavesItsSourceCity();
        DirectedReserveExhaustionIsArmyScoped();
        PreparationCreatesAndReplenishesCityArmy();
        SyntheticRecruitmentStartsOnlyAfterFormalWar();
    }

    private static void SyntheticRecruitmentStartsOnlyAfterFormalWar()
    {
        Assert(TemporaryLevyRules.SyntheticFallbackRequest(
                   ArmyMobilizationPhase.Notice, approvedShortage: 10,
                   availableWarReserve: 10) == 0,
            "notice preparation cannot generate actors");
        Assert(TemporaryLevyRules.SyntheticFallbackRequest(
                   ArmyMobilizationPhase.War, approvedShortage: 10,
                   availableWarReserve: 7) == 7,
            "formal war creates only the available replacement reserve");
        Assert(TemporaryLevyRules.SyntheticFallbackRequest(
                   ArmyMobilizationPhase.War, approvedShortage: 10,
                   availableWarReserve: 2) == 2,
            "war creation remains capped by the source city reserve");
        Assert(TemporaryLevyRules.SyntheticFallbackRequest(
                   ArmyMobilizationPhase.War, approvedShortage: 0,
                   availableWarReserve: 10) == 0,
            "a full army creates no synthetic soldiers");
    }

    private static void PreparationCreatesAndReplenishesCityArmy()
    {
        Assert(TemporaryLevyRules.PreparationRequest(
                   establishmentAccepted: true, living: 0,
                   target: 20, workItemLimit: 8) == 8,
            "an accepted Create disposition recruits the first army batch");
        Assert(TemporaryLevyRules.PreparationRequest(
                   establishmentAccepted: true, living: 5,
                   target: 20, workItemLimit: 8) == 8,
            "an accepted Replenish disposition recruits the next batch");
        Assert(TemporaryLevyRules.PreparationRequest(
                   establishmentAccepted: false, living: 0,
                   target: 20, workItemLimit: 8) == 0,
            "a rejected establishment cannot consume reserves");
        Assert(TemporaryLevyRules.CanRecruitBeforeFrontierReady(
                activeNotice: true, sourceCityValid: true),
            "frontier projection readiness cannot block city recruitment");
        Assert(!TemporaryLevyRules.CanRecruitBeforeFrontierReady(
                activeNotice: false, sourceCityValid: true),
            "an ended notice cannot continue preparation recruitment");
    }

    private static void AnnualRecruitmentStopsAfterWarStarts()
    {
        Assert(!TemporaryLevyRules.ShouldRunAnnualRecruitment(
                emergencyActive: true, activeNotice: false,
                activeWar: true),
            "a formal war must not start a new annual recruitment batch");
        Assert(TemporaryLevyRules.ShouldRunAnnualRecruitment(
                emergencyActive: true, activeNotice: false,
                activeWar: false),
            "annual recruitment remains available during a non-war emergency");
    }

    private static void DirectedReplenishmentNeverLeavesItsSourceCity()
    {
        Assert(TemporaryLevyRules.ShouldUseDirectedReplenishmentAnchor(
                completedWorkItems: 12),
            "every directed replenishment batch must stay in its Army source city");
    }

    private static void DirectedReserveExhaustionIsArmyScoped()
    {
        Assert(!TemporaryLevyRules.IsReplenishmentDemandExhausted(
                directedReplenishment: true,
                realmReserveExhausted: true,
                targetReserveExhausted: false),
            "another city exhausting its reserves must not block this Army");
        Assert(TemporaryLevyRules.IsReplenishmentDemandExhausted(
                directedReplenishment: true,
                realmReserveExhausted: false,
                targetReserveExhausted: true),
            "a directed Army must stop when its own source city is exhausted");
        Assert(TemporaryLevyRules.IsReplenishmentDemandExhausted(
                directedReplenishment: false,
                realmReserveExhausted: true,
                targetReserveExhausted: false),
            "realm-wide establishment stops only after realm reserves exhaust");
    }

    private static void RecoversEligibleOrphanWarriorsDuringWar()
    {
        Assert(TemporaryLevyRules.ShouldRecoverOrphanWarrior(
                emergencyActive: true, isWarrior: true, hasArmy: false,
                localResident: true, protectedIdentity: false,
                activeGarrison: false),
            "an unassigned local warrior must be recovered into a field army during war");
    }

    private static void PreservesGarrisonAndAssignedSoldiers()
    {
        Assert(!TemporaryLevyRules.ShouldRecoverOrphanWarrior(
                emergencyActive: true, isWarrior: true, hasArmy: false,
                localResident: true, protectedIdentity: false,
                activeGarrison: true),
            "a city garrison must not be reassigned as an ordinary field soldier");
        Assert(!TemporaryLevyRules.ShouldRecoverOrphanWarrior(
                emergencyActive: true, isWarrior: true, hasArmy: true,
                localResident: true, protectedIdentity: false,
                activeGarrison: false),
            "a soldier already assigned to an army must not be reprocessed");
    }

    private static void NotifiesRtsForEveryRosterMembershipChange()
    {
        Assert(TemporaryLevyRules.ShouldNotifyRtsRosterChanged(
                actorArmyChanged: true, armyListChanged: false),
            "joining an existing army must reopen the RTS member work");
        Assert(TemporaryLevyRules.ShouldNotifyRtsRosterChanged(
                actorArmyChanged: false, armyListChanged: true),
            "repairing a missing roster entry must reopen the RTS member work");
        Assert(!TemporaryLevyRules.ShouldNotifyRtsRosterChanged(
                actorArmyChanged: false, armyListChanged: false),
            "unchanged membership must not create RTS roster work");
    }

    private static void PreparationScanRunsOnlyOnMonthTickOrInvalidation()
    {
        MethodInfo method = typeof(TemporaryLevyRules).GetMethod(
            "ShouldProcessPreparationMonth",
            BindingFlags.Public | BindingFlags.Static);
        if (method == null)
            throw new InvalidOperationException(
                "war-preparation scanning must be gated by game month");
        Assert((bool)method.Invoke(null, new object[]
            { 24288, int.MinValue }),
            "the first observed game month must initialize war preparation");
        Assert(!(bool)method.Invoke(null, new object[]
            { 24288, 24288 }),
            "war preparation must not rescan every rendered frame in the same month");
        Assert((bool)method.Invoke(null, new object[]
            { 24289, 24288 }),
            "the next game month must permit a new preparation scan");
    }

    private static void Assert(bool condition, string name)
    {
        if (!condition) throw new InvalidOperationException(name);
    }
}
