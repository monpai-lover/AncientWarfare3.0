using System;
using System.IO;
using AncientWarfare3.core.lineage;
using AncientWarfare3.core.performance;

internal static class ArmyRtsSchedulingRulesTests
{
    internal static void Run()
    {
        var aw3 = new ArmyRtsSchedulingGate();
        aw3.StartSession(configAw3: true);
        Assert(!aw3.TryEnter(ArmyRtsSchedulerOwner.NativeArmyManager, 1L,
                allowed: true), "AW3 mode rejects the native caller");
        Assert(aw3.TryEnter(ArmyRtsSchedulerOwner.Aw3Authority, 1L,
                allowed: true), "AW3 mode accepts its first logical token");
        Assert(!aw3.TryEnter(ArmyRtsSchedulerOwner.Aw3Authority, 1L,
                allowed: true), "a logical token is consumed exactly once");
        Assert(aw3.TryEnter(ArmyRtsSchedulerOwner.Aw3Authority, 2L,
                allowed: true), "the next logical token is accepted");
        Assert(!aw3.TryEnter(ArmyRtsSchedulerOwner.Aw3Authority, 1L,
                allowed: true), "an older token cannot replay RTS work");

        var native = new ArmyRtsSchedulingGate();
        native.StartSession(configAw3: false);
        Assert(native.TryEnter(ArmyRtsSchedulerOwner.NativeArmyManager, 1L,
                allowed: true), "native mode accepts the native caller");
        Assert(!native.TryEnter(ArmyRtsSchedulerOwner.Aw3Authority, 2L,
                allowed: true),
            "startup ownership remains frozen for the session");

        string root = Directory.GetCurrentDirectory();
        string service = File.ReadAllText(Path.Combine(root, "Code", "core",
            "performance", "ArmyRtsSchedulingService.cs"));
        Assert(service.Contains("SharedGate", StringComparison.Ordinal),
            "both entry paths share one gate");
        Assert(!service.Contains("NativeGate", StringComparison.Ordinal) &&
               !service.Contains("Aw3Gate", StringComparison.Ordinal),
            "independent gates cannot duplicate logical work");
        Assert(service.Contains("ProcessLogicalPass(",
                StringComparison.Ordinal),
            "large-step passes expose one RTS pulse boundary");
        Assert(service.Contains("ArmyRtsExecutionBudgetRules.Capture(",
                StringComparison.Ordinal) &&
               service.Contains("AWPerformanceSettings.Mode",
                   StringComparison.Ordinal),
            "each RTS pulse captures one mode-aware execution budget");
        Assert(service.Contains(
                "budget.ControllerArmies, budget.ReplenishmentArrivals",
                StringComparison.Ordinal),
            "controller and replenishment drains use the captured budget");
        Assert(service.Contains("budget.WatchdogArmies",
                StringComparison.Ordinal) &&
               service.Contains("budget.LifecycleDiscoveries",
                   StringComparison.Ordinal) &&
               service.Contains("budget.AssignmentReconciliations",
                   StringComparison.Ordinal) &&
               service.Contains("budget.SuccessionRecoveries",
                   StringComparison.Ordinal),
            "Large recovery drains all use the shared snapshot");
        Assert(service.Contains(
                "ArmyRouteProviderService.ProcessFrame",
                StringComparison.Ordinal) &&
               !service.Contains(
                   "ArmyRouteProviderService.ProcessFrame(int.MaxValue",
                   StringComparison.Ordinal),
            "global route planning retains its bounded worker pulse");
        Assert(!service.Contains("int.MaxValue",
                StringComparison.Ordinal),
            "the scheduler never drains a dynamically growing queue without a snapshot");

        var work = new ArmyRtsControllerWorkIndex();
        work.AssignMission(new ArmyRtsMission
        {
            ArmyId = 101L,
            KingdomId = 201L,
            WarId = 301L,
            TargetCityId = 401L
        });
        Assert(work.PendingCount == 1,
            "controller pending count includes priority wartime missions");

        Assert(ArmyRtsTaskOwnershipRules.ShouldRecoverMissingCaptainTarget(
                ArmyRtsState.March, missionActive: true,
                targetAvailable: false, transportOwned: false,
                captainPresent: true),
            "a live marching captain with a missing target requests bounded recovery");
        Assert(!ArmyRtsTaskOwnershipRules.ShouldRecoverMissingCaptainTarget(
                ArmyRtsState.Rally, missionActive: true,
                targetAvailable: false, transportOwned: false,
                captainPresent: true),
            "rally waits for formation instead of repeatedly clearing its route");
        Assert(!ArmyRtsTaskOwnershipRules.ShouldRecoverMissingCaptainTarget(
                ArmyRtsState.March, missionActive: true,
                targetAvailable: false, transportOwned: true,
                captainPresent: true),
            "transport ownership keeps the captain target recovery dormant");
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
}
