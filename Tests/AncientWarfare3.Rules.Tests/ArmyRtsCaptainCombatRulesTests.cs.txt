using System;
using System.IO;
using AncientWarfare3.core.lineage;

internal static class ArmyRtsCaptainCombatRulesTests
{
    public static void Run()
    {
        Assert(ArmyRtsCaptainCombatRules.ShouldRetainTarget(
                targetAlive: true, targetHostile: true,
                withinEnvelope: true),
            "a live hostile target inside the encounter envelope is retained");
        Assert(!ArmyRtsCaptainCombatRules.ShouldRetainTarget(
                targetAlive: false, targetHostile: true,
                withinEnvelope: true),
            "a dead combat target is cleared");
        Assert(!ArmyRtsCaptainCombatRules.ShouldRetainTarget(
                targetAlive: true, targetHostile: false,
                withinEnvelope: true),
            "a friendly target is cleared");
        Assert(!ArmyRtsCaptainCombatRules.ShouldRetainTarget(
                targetAlive: true, targetHostile: true,
                withinEnvelope: false),
            "a target outside the bounded encounter envelope is cleared");
        Assert(ArmyRtsCaptainCombatRules.ShouldRetainMemberTarget(
                targetAlive: true, targetHostile: true,
                sameIsland: true, combatOwned: true),
            "an RTS member retains a hostile target outside the captain envelope");
        Assert(!ArmyRtsCaptainCombatRules.ShouldRetainMemberTarget(
                targetAlive: false, targetHostile: true,
                sameIsland: true, combatOwned: true),
            "a dead member target is rejected");
        Assert(!ArmyRtsCaptainCombatRules.ShouldRetainMemberTarget(
                targetAlive: true, targetHostile: false,
                sameIsland: true, combatOwned: true),
            "a friendly member target is rejected");
        Assert(!ArmyRtsCaptainCombatRules.ShouldRetainMemberTarget(
                targetAlive: true, targetHostile: true,
                sameIsland: false, combatOwned: true),
            "a cross-island member target is rejected");
        Assert(!ArmyRtsCaptainCombatRules.ShouldRetainMemberTarget(
                targetAlive: true, targetHostile: true,
                sameIsland: true, combatOwned: false),
            "a target is released after RTS combat ownership ends");
        Assert(ArmyRtsCaptainCombatRules.ShouldEnterCombat(
                alreadyInCombat: false, engagedCombatants: 3,
                liveCombatants: 10, captainEngaged: false),
            "a 25 percent field contact enters captain combat");
        Assert(ArmyRtsCaptainCombatRules.ShouldEnterCombat(
                alreadyInCombat: false, engagedCombatants: 0,
                liveCombatants: 10, captainEngaged: true),
            "captain contact releases followers into the field battle");
        Assert(!ArmyRtsCaptainCombatRules.ShouldEnterCombat(
                alreadyInCombat: false, engagedCombatants: 2,
                liveCombatants: 10, captainEngaged: false),
            "sub-threshold contact does not interrupt strategic movement");
        Assert(ArmyRtsCaptainCombatRules.ShouldUseMemberCombatTask(
                missionActive: true, actorIsCaptain: false,
                fieldCombatReleased: false, hasValidCombatTarget: true),
            "a marching soldier with a valid close enemy immediately enters dedicated combat");
        Assert(ArmyRtsCaptainCombatRules.ShouldUseMemberCombatTask(
                missionActive: true, actorIsCaptain: false,
                fieldCombatReleased: true, hasValidCombatTarget: true),
            "a soldier enters the dedicated RTS combat task only after field combat begins");
        Assert(!ArmyRtsCaptainCombatRules.ShouldUseMemberCombatTask(
                missionActive: true, actorIsCaptain: true,
                fieldCombatReleased: true, hasValidCombatTarget: true),
            "the captain keeps the captain-specific combat task");
        Assert(!ArmyRtsCaptainCombatRules.ShouldUseMemberCombatTask(
                missionActive: true, actorIsCaptain: false,
                fieldCombatReleased: true, hasValidCombatTarget: false),
            "a targetless soldier remains on native follow until vanilla acquisition finds an enemy");
        Assert(ArmyRtsCaptainCombatRules.ShouldSuppressVanillaMemberFight(
                missionActive: true, actorIsCaptain: false,
                fieldCombatReleased: true, hasValidCombatTarget: false),
            "an active field battle does not put a targetless soldier into an empty combat task");
        Assert(ArmyRtsCaptainCombatRules.ShouldRestoreVanillaMemberFollow(
                suppressVanillaFight: true,
                isDedicatedMemberCombatTask: true),
            "a lost native target restores an RTS member to original follow");
        Assert(!ArmyRtsCaptainCombatRules.ShouldRestoreVanillaMemberFollow(
                suppressVanillaFight: true,
                isDedicatedMemberCombatTask: false),
            "a member already following needs no task rewrite after a target miss");
        Assert(!ArmyRtsCaptainCombatRules.ShouldSuppressVanillaMemberFight(
                missionActive: true, actorIsCaptain: false,
                fieldCombatReleased: true, hasValidCombatTarget: true),
            "a field soldier with a close target receives member combat rather than suppressing combat");
        Assert(ArmyRtsCaptainCombatRules.ShouldSuppressVanillaMemberFight(
                missionActive: true, actorIsCaptain: false,
                fieldCombatReleased: false, hasValidCombatTarget: false),
            "an RTS soldier without a valid close enemy cannot remain in vanilla fighting during a march");
        Assert(ArmyRtsCaptainCombatRules.ShouldUseSiegeCombatTask(
                siegeCombatActive: true,
                actorInsideTargetCityCombatZone: true),
            "a soldier inside the active target-city combat zone joins siege combat");
        Assert(!ArmyRtsCaptainCombatRules.ShouldUseSiegeCombatTask(
                siegeCombatActive: true,
                actorInsideTargetCityCombatZone: false),
            "an outlying soldier keeps following the captain instead of attacking a distant city target");
        Assert(!ArmyRtsCaptainCombatRules.ShouldUseSiegeCombatTask(
                siegeCombatActive: false,
                actorInsideTargetCityCombatZone: true),
            "standing inside a city cannot create siege combat without an active army siege");

        string root = Directory.GetCurrentDirectory();
        string content = File.ReadAllText(Path.Combine(root, "Code",
            "content", "ArmyRtsContent.cs"));
        string controller = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsControllerService.cs"));
        string wartimeTaskPatch = File.ReadAllText(Path.Combine(root,
            "Code", "patch", "AW_WartimeMilitaryTaskPatch.cs"));
        string behaviour = File.ReadAllText(Path.Combine(root, "Code",
            "ai", "behaviours", "actor", "BehArmyRtsCaptainCombat.cs"))
            .Replace("\r\n", "\n");
        Assert(content.Contains("CaptainCombatTaskId", StringComparison.Ordinal) &&
               content.Contains("new BehArmyRtsCaptainCombat()",
                   StringComparison.Ordinal),
            "the captain combat task is registered in the RTS captain job");
        Assert(content.Contains("MemberCombatTaskId", StringComparison.Ordinal) &&
               content.Contains("new BehArmyRtsMemberCombat()",
                   StringComparison.Ordinal) &&
               content.Contains("GoToActorTargetType.SameTile",
                   StringComparison.Ordinal),
            "field soldiers use the original fighting chain after RTS selects a native target");
        Assert(content.Split(new[]
                   {
                       "task.addBeh(new BehRestartTask());"
                   }, StringSplitOptions.None).Length - 1 >= 4,
            "captain, member and siege combat all restart target acquisition after each attack pass");
        string warLocale = File.ReadAllText(Path.Combine(root, "Locales",
            "war.csv"));
        Assert(warLocale.Contains("task_unit_aw_army_rts_captain_combat",
                StringComparison.Ordinal) &&
               warLocale.Contains("task_unit_aw_army_rts_member_combat",
                StringComparison.Ordinal),
            "both RTS combat task labels are localized");
        Assert(controller.Contains("SetCaptainCombatTask",
                   StringComparison.Ordinal) &&
               !controller.Contains("ReleaseArmyActors(pArmy);\n            pRuntime.FieldCombatReleased",
                   StringComparison.Ordinal),
            "field contact does not release the army into an unowned vanilla gap");
        Assert(controller.Contains("RepairCaptainVanillaFight",
                   StringComparison.Ordinal) &&
               controller.Contains("ReassertCaptainMissionTask",
                   StringComparison.Ordinal),
            "a captain is reclaimed from vanilla fighting into bounded combat or its strategic mission");
        Assert(controller.Contains(
                   "internal static bool TryRedirectVanillaFight(",
                   StringComparison.Ordinal),
            "RTS ownership exposes an immediate vanilla-fight redirect");
        int redirectBoundaries = wartimeTaskPatch.Split(new[]
            {
                "ArmyRtsControllerService.TryRedirectVanillaFight("
            }, StringSplitOptions.None).Length - 1;
        Assert(redirectBoundaries >= 2,
            "both vanilla fighting assignment and active-task update redirect immediately into RTS combat");
        int memberMissionStart = controller.IndexOf(
            "private static void SetNativeMemberMissionTask(",
            StringComparison.Ordinal);
        int memberMissionEnd = controller.IndexOf(
            "internal static bool HasImmediateCombatPriority(",
            memberMissionStart, StringComparison.Ordinal);
        string memberMissionBody = memberMissionStart >= 0 &&
            memberMissionEnd > memberMissionStart
            ? controller.Substring(memberMissionStart,
                memberMissionEnd - memberMissionStart)
            : string.Empty;
        Assert(memberMissionBody.Contains(
                   "HasValidMemberCombatTarget(pActor)",
                   StringComparison.Ordinal),
            "job repair cannot replace an individually engaged member combatant with follow before the army-wide threshold is reached");
        Assert(memberMissionBody.Contains(
                   "runtime?.FieldCombatReleased == true",
                   StringComparison.Ordinal),
            "active field combat gives targetless soldiers the bounded RTS combat task");
        Assert(memberMissionBody.Contains(
                   "ArmyRtsCaptainCombatRules.ShouldUseSiegeCombatTask(",
                   StringComparison.Ordinal) &&
               memberMissionBody.Contains(
                   "IsInsideCityCombatZone(pActor, activeSiegeCity)",
                   StringComparison.Ordinal),
            "routine member job repair gates siege combat by the individual soldier's target-city position");
        int siegeRegisterStart = controller.IndexOf(
            "private static void RegisterTargetCitySiegeMembers(",
            StringComparison.Ordinal);
        int siegeRegisterEnd = controller.IndexOf(
            "private static void ExitTargetCitySiegeCombat(",
            Math.Max(0, siegeRegisterStart), StringComparison.Ordinal);
        string siegeRegisterBody = siegeRegisterStart >= 0 &&
            siegeRegisterEnd > siegeRegisterStart
            ? controller.Substring(siegeRegisterStart,
                siegeRegisterEnd - siegeRegisterStart)
            : string.Empty;
        Assert(siegeRegisterBody.Contains(
                   "SetNativeMemberMissionTask(actor, pArmy",
                   StringComparison.Ordinal) &&
               !siegeRegisterBody.Contains(
                   "ArmyRtsContent.SiegeCombatTaskId",
                   StringComparison.Ordinal),
            "siege entry evaluates each member's city position instead of assigning the siege task army-wide");
        Assert(behaviour.Contains("BehArmyRtsMemberCombat",
                   StringComparison.Ordinal) &&
               controller.Contains("FindCaptainCombatTarget(",
                   StringComparison.Ordinal),
            "member combat reuses the bounded RTS target search for civilian targets");
        int assignedTargetStart = controller.IndexOf(
            "internal static bool IsValidAssignedCombatTarget(",
            StringComparison.Ordinal);
        int assignedTargetEnd = controller.IndexOf(
            "internal static bool IsValidSiegeCombatTarget(",
            Math.Max(0, assignedTargetStart), StringComparison.Ordinal);
        string assignedTargetBody = assignedTargetStart >= 0 &&
            assignedTargetEnd > assignedTargetStart
            ? controller.Substring(assignedTargetStart,
                assignedTargetEnd - assignedTargetStart)
            : string.Empty;
        Assert(assignedTargetBody.Contains(
                   "ArmyRtsContent.MemberCombatTaskId",
                   StringComparison.Ordinal) &&
               assignedTargetBody.Contains("IsValidMemberCombatTarget(",
                   StringComparison.Ordinal) &&
               assignedTargetBody.Contains("IsValidCaptainCombatTarget(",
                   StringComparison.Ordinal),
            "the shared attack stage keeps separate captain and member target contracts");
        Assert(controller.Contains(
                   "bool behaviourTarget = IsValidMemberCombatTarget(actor,",
                   StringComparison.Ordinal),
            "field engagement counts a member target outside the captain envelope");
        int enterFieldStart = controller.IndexOf(
            "private static void EnterFieldCombat(",
            StringComparison.Ordinal);
        int enterFieldEnd = controller.IndexOf(
            "private static bool EnterTargetCitySiegeCombat(",
            enterFieldStart, StringComparison.Ordinal);
        string enterFieldBody = enterFieldStart >= 0 &&
            enterFieldEnd > enterFieldStart
            ? controller.Substring(enterFieldStart,
                enterFieldEnd - enterFieldStart)
            : string.Empty;
        int publishFieldCombat = enterFieldBody.IndexOf(
            "pRuntime.FieldCombatReleased = true",
            StringComparison.Ordinal);
        int assignFieldMembers = enterFieldBody.IndexOf(
            "RegisterFieldCombatMembers(pArmy)",
            StringComparison.Ordinal);
        Assert(publishFieldCombat >= 0 &&
               assignFieldMembers > publishFieldCombat,
            "field combat ownership must be published before member tasks are assigned");
        string normalizedBehaviour = behaviour.Replace("\r\n", "\n");
        Assert(normalizedBehaviour.Contains("beh_actor_target", StringComparison.Ordinal) &&
               normalizedBehaviour.Contains("tryToAttack", StringComparison.Ordinal) &&
               normalizedBehaviour.Contains("pActor?.makeWait(0.15f);\n                return BehResult.Continue;",
                    StringComparison.Ordinal),
            "an attack-stage target miss advances into task restart instead of repeating forever on an empty target");
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
}
