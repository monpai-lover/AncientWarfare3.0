using System;
using System.Collections.Generic;
using System.Linq;
using AncientWarfare3.core.court;

internal static class CountyShortlistRulesTests
{
    internal static void Run()
    {
        // 择优顺序是全序:门第档次升序,同档主属性降序,同分按 id 升序。
        // 门第档次压过能力 —— 世家子弟不经科举也能从县令起步,这一条比
        // 「谁属性高」优先。
        True(CountyShortlistRules.SortsBefore(
                (int)LocalLowOfficeCandidateTier.Qualified, 1, 999L,
                (int)LocalLowOfficeCandidateTier.Clan, 99, 1L),
            "an exam-qualified candidate outranks a stronger clan candidate");
        True(CountyShortlistRules.SortsBefore(
                (int)LocalLowOfficeCandidateTier.Clan, 1, 999L,
                (int)LocalLowOfficeCandidateTier.Ordinary, 99, 1L),
            "a clan candidate outranks a stronger commoner");
        False(CountyShortlistRules.SortsBefore(
                (int)LocalLowOfficeCandidateTier.Ordinary, 99, 1L,
                (int)LocalLowOfficeCandidateTier.Clan, 1, 999L),
            "a commoner never outranks a clan candidate on ability alone");
        True(CountyShortlistRules.SortsBefore(1, 9, 100L, 1, 5, 1L),
            "within a tier, higher ability sorts first regardless of id");
        True(CountyShortlistRules.SortsBefore(1, 5, 1L, 1, 5, 2L),
            "equal tier and ability breaks the tie by ascending id");
        False(CountyShortlistRules.SortsBefore(1, 5, 2L, 1, 5, 1L),
            "the tie-break must be antisymmetric");

        // 满员时,排在末位之后的人可以直接丢 —— 他永远进不了前 K。
        True(CountyShortlistRules.CanSkipWhenFull(2, 99, 1L, 1, 3, 50L),
            "a lower-tier candidate cannot enter a full shortlist");
        True(CountyShortlistRules.CanSkipWhenFull(1, 3, 50L, 1, 7, 9L),
            "a worse candidate cannot enter a full shortlist");
        True(CountyShortlistRules.CanSkipWhenFull(1, 7, 50L, 1, 7, 9L),
            "an equal-ability higher id cannot displace the last entry");
        False(CountyShortlistRules.CanSkipWhenFull(1, 8, 50L, 1, 7, 9L),
            "a better candidate must displace the last entry");

        // 核心性质:有界插入 == 全量排序后取前 K。这是把两千多人的
        // OrderByDescending 换成有界插入的全部依据,所以随机对拍。
        var random = new Random(20260830);
        for (int trial = 0; trial < 2000; trial++)
        {
            int count = random.Next(0, 60);
            int limit = 1 + random.Next(0, 12);
            // 档次与属性取值范围故意很窄,好让同档同分大量出现,把 tie-break 压满。
            var tiers = new List<int>();
            var abilities = new List<int>();
            var ids = new List<long>();
            var usedIds = new HashSet<long>();
            for (int index = 0; index < count; index++)
            {
                tiers.Add(random.Next(0, 3));
                abilities.Add(random.Next(0, 5));
                long id;
                do { id = random.Next(0, 200); } while (!usedIds.Add(id));
                ids.Add(id);
            }

            List<long> actual = CountyShortlistRules.TopIds(tiers, abilities,
                ids, limit);
            List<long> expected = Enumerable.Range(0, count)
                .OrderBy(index => tiers[index])
                .ThenByDescending(index => abilities[index])
                .ThenBy(index => ids[index])
                .Take(limit)
                .Select(index => ids[index])
                .ToList();
            if (!actual.SequenceEqual(expected))
                throw new InvalidOperationException(
                    "bounded insertion must equal the sorted prefix: trial " +
                    trial + " limit " + limit +
                    " got [" + string.Join(",", actual) +
                    "] want [" + string.Join(",", expected) + "]");
        }

        // 上限用尽时结果就是完整排序 —— 全量路径与短名单路径同源。
        var flatTiers = new List<int> { 0, 1, 1, 0 };
        var flatAbilities = new List<int> { 1, 3, 3, 2 };
        var flatIds = new List<long> { 40L, 10L, 5L, 99L };
        Equal("99,40,5,10",
            string.Join(",", CountyShortlistRules.TopIds(flatTiers,
                flatAbilities, flatIds, int.MaxValue)),
            "an unbounded shortlist is the full ordering");
        Equal("99,40", string.Join(",",
                CountyShortlistRules.TopIds(flatTiers, flatAbilities, flatIds,
                    2)),
            "a bounded shortlist keeps exactly the best entries");
        Equal("", string.Join(",",
                CountyShortlistRules.TopIds(flatTiers, flatAbilities, flatIds,
                    0)),
            "a non-positive limit yields nothing");

        Console.WriteLine("CountyShortlistRulesTests passed.");
    }

    private static void Equal(string pExpected, string pActual,
        string pMessage)
    {
        if (!string.Equals(pExpected, pActual, StringComparison.Ordinal))
            throw new InvalidOperationException(pMessage + ": expected [" +
                pExpected + "] got [" + pActual + "]");
    }

    private static void True(bool pValue, string pMessage)
    {
        if (!pValue) throw new InvalidOperationException(pMessage);
    }

    private static void False(bool pValue, string pMessage)
    {
        if (pValue) throw new InvalidOperationException(pMessage);
    }
}
