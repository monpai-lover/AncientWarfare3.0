using AncientWarfare3.core.lineage;

internal static class WarForceEliminationRulesTests
{
    public static void Run()
    {
        Equal(1, WarForceEliminationRules.NextZeroStreak(0, 0),
            "first zero observation starts confirmation");
        Equal(1, WarForceEliminationRules.NextZeroStreak(0, 1),
            "zero confirmation saturates after one observation");
        Equal(0, WarForceEliminationRules.NextZeroStreak(1, 2),
            "positive potential clears the zero streak");

        Equal(WarForceEliminationDecisionKind.ScoreSettlement,
            WarForceEliminationRules.Resolve(0, 0, 1, 1, 40).Kind,
            "one zero observation settles immediately");
        Equal(WarForceEliminationDecisionKind.DefendersSurrender,
            WarForceEliminationRules.Resolve(3, 0, 0, 2, -20).Kind,
            "confirmed defender exhaustion overrides score");
        Equal(WarForceEliminationDecisionKind.AttackersSurrender,
            WarForceEliminationRules.Resolve(0, 4, 2, 0, 20).Kind,
            "confirmed attacker exhaustion overrides score");

        Equal(WarForceEliminationDecisionKind.DefendersSurrender,
            WarForceEliminationRules.ResolveFullOccupation(
                pAttackerInitialCities: 2, pAttackerOccupiedCities: 0,
                pDefenderInitialCities: 3, pDefenderOccupiedCities: 3).Kind,
            "full defender occupation forces surrender even with a live army");
        Equal(WarForceEliminationDecisionKind.None,
            WarForceEliminationRules.ResolveFullOccupation(
                pAttackerInitialCities: 2, pAttackerOccupiedCities: 1,
                pDefenderInitialCities: 3, pDefenderOccupiedCities: 2).Kind,
            "partial occupation does not force terminal settlement");

        WarForceEliminationDecision attackerBenefit =
            WarForceEliminationRules.Resolve(0, 0, 2, 2, 35);
        Equal(WarForceEliminationDecisionKind.ScoreSettlement,
            attackerBenefit.Kind,
            "mutual exhaustion at positive score uses score settlement");
        Equal(WarScoreSide.Attackers, attackerBenefit.Beneficiary,
            "positive score benefits attackers");

        WarForceEliminationDecision defenderBenefit =
            WarForceEliminationRules.Resolve(0, 0, 2, 2, -35);
        Equal(WarForceEliminationDecisionKind.ScoreSettlement,
            defenderBenefit.Kind,
            "mutual exhaustion at negative score uses score settlement");
        Equal(WarScoreSide.Defenders, defenderBenefit.Beneficiary,
            "negative score benefits defenders");
        Equal(35, defenderBenefit.Score,
            "benefit score is exposed as a positive budget");

        Equal(WarForceEliminationDecisionKind.WhitePeace,
            WarForceEliminationRules.Resolve(0, 0, 2, 2, 0).Kind,
            "mutual exhaustion at zero score creates white peace");
        Equal(int.MaxValue,
            WarForceEliminationRules.AddPotential(int.MaxValue, 10),
            "military potential clamps overflow");
        Equal(5, WarForceEliminationRules.AddPotential(-2, 5),
            "negative facts do not reduce military potential");
        MilitaryPotentialReusesVanillaStatistics();
        NativeKingdomCityRosterIsAuthoritative();
        DuplicateObservationsInOneMonthDoNotConfirmExhaustion();
        RuntimeWiresImmediateTerminalCoordinator();
        OrdinarySettlementUsesTheRequiredOfferMode();
        SpecialWarsUseBoundedAdapters();
    }

    private static void SpecialWarsUseBoundedAdapters()
    {
        Equal(WarForceSpecialSettlementKind.Rebellion,
            WarForceEliminationRules.SpecialKind(true),
            "rebellion selects its dedicated adapter");
        Equal(WarForceSpecialSettlementKind.None,
            WarForceEliminationRules.SpecialKind(false),
            "ordinary war has no special adapter");

        string root = Directory.GetCurrentDirectory();
        string central = File.ReadAllText(Path.Combine(root, "Code", "core",
            "lineage", "WarForceEliminationSettlementService.cs"));
        string router = File.ReadAllText(Path.Combine(root, "Code", "core",
            "lineage", "WarForceSpecialSettlementService.cs"));
        string coordinator = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "WarTerminalSettlementCoordinator.cs"));
        string rebellion = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "RebellionCollapseSettlementService.cs"));
        True(central.Contains(
                "WarForceSpecialSettlementService.TrySettle"),
            "special router runs before ordinary settlement");
        True(central.Contains("pWar.countAttackersWarriors()") &&
             central.Contains("pWar.countDefendersWarriors()") &&
             !central.Contains("CountSideReserves") &&
             !central.Contains("CityReservePoolService.CountAvailable"),
            "ordinary force exhaustion uses only native War force facts");
        True(coordinator.Contains("TrySettleZhuluZeroForce(war)") &&
             router.Contains("ResolveZeroForceFallback(") &&
             !router.Contains("ZhuluWarSettlementService"),
            "Zhulu uses only its direct zero-force liveness fallback");
        True(rebellion.Contains("QueueForceElimination") &&
             rebellion.Contains("WarForceEliminationDecision"),
            "rebellion exposes a force-elimination adapter");
    }

    private static void OrdinarySettlementUsesTheRequiredOfferMode()
    {
        Equal(WarForceSettlementOfferMode.Surrender,
            WarForceEliminationRules.OfferMode(
                WarForceEliminationDecisionKind.DefendersSurrender),
            "one-sided exhaustion identifies the losing side");
        Equal(WarForceSettlementOfferMode.MaximumBenefit,
            WarForceEliminationRules.OfferMode(
                WarForceEliminationDecisionKind.ScoreSettlement),
            "mutual exhaustion maximizes current-score benefit");
        Equal(WarForceSettlementOfferMode.WhitePeace,
            WarForceEliminationRules.OfferMode(
                WarForceEliminationDecisionKind.WhitePeace),
            "zero-score mutual exhaustion uses white peace");

        string service = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "WarPeaceSettlementService.cs"));
        True(service.Contains("ForceMilitaryEliminationSettlement"),
            "ordinary settlement exposes an elimination entry point");
        True(service.Contains("WarPeaceTermsRules.MaximumWarScore") &&
             !service.Contains("? -WarPeaceTermsRules.MaximumWarScore"),
            "force elimination gives the winner a positive full score");

        string elimination = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "WarForceEliminationSettlementService.cs"));
        True(elimination.Contains("Kingdom requester = winner") &&
             elimination.Contains("WarPeaceDefaultOfferMode.ExhaustionMaximumBenefit"),
            "force exhaustion negotiates from the winner at maximum benefit");
        True(elimination.Contains(
                 "if (attackers == 0 && defenders == 0") &&
             elimination.Contains("TryReadAttackerScore"),
            "only mutual exhaustion requires a current score read");
    }

    private static void DuplicateObservationsInOneMonthDoNotConfirmExhaustion()
    {
        var state = new WarForceObservationState();
        True(state.Observe(1201, 0, 0),
            "first observation in a month is accepted");
        Equal(1, state.AttackerZeroStreak,
            "first month records one attacker zero");
        False(state.Observe(1201, 0, 0),
            "duplicate observation in one month is ignored");
        Equal(1, state.AttackerZeroStreak,
            "duplicate event cannot confirm attacker exhaustion");
        True(state.Observe(1202, 0, 3),
            "next month is accepted");
        Equal(1, state.AttackerZeroStreak,
            "zero observation remains confirmed without another month");
        Equal(0, state.DefenderZeroStreak,
            "recovered defender clears its zero streak");
        False(state.Observe(1201, 0, 0),
            "late work from an older month is ignored");
        Equal(1, state.AttackerZeroStreak,
            "out-of-order work cannot create another zero month");
    }

    private static void RuntimeWiresImmediateTerminalCoordinator()
    {
        string root = Directory.GetCurrentDirectory();
        string authority = File.ReadAllText(Path.Combine(root, "Code",
            "core", "performance", "AWAuthorityCycleService.cs"));
        string bridge = File.ReadAllText(Path.Combine(root, "Code", "core",
            "lineage", "WarScoreRuntimeBridge.cs"));
        True(System.Text.RegularExpressions.Regex.IsMatch(authority,
                @"WarTerminalSettlementCoordinator\s*\.\s*" +
                @"ProcessAuthorityCycle"),
            "authority cycle drains bounded terminal recovery work");
        True(authority.Contains(
                "WarTerminalSettlementCoordinator.ClearRuntime"),
            "world reset clears terminal recovery state");
        True(bridge.Contains(
                "WarTerminalSettlementCoordinator." +
                "NotifyWarChanged(pWar)"),
            "war mutations notify immediate terminal settlement");
    }

    private static void MilitaryPotentialReusesVanillaStatistics()
    {
        string source = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "WartimeMilitaryPotentialService.cs"));
        True(source.Contains("pKingdom.countTotalWarriors()"),
            "kingdom potential reuses vanilla warrior aggregation");
        False(source.Contains(
                "ArmyFieldIndexService.CreateSnapshotCursor"),
            "kingdom potential does not rescan Army indexes");
        True(source.Contains("TryCountAvailable"),
             "AW3 reserves remain additive with explicit readiness");
    }

    private static void NativeKingdomCityRosterIsAuthoritative()
    {
        string root = Directory.GetCurrentDirectory();
        string baseline = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "WarParticipantCityBaselineService.cs"));
        string elimination = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage",
            "WarForceEliminationSettlementService.cs"));
        True(baseline.Contains("pKingdom.cities.Count"),
            "war baseline reads the native kingdom city roster");
        False(baseline.Contains("pKingdom.countCities()"),
            "war baseline does not use the non-native city counter");
        True(elimination.Contains("pKingdom.cities.Count"),
            "full occupation reads the native kingdom city roster");
        False(elimination.Contains("pKingdom?.countCities()"),
            "full occupation does not fall back to the non-native city counter");
    }

    private static void True(bool value, string name)
    {
        if (!value) throw new InvalidOperationException(
            name + ": expected true");
    }

    private static void False(bool value, string name)
    {
        if (value) throw new InvalidOperationException(
            name + ": expected false");
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(name + ": expected " +
                                                expected + ", got " +
                                                actual);
    }
}
