using System;
using AncientWarfare3.core.lineage;

internal static class CityArmyReinforcementRulesTests
{
    public static void Run()
    {
        Equal(35, CityArmyReinforcementRules.CityCapacity(100, 80),
            "city capacity uses the thirty-five percent population cap");
        Equal(20, CityArmyReinforcementRules.CityCapacity(100, 20),
            "warrior slots remain a city capacity ceiling");

        CityArmyReinforcementAllocation[] allocations =
            CityArmyReinforcementRules.Allocate(60, new[]
            {
                new CityArmyReinforcementRequest(9L, 10, 40,
                    CityArmyPriority.Reserve),
                new CityArmyReinforcementRequest(2L, 10, 40,
                    CityArmyPriority.Frontline),
                new CityArmyReinforcementRequest(1L, 10, 40,
                    CityArmyPriority.Frontline)
            });

        Equal(40, TargetFor(allocations, 1L),
            "stable army identity resolves equal priority allocation");
        Equal(10, TargetFor(allocations, 2L),
            "higher priority capacity is allocated before later peers");
        Equal(10, TargetFor(allocations, 9L),
            "lower priority keeps its existing soldiers without new capacity");
        Equal(12, CityArmyReinforcementRules.Shortage(28, 40),
            "approved target alone defines displayed reinforcement shortage");
        Equal(0, CityArmyReinforcementRules.Shortage(40, 35),
            "a population decline never creates a negative shortage");
        Equal(30, ArmyMapInformationRules.ResolvePendingShortage(
                hasMission: false, missionTargetStrength: 0,
                activeDeployment: false, standingTargetStrength: 50,
                memberCount: 20, minimumOperationalForce: 10),
            "an idle ordinary army reports shortage against its city capacity");
        Equal(30, ArmyMapInformationRules.ResolvePendingShortage(
                hasMission: true, missionTargetStrength: 0,
                activeDeployment: false, standingTargetStrength: 50,
                memberCount: 20, minimumOperationalForce: 10),
            "a zero legacy mission target falls back to city capacity");

        Equal(ArmyRecruitmentDisposition.Replenish,
            ArmyEstablishmentRules.DecideCityRecruitment(
                existingOrdinaryArmyCount: 1,
                kingdomFieldArmyCount: 99),
            "an existing city army is always reused even when full");
        Equal(ArmyRecruitmentDisposition.Create,
            ArmyEstablishmentRules.DecideCityRecruitment(
                existingOrdinaryArmyCount: 0,
                kingdomFieldArmyCount: 99),
            "a city without an army may create one above the old realm cap");
        Equal(10L, ArmyEstablishmentRules.SelectCanonicalCityArmy(
                firstArmyId: 10L, firstCaptainStable: true,
                secondArmyId: 5L, secondCaptainStable: false),
            "a stable living captain preserves command continuity");
        Equal(5L, ArmyEstablishmentRules.SelectCanonicalCityArmy(
                firstArmyId: 10L, firstCaptainStable: true,
                secondArmyId: 5L, secondCaptainStable: true),
            "equal command continuity uses deterministic army identity");
        Equal(false, ArmyEstablishmentRules.ShouldMaintainExcessFieldArmies(
                fieldArmyCount: 99),
            "realm army count is a scan budget and never a cross-city merge authority");
    }

    private static int TargetFor(CityArmyReinforcementAllocation[] pAllocations,
        long pArmyId)
    {
        foreach (CityArmyReinforcementAllocation allocation in pAllocations)
            if (allocation.ArmyId == pArmyId) return allocation.ApprovedTarget;
        throw new InvalidOperationException("missing allocation for army " + pArmyId);
    }

    private static void Equal<T>(T pExpected, T pActual, string pName)
    {
        if (!Equals(pExpected, pActual))
            throw new InvalidOperationException(pName + ": expected " +
                pExpected + ", got " + pActual);
    }
}
