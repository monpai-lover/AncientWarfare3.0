using System;
using System.IO;
using AncientWarfare3.core.lineage;
using AncientWarfare3.core.performance;

internal static class ArmyRtsCombatHandoffRulesTests
{
    public static void Run()
    {
        Assert(ArmyRtsP0Rules.ShouldTraceDiagnosticActor(
                isCaptain: true, isRoyalGuard: false, actorId: 17,
                anomaly: false, memberSampleModulo: 16),
            "captain diagnostics are never sampled out");
        Assert(ArmyRtsP0Rules.ShouldTraceDiagnosticActor(
                isCaptain: false, isRoyalGuard: false, actorId: 224,
                anomaly: false, memberSampleModulo: 16) &&
               !ArmyRtsP0Rules.ShouldTraceDiagnosticActor(
                isCaptain: false, isRoyalGuard: false, actorId: 225,
                anomaly: false, memberSampleModulo: 16),
            "ordinary followers use a stable one-in-sixteen diagnostic sample");
        Assert(ArmyRtsP0Rules.ShouldTraceDiagnosticActor(
                isCaptain: false, isRoyalGuard: false, actorId: 225,
                anomaly: true, memberSampleModulo: 16),
            "anomalies bypass follower diagnostic sampling");
        Assert(ArmyMilitaryMovementPriorityRules.ShouldRetainCombatP0(
                largeSchedulerActive: true, militaryOwnerActive: true,
                immediateCombat: true),
            "an RTS combatant keeps military P0 while vanilla combat owns its actions");
        Assert(!ArmyMilitaryMovementPriorityRules.ShouldRetainCombatP0(
                largeSchedulerActive: true, militaryOwnerActive: false,
                immediateCombat: true),
            "an unrelated combatant is not admitted to military P0");
        Assert(!ArmyMilitaryMovementPriorityRules.ShouldRetainCombatP0(
                largeSchedulerActive: false, militaryOwnerActive: true,
                immediateCombat: true),
            "native scheduling does not need the large-step combat lane");
        Assert(ArmyMilitaryMovementPriorityRules.
                ShouldAdvanceNewFightTaskInSameP0(
                    hasAttackTarget: true, isFightingTask: true,
                    actionIndexBeforeAi: 0, actionIndexAfterAi: 1,
                    behaviourSkipped: false, alreadyMoving: false),
            "a newly acquired distant enemy advances from validation into the native approach action in one P0 slice");
        Assert(!ArmyMilitaryMovementPriorityRules.
                ShouldAdvanceNewFightTaskInSameP0(
                    hasAttackTarget: true, isFightingTask: true,
                    actionIndexBeforeAi: 0, actionIndexAfterAi: 1,
                    behaviourSkipped: true, alreadyMoving: false),
            "an in-range combat action that already consumed the native slice is not advanced again");
        Assert(!ArmyMilitaryMovementPriorityRules.
                ShouldAdvanceNewFightTaskInSameP0(
                    hasAttackTarget: true, isFightingTask: true,
                    actionIndexBeforeAi: 1, actionIndexAfterAi: 2,
                    behaviourSkipped: false, alreadyMoving: false),
            "only the fighting validation-to-approach transition is coalesced");
        Assert(ArmyMilitaryMovementPriorityRules.
                ShouldAdvanceMemberCombatApproachInSameP0(
                    hasAttackTarget: true, isMemberCombatTask: true,
                    actionIndexBeforeAi: 0, actionIndexAfterAi: 1,
                    behaviourSkipped: false, alreadyMoving: false),
            "the custom soldier combat task reaches its original approach action without waiting for another large step");
        Assert(ArmyMilitaryMovementPriorityRules.
                ShouldResumeNativeCombatAfterEnemyAcquisition(
                    hadAttackTargetBeforeSearch: false,
                    hasAttackTargetAfterSearch: true,
                    behaviourSkippedBySearch: true),
            "a P0 enemy acquisition must advance the native fighting task instead of leaving it skipped");
        Assert(!ArmyMilitaryMovementPriorityRules.
                ShouldResumeNativeCombatAfterEnemyAcquisition(
                    hadAttackTargetBeforeSearch: true,
                    hasAttackTargetAfterSearch: true,
                    behaviourSkippedBySearch: true),
            "an existing combat target keeps native cooldown and attack skip semantics");
        Assert(ArmyRtsP0Rules.ShouldWriteDiagnosticStage(
                "combat_p0", anomaly: false),
            "combat ownership transitions remain visible");
        Assert(ArmyRtsP0Rules.ShouldWriteDiagnosticStage(
                "combat_after_move_command", anomaly: false),
            "the native fighting approach command remains visible during large-step diagnosis");
        Assert(ArmyRtsP0Rules.ShouldWriteDiagnosticStage(
                "follower_after_ai", anomaly: false),
            "native follower target publication remains visible during large-step diagnosis");
        Assert(!ArmyRtsP0Rules.ShouldWriteDiagnosticStage(
                "native_enemy_check", anomaly: false),
            "normal per-stage actor diagnostics are suppressed");
        Assert(ArmyRtsP0Rules.ShouldWriteDiagnosticStage(
                "native_enemy_check", anomaly: true),
            "anomalies bypass stage suppression");
        Assert(ArmyRtsP0Rules.ShouldRateLimitDiagnostic(
                anomaly: false, elapsedSeconds: 0.25d,
                minimumIntervalSeconds: 2d),
            "normal repeated diagnostics are rate limited");
        Assert(!ArmyRtsP0Rules.ShouldRateLimitDiagnostic(
                anomaly: true, elapsedSeconds: 0.25d,
                minimumIntervalSeconds: 2d),
            "anomalies are emitted immediately");
        Assert(ArmyRtsControllerRules.ShouldUseVanillaArmyMovement(
                ArmyRtsProposalKind.Attack, targetIsEnemy: true),
            "an open enemy-city attack uses vanilla army movement");
        Assert(!ArmyRtsControllerRules.ShouldUseVanillaArmyMovement(
                ArmyRtsProposalKind.Defend, targetIsEnemy: false),
            "friendly defense does not publish a vanilla city attack order");
        Assert(!ArmyRtsControllerRules.ShouldUseVanillaArmyMovement(
                ArmyRtsProposalKind.Retreat, targetIsEnemy: false),
            "retreat cannot be published as a vanilla enemy-city attack");
        Assert(ArmyRtsControllerRules.ShouldUseVanillaRetreatMovement(
                ArmyRtsProposalKind.Retreat, targetIsEnemy: false),
            "a friendly retreat uses the vanilla-style return-city task");
        Assert(ArmyRtsControllerRules.ShouldUseVanillaFollowerMovement(
                ArmyRtsProposalKind.Retreat, targetIsEnemy: false),
            "retreat soldiers use vanilla follow-leader movement");
        Assert(ArmyRtsControllerRules.ShouldUseVanillaFollowerMovement(
                ArmyRtsProposalKind.Attack, targetIsEnemy: true),
            "attack soldiers use vanilla follow-leader movement");
        Assert(ArmyRtsControllerRules.ShouldUseVanillaFollowerMovement(
                ArmyRtsProposalKind.Defend, targetIsEnemy: false),
            "defense soldiers use vanilla follow-leader movement");
        Assert(ArmyRtsControllerRules.ShouldUseVanillaFollowerMovement(
                ArmyRtsProposalKind.Attack, targetIsEnemy: false),
            "a transient target ownership change cannot reopen custom follower movement");
        Assert(!ArmyRtsControllerRules.ShouldUseVanillaFollowerMovement(
                ArmyRtsProposalKind.FrontHold, targetIsEnemy: false),
            "a front-hold task remains stationary rather than following a leader");
        Assert(ArmyRtsControllerRules.ShouldUseNativeMissionExecution(
                ArmyRtsProposalKind.Attack, ArmyRtsState.March,
                targetIsEnemy: true),
            "attack marches must use the original captain attack path");
        Assert(ArmyRtsControllerRules.ShouldUseNativeMissionExecution(
                ArmyRtsProposalKind.Defend, ArmyRtsState.Deploy,
                targetIsEnemy: false),
            "defense deployment must use original army movement");
        Assert(ArmyRtsControllerRules.ShouldUseNativeMissionExecution(
                ArmyRtsProposalKind.Retreat, ArmyRtsState.Retreat,
                targetIsEnemy: false),
            "retreat movement must use original movement and taxi ownership");
        Assert(!ArmyRtsControllerRules.ShouldUseNativeMissionExecution(
                ArmyRtsProposalKind.Attack, ArmyRtsState.Rally,
                targetIsEnemy: true),
            "rally remains an RTS-owned preparation phase");
        Assert(ArmyRtsControllerRules.ShouldResetNativeMovement(
                previousMissionExists: true,
                operationalProgressChanged: true),
            "a replaced strategic command clears its old native path once");
        Assert(!ArmyRtsControllerRules.ShouldResetNativeMovement(
                previousMissionExists: true,
                operationalProgressChanged: false),
            "an unchanged strategic command preserves its active native path");
        Assert(ArmyRtsControllerRules.
                ShouldPrimeForcedTransportBeforeNativeAttack(
                    isCaptain: true, usesVanillaAttack: true,
                    sameIsland: false, transportOwned: false),
            "a cross-island attack captain must start transport before the land-only native selector");
        Assert(!ArmyRtsControllerRules.
                ShouldPrimeForcedTransportBeforeNativeAttack(
                    isCaptain: true, usesVanillaAttack: true,
                    sameIsland: true, transportOwned: false),
            "a same-island attack captain must continue through native land movement");
        Assert(ArmyRtsWarLifecycleRules.ResolveCombatControl(
                ArmyRtsWarPhase.StrategicMovement,
                withdrawalRequired: false, insideTargetTerritory: true,
                hostileCombatUnitNearby: true, objectiveOpen: true) ==
               ArmyRtsCombatControlDecision.ReleaseToVanilla,
            "target-territory hostile contact releases tactical control");

        Assert(ArmyRtsWarLifecycleRules.ShouldTreatAsTargetTerritory(
                exactZone: false, borderZone: true,
                adjacentTargetZone: false),
            "a target city border tile must count as target territory");

        Assert(ArmyRtsWarLifecycleRules.ResolveCombatControl(
                ArmyRtsWarPhase.StrategicMovement,
                withdrawalRequired: false, insideTargetTerritory: false,
                hostileCombatUnitNearby: true, objectiveOpen: true) ==
               ArmyRtsCombatControlDecision.KeepStrategicControl,
            "passing through enemy territory retains RTS movement");

        Assert(ArmyRtsWarLifecycleRules.ResolveCombatControl(
                ArmyRtsWarPhase.VanillaCombat,
                withdrawalRequired: true, insideTargetTerritory: true,
                hostileCombatUnitNearby: true, objectiveOpen: true) ==
               ArmyRtsCombatControlDecision.ReacquireForWithdrawal,
            "the 20 percent threshold overrides vanilla combat");

        Assert(ArmyRtsWarLifecycleRules.ResolveCombatControl(
                ArmyRtsWarPhase.Withdrawal,
                withdrawalRequired: true, insideTargetTerritory: false,
                hostileCombatUnitNearby: false, objectiveOpen: true) ==
               ArmyRtsCombatControlDecision.KeepStrategicControl,
            "an active withdrawal must not be reassigned every frame");
        Assert(ArmyRtsWarLifecycleRules.ResolveCombatControl(
                ArmyRtsWarPhase.Withdrawal,
                withdrawalRequired: true, insideTargetTerritory: true,
                hostileCombatUnitNearby: true, objectiveOpen: true) ==
               ArmyRtsCombatControlDecision.KeepStrategicControl,
            "withdrawal ownership must not hand control back to vanilla combat");

        Assert(ArmyRtsWarLifecycleRules.ResolveCombatControl(
                ArmyRtsWarPhase.VanillaCombat,
                withdrawalRequired: false, insideTargetTerritory: true,
                hostileCombatUnitNearby: false, objectiveOpen: true) ==
               ArmyRtsCombatControlDecision.KeepVanillaControl,
            "a temporary lull must not let RTS reclaim actors before the target city is occupied");

        Assert(ArmyRtsWarLifecycleRules.ResolveCombatControl(
                ArmyRtsWarPhase.VanillaCombat,
                withdrawalRequired: false, insideTargetTerritory: false,
                hostileCombatUnitNearby: false, objectiveOpen: true) ==
               ArmyRtsCombatControlDecision.KeepVanillaControl,
            "vanilla combat ownership remains locked while the strategic objective is open");

        Assert(ArmyRtsWarLifecycleRules.ResolveCombatControl(
                ArmyRtsWarPhase.VanillaCombat,
                withdrawalRequired: false, insideTargetTerritory: true,
                hostileCombatUnitNearby: true, objectiveOpen: false) ==
               ArmyRtsCombatControlDecision.ReacquireStrategicControl,
            "objective completion leaves vanilla combat ownership");

        Assert(ArmyRtsWarLifecycleRules.ResolveCombatControl(
                ArmyRtsWarPhase.VanillaCombat,
                withdrawalRequired: false, insideTargetTerritory: true,
                hostileCombatUnitNearby: true, objectiveOpen: true) ==
               ArmyRtsCombatControlDecision.KeepVanillaControl,
            "active target-city combat remains under vanilla AI");

        Assert(!ArmyRtsWarLifecycleRules.OwnsTacticalActors(
                ArmyRtsWarPhase.VanillaCombat),
            "RTS actor patches must remain disabled during vanilla combat");

        Assert(ArmyRtsWarLifecycleRules.ShouldAllowVanillaCityAttack(
                ArmyRtsWarPhase.VanillaCombat),
            "vanilla city attack selection must resume after RTS handoff");

        Assert(ArmyRtsTaskOwnershipRules.ShouldReassertMissionTask(
                ArmyRtsMode.On, pOwnsActor: false, pActorAlive: true,
                pExpectedJobActive: false, pExpectedTaskActive: false,
                pImmediateCombat: false, pRequiredBoatWork: false,
                pForceRecovery: true),
            "watchdog recovery must repair a task after ownership was lost");
        Assert(ArmyRtsWarLifecycleRules.OwnsTacticalActors(
                ArmyRtsWarPhase.Withdrawal),
            "RTS actor patches resume ownership for withdrawal");

        Assert(ArmyRtsRuntimeModeRules.
                ShouldAllowVanillaStrategicDecision(
                    ArmyRtsMode.On,
                    "warrior_army_leader_move_to_attack_target",
                    rtsOwnsActor: false),
            "released combat actors retain vanilla attack decisions");
        Assert(ArmyRtsRuntimeModeRules.
                ShouldAllowVanillaStrategicDecision(
                    ArmyRtsMode.On,
                    "warrior_army_leader_move_to_attack_target",
                    rtsOwnsActor: true),
            "RTS-owned captains use vanilla attack-target movement");
        Assert(ArmyRtsRuntimeModeRules.
                ShouldAllowVanillaStrategicDecision(
                    ArmyRtsMode.On, "warrior_army_follow_leader",
                    rtsOwnsActor: true),
            "RTS-owned soldiers use vanilla follow-leader movement");
        Assert(ArmyRtsRuntimeModeRules.
                ShouldAllowVanillaStrategicDecision(
                    ArmyRtsMode.On, "check_warrior_transport",
                    rtsOwnsActor: true),
            "RTS-owned armies use vanilla cross-island transport");
        Assert(ArmyRtsRuntimeModeRules.
                ShouldAllowVanillaDecisionEvaluation(

        Assert(!ArmyRtsRuntimeModeRules.
                ShouldAllowVanillaDecisionEvaluation(

                    nativeMilitaryMovement: false),
            "custom RTS actors remain protected from unrelated vanilla decisions");

    // AWPerformanceSettings.SwitchArmyRts changes the configuration while a
    // world runs, so the runtime mode must be resolved on each read rather
    // than captured once during type initialization.
    private static void RuntimeModeFollowsTheLiveConfiguration(string pRoot)
    {
        Assert(ArmyRtsRuntimeModeRules.ResolveConfiguredMode(true) ==
               ArmyRtsMode.On,
            "an enabled configuration resolves to commit mode");
        Assert(ArmyRtsRuntimeModeRules.ResolveConfiguredMode(false) ==
               ArmyRtsMode.Off,
            "a disabled configuration resolves to off");

        string source = File.ReadAllText(Path.Combine(pRoot, "Code",
            "core", "lineage", "ArmyRtsRuntimeMode.cs"));
        int start = source.IndexOf(
            "internal static class ArmyRtsRuntimeMode",
            StringComparison.Ordinal);
        Assert(start >= 0,
            "the runtime accessor must remain in ArmyRtsRuntimeMode.cs");
        string body = source.Substring(start);
        Assert(!body.Contains("static readonly ArmyRtsMode",
                StringComparison.Ordinal),
            "a cached startup snapshot would ignore the runtime RTS toggle");
        Assert(body.Contains("AWPerformanceSettings.EnableArmyRts",
                StringComparison.Ordinal),
            "the runtime mode must read the live configuration value");
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
