using System;
using System.IO;
using AncientWarfare3.core.lineage;

internal static class SyntheticMobilizationRulesTests
{
    internal static void Run()
    {
        SyntheticToggleControlsGenerationAndVanillaRecruitment();
        Equal(50, SyntheticMobilizationRules.Quota(100, 0, 50),
            "a 50 percent law mobilizes half the city population");
        Equal(40, SyntheticMobilizationRules.Quota(100, 20, 50),
            "existing synthetic actors cannot inflate real population");
        Equal(0, SyntheticMobilizationRules.Quota(3, 0, 30),
            "small-city quotas use floor without a forced minimum");
        Equal(30, SyntheticMobilizationRules.Quota(100, 0, 30),
            "the 30 percent law is exact");
        Equal(70, SyntheticMobilizationRules.Quota(100, 0, 70),
            "the 70 percent law is exact");
        Equal(100, SyntheticMobilizationRules.Quota(100, 0, 100),
            "the 100 percent law is exact");
        Equal(8, SyntheticMobilizationRules.Batch(23,
                SyntheticMobilizationRules.SpawnBatchLimit),
            "initial spawning is bounded");
        Equal(12, SyntheticMobilizationRules.ReplacementDemand(
                target: 40, living: 28, remainingReserve: 15),
            "casualties consume numeric reserve");
        Equal(15, SyntheticMobilizationRules.ReplacementDemand(
                target: 40, living: 10, remainingReserve: 15),
            "replacement demand cannot exceed reserve");
        Equal(16, SyntheticMobilizationRules.DemobilizationBatch(41),
            "demobilization removes at most sixteen actors per work item");
        Equal(9, SyntheticMobilizationRules.DemobilizationBatch(9),
            "the final demobilization batch removes the remainder");
        Equal(true, SyntheticMobilizationRules.ShouldRestartCityCursor(
            expectedCityCount: 4, currentCityCount: 3),
            "city removal restarts a participant cursor");
        Equal(false, SyntheticMobilizationRules.ShouldRestartCityCursor(
            expectedCityCount: 4, currentCityCount: 4),
            "a stable city generation preserves cursor progress");
        Equal(7, SyntheticMobilizationRules.ExpandLifecycleEnd(
            currentEndExclusive: 5, eventRecordCount: 7),
            "a repeated event extends its existing record boundary");
        Equal(7, SyntheticMobilizationRules.ExpandLifecycleEnd(
            currentEndExclusive: 7, eventRecordCount: 5),
            "a later smaller view cannot retract pending lifecycle work");
        Equal(true, SyntheticMobilizationRules.ShouldDeferOrphanScan(
                loadReconciliationPending: true),
            "orphan synthetic cleanup waits for the bounded load scan");
        Equal(false, SyntheticMobilizationRules.ShouldDeferOrphanScan(
                 loadReconciliationPending: false),
            "orphan synthetic cleanup runs after the load scan");
        Equal(true, SyntheticMobilizationRules.ShouldDeferDemobilization(
                syntheticActor: true, returnArrivalConfirmed: false,
                armyReturnActive: false, wartimeOwned: false,
                insideFriendlySafeCity: true),
            "synthetic army members survive cancel, discard, and active return until their own arrival is confirmed");
        Equal(false, SyntheticMobilizationRules.ShouldDeferDemobilization(
                syntheticActor: true, returnArrivalConfirmed: true,
                armyReturnActive: false, wartimeOwned: false,
                insideFriendlySafeCity: true),
            "synthetic army members demobilize only after confirmed arrival");
        Equal(true, SyntheticMobilizationRules.ShouldDeferDemobilization(
                syntheticActor: true, returnArrivalConfirmed: true,
                armyReturnActive: true, wartimeOwned: false,
                insideFriendlySafeCity: true),
            "a stale prior arrival fact cannot remove a soldier during a new return");
        Equal(true, SyntheticMobilizationRules.ShouldDeferDemobilization(
                syntheticActor: true, returnArrivalConfirmed: true,
                armyReturnActive: false, wartimeOwned: true,
                insideFriendlySafeCity: true),
            "a stale prior arrival fact cannot remove a soldier from a replacement mission");
        Equal(true, SyntheticMobilizationRules.ShouldDeferDemobilization(
                syntheticActor: true, returnArrivalConfirmed: true,
                armyReturnActive: false, wartimeOwned: false,
                insideFriendlySafeCity: false),
            "a historical arrival fact cannot authorize removal while the actor is abroad");
        Equal(false, SyntheticMobilizationRules.ShouldDeferDemobilization(
                syntheticActor: false, returnArrivalConfirmed: false,
                armyReturnActive: true, wartimeOwned: true,
                insideFriendlySafeCity: false),
            "permanent soldiers are never routed through synthetic cleanup");
        int remaining = 41;
        int[] expectedBatches = { 16, 16, 9 };
        for (int i = 0; i < expectedBatches.Length; i++)
        {
            int batch = SyntheticMobilizationRules.DemobilizationBatch(
                remaining);
            Equal(expectedBatches[i], batch,
                "demobilization sequence batch " + i);
            remaining -= batch;
        }
        Equal(0, remaining,
            "bounded demobilization eventually drains every actor");

        string source = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "SyntheticMobilizationLedgerService.cs"));
        Contains(source, "OnWarStarted(War pWar)",
            "the ledger observes war creation");
        Contains(source,
            "OnKingdomJoinedWar(War pWar, Kingdom pKingdom)",
            "the ledger observes later participants");
        Contains(source, "OnWarEnded(War pWar)",
            "the ledger observes war termination");
        Contains(source, "ProcessAuthorityCycle()",
            "ledger work is authority-cycle bounded");
        Contains(source, "ProcessWarEnrollmentScan();",
            "war enrollment continues while actor reconciliation is pending");
        DoesNotContain(source,
            "if (_loadActorReconciliationPending) return;",
            "actor reconciliation must not block war and replenishment work");
        Contains(source, "city.getPopulationPeople()",
            "quota snapshots use the native aggregate population API");
        DoesNotContain(source, "city.units[",
            "quota snapshots must not enumerate resident actors");
        Contains(source, "SortedSet<long> ActorIds",
            "each city-war record owns stable synthetic actor ids");
        Contains(source,
            "SyntheticMobilizationRules.DemobilizationBatch(",
            "post-war removal is bounded");
        Contains(source,
            "SyntheticLevyService.RemoveWithoutPersonalHistory(",
            "demobilization destroys generated actors without biographies");
        Contains(source,
            "SyntheticMobilizationRules.ShouldDeferDemobilization(",
            "demobilization yields while a synthetic actor's army is returning home");
        Contains(source,
            "HasReturnArrivalConfirmed(",
            "the synthetic cleanup boundary reads the per-actor arrival fact");
        string levySource = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "SyntheticLevyService.cs"));
        Contains(levySource, "ConfirmReturnArrival(Actor actor)",
            "successful army return can persist each synthetic actor's arrival");
        Contains(levySource, "ConfirmReturnArrivalIfSafe(Actor actor)",
            "discard only releases synthetic actors already inside a safe city");
        Contains(levySource, "ResetReturnArrival(Actor actor)",
            "a new bounded return claim clears a stale prior arrival fact");
        Contains(source, "WarArmyReturnService.IsActive(actor.army)",
            "active return ownership blocks demobilization despite stale arrival state");
        Contains(source,
            "HasWartimeMilitaryLock(actor)",
            "replacement wartime ownership blocks stale-arrival demobilization");
        Contains(source,
            "IsInsideFriendlySafeCity(actor)",
            "bounded ledger work observes each actor's current safe-city arrival");
        Contains(source,
            "SyntheticLevyService.ConfirmReturnArrivalIfSafe(actor)",
            "discarded actors become eligible when they later reach safety");
        Contains(source, "OnSyntheticMaterialized(Actor pActor)",
            "materialization registers actor identity atomically");
    }

    private static void SyntheticToggleControlsGenerationAndVanillaRecruitment()
    {
        Equal(false, SyntheticMobilizationRules.ShouldUseSyntheticMobilization(
                enabled: false),
            "disabled synthetic mobilization must not generate synthetic units");
        Equal(true, SyntheticMobilizationRules.ShouldUseSyntheticMobilization(
                enabled: true),
            "enabled synthetic mobilization may generate synthetic units");
        Equal(true, SyntheticMobilizationRules.
            ShouldBypassAw3RecruitmentRestrictions(syntheticEnabled: false),
            "vanilla recruitment must bypass AW3 restrictions when synthetic units are off");
        Equal(false, SyntheticMobilizationRules.
            ShouldBypassAw3RecruitmentRestrictions(syntheticEnabled: true),
            "synthetic mode keeps the AW3 recruitment boundary");
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(message + ": expected " +
                                                expected + ", got " + actual);
    }

    private static void Contains(string source, string value, string message)
    {
        if (!source.Contains(value, StringComparison.Ordinal))
            throw new InvalidOperationException(message + ": missing " +
                                                value);
    }

    private static void DoesNotContain(string source, string value,
        string message)
    {
        if (source.Contains(value, StringComparison.Ordinal))
            throw new InvalidOperationException(message + ": found " +
                                                value);
    }
}
