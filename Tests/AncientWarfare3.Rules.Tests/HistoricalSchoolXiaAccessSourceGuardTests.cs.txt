    private static string Read(string pRelative)
    {
        return File.ReadAllText(Path.Combine(Directory.GetCurrentDirectory(),
                pRelative.Replace('/', Path.DirectorySeparatorChar)))
            .Replace("\r\n", "\n");
    }

    /// <summary>
    /// 在 pScope 起始的那段源码里断言 pFirst 出现在 pSecond 之前。必须限定
    /// 作用域:HistoricalSchoolAcademyConstructionService 在 LoadState /
    /// ClearRuntime 里也出现,整文件比较会撞上那些更早的引用。
    /// </summary>
    private static void OrderedWithin(string pText, string pScope,
        string pFirst, string pSecond, string pMessage)
    {
        int scope = pText.IndexOf(pScope, StringComparison.Ordinal);
        if (scope < 0) throw new InvalidOperationException(pMessage);
        int first = pText.IndexOf(pFirst, scope, StringComparison.Ordinal);
        if (first < 0) throw new InvalidOperationException(pMessage);
        int second = pText.IndexOf(pSecond, first, StringComparison.Ordinal);
        if (second < 0) throw new InvalidOperationException(pMessage);
    }

    private static void Contains(string pText, string pNeedle,
        string pMessage)
    {
        if (!pText.Contains(pNeedle, StringComparison.Ordinal))
            throw new InvalidOperationException(pMessage);
    }
}
