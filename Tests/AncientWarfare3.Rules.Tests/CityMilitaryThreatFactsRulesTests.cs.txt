using AncientWarfare3.core.lineage;

internal static class CityMilitaryThreatFactsRulesTests
{
    public static void Run()
    {
        CacheEntryPointIsInstalledWithoutTileClosures();
        PresenceSnapshotIsSharedAcrossObservingKingdoms();
        LiveInvalidationUsesTheRevisionRule();
        CacheRotationDoesNotInvalidateAsyncFacts();

        var key = new CityMilitaryThreatKey(17L, 9L, 3L);
        Assert(CityMilitaryThreatFactsRules.CanCache(17L, 9L, 3L),
            "a valid city threat query is cacheable");
        Assert(CityMilitaryThreatFactsRules.KeyMatches(key,
                new CityMilitaryThreatKey(17L, 9L, 3L)),
            "identical war/city/kingdom facts share one entry");
        Assert(!CityMilitaryThreatFactsRules.KeyMatches(key,
                new CityMilitaryThreatKey(18L, 9L, 3L)),
            "a changed war never reuses a city fact");
        Assert(CityMilitaryThreatFactsRules.ShouldInvalidate(9L, 9L),
            "city-control changes invalidate that city");
        Assert(!CityMilitaryThreatFactsRules.ShouldInvalidate(9L, 10L),
            "an unrelated city remains cacheable in the current cycle");
        Assert(CityMilitaryThreatFactsRules.ShouldAdvanceRevisionForInvalidation(
                9L),
            "a live city or war state change invalidates async snapshots even without a cache hit");
        Assert(!CityMilitaryThreatFactsRules.ShouldAdvanceRevisionForInvalidation(
                -1L),
            "an invalid state-change identifier cannot advance a fact revision");
        Assert(CityMilitaryThreatFactsRules.ShouldReuse(4L, 4L,
                10.20d, 10.00d),
            "a recent fact from the same revision may be reused across callers");
        Assert(!CityMilitaryThreatFactsRules.ShouldReuse(4L, 4L,
                10.36d, 10.00d),
            "an expired fact must trigger a fresh physical scan");
        Assert(!CityMilitaryThreatFactsRules.ShouldReuse(4L, 5L,
                10.20d, 10.00d),
            "a changed revision must never reuse an old physical scan");
        Assert(CityMilitaryThreatFactsRules.ShouldReuse(4L, 4L,
                10.20d, 10.00d),
            "facts remain reusable when the query comes from the actor RTS path");
        Assert(CityMilitaryThreatFactsRules.ShouldReuseAuthorityCycle(
                true, 12L, 12L),
            "an unchanged fact is reusable for the remainder of its authority cycle");
        Assert(!CityMilitaryThreatFactsRules.ShouldReuseAuthorityCycle(
                true, 12L, 13L),
            "a fact from an earlier authority cycle is never reused");
        Assert(!CityMilitaryThreatFactsRules.ShouldReuseAuthorityCycle(
                false, 12L, 12L),
            "authority-cycle reuse is disabled outside the active cycle");
    }

    private static void CacheEntryPointIsInstalledWithoutTileClosures()
    {
        string source = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "CityAttackZoneService.cs"));
        Assert(source.Contains("CityMilitaryThreatFacts.TryGet("),
            "city hostile-military queries enter the authority-cycle cache");
        Assert(!source.Contains("tile.doUnits(actor =>"),
            "a physical city scan does not allocate one captured callback per tile");
        Assert(source.Contains("pCity.isGettingCaptured()"),
            "city threat queries read the vanilla active-occupation signal");
        Assert(source.Contains("pCity.being_captured_by"),
            "city threat queries use the vanilla occupation controller");
        Assert(!source.Contains("getUnits(kingdomId)"),
            "city threat queries do not enumerate chunk units");
    }

    private static void PresenceSnapshotIsSharedAcrossObservingKingdoms()
    {
        Assert(CityMilitaryThreatFactsRules.CanCachePresence(17L, 9L),
            "a valid war/city presence query is cacheable");
        Assert(CityMilitaryThreatFactsRules.ShouldReusePresence(4L, 4L,
                10.20d, 10.00d),
            "one physical presence scan can serve multiple observers");
        Assert(!CityMilitaryThreatFactsRules.ShouldReusePresence(5L, 4L,
                10.20d, 10.00d),
            "a changed presence revision requires a fresh physical scan");

        string source = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "CityMilitaryThreatFacts.cs"));
        Assert(source.Contains("TryGetPresence("),
            "threat facts expose a shared war/city presence snapshot");
        Assert(source.Contains("StorePresence("),
            "threat facts persist the shared physical presence snapshot");
        Assert(source.Contains("_cacheEpoch"),
            "cache freshness is independent from async planner revision");
        Assert(source.Contains("CanReuse(entry.CacheEpoch, entry.CachedAt)"),
            "city and war invalidation do not expire unrelated city facts");
        Assert(source.Contains("ShouldReuseAuthorityCycle"),
            "city threat facts use the authority-cycle cache when active");
    }

    private static void LiveInvalidationUsesTheRevisionRule()
    {
        string source = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "CityMilitaryThreatFacts.cs"));
        Assert(source.Contains(
                "ShouldAdvanceRevisionForInvalidation(cityId)"),
            "a valid city-control change advances the async fact revision");
        Assert(source.Contains(
                "ShouldAdvanceRevisionForInvalidation(warId)"),
            "a valid war-state change advances the async fact revision");
    }

    private static void CacheRotationDoesNotInvalidateAsyncFacts()
    {
        string source = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "CityMilitaryThreatFacts.cs"));
        int begin = source.IndexOf("internal static void BeginAuthorityCycle()",
            StringComparison.Ordinal);
        int end = source.IndexOf("internal static void EndAuthorityCycle()",
            begin, StringComparison.Ordinal);
        Assert(begin >= 0 && end > begin && !source.Substring(begin,
                end - begin).Contains("AdvanceRevision()"),
            "clearing a per-cycle cache does not invalidate an unchanged async front snapshot");
    }

    private static void Assert(bool pCondition, string pMessage)
    {
        if (!pCondition) throw new InvalidOperationException(pMessage);
    }
}
