using System.Collections.Generic;
using AncientWarfare3.core.policy;

internal static class KingdomPolicyInheritanceDiagnosticRulesTests
{
    public static void Run()
    {
        string key = KingdomPolicyInheritanceDiagnosticRules.BuildKey(
            pWorldGeneration: 4, pChildId: 11, pSourceId: 7,
            pState: "profile-a|profile-b");
        Equal("4:11:7:profile-a|profile-b", key,
            "inheritance diagnostic keys are stable");
        NotEqual(key, KingdomPolicyInheritanceDiagnosticRules.BuildKey(
                pWorldGeneration: 5, pChildId: 11, pSourceId: 7,
                pState: "profile-a|profile-b"),
            "new world generations create new diagnostic keys");
        NotEqual(key, KingdomPolicyInheritanceDiagnosticRules.BuildKey(
                pWorldGeneration: 4, pChildId: 11, pSourceId: 7,
                pState: "profile-c|profile-b"),
            "state changes create new diagnostic keys");

        var emitted = new HashSet<string>();
        True(KingdomPolicyInheritanceDiagnosticRules.ShouldLog(emitted, key),
            "first inheritance state is logged");
        False(KingdomPolicyInheritanceDiagnosticRules.ShouldLog(emitted, key),
            "duplicate inheritance state is suppressed");

        string source = System.IO.File.ReadAllText(System.IO.Path.Combine(
            System.IO.Directory.GetCurrentDirectory(), "Code", "core",
            "policy", "KingdomPolicyInheritanceService.cs"));
        Contains(source, "InheritanceDiagnosticKeys.Clear();",
            "runtime reset clears inheritance diagnostic state");
        Contains(source,
            "KingdomPolicyInheritanceDiagnosticRules.ShouldLog(",
            "policy inheritance logging uses the de-duplication rule");
        int applyIndex = source.IndexOf(
            "KingdomPolicyService.ApplySnapshot",
            System.StringComparison.Ordinal);
        int logIndex = source.IndexOf(
            "ModClass.LogInfo(\"[policy inheritance]",
            System.StringComparison.Ordinal);
        True(applyIndex >= 0 && logIndex > applyIndex,
            "policy inheritance is logged only after applying the snapshot");
    }

    private static void True(bool pValue, string pMessage)
    {
        if (!pValue) throw new InvalidOperationException(
            pMessage + ": expected true");
    }

    private static void False(bool pValue, string pMessage)
    {
        if (pValue) throw new InvalidOperationException(
            pMessage + ": expected false");
    }

    private static void Equal(string pExpected, string pActual,
        string pMessage)
    {
        if (!string.Equals(pExpected, pActual,
                System.StringComparison.Ordinal))
            throw new InvalidOperationException(pMessage + ": expected=" +
                pExpected + ", actual=" + pActual);
    }

    private static void NotEqual(string pExpected, string pActual,
        string pMessage)
    {
        if (string.Equals(pExpected, pActual,
                System.StringComparison.Ordinal))
            throw new InvalidOperationException(pMessage + ": values matched");
    }

    private static void Contains(string pSource, string pExpected,
        string pMessage)
    {
        if (!pSource.Contains(pExpected, System.StringComparison.Ordinal))
            throw new InvalidOperationException(pMessage + ": missing=" +
                pExpected);
    }
}
