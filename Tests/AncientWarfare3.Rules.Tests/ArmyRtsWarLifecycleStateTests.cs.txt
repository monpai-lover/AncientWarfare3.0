using System;
using AncientWarfare3.core.lineage;

internal static class ArmyRtsWarLifecycleStateTests
{
    public static void Run()
    {
        var store = new ArmyRtsWarLifecycleStateStore();
        ArmyRtsWarLifecycleRecord first = store.Ensure(11, 101, 75,
            ArmyRtsWarPhase.StrategicMovement);
        Assert(first != null && first.BaselineStrength == 75,
            "war start captures the existing living strength");

        ArmyRtsWarLifecycleRecord afterCasualties = store.Ensure(11, 101,
            19, ArmyRtsWarPhase.Withdrawal);
        Assert(ReferenceEquals(first, afterCasualties) &&
               afterCasualties.BaselineStrength == 75,
            "later casualties and captain replacement preserve the fixed baseline");

        ArmyRtsWarLifecycleRecord lateArmy = store.Ensure(11, 102, 48,
            ArmyRtsWarPhase.StrategicMovement);
        Assert(lateArmy != null && lateArmy.BaselineStrength == 48,
            "an army created during war captures baseline on first assignment");

        var unordered = new ArmyRtsWarLifecycleStateStore();
        unordered.Ensure(12, 9, 20, ArmyRtsWarPhase.StrategicMovement);
        unordered.Ensure(11, 8, 20, ArmyRtsWarPhase.StrategicMovement);
        unordered.Ensure(11, 7, 20, ArmyRtsWarPhase.StrategicMovement);
        var snapshot = unordered.Snapshot();
        Assert(snapshot.Count == 3 && snapshot[0].WarId == 11 &&
               snapshot[0].ArmyId == 7 && snapshot[1].WarId == 11 &&
               snapshot[1].ArmyId == 8 && snapshot[2].WarId == 12,
            "reconciliation receives a stable war and army ordered snapshot");
        Assert(ReferenceEquals(snapshot, unordered.Snapshot()),
            "unchanged lifecycle state reuses the ordered reconciliation snapshot");
        unordered.RemoveArmy(7);
        Assert(snapshot.Count == 3,
            "a reconciliation snapshot is isolated from later store changes");
        var refreshedSnapshot = unordered.Snapshot();
        Assert(!ReferenceEquals(snapshot, refreshedSnapshot) &&
               refreshedSnapshot.Count == 2,
            "structural lifecycle changes invalidate the cached snapshot");

        Assert(store.Ensure(11, 103, 0,
                   ArmyRtsWarPhase.StrategicMovement) == null,
            "an empty army does not create an unusable baseline");

        Assert(store.TrySetPhase(11, 101, ArmyRtsWarPhase.VanillaCombat) &&
               store.TryGet(11, 101, out ArmyRtsWarLifecycleRecord combat) &&
               combat.Phase == ArmyRtsWarPhase.VanillaCombat,
            "the lifecycle phase changes without replacing baseline state");

        Assert(store.RemoveArmy(101) == 1 &&
               !store.TryGet(11, 101, out _),
            "destroying an army removes all of its lifecycle state");

        store.Ensure(12, 102, 48, ArmyRtsWarPhase.StrategicMovement);
        Assert(store.ClearWar(11) == 1 && store.Count == 1 &&
               store.TryGet(12, 102, out _),
            "war end clears only records belonging to the ended war");

        store.Clear();
        Assert(store.Count == 0, "runtime reset clears lifecycle state");
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
}
