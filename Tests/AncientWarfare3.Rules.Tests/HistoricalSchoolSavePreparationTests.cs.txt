using System;
using System.Collections.Generic;
using AncientWarfare3.core.schools;

internal static class HistoricalSchoolSavePreparationTests
{
    public static void Run()
    {
        RuntimeStateCommitsAfterEveryDatabaseWriter();
        FailedWriterPreventsRuntimeCommit();
        StableTeachingReplayIgnoresVolatilePresentationFields();
        SaveDrainReachesQuiescenceAfterChainedWrites();
    }

    private static void StableTeachingReplayIgnoresVolatilePresentationFields()
    {
        var persisted = TeachingRow(worldTime: 10d, payload: "Old Name");
        var retried = TeachingRow(worldTime: 12d, payload: "New Name");

        True(persisted.MatchesStableReplay(retried,
                pRequireEventId: false),
            "a stable teaching operation replays despite changed time/name");

        retried.TargetActorId = 99L;
        False(persisted.MatchesStableReplay(retried,
                pRequireEventId: false),
            "a structurally different teaching event is not a replay");
    }

    private static void SaveDrainReachesQuiescenceAfterChainedWrites()
    {
        int pending = 3;
        int passes = 0;
        bool resolved = HistoricalSchoolSavePreparation.DrainUntilQuiescent(
            () =>
            {
                passes++;
                pending--;
                return pending == 0;
            },
            () => pending,
            pMaxPasses: 8);

        True(resolved,
            "save drain continues through callback-created chained writes");
        Equal(3, passes,
            "save drain stops immediately after the persistence graph is quiet");
    }

    private static HistoricalSchoolTeachingEventRow TeachingRow(
        double worldTime, string payload)
    {
        return new HistoricalSchoolTeachingEventRow
        {
            EventId = 7L,
            OperationKey = "school-teaching:v1|actor=42|year=100|school=ru|city=3|event=lecture",
            EventType = "lecture",
            ActorId = 42L,
            TargetActorId = -1L,
            SchoolId = "ru",
            CityId = 3L,
            KingdomId = 5L,
            EventYear = 100,
            Payload = payload,
            Importance = 1,
            WorldTime = worldTime
        };
    }

    private static void RuntimeStateCommitsAfterEveryDatabaseWriter()
    {
        var order = new List<string>();
        HistoricalSchoolSavePreparationResult result =
            HistoricalSchoolSavePreparation.Run(
                Step("descents", order),
                Step("deaths", order),
                Step("noble_deaths", order),
                Step("prior_writes", order),
                Step("activities", order),
                Step("writes", order),
                Step("death_archives", order),
                Step("async_writes", order),
                Step("runtime", order));

        True(result.AllResolved, "all successful save stages resolve");
        Equal("runtime", order[order.Count - 1],
            "runtime state is the final database write");
        True(order.IndexOf("async_writes") < order.IndexOf("runtime"),
            "background writes drain before runtime state commits");
    }

    private static void FailedWriterPreventsRuntimeCommit()
    {
        bool runtimeCalled = false;
        HistoricalSchoolSavePreparationResult result =
            HistoricalSchoolSavePreparation.Run(
                () => true, () => true, () => true, () => true,
                () => true, () => true, () => true,
                () => false,
                () =>
                {
                    runtimeCalled = true;
                    return true;
                });

        False(result.AsyncWritesResolved,
            "the failed background writer is reported");
        False(result.RuntimeStateAttempted,
            "runtime state is not written while another writer is unresolved");
        False(runtimeCalled,
            "the runtime persistence delegate is not invoked after failure");
        False(result.AllResolved,
            "an unresolved writer blocks the save boundary");
    }

    private static Func<bool> Step(string name, List<string> order)
    {
        return () =>
        {
            order.Add(name);
            return true;
        };
    }

    private static void True(bool value, string message)
    {
        if (!value) throw new InvalidOperationException(message);
    }

    private static void False(bool value, string message)
    {
        if (value) throw new InvalidOperationException(message);
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(message + ": expected=" +
                expected + " actual=" + actual);
    }
}
