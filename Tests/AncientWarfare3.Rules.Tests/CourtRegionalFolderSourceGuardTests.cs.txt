using System;
using System.IO;

internal static class CourtRegionalFolderSourceGuardTests
{
    internal static void Run()
    {
        string folder = Read("Code/ui/components/CourtRegionalGovernmentFolder.cs");
        Contains(folder, "ToggleExpanded", "regional court folder must toggle expanded state");
        Contains(folder, "CourtActorNodeView.Create", "expanded folder must render miniature official cards");
        Contains(folder, "SetExpanded(false)", "regional court folder must default to collapsed");
        Contains(folder, "aw_court_regional_folder_count", "regional folder count must be localized");
        Contains(folder, "aw_court_regional_folder_desc_collapsed", "regional folder toggle must explain expansion");

        string window = Read("Code/ui/windows/CourtWindow.cs");
        Contains(window, "CourtRegionalGovernmentFolder", "court window must own regional folder views");
        Contains(window, "_expandedRegionIds", "court window must preserve folder expansion by region");
        Contains(window, "RenderRegionalGovernmentFolder", "court window must render one folder per region");
        Contains(window, "folderRect", "regional link target must be the folder container");
        Contains(window, "aw_court_regional_folder", "court window must use the localized folder label");
        string readModel = Read("Code/core/court/CourtReadModelService.cs");
        string pyramidModel = Read("Code/core/court/CourtPyramidRules.cs");
        string actorNode = Read("Code/ui/items/CourtActorNodeView.cs");
        string historyWindow = Read("Code/ui/windows/CourtOfficeHistoryWindow.cs");
        string restore = Read("Code/core/multiplayer/AW3RuntimeRestorePipeline.cs");
        Contains(readModel, "City officials are rendered under their regional folder",
            "central court must not add city officials beside the regional folder");
        Contains(readModel, "string.Equals(officer.layer, CourtOfficeLayer.City",
            "city-layer officers must be filtered from the central node pool");
        Contains(pyramidModel, "public string HistoryOfficeLayer");
        Contains(pyramidModel, "public string HistoryOfficeId");
        Contains(pyramidModel, "public long HistoryCityId");
        Contains(actorNode, "pNode.HistoryOfficeId",
            "history action uses the card's exact appointment scope");
        False(actorNode.Contains("GetActiveOfficers(",
                StringComparison.Ordinal),
            "history action cannot choose another office by actor identity");
        False(historyWindow.Contains("EnsureLocalOfficerHistory(",
                StringComparison.Ordinal),
            "opening office history cannot write an appointment");
        False(readModel.Contains("\"regional-chief:\"",
                StringComparison.Ordinal),
            "new projections cannot recreate the legacy persisted identity");
        Contains(readModel, "\"regional-folder:\"",
            "regional nodes use a presentation-only namespace");
        Contains(restore, "\"legacy_projected_office_repair\"",
            "legacy rows are repaired before career projection rebuild");
    }

    private static string Read(string pRelative) => File.ReadAllText(
        Path.Combine(Directory.GetCurrentDirectory(),
            pRelative.Replace('/', Path.DirectorySeparatorChar)))
        .Replace("\r\n", "\n");

    private static void Contains(string pText, string pNeedle, string pMessage)
    {
        if (!pText.Contains(pNeedle, StringComparison.Ordinal))
            throw new InvalidOperationException(pMessage);
    }

    private static void Contains(string pText, string pNeedle)
    {
        Contains(pText, pNeedle, "source guard failed");
    }

    private static void False(bool pValue, string pMessage)
    {
        if (pValue) throw new InvalidOperationException(pMessage);
    }
}
