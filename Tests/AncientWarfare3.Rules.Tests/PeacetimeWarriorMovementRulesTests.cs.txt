using System;
using System.IO;
using AncientWarfare3.core.lineage;

internal static class PeacetimeWarriorMovementRulesTests
{
    internal static void Run()
    {
        Assert(!ArmyRtsRules.ShouldOwnMilitaryActor(
                authoritative: true,
                actorValid: true,
                currentProfessionIsWarrior: true,
                hasArmyIndex: true,
                armyMissionActive: false,
                isCivilAuthority: false,
                isCurrentCaptain: false),
            "a peaceful warrior without an active RTS mission must not be RTS-owned");
        Assert(ArmyRtsRuntimeModeRules.ShouldAllowVanillaDecisionEvaluation(
                ArmyRtsMode.On, rtsOwnsActor: false),
            "a peaceful warrior outside RTS ownership must retain vanilla decisions");

        string root = Directory.GetCurrentDirectory();
        string enlist = Read(root, "Code", "patch", "AW_EnlistPatch.cs");
        string safety = Read(root, "Code", "patch", "AW_ArmySafetyPatch.cs");
        string peacetime = Read(root, "Code", "core", "lineage",
            "StandingArmyPeacetimeService.cs");
        string diagnostic = Read(root, "Code", "core", "lineage",
            "ArmyRtsMovementDiagnostic.cs");

        Assert(enlist.Contains("StandingArmyPeacetimeService.GetJob(__instance)",
                StringComparison.Ordinal),
            "peaceful standing soldiers are evaluated for their civilian job");
        Assert(safety.Contains("ShouldAllowVanillaDecisionEvaluation(",
                StringComparison.Ordinal) &&
               safety.Contains("ArmyRtsRuntimeMode.Current, rtsOwnsActor,",
                StringComparison.Ordinal),
            "vanilla decisions use live RTS ownership instead of warrior profession alone");
        Assert(peacetime.Contains("ArmyRtsControllerService.HasActiveMission(army.id)",
                StringComparison.Ordinal) &&
               peacetime.Contains("WarArmyReturnRules.ShouldAllowPeacetimeJob(",
                StringComparison.Ordinal),
            "peacetime job selection yields to active missions and army returns");
        Assert(peacetime.Contains("ArmyRtsMovementDiagnostic.Log(\"peacetime\",",
                StringComparison.Ordinal) &&
               diagnostic.Contains("if (!AWPerformanceSettings.ArmyRtsDiagnosticsEnabled) return;",
                StringComparison.Ordinal),
            "peacetime diagnostics use the shared opt-in movement diagnostic");
        Assert(!diagnostic.Contains("cancelAllBeh(", StringComparison.Ordinal) &&
               !diagnostic.Contains("stopMovement(", StringComparison.Ordinal) &&
               !diagnostic.Contains(".setTask(", StringComparison.Ordinal) &&
               !diagnostic.Contains(".goTo(", StringComparison.Ordinal),
            "the shared movement diagnostic must be read-only");
    }

    private static string Read(string root, params string[] parts)
    {
        string path = root;
        for (int i = 0; i < parts.Length; i++) path = Path.Combine(path, parts[i]);
        return File.ReadAllText(path);
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
}
