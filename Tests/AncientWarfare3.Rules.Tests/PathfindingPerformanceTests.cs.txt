    private static void SchedulerStageDiagnosticsAccumulateStableSnapshots()
    {
        AWSchedulerStageDiagnostics.BeginFrame(pSampling: true);
        long frameStarted = AWSchedulerStageDiagnostics.BeginSchedulerFrame();
        long actorStarted = AWSchedulerStageDiagnostics.Begin(
            AWSchedulerStageBucket.Actors);
        Thread.SpinWait(128);
        AWSchedulerStageDiagnostics.End(
            AWSchedulerStageBucket.Actors, actorStarted);
        AWSchedulerStageDiagnostics.EndSchedulerFrame(frameStarted);

        AWSchedulerStageDiagnosticSnapshot snapshot =
            AWSchedulerStageDiagnostics.TakeSnapshot().WithFrameTicks(
                long.MaxValue);
        Assert(snapshot.Calls[(int)AWSchedulerStageBucket.Actors] == 1,
            "actor stage diagnostics record one call");
        Assert(snapshot.TotalTicks > 0L && snapshot.SchedulerTicks > 0L,
            "scheduler diagnostics retain stage and frame totals");
        Assert(snapshot.HostUnaccountedTicks >= 0L &&
               snapshot.UnaccountedTicks >= 0L,
            "scheduler diagnostics clamp unaccounted time");
        Assert(snapshot.FormatMilliseconds().Contains("actors:",
                StringComparison.Ordinal),
            "actor stage diagnostics expose a stable label");

        AWSchedulerStageDiagnosticSnapshot consumed =
            AWSchedulerStageDiagnostics.TakeSnapshot();
        Assert(consumed.Calls[(int)AWSchedulerStageBucket.Actors] == 0,
            "taking a scheduler snapshot consumes the current frame buckets");
    }

    private static void RequestKeysMatchWithoutAllocating()
    {
        var key = new AWPathRequestKey(91, true, false, true, -4);
        Assert(key.RegionLimit == 0, "request key clamps a negative region limit");
        Assert(key.Matches(91, true, false, true, 0),
            "request key matches the same normalized inputs");

        for (int i = 0; i < 128; i++)
            _ = key.Matches(91, true, false, true, 0);

            "10,000 warmed request-key matches allocate zero bytes");
    }

    private static void LongRangeContinuationConsumesCachedWaypointRoute()
    {
        const int width = 140;
        const int height = 3;
        var tiles = new AWTileTraversalSnapshot[width * height];
        for (int y = 0; y < height; y++)
        for (int x = 0; x < width; x++)
        {
            int id = y * width + x;
            var neighbors = new List<int>(4);
            if (x > 0) neighbors.Add(id - 1);
            if (x + 1 < width) neighbors.Add(id + 1);
            if (y > 0) neighbors.Add(id - width);
            if (y + 1 < height) neighbors.Add(id + width);
            bool blocked = y == 1 && x == 10;
            tiles[id] = new AWTileTraversalSnapshot(id, x, y,
                ground: !blocked, block: blocked, regionId: x / 20,
                islandId: 0, pNeighbors: neighbors.ToArray());
        }

        using var generation = AWTraversalGeneration.FromTiles(
            907, width, height, tiles);
        using var request = new AWPathRequest(907, width, width * 2 - 1,
            AWPathRequestOptions.Default,
            AWActorTraversalProfile.CreateWalker(100f, 100f, 5f),
            generation, 0d);
        var config = new AWPathfindingConfig
        {
            ShortRangeTiles = 24,
            LongRangeTiles = 96,
            RegionCorridorLookaheadTiles = 64,
            SegmentTargetSteps = 10,
            MaxNodesLong = 12000,
            MaxNodesLongFallback = 60000
        };
        var generator = new AWStreamingPathGenerator(config);
        AWPathGenerationResult first = generator.GenerateSegment(request,
            CancellationToken.None, 10);
        Assert(first.Succeeded && !first.ReachedTarget && first.Steps.Count == 10,
            "long-range generation returns a bounded intermediate segment");
        int firstEnd = first.EndTileId;
        request.AdvanceStartTile(firstEnd);
        AWPathGenerationResult second = generator.GenerateSegment(request,
            CancellationToken.None, 10);
        Assert(second.Succeeded && second.Steps.Count > 0 &&
               second.EndTileId != firstEnd,
            "the next segment consumes the cached route instead of rerunning A*");
        request.AdvanceStartTile(second.EndTileId);
        AWPathGenerationResult third = generator.GenerateSegment(request,
            CancellationToken.None, 10);
        Assert(third.Succeeded && third.Steps.Count > 0 &&
               third.EndTileId != second.EndTileId,
            "a third segment continues the same cached route cursor");
    }

    private static void AsyncBackgroundCommitRunsOffCallerThread()
    {
        using var coordinator = new AWAsyncWorkCoordinator(true);
        long generation = coordinator.StartWorld();
        int callerThread = Thread.CurrentThread.ManagedThreadId;
        int commitThread = 0;
        using var committed = new ManualResetEventSlim(false);
        var request = new AWAsyncWorkRequest("runtime-background-commit",
            AWAsyncLane.Ai,
            new AWAsyncStamp(generation, 1L, 1L),
            token =>
            {
                token.ThrowIfCancellationRequested();
                return new object();
            },
            result =>
            {
                commitThread = Thread.CurrentThread.ManagedThreadId;
                committed.Set();
            },
            pCommitMode: AWAsyncCommitMode.Background);

        Assert(coordinator.TrySchedule(request),
            "background runtime request is admitted");
        Assert(committed.Wait(TimeSpan.FromSeconds(2)),
            "background commit completes without a main-thread drain");
        Assert(commitThread != 0 && commitThread != callerThread,
            "background commit executes on the async worker thread");
        Assert(coordinator.PendingCompletionCount == 0,
            "background commit does not enqueue a main-thread completion");
    }

    private static void AsyncWorkerPoolRunsIndependentRequestsConcurrently()
    {
        using var coordinator = new AWAsyncWorkCoordinator(true, 2);
        long generation = coordinator.StartWorld();
        using var release = new ManualResetEventSlim(false);
        using var bothStarted = new ManualResetEventSlim(false);
        using var bothCommitted = new ManualResetEventSlim(false);
        int started = 0;
        int committed = 0;

        Func<object> execute = () =>
        {
            if (Interlocked.Increment(ref started) == 2)
                bothStarted.Set();
            Assert(release.Wait(TimeSpan.FromSeconds(2)),
                "independent async work is allowed to finish after both workers start");
            return new object();
        };
        Action<object> commit = _ =>
        {
            if (Interlocked.Increment(ref committed) == 2)
                bothCommitted.Set();
        };

        Assert(coordinator.TrySchedule(new AWAsyncWorkRequest(
                "runtime-concurrent-a", AWAsyncLane.Ai,
                new AWAsyncStamp(generation, 1L, 1L),
                _ => execute(), commit,
                pCommitMode: AWAsyncCommitMode.Background)),
            "first independent async request is admitted");
        Assert(coordinator.TrySchedule(new AWAsyncWorkRequest(
                "runtime-concurrent-b", AWAsyncLane.Ai,
                new AWAsyncStamp(generation, 1L, 2L),
                _ => execute(), commit,
                pCommitMode: AWAsyncCommitMode.Background)),
            "second independent async request is admitted");

        Assert(bothStarted.Wait(TimeSpan.FromSeconds(2)),
            "independent async requests do not serialize behind one worker");
        release.Set();
        Assert(bothCommitted.Wait(TimeSpan.FromSeconds(2)),
            "both background commits finish without a main-thread drain");
        Assert(coordinator.WorkerCount == 2 &&
               coordinator.ActiveWorkerCount == 0,
               "the coordinator reports a bounded fixed worker pool");
    }

    private static void BranchingSearchCannotFillTheSparseLabelTable()
    {
        var tiles = new AWTileTraversalSnapshot[70];
        for (int i = 0; i < tiles.Length; i++)
            tiles[i] = new AWTileTraversalSnapshot(i, i, 0, ground: true,
                pNeighbors: Array.Empty<int>());
        // Force this case through A* instead of the valid straight-line
        // shortcut; the assertion exercises sparse-label budget handling.
        tiles[65] = new AWTileTraversalSnapshot(65, 65, 0, ground: true,
            damageUnits: true, pNeighbors: Array.Empty<int>());
        tiles[69] = new AWTileTraversalSnapshot(69, 69, 0, ground: true,
            pNeighbors: Enumerable.Range(61, 8).ToArray());
        tiles[68] = new AWTileTraversalSnapshot(68, 68, 0, ground: true,
            pNeighbors: new[] { 69, 54, 55, 56, 57, 58, 59, 60 });
        tiles[67] = new AWTileTraversalSnapshot(67, 67, 0, ground: true,
            pNeighbors: new[] { 69, 46, 47, 48, 49, 50, 51, 52 });
        tiles[66] = new AWTileTraversalSnapshot(66, 66, 0, ground: true,
            pNeighbors: new[] { 69, 38, 39, 40, 41, 42, 43, 44 });

        using AWTraversalGeneration generation =
            AWTraversalGeneration.FromTiles(400, 70, 1, tiles);
        var config = new AWPathfindingConfig
        {
            ShortRangeTiles = 1000,
            MaxNodesShort = 4,
            MaxLabelsPerTile = 4
        };
        using var request = new AWPathRequest(54, 69, 0,
            AWPathRequestOptions.Default,
            AWActorTraversalProfile.CreateWalker(100f, 100f, 5f),
            generation, 0d);

        Exception workerError = null;
        var worker = new Thread(() =>
        {
            try
            {
                new AWStreamingPathGenerator(config).Generate(request,
                    CancellationToken.None);
            }
            catch (Exception error)
            {
                workerError = error;
            }
        }) { IsBackground = true };

            "a highly branching search cannot deadlock after filling its initial sparse label table");
        if (workerError != null) throw workerError;

               request.Stream.FailureReason ==
               AWPathFailureReason.SearchLimitExceeded,
            "a highly branching search terminates at its node budget instead of filling the sparse label table");
    }

    private static AWTraversalGeneration LinearWaterGeneration(int waterTiles)
    {
        int length = waterTiles + 2;
        var tiles = new AWTileTraversalSnapshot[length];
        for (int i = 0; i < length; i++)
        {
            var neighbors = new List<int>(2);
            if (i > 0) neighbors.Add(i - 1);
            if (i + 1 < length) neighbors.Add(i + 1);
            bool water = i > 0 && i <= waterTiles;
            tiles[i] = new AWTileTraversalSnapshot(i, i, 0,
                ground: !water, liquid: water, ocean: water,
                pNeighbors: neighbors.ToArray());
        }
        return AWTraversalGeneration.FromTiles(90 + waterTiles,
            length, 1, tiles);
    }
