    private static void ActorPostPipelineKeepsEnemySearchOrdered()
    {
        string root = Directory.GetCurrentDirectory();
        string postRunnerPath = Path.Combine(root, "Code", "core",
            "performance", "AWCooperativeActorPostRunner.cs");
        Assert(File.Exists(postRunnerPath),
            "Actor scheduler has an ordered post-stage runner");

        string simulation = File.ReadAllText(Path.Combine(root, "Code",
            "core", "performance", "AWCooperativeSimulationRunner.cs"));
        Assert(simulation.Contains(

            "Actor scheduler injects the ordered post-stage runner");

        string postRunner = File.ReadAllText(postRunnerPath);
        string[] orderedStages = {
            "BeforeEnemySearch",
            "PrepareEnemySearch",
            "ScheduleEnemySearch",
            "AwaitEnemySearch",
            "CommitEnemySearch"
        };
        int previousStage = -1;
        foreach (string stage in orderedStages)
        {
            int stageIndex = postRunner.IndexOf(stage,
                StringComparison.Ordinal);
            Assert(stageIndex > previousStage,
                "Actor post runner preserves ordered enemy-search stages");
            previousStage = stageIndex;
        }

        int workerStart = postRunner.IndexOf("private void SearchWorkItemAt(",
            StringComparison.Ordinal);
        int commitStart = postRunner.IndexOf("private int CommitEnemySearchClassification(",
            StringComparison.Ordinal);
        int nextGroupStart = postRunner.IndexOf("private bool TryCommitNextGroup(",
            StringComparison.Ordinal);
        Assert(workerStart >= 0 && commitStart >= 0,
            "Actor enemy worker and commit phases are isolated");
        string worker = postRunner.Substring(workerStart,
            Math.Max(0, nextGroupStart - workerStart));
        Assert(!worker.Contains("startFightingWith") &&
               !worker.Contains("stopMovement") &&
               !worker.Contains("skipBehaviour"),
            "enemy workers never mutate live Actor combat state");
        Assert(!worker.Contains("EnemiesFinder.findEnemiesFrom"),
            "enemy workers never touch the vanilla finder cache");
        Assert(postRunner.Substring(commitStart).Contains(
                "actor.canAttackTarget("),
            "main-thread commit revalidates a worker-selected target");
    }

    private static void ActorLifecycleGateClosesIngressAndReclaimsRetiredActors()
    {
        var gates = new AWPathActorGateIndex();
        Assert(gates.TryAcquire(41, out AWPathActorGateLease lease),
            "an open lifecycle gate accepts Actor work");
        Assert(gates.Count == 1,
            "one Actor publishes one lifecycle gate");
        lease.Retire();
        Assert(gates.TryAcquire(41, out AWPathActorGateLease replacement),
            "a new task reuses the retiring Actor gate instead of being lost");
        Assert(gates.Count == 1,
            "retargeting cannot create a second gate for the same Actor");
        replacement.Retire();
        replacement.Dispose();
        lease.Dispose();
        Assert(gates.Count == 0,
            "the last retired Actor lease removes its lifecycle gate");

        Assert(gates.BeginClear(),
            "world clear closes lifecycle ingress once");
        Assert(!gates.TryAcquire(42, out _),
            "no Actor state can enter while world clear owns the barrier");
        gates.EndClear();
        Assert(gates.TryAcquire(42, out AWPathActorGateLease afterClear),
            "a new world can enter after the clear barrier reopens");
        afterClear.Retire();
        afterClear.Dispose();
    }

    private static void ConcurrentDisposeWaitsForTheActiveDrain()
    {
        var generator = new BlockingStopGenerator();
        var finder = new AWPathFinder(generator);
        finder.Start(1);
        AWPathRequest Request(long actorId)
        {
            AWTraversalGeneration generation = LinearLandGeneration(3);
            try
            {
                return new AWPathRequest(actorId, 0, 2,
                    AWPathRequestOptions.Default,
                    AWActorTraversalProfile.CreateWalker(100f, 100f, 5f),
                    generation, 0d);
            }
            finally
            {
                generation.Dispose();
            }
        }

        AWPathRequest request = Request(404);
        Assert(finder.Request(request, out bool _),
            "the shutdown test request is accepted");
        Assert(generator.Started.Wait(TimeSpan.FromSeconds(2)),
            "the shutdown test occupies the worker");

        System.Threading.Tasks.Task first =
            System.Threading.Tasks.Task.Run(() => finder.StopAndDrain());
        Assert(SpinUntil(() => !finder.Request(Request(405), out bool _)),
            "stopping closes request admission before draining workers");
        System.Threading.Tasks.Task second =
            System.Threading.Tasks.Task.Run(() => finder.Dispose());
        bool secondReturnedBeforeWorker = second.Wait(100);

        generator.Release.Set();
        Assert(first.Wait(TimeSpan.FromSeconds(3)),
            "the first drain completes after the worker returns");
        Assert(second.Wait(TimeSpan.FromSeconds(3)),
            "a concurrent disposer joins the completed drain");
        Assert(!secondReturnedBeforeWorker,
            "a second dispose cannot release worker resources before drain completes");
    }

            "a region route is reused for the same immutable generation");

        using var secondGeneration = LinearRegionLandGeneration(16);
        int[] changed = cache.GetOrBuild(secondGeneration, 0, 15, 0);
        Assert(changed != null && !ReferenceEquals(first, changed),
            "a new traversal generation cannot reuse an old region route");
    }

    private static void FastSwimmingProfileChangesSpeedAndStaminaCost()
    {
        var start = new AWTileTraversalSnapshot(0, 0, 0, ground: true,
            pNeighbors: new[] { 1 });
        var water = new AWTileTraversalSnapshot(1, 1, 0, liquid: true,
            ocean: true, pNeighbors: new[] { 0 });
        var options = new AWPathRequestOptions(true, false, false, 0);
        var normal = new AWActorTraversalProfile(false, false, false, false,
            false, false, true, false, false, false,
            100f, 100f, 100f, 100f, 5f, 2f, 0f);
        var fast = new AWActorTraversalProfile(false, false, false, false,
            false, false, true, false, false, false,
            100f, 100f, 100f, 100f, 5f, 2f, 0f,
            pHasFastSwimming: true);

        AWTraversalEstimate normalEstimate = AWTraversalRules.Estimate(
            start, water, normal, options);
        AWTraversalEstimate fastEstimate = AWTraversalRules.Estimate(
            start, water, fast, options);
        Assert(fastEstimate.TimeSeconds < normalEstimate.TimeSeconds &&
               fastEstimate.StaminaCost < normalEstimate.StaminaCost,
               "fast swimming improves speed and reduces water stamina cost");
    }

    private static void RetainAfterDisposeDoesNotResurrectSnapshot()
    {
        var generation = LinearLandGeneration(2);
        generation.Dispose();
        bool threw = false;
        try
        {
            generation.Retain();
        }
        catch (ObjectDisposedException)
        {
            threw = true;
        }

        Assert(threw && generation.ReferenceCount == 0,
            "retaining a released traversal snapshot cannot resurrect its reference count");
    }

    private static void RegionTopologyUsesDeclaredHeightForPartialChunkArrays()
    {
        const int width = 3;
        const int height = 3;
        const int chunkSize = 2;
        var chunks = new AWTileTraversalSnapshot[3][];
        for (int i = 0; i < chunks.Length; i++)
            chunks[i] = new AWTileTraversalSnapshot[chunkSize * chunkSize];

        chunks[2][0] = new AWTileTraversalSnapshot(6, 0, 2,
            ground: true, regionId: 42);
        AWRegionTopologySnapshot topology = AWRegionTopologySnapshot.Build(
            chunks, width, height, chunkSize);

        Assert(topology.TryGetRegion(42, out _),
            "region topology honors the declared map height when chunk storage is partial");
    }

    private static AWTraversalGeneration LinearWaterGeneration(int waterTiles)
    {
        int length = waterTiles + 2;
        var tiles = new AWTileTraversalSnapshot[length];
        for (int i = 0; i < length; i++)
        {
            var neighbors = new List<int>(2);
            if (i > 0) neighbors.Add(i - 1);
            if (i + 1 < length) neighbors.Add(i + 1);
            bool water = i > 0 && i <= waterTiles;
            tiles[i] = new AWTileTraversalSnapshot(i, i, 0,
                ground: !water, liquid: water, ocean: water,
                pNeighbors: neighbors.ToArray());
        }
        return AWTraversalGeneration.FromTiles(90 + waterTiles,
            length, 1, tiles);
    }

    private sealed class BlockingSecondSegmentGenerator :
        IAWPathSegmentGenerator
    {
        private int _calls;
        private int _generationDisposedWhileRunning;

        internal ManualResetEventSlim SecondSegmentStarted { get; } =
            new ManualResetEventSlim(false);
        internal ManualResetEventSlim AllowSecondSegment { get; } =
            new ManualResetEventSlim(false);
        internal ManualResetEventSlim SecondSegmentFinished { get; } =
            new ManualResetEventSlim(false);
        internal bool GenerationDisposedWhileRunning =>
            Volatile.Read(ref _generationDisposedWhileRunning) != 0;

        public void Generate(AWPathRequest pRequest,
            CancellationToken pCancellation)
        {
            throw new NotSupportedException();
        }

        public AWPathGenerationResult GenerateSegment(AWPathRequest pRequest,
            CancellationToken pCancellation, int pMaximumSteps)
        {
            if (Interlocked.Increment(ref _calls) == 1)
                return AWPathGenerationResult.Success(1,
                    pReachedTarget: false, Array.Empty<AWPathStep>());

            SecondSegmentStarted.Set();
            AllowSecondSegment.Wait(TimeSpan.FromSeconds(2));
            try
            {
                AWTraversalGeneration retained = pRequest.Generation.Retain();
                retained.Dispose();
            }
            catch (ObjectDisposedException)
            {
                Volatile.Write(ref _generationDisposedWhileRunning, 1);
            }
            finally
            {
                SecondSegmentFinished.Set();
            }
            return AWPathGenerationResult.Success(pRequest.TargetTileId,
                pReachedTarget: true, Array.Empty<AWPathStep>());
        }
    }

    private sealed class BlockingStopGenerator : IAWPathGenerator
    {
        internal ManualResetEventSlim Started { get; } = new(false);
        internal ManualResetEventSlim Release { get; } = new(false);

        public void Generate(AWPathRequest pRequest,
            CancellationToken pCancellation)
        {
            Started.Set();
            Release.Wait(TimeSpan.FromSeconds(3));
            pRequest.Stream.Complete();
        }
    }

    private static bool Near(float left, float right) => Math.Abs(left - right) < 0.0001f;

    private static bool SpinUntil(Func<bool> condition)
    {
        var timeout = System.Diagnostics.Stopwatch.StartNew();
        while (timeout.Elapsed < TimeSpan.FromSeconds(3))
        {
            if (condition()) return true;
            Thread.Sleep(1);
        }
        return false;
    }

    private static int CountOccurrences(string source, string value)
    {
        int count = 0;
        int offset = 0;
        while ((offset = source.IndexOf(value, offset,
                   StringComparison.Ordinal)) >= 0)
        {
            count++;
            offset += value.Length;
        }
        return count;
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
