using System.Reflection;
using AncientWarfare3.core.lineage;

internal static class RoyalGuardCaptainHandoffRulesTests
{
    public static void Run()
    {
        var failures = new List<string>();

        Check(failures,
            ArmyCaptainContinuityRules.ShouldRejectCaptainMutation(
                ArmyRtsMode.On,
                replicaApplying: false,
                armyLive: true,
                currentCaptainExists: true,
                currentCaptainAlive: true,
                currentCaptainIsMember: true,
                requestedSameCaptain: false,
                currentCaptainIsCivilAuthority: false),
            "RTS keeps rejecting a live ordinary army captain mutation");

        Check(failures,
            !ArmyCaptainContinuityRules.ShouldRejectCaptainMutation(
                ArmyRtsMode.On,
                replicaApplying: false,
                armyLive: true,
                currentCaptainExists: true,
                currentCaptainAlive: true,
                currentCaptainIsMember: true,
                requestedSameCaptain: false,
                currentCaptainIsCivilAuthority: false,
                royalGuardRoleOwnsCaptain: true),
            "RTS permits the intended live Royal Guard captain handoff");

        Check(failures, RoyalGuardMaintenanceRules.ShouldRebindGuardJob(
                pCitizenJobMatches: true,
                pRuntimeJobMatches: false,
                pCaptainStateChanged: false),
            "a correct citizen job with a stale actor AI job is runtime drift");
        Check(failures, RoyalGuardMaintenanceRules.ShouldRebindGuardJob(
                pCitizenJobMatches: true,
                pRuntimeJobMatches: true,
                pCaptainStateChanged: true),
            "captain promotion or demotion immediately rebinds the guard job");
        Check(failures, !RoyalGuardMaintenanceRules.ShouldRebindGuardJob(
                pCitizenJobMatches: true,
                pRuntimeJobMatches: true,
                pCaptainStateChanged: false),
            "a stable guard with both jobs bound is not force-reset");
        Check(failures, RoyalGuardMaintenanceRules.ShouldRepairGuardTask(
                pProtectTaskActive: false, pFollowTaskActive: false),
            "a social or idle task is Royal Guard task drift");
        Check(failures, !RoyalGuardMaintenanceRules.ShouldRepairGuardTask(
                pProtectTaskActive: true, pFollowTaskActive: false),
            "an active protect task remains stable");
        Check(failures, !RoyalGuardMaintenanceRules.ShouldRepairGuardTask(
                pProtectTaskActive: false, pFollowTaskActive: true),
            "an active follow task remains stable");

        Check(failures,
            RoyalGuardActionRules.FollowIdleWaitMax <= 0.1f &&
            RoyalGuardActionRules.NoThreatWaitMax <= 0.1f,
            "royal guards recheck their king immediately instead of idling for seconds");
        Check(failures,
            RoyalGuardActionRules.ResolveFollowOffsetIndex(17L, 8) ==
            RoyalGuardActionRules.ResolveFollowOffsetIndex(17L, 8),
            "a guard keeps one deterministic formation offset around the king");
        Check(failures,
            RoyalGuardActionRules.ShouldProbeProtectionInP0(
                isProtectTask: true, hasThreatTarget: false),
            "a protect task receives one prioritized native AI probe");
        Check(failures,
            !RoyalGuardActionRules.ShouldProbeProtectionInP0(
                isProtectTask: true, hasThreatTarget: true),
            "a guard with a threat leaves follow P0 for combat");
        Check(failures,
            RoyalGuardActionRules.ShouldUseFollowP0(
                isFollowTask: true, isProtectTask: false,
                hasThreatTarget: false),
            "only the follow task is owned by movement P0");
        Check(failures,
            !RoyalGuardActionRules.ShouldUseFollowP0(
                isFollowTask: false, isProtectTask: true,
                hasThreatTarget: false),
            "protect mode must not be flattened into follow movement");
        Check(failures,
            RoyalGuardActionRules.ShouldKeepMilitaryP0(
                canFollowKingOnLand: true, hasThreatTarget: false,
                immediateCombat: false),
            "a land guard stays in P0 even between follow task iterations");
        MethodInfo cooldownRule = typeof(RoyalGuardActionRules).GetMethod(
            "ShouldIssueFollowMoveAfterCooldown",
            new[] { typeof(bool), typeof(bool), typeof(bool) });
        Check(failures, cooldownRule != null &&
            !(bool)cooldownRule.Invoke(null,
                new object[] { true, true, true }),
            "a guard at its offset must pause instead of republishing movement during cooldown");
        Check(failures, cooldownRule != null &&
            (bool)cooldownRule.Invoke(null,
                new object[] { true, true, false }),
            "a guard may re-evaluate its offset after the pause expires");
        Check(failures,
            RoyalGuardActionRules.ShouldKeepMilitaryP0(
                canFollowKingOnLand: true, hasThreatTarget: true,
                immediateCombat: true),
            "a fighting guard keeps the military P0 combat lane");
        Check(failures,
            !RoyalGuardActionRules.ShouldKeepMilitaryP0(
                canFollowKingOnLand: false, hasThreatTarget: false,
                immediateCombat: false),
            "a guard without a valid land king target leaves movement P0");

        Check(failures,
            !AWArmyRoleRules.ShouldRtsOwnCaptain(
                AWArmyRole.RoyalGuard, pCaptainRoyalGuard: true),
            "RTS must never replace a Royal Guard captain's protect-king task");
        Check(failures,
            !AWArmyRoleRules.ShouldRtsOwnCaptain(
                AWArmyRole.RoyalGuard, pCaptainRoyalGuard: false),
            "a Royal Guard role army remains outside RTS captain ownership");
        Check(failures,
            AWArmyRoleRules.ShouldRtsOwnCaptain(
                "", pCaptainRoyalGuard: false),
            "ordinary army captains remain eligible for RTS ownership");

        Check(failures,
            RoyalGuardMaintenanceRules.ShouldRefreshGuardInMaintenancePass(
                pIsCaptain: false,
                pIsNewlyAppointed: false,
                pCaptainStateChanged: true,
                pActorIndex: 7,
                pCursor: 0,
                pBatchLimit: 1,
                pActiveCount: 10),
            "a former captain refreshes immediately outside the member cursor");
        Check(failures,
            !RoyalGuardMaintenanceRules.ShouldRefreshGuardInMaintenancePass(
                pIsCaptain: false,
                pIsNewlyAppointed: false,
                pCaptainStateChanged: false,
                pActorIndex: 7,
                pCursor: 0,
                pBatchLimit: 1,
                pActiveCount: 10),
            "a stable member outside the cursor remains bounded");

        string root = Directory.GetCurrentDirectory();
        string threatBehaviour = File.ReadAllText(Path.Combine(root, "Code",
            "ai", "behaviours", "actor", "BehFindRoyalGuardThreat.cs"));
        Check(failures,
            threatBehaviour.Contains("return BehResult.RepeatStep;") &&
            !threatBehaviour.Contains("WaitAfterGuardNoThreat(pActor);\n                return BehResult.Stop;"),
            "a guard without a threat keeps its follow task alive");
        string followTaskBehaviour = File.ReadAllText(Path.Combine(root, "Code",
            "ai", "behaviours", "actor", "BehRoyalGuardFollowKing.cs"));
        Check(failures,
            followTaskBehaviour.Contains("WaitAfterGuardFollowIdle(pActor);") &&
            followTaskBehaviour.Contains("return BehResult.RepeatStep;"),
            "a guard without a follow tile must keep the follow task alive");
        string armyService = File.ReadAllText(Path.Combine(root, "Code", "core",
            "lineage", "AWArmyService.cs"));
        string guardService = File.ReadAllText(Path.Combine(root, "Code", "core",
            "lineage", "RoyalGuardService.cs"));
        string rtsService = File.ReadAllText(Path.Combine(root, "Code", "core",
            "lineage", "ArmyRtsControllerService.cs"));
        string armySafetyPatch = File.ReadAllText(Path.Combine(root, "Code", "patch",
            "AW_ArmySafetyPatch.cs"));
        Check(failures,
            armyService.Contains("SetRoyalGuardCaptainIfChanged") &&
            armyService.Contains("IsRoleArmy(pArmy, AWArmyRole.RoyalGuard)"),
            "the captain bypass is restricted to an explicitly role-owned Royal Guard army");
        Check(failures,
            guardService.Contains("AWArmyService.SetRoyalGuardCaptainIfChanged(army, pCaptain)"),
            "logical Royal Guard selection uses the role-owned actual-captain handoff");
        int scopedPatchChecks = armySafetyPatch.Split(
            "RoyalGuardCaptainHandoffScope.IsActive(__instance)").Length - 1;
        Check(failures,
            armyService.Contains("RoyalGuardCaptainHandoffScope.Open(pArmy)") &&
            scopedPatchChecks >= 2 &&
            armySafetyPatch.Contains(
                "royalGuardRoleOwnsCaptain: royalGuardHandoff"),
            "the Royal Guard handoff scope crosses both captain Harmony guards");
        Check(failures,
            guardService.Contains("pActor.ai.job?.id") &&
            guardService.Contains("GuardContent.ACTOR_JOB_KING_GUARD") &&
            guardService.Contains("pActor.stopMovement()") &&
            guardService.Contains("pActor.cancelAllBeh()") &&
            guardService.Contains("pActor.setCitizenJob(GuardContent.KingGuardJob)") &&
            guardService.Contains("pActor.ai.setJob(GuardContent.ACTOR_JOB_KING_GUARD)"),
            "job drift repair clears stale movement/tasks and rebinds citizen and actor jobs");
        int ensureGuardTaskStart = guardService.IndexOf(
            "internal static void EnsureProtectKingTask(Actor pActor)",
            StringComparison.Ordinal);
        int ensureGuardTaskEnd = guardService.IndexOf(
            "private static void RepairProtectKingTaskIfNeeded",
            ensureGuardTaskStart, StringComparison.Ordinal);
        string ensureGuardTask = ensureGuardTaskStart >= 0 &&
            ensureGuardTaskEnd > ensureGuardTaskStart
            ? guardService.Substring(ensureGuardTaskStart,
                ensureGuardTaskEnd - ensureGuardTaskStart)
            : "";
        Check(failures,
            ensureGuardTask.Contains("TryRegisterMilitaryPriority(pActor)"),
            "guard task repair immediately registers follow or protection priority for the large scheduler");
        Check(failures,
            guardService.Contains("RepairProtectKingTaskIfNeeded(captain);") &&
            guardService.Contains("TryRegisterMilitaryPriority(captain);") &&
            guardService.Contains("RepairProtectKingTaskIfNeeded(guard);") &&
            guardService.Contains("TryRegisterMilitaryPriority(guard);"),
            "scheduled guard identity refresh must proactively seed military P0");
        string postRunner = File.ReadAllText(Path.Combine(root, "Code", "core",
            "performance", "AWCooperativeActorPostRunner.cs"));
        int militaryActorStart = postRunner.IndexOf(
            "private void ProcessMilitaryP0Actor(",
            StringComparison.Ordinal);
        int militaryActorEnd = postRunner.IndexOf(
            "private static bool HasLiveMilitaryP0Objective",
            militaryActorStart, StringComparison.Ordinal);
        string militaryActorBody = militaryActorStart >= 0 &&
            militaryActorEnd > militaryActorStart
            ? postRunner.Substring(militaryActorStart,
                militaryActorEnd - militaryActorStart)
            : string.Empty;
        Check(failures,
            militaryActorBody.Contains(
                "kind == ArmyMilitaryMovementPriorityKind.RoyalGuard") &&
            militaryActorBody.Contains("actor.b6_updateAI(cycleElapsed)") &&
            militaryActorBody.Contains(
                "kind == ArmyMilitaryMovementPriorityKind.RoyalGuard &&") &&
            militaryActorBody.Contains("actor.beh_actor_target != null") &&
            !militaryActorBody.Contains("ShouldIssueNativeFollowMove(") &&
            !militaryActorBody.Contains("actor.goTo("),
            "Royal Guard P0 uses native AI, yields a discovered threat immediately, and never moves directly");
        int followPriorityStart = guardService.IndexOf(
            "internal static bool HasLandFollowPriority(Actor pGuard)",
            StringComparison.Ordinal);
        int followPriorityEnd = guardService.IndexOf(
            "internal static bool HasLandProtectionProbePriority",
            followPriorityStart, StringComparison.Ordinal);
        string followPriorityBody = followPriorityStart >= 0 &&
            followPriorityEnd > followPriorityStart
            ? guardService.Substring(followPriorityStart,
                followPriorityEnd - followPriorityStart)
            : string.Empty;
        Check(failures,
            followPriorityBody.Contains(
                "RoyalGuardActionRules.ShouldKeepMilitaryP0(") &&
            followPriorityBody.Contains("CanUseLandKingPriority(pGuard)"),
            "Royal Guard P0 survives task completion while a land king target remains valid");
        int releasePathStart = guardService.IndexOf(
            "internal static void ReleaseFollowPathForCombat(",
            StringComparison.Ordinal);
        int releasePathEnd = guardService.IndexOf(
            "private static WorldTile ResolveGuardOffsetTile(",
            releasePathStart, StringComparison.Ordinal);
        string releasePathBody = releasePathStart >= 0 &&
            releasePathEnd > releasePathStart
            ? guardService.Substring(releasePathStart,
                releasePathEnd - releasePathStart)
            : string.Empty;
        Check(failures,
            releasePathBody.Contains(
                "ArmyMilitaryMovementPriorityIndex.Register(") &&
            !releasePathBody.Contains(
                "ArmyMilitaryMovementPriorityIndex.Unregister("),
            "entering guard combat releases the follow path without dropping P0 priority");
        int rtsGuardOwnershipChecks = rtsService.Split(
            "AWArmyRoleRules.ShouldRtsOwnCaptain(").Length - 1;
        Check(failures,
            rtsGuardOwnershipChecks >= 2 &&
            rtsService.Contains("RoyalGuardService.EnsureProtectKingTask(pActor)") &&
            guardService.Contains("internal static void EnsureProtectKingTask(Actor pActor)") &&
            guardService.Contains("pActor.ai.setTask(GuardContent.TASK_FOLLOW_KING)"),
            "every RTS task write and release path preserves the Royal Guard protect-king task");
        Check(failures,
            guardService.Contains(
                "RoyalGuardMaintenanceRules.ShouldRepairGuardTask(") &&
            guardService.Split("RepairProtectKingTaskIfNeeded(").Length - 1 >= 3,
            "bounded Royal Guard maintenance repairs task drift for the captain and member batch");
        string guardContent = File.ReadAllText(Path.Combine(root, "Code", "content",
            "GuardContent.cs"));
        string followBehaviour = File.ReadAllText(Path.Combine(root, "Code",
            "ai", "behaviours", "actor", "BehRoyalGuardFollowKing.cs"));
        Check(failures,
            !guardContent.Contains("new BehRandomWait(RoyalGuardActionRules.FollowIdleWaitMin"),
            "the guard follow task does not append a long random idle step");
        Check(failures,
            followBehaviour.Contains("TryPublishKingFollowTarget(pActor)") &&
            followBehaviour.Contains("return BehResult.Continue;") &&
            !followBehaviour.Contains("pActor.goTo("),
            "the guard behaviour publishes a formation target without issuing movement");
        Check(failures,
            guardContent.IndexOf("follow.addBeh(new BehRoyalGuardFollowKing())", StringComparison.Ordinal) >= 0 &&
            guardContent.IndexOf("new BehGoToTileTarget", StringComparison.Ordinal) >
                guardContent.IndexOf("follow.addBeh(new BehRoyalGuardFollowKing())", StringComparison.Ordinal) &&
            guardContent.Contains("limit_pathfinding_regions = 6"),
            "the follow task publishes its target before native tile movement runs");
        Check(failures,
            guardService.Contains("TryPublishKingFollowTarget") &&
            guardService.Contains("ResolveGuardOffsetTile") &&
            guardService.Contains("pGuard.beh_tile_target?.data?.tile_id !=") &&
            !guardService.Contains("current_path.Add") &&
            !guardService.Contains("updatePathMovement()") &&
            !guardService.Contains("TryInstallMirroredKingPath") &&
            !guardService.Contains("InstalledKingPathTargetByGuard"),
            "the guard service keeps the stable offset but never mirrors or drives the king path");
        Check(failures,
            guardService.Contains("pGuard.clearTileTarget()") &&
            guardService.Contains("pGuard.beh_tile_target = null"),
            "combat release removes the old follow target before combat takes movement ownership");

        if (failures.Count > 0)
            throw new InvalidOperationException(string.Join(Environment.NewLine, failures));
    }

    private static void Check(List<string> failures, bool condition, string name)
    {
        if (!condition) failures.Add(name);
    }
}
