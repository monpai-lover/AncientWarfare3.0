using System.Data.SQLite;
using AncientWarfare3.core.lineage;

internal static class XiaizedFamilyBranchTransitionPersistenceSqlTests
{
    public static void Run()
    {
        CreatesChildAndMovesOnlyTheNewFounder();
        ReplayReusesTheSameChild();
        FailureRollsBackEveryWrite();
    }

    private static void CreatesChildAndMovesOnlyTheNewFounder()
    {
        using SQLiteConnection db = OpenDatabase();
        SeedWesternBranch(db);
        XiaizedFamilyBranchTransitionResult result =
            XiaizedFamilyBranchTransitionPersistence.TryCommit(db, Request());

        Equal(true, result.Success, "transition commits");
        Equal(11L, result.NewShiId, "one child Shi is allocated");
        Equal("western|de Paris", ScalarText(db,
            "SELECT NAMING_PROFILE||'|'||DISPLAY_STEM FROM ShiBranch " +
            "WHERE SHI_ID=10"), "historical branch remains western");
        Equal(11L, ScalarLong(db,
            "SELECT SHI_ID FROM ActorArchive WHERE ID=101"),
            "new founder moves to Xia branch");
        Equal(10L, ScalarLong(db,
            "SELECT SHI_ID FROM ActorArchive WHERE ID=102"),
            "dead relative remains on historical branch");
        Equal(10L, ScalarLong(db,
            "SELECT SHI_ID FROM ActorArchive WHERE ID=103"),
            "older living relative remains on historical branch");
    }

    private static void ReplayReusesTheSameChild()
    {
        using SQLiteConnection db = OpenDatabase();
        SeedWesternBranch(db);
        XiaizedFamilyBranchTransitionPersistence.TryCommit(db, Request());
        XiaizedFamilyBranchTransitionResult replay =
            XiaizedFamilyBranchTransitionPersistence.TryCommit(db, Request());

        Equal(true, replay.Success, "replay succeeds");
        Equal(11L, replay.NewShiId, "replay reuses child branch");
        Equal(2L, ScalarLong(db, "SELECT COUNT(*) FROM ShiBranch"),
            "replay creates no duplicate branch");
    }

    private static void FailureRollsBackEveryWrite()
    {
        using SQLiteConnection db = OpenDatabase();
        SeedWesternBranch(db);
        XiaizedFamilyBranchTransitionResult failed =
            XiaizedFamilyBranchTransitionPersistence.TryCommit(db, Request(),
                stage => stage ==
                    XiaizedFamilyBranchTransitionStage.AfterBranchWrite);

        Equal(false, failed.Success, "injected failure is reported");
        Equal(1L, ScalarLong(db, "SELECT COUNT(*) FROM ShiBranch"),
            "failed transition leaves no child branch");
        Equal(10L, ScalarLong(db,
            "SELECT SHI_ID FROM ActorArchive WHERE ID=101"),
            "failed transition leaves founder on old branch");
    }

    private static XiaizedFamilyBranchTransitionRequest Request()
    {
        return new XiaizedFamilyBranchTransitionRequest
        {
            FounderActorId = 101L,
            LineageId = 5L,
            OldShiId = 10L,
            OldNamingProfile = "western",
            FamilyName = "Zhou",
            ClanName = "Lu",
            DisplayStem = "de Paris",
            OriginKingdomId = 7L,
            OriginCityId = 8L,
            OriginCityChineseName = "Luoyang",
            CreatedTime = 100d
        };
    }

    private static SQLiteConnection OpenDatabase()
    {
        var db = new SQLiteConnection("Data Source=:memory:");
        db.Open();
        Execute(db, "CREATE TABLE ShiBranch (SHI_ID INTEGER PRIMARY KEY," +
            "LINEAGE_ID INTEGER,CLAN_NAME TEXT,PARENT_SHI_ID INTEGER," +
            "NAMING_PROFILE TEXT,WESTERN_NAMING_TRADITION TEXT," +
            "ORIGIN_CITY_CHINESE_NAME TEXT,DISPLAY_STEM TEXT," +
            "FOUNDER_ACTOR_ID INTEGER,SOURCE_TYPE TEXT," +
            "ORIGIN_KINGDOM_ID INTEGER,ORIGIN_CITY_ID INTEGER," +
            "ORIGIN_ORIGINAL_CLAN_ID INTEGER,CREATED_TIME REAL," +
            "IS_EXTINCT INTEGER)");
        Execute(db, "CREATE TABLE ActorArchive (ID INTEGER PRIMARY KEY," +
            "SHI_ID INTEGER,LINEAGE_ID INTEGER,FAMILY_NAME TEXT," +
            "CLAN_NAME TEXT,GIVEN_NAME TEXT,DISPLAY_NAME TEXT," +
            "NAME_INTEGRATED INTEGER,IS_ALIVE INTEGER)");
        return db;
    }

    private static void SeedWesternBranch(SQLiteConnection db)
    {
        Execute(db, "INSERT INTO ShiBranch VALUES " +
            "(10,5,'Paris',-1,'western','de','Paris','de Paris'," +
            "101,'western_admission',7,8,-1,1,0)");
        Execute(db, "INSERT INTO ActorArchive VALUES " +
            "(101,10,5,'Paris','Paris','Louis','Louis de Paris',0,1)");
        Execute(db, "INSERT INTO ActorArchive VALUES " +
            "(102,10,5,'Paris','Paris','Charles','Charles de Paris',0,0)");
        Execute(db, "INSERT INTO ActorArchive VALUES " +
            "(103,10,5,'Paris','Paris','Alex','Alex de Paris',0,1)");
    }

    private static void Execute(SQLiteConnection db, string sql)
    {
        using var command = new SQLiteCommand(sql, db);
        command.ExecuteNonQuery();
    }

    private static long ScalarLong(SQLiteConnection db, string sql)
    {
        using var command = new SQLiteCommand(sql, db);
        return Convert.ToInt64(command.ExecuteScalar());
    }

    private static string ScalarText(SQLiteConnection db, string sql)
    {
        using var command = new SQLiteCommand(sql, db);
        return Convert.ToString(command.ExecuteScalar()) ?? string.Empty;
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(
                $"{message}: expected {expected}, got {actual}");
    }
}
