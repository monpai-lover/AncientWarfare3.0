    private static void CustomHierarchyUsesManagementDepthInsteadOfEditorCoordinates()
    {
        var offices = new[]
        {
            new AncientWarfare3.core.court.CustomCourtOffice
            {
                Id = "root", Layer = AncientWarfare3.core.court.CourtOfficeLayer.Central,
                Grade = 10,
                Layout = new AncientWarfare3.core.court.CustomCourtOfficeLayout
                    { X = 900f, Y = 900f, Lane = 9 }
            },
            new AncientWarfare3.core.court.CustomCourtOffice
            {
                Id = "child", Layer = AncientWarfare3.core.court.CourtOfficeLayer.Central,
                Grade = 10,
                Layout = new AncientWarfare3.core.court.CustomCourtOfficeLayout
                    { X = -900f, Y = -900f, Lane = -9 }
            },
            new AncientWarfare3.core.court.CustomCourtOffice
            {
                Id = "specialist", Layer = AncientWarfare3.core.court.CourtOfficeLayer.Central,
                Grade = 30
            }
        };
        var edges = new[]
        {
            new AncientWarfare3.core.court.CustomCourtEdge
            {
                FromOfficeId = "root", ToOfficeId = "child",
                Kind = AncientWarfare3.core.court.CustomCourtEdgeKind.Management
            }
        };
        var ranks = AncientWarfare3.core.court.CustomCourtHierarchyLayoutRules
            .BuildRanks(offices, edges);
        Equal(10, ranks["root"], "root office keeps its grade-based tier");
        Equal(11, ranks["child"],
            "managed office is automatically placed below its manager");
        Equal(30, ranks["specialist"],
            "unconnected office uses its grade instead of editor coordinates");
    }

    private static void Contains(string text, string expected)
    {
        if (text.IndexOf(expected, StringComparison.Ordinal) < 0)
            throw new InvalidOperationException("missing source contract: " + expected);
    }

    private static void DoesNotContain(string text, string unexpected)
    {
        if (text.IndexOf(unexpected, StringComparison.Ordinal) >= 0)
            throw new InvalidOperationException("forbidden source contract: " +
                unexpected);
    }

    private static void Equal(int expected, int actual, string message)
    {
        if (expected != actual)
            throw new InvalidOperationException(message + ": expected " +
                expected + ", got " + actual);
    }

    private static void HasLocalizationKey(string text, string key)
    {
        string prefix = key + ",";
        foreach (string line in text.Split(new[] { "\r\n", "\n" },
                     StringSplitOptions.None))
            if (line.StartsWith(prefix, StringComparison.Ordinal)) return;
        throw new InvalidOperationException("missing localization row: " + key);
    }

    private static void CsvRowsHaveFourColumns(string text, string fileName)
    {
        string[] rows = text.Split(new[] { "\r\n", "\n" },
            StringSplitOptions.None);
        for (int index = 1; index < rows.Length; index++)
        {
            if (string.IsNullOrEmpty(rows[index])) continue;
            if (rows[index].Split(',').Length != 4)
                throw new InvalidOperationException("invalid locale row in " +
                    fileName + ": " + (index + 1));
        }
    }

    private static void RequiresFile(string path)
    {
        if (!File.Exists(path))
            throw new InvalidOperationException("missing required asset: " + path);
    }
