using System;
using System.Collections.Generic;
using AncientWarfare3.core.lineage;

internal static class ArmyAbstractBattleRulesTests
{
    public static void Run()
    {
        ArmyAbstractBattleResult empty = ArmyAbstractBattleRules.Resolve(
            new ArmyAbstractBattleFacts
            {
                WarId = 1,
                TargetCityId = 2,
                ResolutionSequence = 3,
                Attackers = Array.Empty<ArmyAbstractBattleParticipant>(),
                Defenders = Array.Empty<ArmyAbstractBattleParticipant>()
            });
        Equal(ArmyAbstractBattleOutcome.NoBattle, empty.Outcome,
            "zero versus zero does not resolve a battle");

        Equal(ArmyAbstractBattleOutcome.AttackVictory,
            ArmyAbstractBattleRules.Resolve(OneSided(10, 0)).Outcome,
            "an attacker-only battle is an attack victory");
        Equal(ArmyAbstractBattleOutcome.DefenseVictory,
            ArmyAbstractBattleRules.Resolve(OneSided(0, 10)).Outcome,
            "a defender-only battle is a defense victory");

        Equal(ArmyAbstractBattleOutcome.AttackVictory,
            ArmyAbstractBattleRules.Resolve(OneSided(125, 100)).Outcome,
            "an exact 1.25 ratio resolves directly for the stronger side");
        Equal(ArmyAbstractBattleOutcome.DefenseVictory,
            ArmyAbstractBattleRules.Resolve(OneSided(100, 125)).Outcome,
            "an exact 1.25 ratio resolves directly for the stronger side");

        ArmyAbstractBattleFacts manyDefenders = new ArmyAbstractBattleFacts
        {
            WarId = 5,
            TargetCityId = 8,
            ResolutionSequence = 1,
            Attackers = new[]
            {
                Participant(101, 100, 0, true),
                Participant(102, 100, 0, true)
            },
            Defenders = new[]
            {
                Participant(201, 120, 0, false),
                Participant(202, 40, 0, false)
            }
        };
        ArmyAbstractBattleResult aggregate =
            ArmyAbstractBattleRules.Resolve(manyDefenders);
        Equal(200, aggregate.AttackValue,
            "one-to-many battle aggregates all attacker cards");
        Equal(160, aggregate.DefenseValue,
            "one-to-many battle aggregates all defender cards");

        ArmyAbstractBattleFacts duplicateFacts = new ArmyAbstractBattleFacts
        {
            WarId = 6,
            TargetCityId = 9,
            ResolutionSequence = 2,
            Attackers = new[]
            {
                Participant(301, 100, 0, true),
                Participant(301, 999, 0, true),
                Participant(-1, 75, 0, true, actorId: 401),
                Participant(-1, 50, 0, true, actorId: 401),
                Participant(302, 0, 0, true),
                null
            },
            Defenders = new[]
            {
                Participant(401, 60, 0, false),
                Participant(402, 40, 50, false),
                Participant(402, 999, 0, false),
                Participant(-1, 40, 0, false, actorId: 501),
                Participant(-1, 20, 0, false, actorId: 501),
                Participant(403, -1, 0, false)
            }
        };
        ArmyAbstractBattleResult duplicate =
            ArmyAbstractBattleRules.Resolve(duplicateFacts);
        Equal(175, duplicate.AttackValue,
            "duplicate army and fallback actor cards count once");
        Equal(160, duplicate.DefenseValue,
            "duplicate defender army and fallback actor cards count once");

        ArmyAbstractBattleResult namespaced = ArmyAbstractBattleRules.Resolve(
            new ArmyAbstractBattleFacts
            {
                Attackers = new[]
                {
                    Participant(5, 10, 0, true),
                    Participant(-1, 20, 0, true, actorId: 5),
                    Participant(0, 30, 0, true),
                    Participant(-1, 40, 0, true, actorId: 0)
                },
                Defenders = Array.Empty<ArmyAbstractBattleParticipant>()
            });
        Equal(100, namespaced.AttackValue,
            "army and actor card namespaces do not collide, including zero IDs");

        Equal(7L, ArmyAbstractBattleRules.SelectPrimaryAttacker(
            new[]
            {
                Participant(7, 100, 0, false),
                Participant(7, 90, 0, true)
            }).ArmyId,
            "primary attacker filters mixed-side records before deduplication");

        ArmyAbstractBattleFacts reorderedFacts = new ArmyAbstractBattleFacts
        {
            WarId = duplicateFacts.WarId,
            TargetCityId = duplicateFacts.TargetCityId,
            ResolutionSequence = duplicateFacts.ResolutionSequence,
            Attackers = new[]
            {
                duplicateFacts.Attackers[0],
                duplicateFacts.Attackers[2],
                duplicateFacts.Attackers[4],
                duplicateFacts.Attackers[1],
                duplicateFacts.Attackers[3],
                duplicateFacts.Attackers[5]
            },
            Defenders = new[]
            {
                duplicateFacts.Defenders[0],
                duplicateFacts.Defenders[1],
                duplicateFacts.Defenders[3],
                duplicateFacts.Defenders[2],
                duplicateFacts.Defenders[4],
                duplicateFacts.Defenders[5]
            }
        };
        ArmyAbstractBattleResult reordered =
            ArmyAbstractBattleRules.Resolve(reorderedFacts);
        Equal(duplicate.AttackValue, reordered.AttackValue,
            "participant aggregation is independent of source ordering");
        Equal(duplicate.DefenseValue, reordered.DefenseValue,
            "defender aggregation is independent of source ordering");
        Equal(ArmyAbstractBattleRules.ParticipantHash(duplicateFacts),
            ArmyAbstractBattleRules.ParticipantHash(reorderedFacts),
            "participant hash is stable after canonical sorting");

        ArmyAbstractBattleResult commanderBuff =
            ArmyAbstractBattleRules.Resolve(OneSided(100, 100,
                attackerCommander: 50, defenderCommander: 0));
        Equal(150, commanderBuff.AttackValue,
            "commander strength contributes a bounded percentage buff");
        Equal(ArmyAbstractBattleOutcome.AttackVictory,
            commanderBuff.Outcome,
            "commander buff can decide a close card duel");

        ArmyAbstractBattleFacts stableFacts = OneSided(101, 100);
        ArmyAbstractBattleResult first =
            ArmyAbstractBattleRules.Resolve(stableFacts);
        ArmyAbstractBattleResult replay =
            ArmyAbstractBattleRules.Resolve(stableFacts);
        Equal(first.Outcome, replay.Outcome,
            "the same operation replays the same outcome");
        Equal(first.Seed, replay.Seed,
            "the same operation reuses the same deterministic seed identity");

        IReadOnlyList<ArmyAbstractBattleParticipant> attackers =
            new[]
            {
                Participant(50, 100, 0, true),
                Participant(40, 100, 0, true)
            };
        Equal(40L, ArmyAbstractBattleRules.SelectPrimaryAttacker(
                attackers).ArmyId,
            "primary attacker tie breaks by the lowest army id");
        Equal(50L, ArmyAbstractBattleRules.SelectPrimaryAttacker(
                new[] { Participant(50, 110, 0, true),
                       Participant(40, 100, 0, true) }).ArmyId,
            "primary attacker chooses the largest adjusted card");
    }

    private static ArmyAbstractBattleFacts OneSided(int pAttack,
        int pDefense, int attackerCommander = 0, int defenderCommander = 0)
    {
        return new ArmyAbstractBattleFacts
        {
            WarId = 10,
            TargetCityId = 20,
            ResolutionSequence = 30,
            Attackers = pAttack > 0
                ? new[] { Participant(1, pAttack, attackerCommander, true) }
                : Array.Empty<ArmyAbstractBattleParticipant>(),
            Defenders = pDefense > 0
                ? new[] { Participant(2, pDefense, defenderCommander, false) }
                : Array.Empty<ArmyAbstractBattleParticipant>()
        };
    }

    private static ArmyAbstractBattleParticipant Participant(long pArmyId,
        int pCount, int pCommander, bool pAttacker, long actorId = -1L)
    {
        return new ArmyAbstractBattleParticipant
        {
            ArmyId = pArmyId,
            ActorId = actorId,
            UnitCount = pCount,
            CommanderStrength = pCommander,
            IsAttacker = pAttacker
        };
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(
                message + $" (expected {expected}, actual {actual})");
    }
}
