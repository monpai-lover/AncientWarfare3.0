    private static void NativeXiaIdentitySurvivesMissingRuntimeIds()
    {
        Equal(WarAiPeopleRelation.SameSpecies,
            WarAiGoalSelectionRules.ResolvePeopleRelation(
                "", "", -1L, -1L,
                attackerIsNativeXia: true,
                defenderIsNativeXia: true),
            "two native Xia realms remain the same people without runtime ids");
        Equal(WarAiPeopleRelation.Foreign,
            WarAiGoalSelectionRules.ResolvePeopleRelation(
                "human", "human", 11L, 12L,
                attackerIsNativeXia: true,
                defenderIsNativeXia: false),
            "native Xia and non-Xia realms do not collapse to one species");
    }

    private static void ShortUninvadedWarStronglyRejectsSurrender()
    {
        var facts = new WarSettlementAiFacts(
            warYears: 1, requesterToOpponentMilitaryRatio: .28f,
            requesterFieldLossRatio: .12f, opponentFieldLossRatio: .05f,
            requesterCitiesLostRatio: 0f, opponentCitiesLostRatio: 0f,
            requesterCapitalThreatened: false,
            opponentCapitalThreatened: false,
            requesterBorderThreatened: true,
            requesterWarFatigue: .08f, opponentWarFatigue: .04f,
            requesterFoodSecurity: .9f, requesterOrder: .85f,
            rulerWeak: false, rulerResolute: false,
            peaceCourtDominant: false, warCourtDominant: false,
            highLegitimacyWar: false);

        Equal(-100, DiplomacyProposalAiRules.BaseWillingness(
                WarSettlementAiDecision.Surrender),
            "surrender keeps the required fixed negative base");
        var surrenderCandidate = new DiplomacyProposalAiCandidate(
            DiplomacyProposalType.Surrender, allowed: true, opinion: 0,
            requesterPowerRatio: .28f, directRoyalMarriage: false,
            targetPowerRatio: 0f, targetHasMandate: false);

            "generic diplomacy selection cannot restore a positive surrender base");
        Equal(WarSettlementAiDecision.None,
            DiplomacyProposalAiRules.SelectWarSettlement(facts,
                earlySurrenderRoll: 0f),
            "a short war with no occupied city does not surrender on ordinary disparity");

    private static void SettlementAvailabilityFollowsWarPosition()
    {
        DiplomacyAvailabilityFacts Facts(WarSettlementPosition position) =>
            new DiplomacyAvailabilityFacts(
                atWar: true, allied: false, requesterIsSubject: false,
                responderIsSubject: false, directSubjectRelation: false,
                hasJoinableWar: false, requesterIsMandate: false,
                activeNonAggression: false, activeTruce: false,
                subjectFailureReason: "", allianceFailureReason: "",
                activeWarPreparation: false, peaceNegotiators: true,
                warPosition: position);

        Equal("", DiplomacyProposalRules.UnavailableReason(
                DiplomacyProposalType.Surrender,
                Facts(WarSettlementPosition.Losing)),
            "only the losing realm may offer surrender");
        Equal("not_winning_war", DiplomacyProposalRules.UnavailableReason(
                DiplomacyProposalType.EnforceDemands,
                Facts(WarSettlementPosition.Losing)),
            "a losing realm cannot enforce demands");
        Equal("", DiplomacyProposalRules.UnavailableReason(
                DiplomacyProposalType.EnforceDemands,
                Facts(WarSettlementPosition.Winning)),
            "only the winning realm may enforce demands");
        Equal("not_losing_war", DiplomacyProposalRules.UnavailableReason(
                DiplomacyProposalType.Surrender,
                Facts(WarSettlementPosition.Winning)),
            "a winning realm cannot offer surrender");
        Equal("not_losing_war", DiplomacyProposalRules.UnavailableReason(
                DiplomacyProposalType.Surrender,
                Facts(WarSettlementPosition.Contested)),
            "a contested realm cannot claim it has already lost");
        Equal("not_winning_war", DiplomacyProposalRules.UnavailableReason(
                DiplomacyProposalType.EnforceDemands,
                Facts(WarSettlementPosition.Contested)),
            "a contested realm cannot claim victory");
    }

    private static void SettlementAcceptanceUsesDetailedReadiness()
    {
        var exhaustedPeace = DiplomacyProposalRules.Assess(
            DiplomacyProposalType.Peace,
            new DiplomacyProposalScoreFacts(opinion: 0,
                requesterPower: 100, responderPower: 100,
                sharedEnemy: false, responderLosingWar: false,
                allied: false, requesterIsMandate: false,
                requesterDiplomacy: 10, responderDiplomacy: 10,
                responderReadyForPeace: true,
                responderReadyToConcede: false,
                hasWarSettlementReadiness: true));
        True(exhaustedPeace.ExpectedAccepted,
            "an exhausted responder accepts peace through the same settlement rule");
        Equal(28, exhaustedPeace.Parts.Single(part =>
                part.Key == "war_situation").Value,
            "the peace tooltip reports the detailed settlement readiness");

        var unyieldingEnforcement = DiplomacyProposalRules.Assess(
            DiplomacyProposalType.EnforceDemands,
            new DiplomacyProposalScoreFacts(opinion: 0,
                requesterPower: 300, responderPower: 100,
                sharedEnemy: false, responderLosingWar: true,
                allied: false, requesterIsMandate: false,
                requesterDiplomacy: 10, responderDiplomacy: 10,
                responderReadyForPeace: false,
                responderReadyToConcede: false,
                hasWarSettlementReadiness: true));
        False(unyieldingEnforcement.ExpectedAccepted,
            "winning alone does not bypass an unyielding responder");
        Equal(-30, unyieldingEnforcement.Parts.Single(part =>
                part.Key == "war_situation").Value,
            "the enforcement tooltip uses the same refusal state as AI execution");

        var concedingEnforcement = DiplomacyProposalRules.Assess(
            DiplomacyProposalType.EnforceDemands,
            new DiplomacyProposalScoreFacts(opinion: 0,
                requesterPower: 300, responderPower: 100,
                sharedEnemy: false, responderLosingWar: true,
                allied: false, requesterIsMandate: false,
                requesterDiplomacy: 10, responderDiplomacy: 10,
                responderReadyForPeace: true,
                responderReadyToConcede: true,
                hasWarSettlementReadiness: true));
        True(concedingEnforcement.ExpectedAccepted,
            "a genuine loser may concede to enforcement");
    }

    private static void SeparatePeacePrefersVulnerableOpposingAllies()

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(
                $"{name}: expected {expected}, got {actual}");
    }

    private static void True(bool actual, string name)
    {
        if (!actual) throw new InvalidOperationException(name);
    }

    private static void False(bool actual, string name)
    {
        if (actual) throw new InvalidOperationException(name);
    }
