    private static void CommittedBirthRevisionSurvivesNewerIdentityFailure()
    {
        var pending = new FamilyTreeProjectionPendingState();
        pending.Publish(100L, 70L,
            FamilyTreeProjectionChange.FamilyStructure);
        pending.Publish(100L, 71L,
            FamilyTreeProjectionChange.IdentityOrTitle);
        long revisionBefore = FamilyTreeProjectionRevision.Current;

        True(pending.TryComplete(100L, 70L,
                out FamilyTreeProjectionChange committedBirth),
            "a committed birth completes while a newer identity is pending");
        Equal(FamilyTreeProjectionChange.FamilyStructure, committedBirth,
            "the birth completion releases its own family structure change");
        FamilyTreeProjectionRevision.Advance(committedBirth);
        pending.Fail(100L, 71L);
        if (pending.TryComplete(100L, 70L,
                out FamilyTreeProjectionChange repeatedBirth))
            FamilyTreeProjectionRevision.Advance(repeatedBirth);
        if (pending.TryComplete(100L, 71L,
                out FamilyTreeProjectionChange failedIdentity))
            FamilyTreeProjectionRevision.Advance(failedIdentity);
        Equal(revisionBefore + 1L, FamilyTreeProjectionRevision.Current,
            "birth success and identity failure advance exactly once");
        True(!pending.TryComplete(100L, 70L, out _),
            "the committed birth change is released exactly once");
        True(!pending.TryComplete(100L, 71L, out _),
            "the failed identity leaves no pending projection owner");
    }

    private static void EarlierCommittedCompletionIsConsumedExactlyOnce()
    {
        var pending = new FamilyTreeProjectionPendingState();
        pending.Publish(101L, 80L,
            FamilyTreeProjectionChange.FamilyStructure);
        pending.Publish(101L, 81L,
            FamilyTreeProjectionChange.IdentityOrTitle);

        True(pending.TryComplete(101L, 80L,
                out FamilyTreeProjectionChange committedBirth),
            "a newer sequence cannot take ownership of an earlier commit");
        Equal(FamilyTreeProjectionChange.FamilyStructure, committedBirth,
            "the earlier committed source remains attributable to its sequence");
        True(!pending.TryComplete(101L, 80L, out _),
            "an earlier committed source cannot be consumed twice");
    }

    private static void FailedIdentityWriteClearsPendingProjection()
    {
        var pending = new FamilyTreeProjectionPendingState();
        pending.Publish(102L, 90L,
            FamilyTreeProjectionChange.IdentityOrTitle);

        pending.Fail(102L, 90L);

        True(!pending.TryComplete(102L, 90L, out _),
            "a failed identity write clears its pending projection");
        pending.Publish(102L, 91L, FamilyTreeProjectionChange.None);
        True(pending.TryComplete(102L, 91L,
                out FamilyTreeProjectionChange nextChange),
            "a later write owns a fresh projection boundary");
        Equal(FamilyTreeProjectionChange.None, nextChange,
            "failed identity metadata cannot leak into a later commit");
    }

    private static void LateBirthPublicationRetainsCommittedProvenance()
    {
        var pending = new FamilyTreeProjectionPendingState();
        pending.Publish(103L, 101L,
            FamilyTreeProjectionChange.IdentityOrTitle);
        pending.Publish(103L, 100L,
            FamilyTreeProjectionChange.FamilyStructure);

        True(pending.TryComplete(103L, 100L,
                out FamilyTreeProjectionChange committedBirth),
            "a late birth publication remains attributable to its sequence");
        Equal(FamilyTreeProjectionChange.FamilyStructure, committedBirth,
            "a newer pending identity cannot discard a late birth source");
        pending.Fail(103L, 101L);
        True(!pending.TryComplete(103L, 101L, out _),
            "the newer failed identity still clears its own source");
    }

    private static void ArchiveFailureCallbacksClearBothPendingStores()
    {
        string archiveWriter = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "LineageArchiveWriter.cs"));
        string birthWriter = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "LineageBirthArchiveService.cs"));

        True(Regex.IsMatch(archiveWriter,
                @"OnAsyncWriteFailed\s*\(\s*long\s+pActorId\s*,\s*" +
                @"long\s+pSequence\s*\).*?" +
                @"ActorArchivePendingStore\.Complete\s*\(\s*pActorId\s*,\s*" +
                @"pSequence\s*\)\s*;.*?" +
                @"FamilyTreeProjectionPendingStore\.Fail\s*\(\s*" +
                @"pActorId\s*,\s*pSequence\s*\)\s*;",
                RegexOptions.Singleline),
            "async archive failure clears actor and projection pending stores");
        True(Regex.Matches(archiveWriter,
                @"TryUpsertState\s*\(.*?OnAsyncWriteFailed\s*\(\s*id\s*,\s*" +
                @"sequence\s*\)", RegexOptions.Singleline).Count >= 2,
            "live and captured archive writes register failure cleanup");
        True(Regex.IsMatch(birthWriter,
                @"OnFailed\s*\(\s*long\s+pChildId.*?" +
                @"ActorArchivePendingStore\.Complete\s*\(\s*pChildId\s*,\s*" +
                @"pSequence\s*\)\s*;.*?" +
                @"FamilyTreeProjectionPendingStore\.Fail\s*\(\s*" +
                @"pChildId\s*,\s*pSequence\s*\)\s*;",
                RegexOptions.Singleline),
            "birth failure clears actor and projection pending stores");
    }

    private static void StructuredResultAcceptsOnlyQueuedOrCommitted()
    {
        var notEligible = new LineageBirthArchiveResult(
            LineageBirthArchiveStatus.NotEligible, 100L, 10L, 20L,
            "not eligible");
        var queued = new LineageBirthArchiveResult(
            LineageBirthArchiveStatus.Queued, 100L, 10L, 20L, "");
        var committed = new LineageBirthArchiveResult(
            LineageBirthArchiveStatus.Committed, 100L, 10L, 20L, "");
        var failed = new LineageBirthArchiveResult(
            LineageBirthArchiveStatus.Failed, 100L, 10L, 20L, "failed");

        True(!notEligible.Accepted,
            "not eligible births are not accepted persistence writes");
        True(queued.Accepted, "queued births are accepted");
        True(committed.Accepted, "synchronous commits are accepted");
        True(!failed.Accepted, "failed births are not accepted");
        Equal(100L, failed.ChildId, "result keeps the child id");
        Equal(10L, failed.Parent1Id, "result keeps parent slot 1");
        Equal(20L, failed.Parent2Id, "result keeps parent slot 2");
        Equal("failed", failed.Reason, "result keeps the failure reason");
    }

    private static void AsyncEnvelopeUsesPerChildStateKeyAndReturnsTypedOutcome()
    {
        using SQLiteConnection db = OpenDatabase();
        var envelope = new LineageBirthArchiveEnvelope(42L,
            new AWAsyncStamp(3L, 42L, 42L),
            Birth("Async snapshot", 10L, 20L, 100d));

        Equal("lineage-birth:v1:child:100", envelope.OperationKey,
            "birth envelope uses a versioned per-child state key");
        True(envelope.Kind == HistoricalWriteKind.State,
            "birth envelope is coalescible state");

        using SQLiteTransaction tx = db.BeginTransaction();
        object rawOutcome = ((IHistoricalCustomWriteEnvelope)envelope)
            .Execute(db, tx);
        True(rawOutcome is LineageBirthArchiveOutcome,
            "birth envelope custom execution returns a typed outcome");
        var outcome = (LineageBirthArchiveOutcome)rawOutcome;
        tx.Commit();

        Equal(100L, outcome.ChildId,
            "birth envelope outcome identifies the newborn");
        True(outcome.ArchiveWritten,
            "birth envelope executes the archive persistence");
        True(outcome.ParentSlot1Written && outcome.ParentSlot2Written,
            "birth envelope executes both parent edges");
    }

    private static void ExplicitParentSlotsNormalizeArchiveSnapshot()
    {
        using SQLiteConnection db = OpenDatabase();
        Commit(db, Birth("Explicit parents", 10L, 11L, 100d,
            999L, 998L));

        Equal(10L, Scalar(db,
                "SELECT PARENT_ID_1 FROM ActorArchive WHERE ID=100"),
            "explicit slot 1 normalizes the child archive");
        Equal(11L, Scalar(db,
                "SELECT PARENT_ID_2 FROM ActorArchive WHERE ID=100"),
            "explicit slot 2 normalizes the child archive");
        Equal(10L, Scalar(db,
                "SELECT PARENT_ID FROM FamilyEdge WHERE EDGE_ID=1001"),
            "explicit slot 1 remains authoritative for its edge");
        Equal(11L, Scalar(db,
                "SELECT PARENT_ID FROM FamilyEdge WHERE EDGE_ID=1002"),
            "explicit slot 2 remains authoritative for its edge");
    }

    private static void BirthWithoutKnownParentsWritesNoEdges()
    {
        using SQLiteConnection db = OpenDatabase();
        LineageBirthArchiveOutcome outcome = Commit(db,
            Birth("No known parents", -1L, -1L, 100d));

        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM ActorArchive WHERE ID=100"),
            "birth without parents still writes the child archive");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM FamilyEdge WHERE CHILD_ID=100"),
            "birth without parents writes no parent edges");
        True(!outcome.ParentSlot1Written,
            "outcome skips unknown parent slot 1");
        True(!outcome.ParentSlot2Written,
            "outcome skips unknown parent slot 2");
    }

    private static void BirthWithOnlyFirstParentWritesOneEdge()
    {
        using SQLiteConnection db = OpenDatabase();
        LineageBirthArchiveOutcome outcome = Commit(db,
            Birth("One known parent", 10L, -1L, 100d));

        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM FamilyEdge WHERE CHILD_ID=100"),
            "birth with one parent writes one edge");
        Equal(10L, Scalar(db,
                "SELECT PARENT_ID FROM FamilyEdge WHERE EDGE_ID=1001"),
            "birth with one parent writes slot 1");
        Equal(0L, Scalar(db,

            "birth with one parent writes no unknown-parent sentinel");
        True(outcome.ParentSlot1Written,
            "outcome reports known parent slot 1");

            "outcome skips unknown parent slot 2");
    }

    private static void ReplayWithoutSecondParentRemovesStaleEdge()
    {
        using SQLiteConnection db = OpenDatabase();
        Commit(db, Birth("Two known parents", 10L, 20L, 100d));
        LineageBirthArchiveOutcome outcome = Commit(db,
            Birth("Latest snapshot", 11L, -1L, 200d));

        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM FamilyEdge WHERE CHILD_ID=100"),
            "replay without slot 2 leaves one current edge");
        Equal(11L, Scalar(db,
                "SELECT PARENT_ID FROM FamilyEdge WHERE EDGE_ID=1001"),
            "replay refreshes parent slot 1");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM FamilyEdge WHERE EDGE_ID=1002"),
            "replay removes the stale parent slot 2 edge");
        True(outcome.ParentSlot1Written,
            "replay reports the refreshed parent slot 1");
        True(!outcome.ParentSlot2Written,
            "replay reports no parent slot 2 write");
    }

    private static void OrdinaryBirthCommitsArchiveAndTwoEdges()
    {
        using SQLiteConnection db = OpenDatabase();
        LineageBirthArchiveWrite write = Birth("First snapshot", 10L, 20L,
            100d);

        using SQLiteTransaction tx = db.BeginTransaction();
        LineageBirthArchiveOutcome outcome =
            LineageBirthArchivePersistence.Execute(db, tx, write);
        tx.Commit();

        Equal(100L, outcome.ChildId, "outcome identifies the newborn");
        True(outcome.ArchiveWritten, "outcome reports the archive write");
        True(outcome.ParentSlot1Written, "outcome reports parent slot 1");
        True(outcome.ParentSlot2Written, "outcome reports parent slot 2");
        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM ActorArchive WHERE ID=100"),
            "birth writes one child archive");
        Equal(2L, Scalar(db,
                "SELECT COUNT(*) FROM FamilyEdge WHERE CHILD_ID=100"),
            "birth writes both parent edges");
    }

    private static void SecondEdgeFailureRollsBackEverything()
    {
        using SQLiteConnection db = OpenDatabase();
        long revisionBefore = FamilyTreeProjectionRevision.Current;
        var pending = new FamilyTreeProjectionPendingState();
        pending.Publish(100L, 60L,
            FamilyTreeProjectionChange.FamilyStructure);
        Execute(db, "CREATE TRIGGER abort_second_parent_edge " +
            "BEFORE INSERT ON FamilyEdge WHEN NEW.PARENT_SLOT=2 " +
            "BEGIN SELECT RAISE(ABORT,'injected slot 2 failure'); END");

        using SQLiteTransaction tx = db.BeginTransaction();
        bool failed = false;
        try
        {
            LineageBirthArchivePersistence.Execute(db, tx,
                Birth("Rolled back", 10L, 20L, 100d));
            tx.Commit();
        }
        catch (SQLiteException exception)
        {
            failed = true;
            True(exception.Message.Contains("injected slot 2 failure",
                    StringComparison.Ordinal),
                "rollback failure comes from the slot 2 trigger");
            tx.Rollback();
            pending.Fail(100L, 60L);
        }

        True(failed, "slot 2 trigger abort reaches the transaction owner");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM ActorArchive WHERE ID=100"),
            "slot 2 failure rolls back the child archive");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM FamilyEdge WHERE CHILD_ID=100"),
            "slot 2 failure rolls back both parent edges");
        bool exposedFailedBirth = pending.TryComplete(100L, 60L,
            out FamilyTreeProjectionChange failedBirthChange);
        if (exposedFailedBirth)
            FamilyTreeProjectionRevision.Advance(failedBirthChange);
        True(!exposedFailedBirth,
            "a rolled-back birth leaves no releasable projection change");
        Equal(revisionBefore, FamilyTreeProjectionRevision.Current,
            "a rolled-back birth cannot advance the family-tree revision");
    }

    private static void RepeatedBirthIsIdempotent()
    {
        using SQLiteConnection db = OpenDatabase();
        Commit(db, Birth("First snapshot", 10L, 20L, 100d));
        Commit(db, Birth("Latest snapshot", 11L, 22L, 200d));

        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM ActorArchive WHERE ID=100"),
            "repeated birth keeps one archive row");
        Equal(2L, Scalar(db,
                "SELECT COUNT(*) FROM FamilyEdge WHERE CHILD_ID=100"),
            "repeated birth keeps two deterministic parent edges");
        Equal("Latest snapshot", TextScalar(db,
                "SELECT DISPLAY_NAME FROM ActorArchive WHERE ID=100"),
            "repeated birth keeps the latest child snapshot");
        Equal(11L, Scalar(db,
                "SELECT PARENT_ID FROM FamilyEdge WHERE EDGE_ID=1001"),
            "repeated birth refreshes parent slot 1");
        Equal(22L, Scalar(db,
                "SELECT PARENT_ID FROM FamilyEdge WHERE EDGE_ID=1002"),
            "repeated birth refreshes parent slot 2");
        Equal(200d, DoubleScalar(db,
                "SELECT CREATED_TIME FROM FamilyEdge WHERE EDGE_ID=1002"),
            "repeated birth keeps the latest edge snapshot");
    }

    private static LineageBirthArchiveOutcome Commit(SQLiteConnection pDb,
        LineageBirthArchiveWrite pWrite)
    {
        using SQLiteTransaction tx = pDb.BeginTransaction();
        LineageBirthArchiveOutcome outcome =
            LineageBirthArchivePersistence.Execute(pDb, tx, pWrite);
        tx.Commit();
        return outcome;
    }

    private static SQLiteConnection OpenDatabase()
    {
        var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        Execute(db, "CREATE TABLE ActorArchive (" +
            "ID INTEGER PRIMARY KEY,GIVEN_NAME TEXT,DISPLAY_NAME TEXT," +
            "FAMILY_NAME TEXT,CLAN_NAME TEXT,LINEAGE_ID INTEGER," +
            "SHI_ID INTEGER,ASSET_ID TEXT,SUBSPECIES_ID INTEGER," +
            "SUBSPECIES_NAME TEXT,SEX INTEGER,STATUS TEXT," +
            "KINGDOM_ID INTEGER,KINGDOM_NAME TEXT,KINGDOM_COLOR TEXT," +
            "CITY_ID INTEGER,CITY_NAME TEXT,SOCIAL_TITLE TEXT," +
            "SOCIAL_TITLE_COLOR TEXT,ORIGINAL_CLAN_ID INTEGER," +
            "CLAN_COLOR_TEXT TEXT,CLAN_COLOR_ID INTEGER," +
            "CLAN_BANNER_ICON_ID INTEGER," +
            "CLAN_BANNER_BACKGROUND_ID INTEGER,PARENT_ID_1 INTEGER," +
            "PARENT_ID_2 INTEGER,GENERATION INTEGER,NOBLE_DISTANCE INTEGER," +
            "EVER_NOBLE_BLOOD INTEGER,NOBLE_ORIGIN_ACTOR_ID INTEGER," +
            "NOBLE_ORIGIN_NAME TEXT,NOBLE_ORIGIN_DISTANCE INTEGER," +
            "BIRTH_TIME REAL,DEATH_TIME REAL,DEATH_CAUSE TEXT," +
            "IS_ALIVE INTEGER,NAME_INTEGRATED INTEGER,HEAD INTEGER," +
            "SKIN INTEGER,SKIN_SET INTEGER,AGE_OVERGROWTH INTEGER," +
            "PHENOTYPE_INDEX INTEGER,PHENOTYPE_SHADE INTEGER," +
            "FOUNDED_BRANCH_SHI_ID INTEGER)");
        Execute(db, "CREATE TABLE FamilyEdge (" +
            "EDGE_ID INTEGER PRIMARY KEY,CHILD_ID INTEGER," +
            "PARENT_ID INTEGER,PARENT_SLOT INTEGER," +
            "CHILD_LINEAGE_ID INTEGER,CREATED_TIME REAL)");
        return db;
    }

    private static void Execute(SQLiteConnection pDb, string pSql)
    {
        using var command = new SQLiteCommand(pSql, pDb);
        command.ExecuteNonQuery();
    }

    private static long Scalar(SQLiteConnection pDb, string pSql)
    {
        using var command = new SQLiteCommand(pSql, pDb);
        return Convert.ToInt64(command.ExecuteScalar());
    }

    private static double DoubleScalar(SQLiteConnection pDb, string pSql)
    {
        using var command = new SQLiteCommand(pSql, pDb);
        return Convert.ToDouble(command.ExecuteScalar());
    }

    private static string TextScalar(SQLiteConnection pDb, string pSql)
    {
        using var command = new SQLiteCommand(pSql, pDb);
        return Convert.ToString(command.ExecuteScalar()) ?? "";
    }

    private static void True(bool pValue, string pName)
    {
        if (!pValue)
            throw new InvalidOperationException(pName + ": expected true");
    }

    private static void Equal(long pExpected, long pActual, string pName)
    {
        if (pExpected != pActual)
            throw new InvalidOperationException(pName + ": expected " +
                                                pExpected + ", got " +
                                                pActual);
    }

    private static void Equal(double pExpected, double pActual,
        string pName)
    {
        if (!pExpected.Equals(pActual))
            throw new InvalidOperationException(pName + ": expected " +
                                                pExpected + ", got " +
                                                pActual);
    }

    private static void Equal(string pExpected, string pActual,
        string pName)
    {
        if (!string.Equals(pExpected, pActual, StringComparison.Ordinal))
            throw new InvalidOperationException(pName + ": expected '" +
                                                pExpected + "', got '" +
                                                pActual + "'");
    }

    private static void Equal(FamilyTreeProjectionChange pExpected,
        FamilyTreeProjectionChange pActual, string pName)
    {
        if (pExpected != pActual)
            throw new InvalidOperationException(pName + ": expected " +
                                                pExpected + ", got " +
                                                pActual);
    }
