using System;
using System.IO;

public static class WesternSurnameRepairRulesTests
{
    public static void Run()
    {
        string lineage = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "LineageService.cs"));
        string birth = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "patch",
            "AW_BirthPatch.cs"));
        int lightweightStart = lineage.IndexOf("if (!pUseFullPath)",
            StringComparison.Ordinal);
        int fullPathStart = lineage.IndexOf(
            "ArchiveTraceableActor(pParent1", lightweightStart,
            StringComparison.Ordinal);
        if (lightweightStart < 0 || fullPathStart <= lightweightStart)
            throw new InvalidOperationException(
                "lightweight birth boundary is missing");

        string lightweightBlock = lineage.Substring(lightweightStart,
            fullPathStart - lightweightStart);
        True(lightweightBlock.Contains("return;"),
            "non-full births stop before the full lineage path");
        Equal(false, lightweightBlock.Contains("InheritFromParents"),
            "lightweight births do not promote commoners into full lineage");
        True(birth.Contains("WesternLineageParentEdgeService.RecordBirth"),
            "lightweight births use the dedicated parent-edge service");
        True(birth.IndexOf("WesternLineageParentEdgeService.RecordBirth",
                StringComparison.Ordinal) <
            birth.IndexOf("LineageService.OnActorBornWithParents",
                StringComparison.Ordinal),
            "lightweight edge ownership is decided before the full path");
    }

    private static void True(bool pValue, string pMessage)
    {
        if (!pValue) throw new InvalidOperationException(pMessage);
    }

    private static void Equal<T>(T pExpected, T pActual, string pMessage)
    {
        if (!Equals(pExpected, pActual))
            throw new InvalidOperationException(
                $"{pMessage}: expected {pExpected}, got {pActual}");
    }
}
