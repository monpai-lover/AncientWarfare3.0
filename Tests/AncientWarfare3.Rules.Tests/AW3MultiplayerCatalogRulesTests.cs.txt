    private static void CatalogsAreCompleteUniqueAndImmutable()
    {
        IReadOnlyList<AW3WindowDescriptor> windows =
            AW3MultiplayerCatalog.Windows;
        IReadOnlyList<AW3CommandDescriptor> commands =
            AW3MultiplayerCatalog.Commands;

        Equal(29, windows.Count, "current AW3 window count");
        Equal(29, windows.Select(value => value.WindowId).Distinct(
                StringComparer.Ordinal).Count(),
            "window IDs are unique");
        Equal(Enum.GetValues<AW3WindowKind>().Length, windows.Count,
            "every window kind has one descriptor");
        Equal(Enum.GetValues<AW3CommandKind>().Length, commands.Count,
            "every command kind has one descriptor");
        Equal(windows.Count,
            windows.Select(value => value.Kind).Distinct().Count(),
            "window kinds are unique");
        Equal(commands.Count,
            commands.Select(value => value.Kind).Distinct().Count(),
            "command kinds are unique");
        True(windows.All(value => value.PayloadSchemaVersion ==
                                  AW3MultiplayerCatalog.PayloadSchemaVersion),
            "window descriptors use the catalog schema");
        True(commands.All(value => value.PayloadSchemaVersion ==
                                   AW3MultiplayerCatalog.PayloadSchemaVersion),
            "command descriptors use the catalog schema");

        Throws<NotSupportedException>(() =>
                ((IList<AW3WindowDescriptor>)windows).Add(windows[0]),
            "window catalog is immutable");
        Throws<NotSupportedException>(() =>
                ((IList<AW3CommandDescriptor>)commands).Add(commands[0]),
            "command catalog is immutable");
    }

    private static void WindowRequestsValidateDeclaredContext()
    {
        AW3WindowDescriptor policy = AW3MultiplayerCatalog.GetWindow(
            AW3WindowKind.PolicyTree);
        var valid = AW3WindowOpenRequest.ForCountry(
            AW3WindowKind.PolicyTree, 7);
        True(valid.IsValidFor(policy), "country context is valid");
        Equal(7L, valid.CountryId, "country identity retained");

        var missing = AW3WindowOpenRequest.Empty(
            AW3WindowKind.PolicyTree);
        Equal(false, missing.IsValidFor(policy),
            "missing country is rejected");

        var diplomacy = AW3WindowOpenRequest.ForCountries(
            AW3WindowKind.DiplomaticWarDeclaration, 7, 8);
        True(diplomacy.IsValidFor(AW3MultiplayerCatalog.GetWindow(
                AW3WindowKind.DiplomaticWarDeclaration)),
            "source and target country context is valid");

        var history = AW3WindowOpenRequest.ForActor(
            AW3WindowKind.History, 31, countryId: 7);
        True(history.IsValidFor(AW3MultiplayerCatalog.GetWindow(
                AW3WindowKind.History)),
            "actor history accepts actor and optional country context");

        Equal("aw_diplomatic_marriage",
            AW3MultiplayerCatalog.GetWindow(
                AW3WindowKind.DiplomaticMarriage).WindowId,
            "the current diplomatic marriage window is catalogued");
    }

    private static void ModelsAreImmutableAndTransportFree()
    {
        Type[] types = typeof(AW3MultiplayerCatalog).Assembly
            .GetExportedTypes()
            .Where(type => type.Namespace ==
                           "AncientWarfare3.api.multiplayer" &&
                           (type.Name.StartsWith("AW3Window",
                                StringComparison.Ordinal) ||
                            type.Name.StartsWith("AW3Command",
                                StringComparison.Ordinal) ||
                            type == typeof(AW3MultiplayerCatalog)))
            .ToArray();
        foreach (Type type in types)
        {
            foreach (PropertyInfo property in type.GetProperties())
                Equal(null, property.SetMethod,
                    type.Name + "." + property.Name + " is immutable");
            foreach (MethodInfo method in type.GetMethods())
            {
                AssertTransportFree(method.ReturnType,
                    type.Name + "." + method.Name + " return type");
                foreach (ParameterInfo parameter in method.GetParameters())
                    AssertTransportFree(parameter.ParameterType,
                        type.Name + "." + method.Name + " parameter");
            }
        }
    }

    private static void WindowSourceAndFacadeCoverEveryCurrentId()
    {
        string root = Directory.GetCurrentDirectory();
        string idsSource = File.ReadAllText(Path.Combine(root, "Code", "ui",

        string[] sourceIds = Regex.Matches(idsSource,
                "public\\s+const\\s+string\\s+[A-Z_]+\\s*=\\s*\"([^\"]+)\"")

            .Select(match => match.Groups[1].Value)
            .OrderBy(value => value, StringComparer.Ordinal)
            .ToArray();
        string[] catalogIds = AW3MultiplayerCatalog.Windows
            .Select(value => value.WindowId)
            .OrderBy(value => value, StringComparer.Ordinal)

            "the live AW3 window constants equal the public catalog");

        string facade = File.ReadAllText(Path.Combine(root, "Code", "api",
            "multiplayer", "AW3MultiplayerUiFacade.cs"));
        foreach (AW3WindowKind kind in Enum.GetValues<AW3WindowKind>())

            facade.Contains("MethodInfo", StringComparison.Ordinal) ||

    private static void CommandDispatcherOwnershipIsReferenceBased()
    {
        AW3AuthoritativeCommandRouter.Reset();
        AW3CommandRequest request = AW3CommandRequest.ConfigurePolicy(
            7, true, false);
        var first = new FakeDispatcher();
        var second = new FakeDispatcher();
        var throwingSubscription = new ThrowingSubscribeDispatcher();

        AW3CommandResult local =
            AW3MultiplayerCommandFacade.DispatchFromUi(request);
        Equal(AW3CommandStatus.Accepted, local.Status,
            "single-player UI uses the authoritative router");
        Equal(1, AW3AuthoritativeCommandRouter.DispatchCalls,
            "the local router executes once");

        Equal(false,
            AW3MultiplayerCommandFacade.Register(throwingSubscription),
            "failed provider event subscription rolls registration back");
        Same(null, AW3MultiplayerCommandFacade.Current,
            "failed subscription cannot leave a half-registered provider");

        var registrationChanges = 0;
        Action registrationChanged = () => registrationChanges++;
        AW3MultiplayerCommandFacade.Changed += registrationChanged;
        Equal(true, AW3MultiplayerCommandFacade.Register(first),
            "the first dispatcher owns UI dispatch");
        Equal(1, registrationChanges,
            "successful registration immediately refreshes UI subscribers");
        AW3MultiplayerCommandFacade.Changed -= registrationChanged;
        Equal(true, AW3MultiplayerCommandFacade.Register(first),
            "same-reference registration is idempotent");
        Same(first, AW3MultiplayerCommandFacade.Current,
            "the current dispatcher is observable");
        Equal(false, AW3MultiplayerCommandFacade.Register(second),
            "a second dispatcher cannot replace the owner");

        var changedCount = 0;
        Action changed = () => changedCount++;
        AW3MultiplayerCommandFacade.Changed += changed;
        first.NotifyChanged();
        Equal(1, changedCount,
            "provider state changes reach AW3 UI subscribers once");

        AW3CommandResult pending =
            AW3MultiplayerCommandFacade.DispatchFromUi(request);
        Equal(AW3CommandStatus.Pending, pending.Status,
            "registered multiplayer dispatcher receives UI commands");
        Equal(1, first.DispatchCalls,
            "the provider receives one command");
        Equal(1, AW3AuthoritativeCommandRouter.DispatchCalls,
            "provider UI dispatch does not execute locally");

        first.ThrowOnDispatch = true;
        AW3CommandResult isolated =
            AW3MultiplayerCommandFacade.DispatchFromUi(request);
        Equal(AW3CommandError.ExecutionFailed, isolated.Error,
            "provider exceptions are isolated");

        AW3CommandResult authoritative =
            AW3MultiplayerCommandFacade.DispatchAuthoritative(request);
        Equal(AW3CommandStatus.Accepted, authoritative.Status,
            "host-authoritative dispatch bypasses the provider");
        Equal(2, AW3AuthoritativeCommandRouter.DispatchCalls,
            "authoritative router executes after provider registration");

        var replicaOwner = new object();
        Equal(true, AW3MultiplayerReplicaScope.Activate(replicaOwner),
            "test replica session activates");
        try
        {
            AW3CommandResult replicaAuthoritative =
                AW3MultiplayerCommandFacade.DispatchAuthoritative(request);
            Equal(AW3CommandError.Unauthorized,
                replicaAuthoritative.Error,
                "a replica cannot bypass its dispatcher through the public authoritative API");
            Equal(2, AW3AuthoritativeCommandRouter.DispatchCalls,
                "replica authoritative dispatch never mutates local world state");
        }
        finally
        {
            Equal(true, AW3MultiplayerReplicaScope.Deactivate(replicaOwner),
                "test replica session deactivates");
        }

        Equal(false, AW3MultiplayerCommandFacade.Unregister(second),
            "a non-owner cannot unregister the dispatcher");
        Equal(true, AW3MultiplayerCommandFacade.Unregister(first),
            "the owner unregisters");
        Equal(2, changedCount,
            "dispatcher removal releases pending AW3 UI state once");
        first.NotifyChanged();
        Equal(2, changedCount,
            "unregistered providers cannot notify AW3 UI");
        AW3MultiplayerCommandFacade.Changed -= changed;
        Same(null, AW3MultiplayerCommandFacade.Current,
            "unregister clears provider ownership");
        Equal(true,
            AW3MultiplayerCommandFacade.DispatchFromUi(request).Accepted,
            "UI returns to local authoritative execution");
    }

    private static void AllPlayerFacingUiWritesUseDispatcher()
    {
        string root = Path.Combine(Directory.GetCurrentDirectory(), "Code",
            "ui");
        var writeCall = new Regex(
            @"\b(?<owner>[A-Za-z_][A-Za-z0-9_]*(?:Service|Writer|Store|Manager))" +
            @"\s*\.\s*(?<method>(?:Try|Force|Set|Start|Toggle|Rename|" +
            @"Respond|Issue|Execute|Apply|Change|Create|Delete|Remove|" +
            @"Grant|Revoke|Declare|Cancel|Commit|Appoint|Abolish|" +
            @"Relocate|Reclaim)[A-Za-z0-9_]*)\s*\(",
            RegexOptions.Compiled);
        var readOnlyExceptions = new HashSet<string>(StringComparer.Ordinal)
        {
            "TabManager.CreateTab",
            "CourtPetitionService.AppointmentFavor",
            "CourtService.TryProjectManualAppointmentCandidate",
            "AWHistoricalReadService.TrySchedule",
            "DiplomacyProposalService.TryGetActiveNonAggression",
            "DiplomacyProposalService.TryGetActiveTruce",
            "SuccessionDisputeService.TryGetMaterializedByKingdom",
            "WarNoticeService.TryGetPreparationSummary",
            "WarScoreService.TryGetSnapshot",
            "WarParticipantRosterService.TryBuildReadOnly",
            "ArmyRtsControllerService.TryGetProjection",
            "SchoolMapModeService.SetFocus"
        };
        var violations = new List<string>();
        foreach (string file in Directory.EnumerateFiles(root, "*.cs",
                     SearchOption.AllDirectories).OrderBy(value => value,
                     StringComparer.Ordinal))
        {
            string relative = Path.GetRelativePath(root, file)
                .Replace('\\', '/');
            string[] lines = File.ReadAllLines(file);
            for (var lineIndex = 0; lineIndex < lines.Length; lineIndex++)
            {
                foreach (Match match in writeCall.Matches(lines[lineIndex]))
                {
                    string call = match.Groups["owner"].Value + "." +
                                  match.Groups["method"].Value;
                    if (!readOnlyExceptions.Contains(call))
                        violations.Add(relative + ":" + (lineIndex + 1) +
                                       ":" + call);
                }
            }

            string source = string.Join("\n", lines);
            foreach ((string token, string label) in new[]
                     {
                         ("System.Reflection", "reflection"),
                         ("MethodInfo", "reflection"),
                         ("BindingFlags", "reflection"),
                         ("SQLiteCommand", "database writer"),
                         ("ExecuteNonQuery(", "database writer"),
                         (".OperatingDB", "database writer"),
                         (".setKing(", "direct ruler write"),
                         ("units.createNewUnit(", "world spawn"),
                         ("units.spawn(", "world spawn")
                     })
                if (source.Contains(token, StringComparison.Ordinal))
                    violations.Add(relative + ":" + label + ":" + token);
        }

        if (violations.Count > 0)
            throw new InvalidOperationException(
                "unclassified AW3 UI writes:\n" +
                string.Join("\n", violations));
    }

    private static void AssertTransportFree(Type type, string message)
    {
        Type current = type;
        while (current.HasElementType) current = current.GetElementType();
        string fullName = current.FullName ?? string.Empty;
        Equal(false, fullName.Contains("Unity", StringComparison.Ordinal) ||
                     fullName.Contains("WorldBox", StringComparison.Ordinal) ||
                     fullName.Contains("Network", StringComparison.Ordinal) ||
                     fullName.Contains("Transport", StringComparison.Ordinal),
            message);
    }

    private static void True(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(message +
                $": expected={expected}, actual={actual}");
    }

    private static void Same(object expected, object actual, string message)
    {
        if (!ReferenceEquals(expected, actual))
            throw new InvalidOperationException(message);
    }

    private static void Contains(string source, string expected,
        string message)
    {
        if (!source.Contains(expected, StringComparison.Ordinal))
            throw new InvalidOperationException(message);
    }

    private static void SequenceEqual<T>(IReadOnlyList<T> expected,
        IReadOnlyList<T> actual, string message)
    {
        if (expected.Count != actual.Count)
            throw new InvalidOperationException(message +
                $": expected={expected.Count}, actual={actual.Count}");
        for (var index = 0; index < expected.Count; index++)
            if (!EqualityComparer<T>.Default.Equals(expected[index],
                    actual[index]))
                throw new InvalidOperationException(message +
                    $": index={index}, expected={expected[index]}, " +
                    $"actual={actual[index]}");
    }

    private static void Throws<T>(Action action, string message)
        where T : Exception
    {
        try { action(); }
        catch (T) { return; }
        throw new InvalidOperationException(message);
    }

    private sealed class FakeDispatcher : IAW3CommandDispatcher
    {
        public event Action Changed;
        public int DispatchCalls { get; private set; }
        public bool ThrowOnDispatch { get; set; }

        public AW3CommandResult Dispatch(AW3CommandRequest request)
        {
            DispatchCalls++;
            if (ThrowOnDispatch)
                throw new InvalidOperationException("provider failure");
            return AW3CommandResult.Pending("aw3_command_pending");
        }

        public void NotifyChanged()
        {
            Changed?.Invoke();
        }
    }

    private sealed class ThrowingSubscribeDispatcher : IAW3CommandDispatcher
    {
        public event Action Changed
        {
            add => throw new InvalidOperationException("subscribe failure");
            remove { }
        }

        public AW3CommandResult Dispatch(AW3CommandRequest request)
        {
            return AW3CommandResult.Pending("aw3_command_pending");
        }
    }
