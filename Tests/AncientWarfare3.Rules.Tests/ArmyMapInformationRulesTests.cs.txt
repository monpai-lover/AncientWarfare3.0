using AncientWarfare3.core.lineage;

internal static class ArmyMapInformationRulesTests
{
    public static void Run()
    {
        Equal("12 / 27", ArmyMapInformationRules.ComposeManpowerText(
                shortage: 12, reserveSupply: 27),
            "shortage and supply remain distinct");
        Equal("0 / 5", ArmyMapInformationRules.ComposeManpowerText(
                shortage: -3, reserveSupply: 5),
            "display clamps invalid counts");
        Equal(ArmyRtsState.March,
            ArmyMapInformationRules.ResolvePendingState(
                hasProjection: true,
                projectionState: ArmyRtsState.March,
                memberCount: 1,
                minimumOperationalForce: 2),
            "a live projection remains authoritative while mission publication catches up");
        Equal(ArmyRtsState.Replenish,
            ArmyMapInformationRules.ResolvePendingState(
                hasProjection: false,
                projectionState: ArmyRtsState.Idle,
                memberCount: 1,
                minimumOperationalForce: 2),
            "a missionless under-strength army remains visible as replenishing");
        Equal(ArmyRtsState.Idle,
            ArmyMapInformationRules.ResolvePendingState(
                hasProjection: false,
                projectionState: ArmyRtsState.Replenish,
                memberCount: 2,
                minimumOperationalForce: 2),
            "a missionless operational army remains visible while awaiting orders");
        Equal(ArmyRtsState.Assault,
            ArmyMapInformationRules.ResolvePendingState(
                hasProjection: false,
                projectionState: ArmyRtsState.Idle,
                memberCount: 2,
                minimumOperationalForce: 2,
                hasCombatActivity: true),
            "a missionless army with an active combat target is shown fighting");
        Equal(ArmyRtsState.Idle,
            ArmyMapInformationRules.ResolvePendingState(
                hasProjection: false,
                projectionState: ArmyRtsState.Idle,
                memberCount: 1,
                minimumOperationalForce: 2,
                hasCombatActivity: false,
                hasReplenishmentOperation: false),
            "a shortage without a real operation must not claim replenishment");
        Equal(ArmyRtsState.Replenish,
            ArmyMapInformationRules.ResolvePendingState(
                hasProjection: false,
                projectionState: ArmyRtsState.Idle,
                memberCount: 1,
                minimumOperationalForce: 2,
                hasCombatActivity: false,
                hasReplenishmentOperation: true),
            "a pending operation keeps the replenishment label");
        Equal("aw_army_rts_state_replenish",
            ArmyMapInformationRules.PendingOperationLocalizationKey(
                ArmyRtsState.Replenish),
            "an under-strength army uses the replenishment label");
        Equal("aw_army_rts_state_awaiting_orders",
            ArmyMapInformationRules.PendingOperationLocalizationKey(
                ArmyRtsState.Idle),
            "an operational missionless army explicitly awaits orders");
        Equal("task_unit_aw_war_deployment",
            ArmyMapInformationRules.PendingOperationLocalizationKey(
                ArmyRtsState.Idle, royalGuardArmy: false,
                activeDeployment: true, deploymentArrived: false),
            "a missionless army marching to its notice frontier must not appear idle");
        Equal("aw_war_deployment_ready",
            ArmyMapInformationRules.PendingOperationLocalizationKey(
                ArmyRtsState.Idle, royalGuardArmy: false,
                activeDeployment: true, deploymentArrived: true),
            "an army assembled at the notice frontier reports frontier readiness");
        Equal(9, ArmyMapInformationRules.ResolvePendingShortage(
                hasMission: false, missionTargetStrength: 0,
                activeDeployment: true, standingTargetStrength: 22,
                memberCount: 13, minimumOperationalForce: 2),
            "notice deployment reports the standing establishment shortage");
        Equal(9, ArmyMapInformationRules.ResolvePendingShortage(
                hasMission: false, missionTargetStrength: 0,
                activeDeployment: false, standingTargetStrength: 22,
                memberCount: 13, minimumOperationalForce: 2),
            "an idle ordinary army reports its city establishment shortage");
        Equal(true, ArmyMapInformationRules.ShouldDisplayReserveManpower(
                royalGuardArmy: false),
            "ordinary army labels display shortage and reserve supply");
        Equal(false, ArmyMapInformationRules.ShouldDisplayReserveManpower(
                royalGuardArmy: true),
            "royal guard labels omit ordinary reserve-pool fields");
        Equal("task_unit_aw_guard_protect_king",
            ArmyMapInformationRules.PendingOperationLocalizationKey(
                ArmyRtsState.Replenish, royalGuardArmy: true),
            "an under-strength royal guard keeps its protect-king task");
        Equal("aw_army_rts_state_replenish",
            ArmyMapInformationRules.PendingOperationLocalizationKey(
                ArmyRtsState.Replenish, royalGuardArmy: false),
            "an ordinary under-strength army keeps its replenishment task");
        Equal(false, RoyalGuardArmyAssignmentRules.IsGuardArmyIdentity(
                roleMarked: false, ordinaryArmy: true,
                legacyNameMarked: false, captainRoyalGuard: true),
            "a royal-guard captain cannot turn an ordinary army into a guard army");
        Equal(true, RoyalGuardArmyAssignmentRules.IsGuardArmyIdentity(
                roleMarked: true, ordinaryArmy: false,
                legacyNameMarked: false, captainRoyalGuard: false),
            "the explicit royal-guard role remains authoritative");
        Equal(true, RoyalGuardArmyAssignmentRules.IsGuardArmyIdentity(
                roleMarked: false, ordinaryArmy: false,
                legacyNameMarked: true, captainRoyalGuard: true),
            "a detached legacy guard army remains recoverable by its name and captain");
        Equal(false, ArmyCaptainContinuityRules.ShouldRetainCaptain(
                structurallyStable: true, leaseEligible: false),
            "captain continuity cannot preserve a guard inside an ordinary army");
        Equal(true, ArmyCaptainContinuityRules.ShouldRetainCaptain(
                structurallyStable: true, leaseEligible: true),
            "a stable eligible ordinary captain remains protected from churn");
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(
                $"{name}: expected {expected}, got {actual}");
    }
}
