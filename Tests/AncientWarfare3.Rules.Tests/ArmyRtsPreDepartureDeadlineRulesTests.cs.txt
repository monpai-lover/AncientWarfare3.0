using System.IO;
using System.Reflection;
using AncientWarfare3.core.lineage;

internal static class ArmyRtsPreDepartureDeadlineRulesTests
{
    public static void Run()
    {
        ReplenishmentBypassNeverMakesAnUnderstrengthArmyReady();
        MissionStrengthSurvivesMissingAnchorsAndCasualties();
        ReplenishmentBypassCannotLeaveTheReplenishmentState();
        UnderstrengthArmiesRetryTheirReplenishmentRequest();
        ForceDepartureEscapesMissingEscortAfterDeadline();
        ForceDeploymentRequiresEscortQuorum();
        CasualtiesCannotHoldAnArmyInStagingForever();
        EscortQuorumComparesTheSamePopulation();
        StagingReleaseUsesTheOperationalForceGate();
        QuorumDoesNotDependOnTheThrottledCorrectionPath();
        StagingDeadlinesUseWorldTimeNotRealtime();
        Assert(ArmyStallWatchdogRules.ShouldExpectFormationProgress(
                ArmyRtsState.Rally, targetComplete: false),
            "watchdog samples stranded formation members during Rally");
        Assert(!ArmyRtsRules.ShouldForcePreDeparture(
                authoritative: true, state: ArmyRtsState.Rally,
                minimumForceReady: true, captainPresent: true,
                issuedWorldTime: 100d, currentWorldTime: 219.9d),
            "rally must retain its readiness checks before two years");
        Assert(ArmyRtsRules.ShouldForcePreDeparture(
                authoritative: true, state: ArmyRtsState.Rally,
                minimumForceReady: true, captainPresent: true,
                issuedWorldTime: 100d, currentWorldTime: 220d),
            "expired Rally timeout releases a viable captain from staging");
        Assert(ArmyRtsRules.ShouldForcePreDeparture(
                authoritative: true, state: ArmyRtsState.Replenish,
                minimumForceReady: true, captainPresent: true,
                issuedWorldTime: 100d, currentWorldTime: 220d),
            "expired replenishment staging also resumes viable movement");
        Assert(!ArmyRtsRules.ShouldForcePreDeparture(
                authoritative: true, state: ArmyRtsState.Rally,
                minimumForceReady: false, captainPresent: true,
                issuedWorldTime: 100d, currentWorldTime: 220d),
            "an understrength army must not be forced into combat");
        Assert(!ArmyRtsRules.ShouldForcePreDeparture(
                authoritative: true, state: ArmyRtsState.March,
                minimumForceReady: true, captainPresent: true,
                issuedWorldTime: 100d, currentWorldTime: 220d),
            "a marching army must not use the pre-departure timeout");
        Assert(!ArmyRtsRules.ShouldForcePreDeparture(
                authoritative: true, state: ArmyRtsState.Rally,
                minimumForceReady: true, captainPresent: true,
                issuedWorldTime: double.NaN, currentWorldTime: 220d),
            "invalid mission time must not force departure");
        Assert(!ArmyRtsRules.ShouldForceDeployment(
                authoritative: true, state: ArmyRtsState.Deploy,
                minimumForceReady: true, captainPresent: true,
                routeArrived: true, deploymentStartedWorldTime: 100d,
                currentWorldTime: 219.9d),
            "deployment must retain its escort quorum before the bounded wait expires");
        Assert(!ArmyRtsRules.ShouldForceDeployment(
                authoritative: true, state: ArmyRtsState.Deploy,
                minimumForceReady: true, captainPresent: true,
                routeArrived: true, deploymentStartedWorldTime: 100d,
                currentWorldTime: 220d),
            "the legacy deployment overload must not send a captain alone into assault");
        Assert(!ArmyRtsRules.ShouldForceDeployment(
                authoritative: true, state: ArmyRtsState.Deploy,
                minimumForceReady: true, captainPresent: true,
                routeArrived: false, deploymentStartedWorldTime: 100d,
                currentWorldTime: 220d),
            "a force without a completed route must not bypass the march state");
        Assert(!ArmyRtsRules.ShouldForceDeployment(
                authoritative: true, state: ArmyRtsState.Deploy,
                minimumForceReady: false, captainPresent: true,
                routeArrived: true, deploymentStartedWorldTime: 100d,
                currentWorldTime: 220d),
            "an understrength force must not be pushed into an assault by deployment timeout");
    }

    // A depleted army keeps max(living, persisted) as its mission target, so
    // departure strength never recovers. The bounded staging window must
    // still release it, or RTS ownership suppresses vanilla movement forever.
    private static void CasualtiesCannotHoldAnArmyInStagingForever()
    {
        Assert(!ArmyRtsRules.HasDepartureStrength(
                living: 30, targetStrength: 48,
                minimumForceReady: true,
                replenishmentBypassActive: false),
            "a depleted army has not recovered its departure strength");
        Assert(ArmyLogisticsRules.HasMinimumOperationalForce(30),
            "a 30-strong remnant is still an operational force");
        Assert(ArmyRtsRules.ShouldForcePreDeparture(
                authoritative: true, state: ArmyRtsState.Rally,
                minimumForceReady: ArmyLogisticsRules.
                    HasMinimumOperationalForce(30),
                captainPresent: true,
                issuedWorldTime: 100d, currentWorldTime: 220d),
            "an operational remnant escapes staging on the bounded window");
        Assert(!ArmyRtsRules.ShouldForcePreDeparture(
                authoritative: true, state: ArmyRtsState.Rally,
                minimumForceReady: ArmyLogisticsRules.
                    HasMinimumOperationalForce(1),
                captainPresent: true,
                issuedWorldTime: 100d, currentWorldTime: 220d),
            "a sub-minimal remnant still must not be forced out of staging");
    }

    // The rallied numerator counts eligible formation members only. Measuring
    // it against the raw roster makes the quorum unreachable for an army that
    // carries members the formation cannot command.
    private static void EscortQuorumComparesTheSamePopulation()
    {
        Equal(20, ArmyRtsRules.ResolveEscortPopulation(
                rosterLiving: 20, eligibleFollowersObserved: 19,
                observationComplete: true, captainPresent: true),
            "an all-military roster keeps its full population");
        Equal(15, ArmyRtsRules.ResolveEscortPopulation(
                rosterLiving: 20, eligibleFollowersObserved: 14,
                observationComplete: true, captainPresent: true),
            "ineligible members leave the escort population");
        Equal(20, ArmyRtsRules.ResolveEscortPopulation(
                rosterLiving: 20, eligibleFollowersObserved: 0,
                observationComplete: false, captainPresent: true),
            "an incomplete observation falls back to the roster");
        Equal(20, ArmyRtsRules.ResolveEscortPopulation(
                rosterLiving: 20, eligibleFollowersObserved: 0,
                observationComplete: true, captainPresent: false),
            "an absent captain cannot collapse the population to zero");

        Assert(!ArmyRtsRules.HasIncrementalEscortQuorum(
                rosterLiving: 20, ralliedFollowers: 14,
                captainPresent: true),
            "the raw roster denominator keeps a mixed army below quorum");
        Assert(ArmyRtsRules.HasIncrementalEscortQuorum(
                rosterLiving: ArmyRtsRules.ResolveEscortPopulation(
                    rosterLiving: 20, eligibleFollowersObserved: 14,
                    observationComplete: true, captainPresent: true),
                ralliedFollowers: 14, captainPresent: true),
            "a fully rallied eligible formation reaches quorum");
    }

    // Under AWSimulationMode.Large the tick rate stays constant while each
    // tick advances multiplier-scaled world time, so the realtime follower
    // correction budget buys progressively less world-time movement. Rally
    // therefore must not depend on that throttled path: a follower reaches
    // the rallied band using ordinary unthrottled pathfinding, which only
    // happens while the rallied radius is not tighter than the distance at
    // which long-range routing hands off to local correction.
    private static void QuorumDoesNotDependOnTheThrottledCorrectionPath()
    {
        const int ralliedRadiusSquared =
            ArmyFormationRules.LocalRadius * ArmyFormationRules.LocalRadius;
        const int escortRadiusSquared =
            ArmyFormationRules.LooseEscortOuterRadius *
            ArmyFormationRules.LooseEscortOuterRadius;
        const int handoffRadiusSquared =
            ArmySharedPathRules.LocalReconnectRadius *
            ArmySharedPathRules.LocalReconnectRadius;

        Assert(ralliedRadiusSquared >= handoffRadiusSquared,
            "a rallied band tighter than the long-range handoff would force followers to close the gap with throttled corrections, which starves quorum under large simulation steps");
        Assert(escortRadiusSquared >= handoffRadiusSquared,
            "the loose escort band must also be reachable by unthrottled pathfinding");

        Assert(!ArmySharedPathRules.ShouldUseLongRangeFollowerRoute(
                targetValid: true,
                targetDistanceSquared: ralliedRadiusSquared),
            "a follower at the rallied boundary has already left long-range routing");
        Assert(ArmySharedPathRules.ShouldUseLongRangeFollowerRoute(
                targetValid: true,
                targetDistanceSquared: ralliedRadiusSquared + 1),
            "a follower outside the rallied band still routes at long range");
    }

    // The staging and deployment windows are denominated in world seconds.
    // Feeding them realtime would decouple the deadline from the simulation
    // whenever a large step advances world time faster than the wall clock,
    // so the readiness clock must stay on the world timeline.
    private static void StagingDeadlinesUseWorldTimeNotRealtime()
    {
        string root = Directory.GetCurrentDirectory();
        string controller = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsControllerService.cs"));
        Assert(controller.Contains("double readinessTime = CurrentWorldTime();",
                StringComparison.Ordinal),
            "the readiness clock must be sampled from world time");

        int start = controller.IndexOf("double readinessTime =",
            StringComparison.Ordinal);
        int end = controller.IndexOf("return facts;", start,
            StringComparison.Ordinal);
        Assert(end > start, "BuildFacts must resolve its transition facts");
        string body = controller.Substring(start, end - start);
        Assert(!body.Contains("CurrentRealtime()", StringComparison.Ordinal),
            "a realtime sample inside the readiness window would drift from world time under a large simulation step");
    }

    // Guard the wiring itself: passing departure strength into the bounded
    // staging window reintroduces the permanent Rally stall.
    private static void StagingReleaseUsesTheOperationalForceGate()
    {
        string root = Directory.GetCurrentDirectory();
        string controller = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsControllerService.cs"));
        int start = controller.IndexOf("bool forcePreDeparture =",
            StringComparison.Ordinal);
        Assert(start >= 0,
            "the controller must resolve a bounded staging release");
        int end = controller.IndexOf(';', start);
        Assert(end > start, "the staging release must be a statement");
        string call = controller.Substring(start, end - start);
        Assert(call.Contains("minimumForceReady", StringComparison.Ordinal),
            "the staging window must be gated on the operational force");
        Assert(!call.Contains("departureStrengthReady",
                StringComparison.Ordinal),
            "departure strength never recovers after casualties, so it must not gate the escape hatch");
    }

    private static void ReplenishmentBypassNeverMakesAnUnderstrengthArmyReady()
    {
        Assert(!ArmyRtsRules.HasDepartureStrength(
                living: 9, targetStrength: 37,
                minimumForceReady: true,
                replenishmentBypassActive: true),
            "a stalled replenishment bypass must not send a 24-percent army into combat");
        Assert(ArmyRtsRules.HasDepartureStrength(
                living: 30, targetStrength: 37,
                minimumForceReady: true,
                replenishmentBypassActive: false),
            "an army at the 80-percent deployment quorum remains eligible to depart");
    }

    private static void MissionStrengthSurvivesMissingAnchorsAndCasualties()
    {
        Equal(48, ResolveMissionTargetStrength(
                persistedTargetStrength: 0, anchorTargetStrength: 2,
                rosterLiving: 48),
            "a detached army must retain its actual initial strength when its anchor has no warrior slots");
        Equal(48, ResolveMissionTargetStrength(
                persistedTargetStrength: 48, anchorTargetStrength: 2,
                rosterLiving: 9),
            "casualties must not lower a mission's recruitment target");
        Equal(60, ResolveMissionTargetStrength(
                persistedTargetStrength: 48, anchorTargetStrength: 60,
                rosterLiving: 9),
            "a larger anchor establishment may raise the retained mission target");
    }

    private static void ReplenishmentBypassCannotLeaveTheReplenishmentState()
    {
        MethodInfo method = typeof(ArmyRtsRules).GetMethod(
            "ShouldRemainInReplenishment",
            BindingFlags.Public | BindingFlags.Static, binder: null,
            types: new[] { typeof(bool), typeof(bool) },
            modifiers: null);
        if (method == null)
            throw new InvalidOperationException(
                "RTS must retain an understrength army in replenishment regardless of route-recovery bypasses");
        Assert((bool)method.Invoke(null, new object[]
            { true, false }),
            "an understrength army must remain in Replenish until it reaches the deployment quorum");
    }

    private static void UnderstrengthArmiesRetryTheirReplenishmentRequest()
    {
        MethodInfo method = typeof(ArmyRtsRules).GetMethod(
            "ShouldRetryStalledReplenishment",
            new[] { typeof(bool), typeof(bool), typeof(bool),
                    typeof(bool), typeof(bool) });
        if (method == null)
            throw new InvalidOperationException(
                "replenishment retry must distinguish an operational remnant from deployment readiness");
        Assert((bool)method.Invoke(null, new object[]
            { true, true, false, true, true }),
            "a 9-of-37 operational remnant must retry a stalled recruitment request");
        Assert(!(bool)method.Invoke(null, new object[]
            { true, true, true, true, true }),
            "an army already at the deployment quorum must not repeatedly enqueue replacement work");
    }

    private static void ForceDepartureEscapesMissingEscortAfterDeadline()
    {
        MethodInfo method = typeof(ArmyRtsRules).GetMethod(
            "ShouldForcePreDeparture", new[]
            {
                typeof(bool), typeof(ArmyRtsState), typeof(bool),
                typeof(bool), typeof(bool), typeof(double), typeof(double)
            });
        if (method == null)
            throw new InvalidOperationException(
                "RTS pre-departure timeout exposes escort readiness facts");

        object[] common =
        {
            true, ArmyRtsState.Rally, true, true, false, 100d, 219.9d
        };
        Assert(!(bool)method.Invoke(null, common),
            "Rally still requires normal readiness before its deadline");
        common[6] = 220d;
        Assert((bool)method.Invoke(null, common),
            "expired Rally leaves staging despite a broken follower quorum");
    }

    private static void ForceDeploymentRequiresEscortQuorum()
    {
        MethodInfo method = typeof(ArmyRtsRules).GetMethod(
            "ShouldForceDeployment", new[]
            {
                typeof(bool), typeof(ArmyRtsState), typeof(bool),
                typeof(bool), typeof(bool), typeof(bool),
                typeof(double), typeof(double)
            });
        if (method == null)
            throw new InvalidOperationException(
                "RTS deployment timeout must require an escort quorum");

        object[] common =
        {
            true, ArmyRtsState.Deploy, true, true, false, true, 100d,
            220d
        };
        Assert(!(bool)method.Invoke(null, common),
            "a captain at the objective must wait when his followers are late");
        common[4] = true;
        Assert((bool)method.Invoke(null, common),
            "a fully deployed army may leave bounded staging after two years");
    }

    private static int ResolveMissionTargetStrength(
        int persistedTargetStrength, int anchorTargetStrength,
        int rosterLiving)
    {
        MethodInfo method = typeof(ArmyRtsRules).GetMethod(
            "ResolveMissionTargetStrength",
            BindingFlags.Public | BindingFlags.Static);
        if (method == null)
            throw new InvalidOperationException(
                "RTS mission strength must survive detached-army anchor loss");
        return (int)method.Invoke(null, new object[]
        {
            persistedTargetStrength, anchorTargetStrength, rosterLiving
        });
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(message + ": expected " +
                                                expected + ", got " + actual);
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
}
