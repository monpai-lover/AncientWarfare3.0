using AncientWarfare3.core.court;

internal static class CityBureauRetryRulesTests
{
    internal static void Run()
    {
        Assert(!CityBureauRetryRules.ShouldRetry(true, 0, 3),
            "a completed city pass must not retry just because it has a vacancy");
        Assert(CityBureauRetryRules.ShouldRetry(false, 0, 3),
            "a failed city write may retry");
        Assert(!CityBureauRetryRules.ShouldRetry(false, 2, 3),
            "a failed city write must stop at the retry limit");
        Assert(CityBureauRetryRules.ShouldSkipSameFrame(120, 120),
            "a deferred vacancy retry must skip duplicate work in the same frame");
        Assert(!CityBureauRetryRules.ShouldSkipSameFrame(120, 121),
            "a deferred vacancy retry may resume on the next frame");
    }

    private static void Assert(bool pCondition, string pMessage)
    {
        if (!pCondition) throw new InvalidOperationException(pMessage);
    }
}
