using System;
using AncientWarfare3.core.performance;

internal static class OperationalPathCadenceRulesTests
{
    internal static void Run()
    {
        foreach (bool large in new[] { false, true })
        foreach (bool militaryPriority in new[] { false, true })
        {
            bool expected = large && militaryPriority;
            Assert(ArmyRtsMovementCadenceRules.ShouldRunDuringSkippedPathBatch(
                    large, militaryPriority) == expected,
                "only active military priority bypasses a skipped Large path batch");
            Assert(ArmyRtsMovementCadenceRules.ShouldRunDuringSkippedMovementBatch(
                    large, militaryPriority) == expected,
                "path and smooth movement retain the same military cadence");
        }
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
}
