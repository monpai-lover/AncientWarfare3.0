using System;
using System.IO;
using AncientWarfare3.core.court;

internal static class DeJureNewCityAssignmentRulesTests
{
    internal static void Run()
    {
        long adjacent = DeJureNewCityAssignmentRules.Select(new[]
        {
            new DeJureNewCityRegionCandidate(2L, true, 1, 900, 900, true),
            new DeJureNewCityRegionCandidate(1L, true, 1, 1200, 1200, true)
        }, pSelector: 0);
        Equal(1L, adjacent,
            "the first adjacent capital is selected by the deterministic selector");

        long adjacentSecond = DeJureNewCityAssignmentRules.Select(new[]
        {
            new DeJureNewCityRegionCandidate(2L, true, 1, 900, 900, true),
            new DeJureNewCityRegionCandidate(1L, true, 1, 1200, 1200, true)
        }, pSelector: 1);
        Equal(2L, adjacentSecond,
            "multiple adjacent capitals can select the other candidate");

        long nearestMember = DeJureNewCityAssignmentRules.Select(new[]
        {
            new DeJureNewCityRegionCandidate(7L, false, 0, 900, 900, true),
            new DeJureNewCityRegionCandidate(3L, false, 0, 1200, 1200, true)
        }, pSelector: 0);
        Equal(7L, nearestMember,
            "nearest adjacent member breaks adjacent-count ties");

        long noAdjacentSeat = DeJureNewCityAssignmentRules.Select(new[]
        {
            new DeJureNewCityRegionCandidate(2L, false, 0, 400, 300, true),
            new DeJureNewCityRegionCandidate(1L, false, 0, 200, 500, true)
        }, pSelector: 0);
        Equal(1L, noAdjacentSeat,
            "a city without an adjacent capital joins the nearest state");

        long seatTie = DeJureNewCityAssignmentRules.Select(new[]
        {
            new DeJureNewCityRegionCandidate(9L, false, 1, 100, 400, true),
            new DeJureNewCityRegionCandidate(4L, false, 1, 100, 200, true)
        }, pSelector: 0);
        Equal(4L, seatTie,
            "seat distance breaks adjacent-seat ties");

        long idTie = DeJureNewCityAssignmentRules.Select(new[]
        {
            new DeJureNewCityRegionCandidate(9L, false, 1, 100, 200, true),
            new DeJureNewCityRegionCandidate(4L, false, 1, 100, 200, true)
        }, pSelector: 0);
        Equal(4L, idTie, "region ID makes ties deterministic");

        long excluded = DeJureNewCityAssignmentRules.Select(new[]
        {
            new DeJureNewCityRegionCandidate(2L, true, 1, 1, 1, false),
            new DeJureNewCityRegionCandidate(1L, false, 0, 1, 1, false)
        }, pSelector: 0);
        Equal(-1L, excluded,
            "foreign, retired, empty, or ineligible candidates are excluded");

        Equal(-1L, DeJureNewCityAssignmentRules.Select(
            Array.Empty<DeJureNewCityRegionCandidate>(), pSelector: 0),
            "no candidate leaves a city unassigned");

        Equal(-1L, DeJureNewCityAssignmentRules.Select(new[]
        {
            new DeJureNewCityRegionCandidate(8L, true, 1, 1, 1, false)
        }, pSelector: 0),
            "a six-city region is excluded from automatic assignment");

        string service = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "court",
            "DeJureNewCityAssignmentService.cs"));
        string store = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "court",
            "DeJureRegionStore.cs"));
        True(service.Contains("StableSelector(pCity.data.id)"),
            "new-city assignment uses a stable city/world selector");
        int membersStart = service.IndexOf("var members", StringComparison.Ordinal);
        int membersEnd = membersStart < 0 ? -1 : service.IndexOf(
            ".ToList();", membersStart, StringComparison.Ordinal);
        True(membersStart >= 0 && membersEnd > membersStart &&
             service.Substring(membersStart, membersEnd - membersStart)
                 .Contains("city.kingdom == pKingdom",
                     StringComparison.Ordinal),
            "new-city candidates are restricted to the founded city's kingdom");
        True(store.Contains("sameKingdomMember", StringComparison.Ordinal) &&
             store.Contains("member.kingdom == pCity.kingdom",
                 StringComparison.Ordinal),
            "automatic state writes reject cross-kingdom targets");
        True(store.Contains("EnsureAllKingdomCapitalSeatsLocked();\n                    Migrate(_store)",
                StringComparison.Ordinal),
            "legacy migration establishes kingdom capitals before grouping");
        False(store.Contains("legacy_single_city", StringComparison.Ordinal),
            "legacy migration never creates an independent state for every city");
        False(service.Contains("CreateState(pCity"),
            "new-city assignment does not implicitly create a state");
    }

    private static void Equal(long expected, long actual, string message)
    {
        if (expected != actual)
            throw new InvalidOperationException(message +
                $": expected {expected}, got {actual}");
    }

    private static void True(bool value, string message)
    {
        if (!value) throw new InvalidOperationException(message +
            ": expected true");
    }

    private static void False(bool value, string message)
    {
        if (value) throw new InvalidOperationException(message +
            ": expected false");
    }
}
