    private static void RetryLimitProducesFailureCompletionForEverySequence()
    {
        using var sink = new RetryableBatchSink("database is busy");
        using var worker = new HistoricalWriteWorker(sink, batchSize: 8,
            retryDelayMilliseconds: 0, maxRetryAttempts: 2);
        for (long sequence = 11L; sequence <= 12L; sequence++)
        {
            string key = "retry-limit-test:" + sequence;
            var envelope = new HistoricalWriteEnvelope(sequence, key,
                string.Empty, Array.Empty<HistoricalSqlValue>(),
                HistoricalWriteKind.Append,
                new AWAsyncStamp(1L, sequence, sequence));
            True(worker.TryEnqueue(envelope, out _),
                "retry-limit test accepts sequence " + sequence);
        }

        worker.Start();
        True(SpinWait.SpinUntil(() => worker.TerminalFaulted,
                TimeSpan.FromSeconds(2)),
            "retry limit faults the worker");
        Equal(2L, sink.ExecuteCount,
            "retryable batch executes through the configured limit");
        True(worker.TryDequeueCompletion(out HistoricalWriteCompletion result),
            "retry limit publishes a completion");
        False(ReadProperty<bool>(result, "IsCommitted"),
            "retry-limit completion is not committed");
        Equal("retry limit 2 reached: database is busy",
            ReadProperty<string>(result, "Error"),
            "retry-limit completion preserves the terminal reason");
        Equal(2, result.Sequences.Count,
            "retry-limit completion covers every in-flight sequence");
        Equal(11L, result.Sequences[0], "retry-limit completion sequence 11");
        Equal(12L, result.Sequences[1], "retry-limit completion sequence 12");
    }

    private static void ConsecutiveQueuedReplacementsTransferProvenance()
    {
        using var sink = new SuccessfulBatchSink();
        using var worker = new HistoricalWriteWorker(sink, batchSize: 1,
            retryDelayMilliseconds: 0, maxRetryAttempts: 1);
        var callbacks = new HistoricalWriteCallbackRegistry();
        var pending = new FamilyTreeProjectionPendingState();
        var replacements = new List<long>();
        FamilyTreeProjectionChange committedChange =
            FamilyTreeProjectionChange.None;
        string key = "replacement-chain:actor";

        for (long sequence = 91L; sequence <= 93L; sequence++)
        {
            FamilyTreeProjectionChange ownChange = sequence == 91L
                ? FamilyTreeProjectionChange.FamilyStructure
                : sequence == 92L
                    ? FamilyTreeProjectionChange.LifeStatus
                    : FamilyTreeProjectionChange.IdentityOrTitle;
            var envelope = new HistoricalWriteEnvelope(sequence, key,
                string.Empty, Array.Empty<HistoricalSqlValue>(),
                HistoricalWriteKind.State,
                new AWAsyncStamp(1L, sequence, sequence));
            True(callbacks.TryEnqueue(worker, envelope,
                    (accepted, replaced) =>
                    {
                        replacements.Add(replaced);
                        pending.TransferOwnership(701L, replaced, accepted);
                        pending.Publish(701L, accepted, ownChange);
                    }, (committed, outcome) =>
                    {
                        pending.TryComplete(701L, committed,
                            out committedChange);
                    }, null, out _),
                "replacement chain accepts sequence " + sequence);
        }

        Equal(0L, replacements[0], "chain A replaces no sequence");
        Equal(91L, replacements[1], "chain B replaces A");
        Equal(92L, replacements[2], "chain C replaces B");
        worker.Start();
        True(SpinWait.SpinUntil(() => worker.PendingCompletionCount > 0,
                TimeSpan.FromSeconds(2)),
            "replacement chain publishes one completion");
        True(worker.TryDequeueCompletion(
                out HistoricalWriteCompletion completion),
            "replacement chain completion is available");
        callbacks.Take(completion.Sequences[0],
            out Action<long, object> committedCallback,
            out Action<long, string> failureCallback);
        committedCallback?.Invoke(completion.Sequences[0],
            completion.Outcomes[0]);

        Equal(93L, completion.Sequences[0],
            "only chain C reaches the worker completion");
        True(committedChange == FamilyTreeProjectionChange.FamilyStructure,
            "A provenance transfers through B into C exactly once");
        True(!pending.TryComplete(701L, 91L, out _) &&
             !pending.TryComplete(701L, 92L, out _),
            "chain A and B retain no orphaned provenance");
    }

    private static void InFlightCommitSurvivesFailedReplacementChain()
    {
        using var sink = new SuccessfulThenTerminalSink(
            "injected replacement chain failure");
        using var worker = new HistoricalWriteWorker(sink, batchSize: 1,
            retryDelayMilliseconds: 0, maxRetryAttempts: 1);
        var callbacks = new HistoricalWriteCallbackRegistry();
        var pending = new FamilyTreeProjectionPendingState();
        int revisionAdvanceCount = 0;
        var commitTaken = new ManualResetEventSlim(false);
        var allowCommit = new ManualResetEventSlim(false);
        var failureFinished = new ManualResetEventSlim(false);
        Exception commitDrainError = null;
        Exception failureDrainError = null;

        var independent = new HistoricalWriteEnvelope(101L,
            "inflight-chain:independent", string.Empty,
            Array.Empty<HistoricalSqlValue>(), HistoricalWriteKind.State,
            new AWAsyncStamp(1L, 101L, 101L));
        True(callbacks.TryEnqueue(worker, independent,
                (sequence, replaced) => pending.Publish(702L, sequence,
                    FamilyTreeProjectionChange.FamilyStructure),
                (sequence, outcome) =>
                {
                    if (pending.TryComplete(702L, sequence,
                            out FamilyTreeProjectionChange change) &&
                        change == FamilyTreeProjectionChange.FamilyStructure)
                        Interlocked.Increment(ref revisionAdvanceCount);
                }, null, out _),
            "independent in-flight sequence is accepted");
        worker.Start();
        bool firstExecuteEntered = sink.FirstExecuteEntered.Wait(
            ConcurrentDrainTimeout);
        try
        {
            True(firstExecuteEntered,
                "independent sequence enters the worker before chain enqueue");

            long expectedReplacement = 0L;
            for (long sequence = 102L; sequence <= 104L; sequence++)
            {
                var envelope = new HistoricalWriteEnvelope(sequence,
                    "inflight-chain:replacement", string.Empty,
                    Array.Empty<HistoricalSqlValue>(),
                    HistoricalWriteKind.State,
                    new AWAsyncStamp(1L, sequence, sequence));
                long expected = expectedReplacement;
                True(callbacks.TryEnqueue(worker, envelope,
                        (accepted, replaced) =>
                        {
                            Equal(expected, replaced,
                                "replacement chain reports explicit owner");
                            pending.TransferOwnership(702L, replaced,
                                accepted);
                            pending.Publish(702L, accepted,
                                FamilyTreeProjectionChange.IdentityOrTitle);
                        }, (committed, outcome) =>
                        {
                            if (pending.TryComplete(702L, committed, out _))
                                Interlocked.Increment(
                                    ref revisionAdvanceCount);
                        }, (failed, error) => pending.Fail(702L, failed),
                        out _),
                    "in-flight replacement chain accepts " + sequence);
                expectedReplacement = sequence;
            }
        }
        finally
        {
            sink.AllowFirstExecute.Set();
        }

        True(SpinWait.SpinUntil(() => worker.PendingCompletionCount >= 2,
                ConcurrentDrainTimeout),
            "worker publishes independent success and chain failure");
        var commitDrain = new Thread(() =>
        {
            try
            {
                True(worker.TryDequeueCompletion(
                        out HistoricalWriteCompletion completion),
                    "commit drain dequeues the independent completion");
                callbacks.Take(completion.Sequences[0],
                    out Action<long, object> committed,
                    out Action<long, string> failed);
                commitTaken.Set();
                True(allowCommit.Wait(ConcurrentDrainTimeout),
                    "commit drain receives the main-thread release signal");
                committed?.Invoke(completion.Sequences[0],
                    completion.Outcomes[0]);
            }
            catch (Exception error)
            {
                commitDrainError = error;
            }
        });
        var failureDrain = new Thread(() =>
        {
            try
            {
                True(commitTaken.Wait(ConcurrentDrainTimeout),
                    "failure drain waits until the commit callback is taken");
                True(worker.TryDequeueCompletion(
                        out HistoricalWriteCompletion completion),
                    "failure drain dequeues the replacement failure");
                callbacks.Take(completion.Sequences[0],
                    out Action<long, object> committed,
                    out Action<long, string> failed);
                failed?.Invoke(completion.Sequences[0], completion.Error);
                failureFinished.Set();
            }
            catch (Exception error)
            {
                failureDrainError = error;
            }
        });
        commitDrain.Start();
        failureDrain.Start();
        bool failureFinishedInTime = false;
        bool commitDrainJoined = false;
        bool failureDrainJoined = false;
        int advanceCountBeforeCommit = -1;
        try
        {
            failureFinishedInTime = failureFinished.Wait(
                ConcurrentDrainTimeout);
            advanceCountBeforeCommit = Volatile.Read(
                ref revisionAdvanceCount);
        }
        finally
        {
            allowCommit.Set();
            commitTaken.Set();
            commitDrainJoined = commitDrain.Join(ConcurrentDrainTimeout);
            failureDrainJoined = failureDrain.Join(ConcurrentDrainTimeout);
        }

        string drainDiagnostics =
            "; commit joined=" + commitDrainJoined +
            ", failure joined=" + failureDrainJoined +
            ", commit error=" + (commitDrainError?.ToString() ?? "none") +
            ", failure error=" + (failureDrainError?.ToString() ?? "none");
        True(failureFinishedInTime,
            "replacement chain failure runs before independent commit" +
            drainDiagnostics);
        True(commitDrainJoined && failureDrainJoined,
            "both completion drains exit after their explicit handshakes" +
            drainDiagnostics);
        True(commitDrainError == null && failureDrainError == null,
            "completion drains finish without background exceptions" +
            drainDiagnostics);
        Equal(0L, advanceCountBeforeCommit,
            "failed replacement chain never advances revision");

        Equal(1L, Volatile.Read(ref revisionAdvanceCount),
            "independent in-flight success advances exactly once");
        True(!pending.TryComplete(702L, 102L, out _) &&
             !pending.TryComplete(702L, 103L, out _) &&
             !pending.TryComplete(702L, 104L, out _),
            "C failure clears only the complete replacement chain");
    }

    private static void LatePrerequisiteFollowsReplacementChain()
    {
        FamilyTreeProjectionChange committedChange =
            RunLatePrerequisiteChain(pFail: false, pActorId: 703L,
                pFirstSequence: 111L);
        True(committedChange == FamilyTreeProjectionChange.FamilyStructure,
            "late prerequisite transfers from B through C on commit");

        FamilyTreeProjectionChange failedChange =
            RunLatePrerequisiteChain(pFail: true, pActorId: 704L,
                pFirstSequence: 121L);
        True(failedChange == FamilyTreeProjectionChange.None,
            "failed C never advances its uncommitted prerequisite");
    }

    private static FamilyTreeProjectionChange RunLatePrerequisiteChain(
        bool pFail, long pActorId, long pFirstSequence)
    {
        using IHistoricalWriteBatchSink sink = pFail
            ? new TerminalBatchSink("injected prerequisite chain failure")
            : new SuccessfulBatchSink();
        using var worker = new HistoricalWriteWorker(sink, batchSize: 1,
            retryDelayMilliseconds: 0, maxRetryAttempts: 1);
        var callbacks = new HistoricalWriteCallbackRegistry();
        var pending = new FamilyTreeProjectionPendingState();
        FamilyTreeProjectionChange observedChange =
            FamilyTreeProjectionChange.None;
        string key = "late-prerequisite:" + pActorId;

        for (long offset = 0L; offset < 3L; offset++)
        {
            long sequence = pFirstSequence + offset;
            var envelope = new HistoricalWriteEnvelope(sequence, key,
                string.Empty, Array.Empty<HistoricalSqlValue>(),
                HistoricalWriteKind.State,
                new AWAsyncStamp(1L, sequence, sequence));
            True(callbacks.TryEnqueue(worker, envelope,
                    (accepted, replaced) =>
                    {
                        pending.TransferOwnership(pActorId, replaced,
                            accepted);
                        pending.Publish(pActorId, accepted,
                            FamilyTreeProjectionChange.None);
                    }, (committed, outcome) =>
                    {
                        if (pending.TryComplete(pActorId, committed,
                                out FamilyTreeProjectionChange change))
                            observedChange = change;
                    }, (failed, error) => pending.Fail(pActorId, failed),
                    out _),
                "late prerequisite chain accepts offset " + offset);
            if (offset == 1L)
                pending.IncludePrerequisite(pActorId,
                    FamilyTreeProjectionChange.FamilyStructure);
        }

        worker.Start();
        True(SpinWait.SpinUntil(() => worker.PendingCompletionCount > 0,
                TimeSpan.FromSeconds(2)),
            "late prerequisite chain publishes a completion");
        True(worker.TryDequeueCompletion(
                out HistoricalWriteCompletion completion),
            "late prerequisite completion is available");
        callbacks.Take(completion.Sequences[0],
            out Action<long, object> committedCallback,
            out Action<long, string> failedCallback);
        if (completion.IsCommitted)
            committedCallback?.Invoke(completion.Sequences[0],
                completion.Outcomes[0]);
        else
            failedCallback?.Invoke(completion.Sequences[0],
                completion.Error);

        True(!pending.TryComplete(pActorId, pFirstSequence, out _) &&
             !pending.TryComplete(pActorId, pFirstSequence + 1L, out _) &&
             !pending.TryComplete(pActorId, pFirstSequence + 2L, out _),
            "late prerequisite chain leaves no orphaned owner");
        pending.Publish(pActorId, pFirstSequence + 3L,
            FamilyTreeProjectionChange.None);
        True(pending.TryComplete(pActorId, pFirstSequence + 3L,
                out FamilyTreeProjectionChange laterChange),
            "later independent owner remains usable");
        True(laterChange == FamilyTreeProjectionChange.None,
            "failed prerequisite cannot leak into later owner");
        return observedChange;
    }

    private static void QueuedReplacementExplicitlyTransfersProjectionSource()
    {
        using var sink = new SuccessfulBatchSink();
        using var worker = new HistoricalWriteWorker(sink, batchSize: 1,
            retryDelayMilliseconds: 0, maxRetryAttempts: 1);
        var callbacks = new HistoricalWriteCallbackRegistry();
        var pending = new FamilyTreeProjectionPendingState();
        long observedReplacement = -1L;
        FamilyTreeProjectionChange committedChange =
            FamilyTreeProjectionChange.None;
        var original = new HistoricalWriteEnvelope(81L,
            "queued-replacement:actor", string.Empty,
            Array.Empty<HistoricalSqlValue>(), HistoricalWriteKind.State,
            new AWAsyncStamp(1L, 81L, 81L));
        var replacement = new HistoricalWriteEnvelope(82L,
            "queued-replacement:actor", string.Empty,
            Array.Empty<HistoricalSqlValue>(), HistoricalWriteKind.State,
            new AWAsyncStamp(1L, 82L, 82L));

        True(callbacks.TryEnqueue(worker, original,
                (sequence, replacedSequence) =>
                {
                    observedReplacement = replacedSequence;
                    pending.Publish(602L, sequence,
                        FamilyTreeProjectionChange.FamilyStructure);
                }, (sequence, outcome) =>
                {
                    pending.TryComplete(602L, sequence,
                        out committedChange);
                }, null, out _),
            "queued replacement test accepts original sequence");
        Equal(0L, observedReplacement,
            "the original queued write replaces no source");
        True(callbacks.TryEnqueue(worker, replacement,
                (sequence, replacedSequence) =>
                {
                    observedReplacement = replacedSequence;
                    pending.TransferOwnership(602L, replacedSequence,
                        sequence);
                    pending.Publish(602L, sequence,
                        FamilyTreeProjectionChange.IdentityOrTitle);
                }, (sequence, outcome) =>
                {
                    pending.TryComplete(602L, sequence,
                        out committedChange);
                }, null, out _),
            "queued replacement test accepts replacement sequence");
        Equal(81L, observedReplacement,
            "registry reports only the actually coalesced queued sequence");

        worker.Start();
        True(SpinWait.SpinUntil(() => worker.PendingCompletionCount > 0,
                TimeSpan.FromSeconds(2)),
            "queued replacement produces one completion");
        True(worker.TryDequeueCompletion(
                out HistoricalWriteCompletion completion),
            "queued replacement completion is available");
        callbacks.Take(completion.Sequences[0],
            out Action<long, object> committed,
            out Action<long, string> failed);
        committed?.Invoke(completion.Sequences[0], completion.Outcomes[0]);

        Equal(82L, completion.Sequences[0],
            "only replacement sequence reaches completion");
        True(committedChange == FamilyTreeProjectionChange.FamilyStructure,
            "replacement commit releases explicitly transferred structure");
        True(!pending.TryComplete(602L, 81L, out _),
            "replaced sequence has no independent completion source");
    }

    private static void NewerFailureCannotConsumeTakenCommittedSource()
    {
        using var sink = new SuccessfulThenTerminalSink(
            "injected newer identity failure");
        using var worker = new HistoricalWriteWorker(sink, batchSize: 1,
            retryDelayMilliseconds: 0, maxRetryAttempts: 1);
        var callbacks = new HistoricalWriteCallbackRegistry();
        var pending = new FamilyTreeProjectionPendingState();
        int revisionAdvanceCount = 0;
        var birthCallbackTaken = new ManualResetEventSlim(false);
        var allowBirthCallback = new ManualResetEventSlim(false);
        var identityFailureFinished = new ManualResetEventSlim(false);

        long birthReplacedSequence = -1L;
        long identityReplacedSequence = -1L;
        var birth = new HistoricalWriteEnvelope(71L,
            "failure-first:actor", string.Empty,
            Array.Empty<HistoricalSqlValue>(), HistoricalWriteKind.State,
            new AWAsyncStamp(1L, 71L, 71L));
        var identity = new HistoricalWriteEnvelope(72L,
            "failure-first:actor", string.Empty,
            Array.Empty<HistoricalSqlValue>(), HistoricalWriteKind.State,
            new AWAsyncStamp(1L, 72L, 72L));
        True(callbacks.TryEnqueue(worker, birth,
                (sequence, replacedSequence) =>
                {
                    birthReplacedSequence = replacedSequence;
                    pending.Publish(601L, sequence,
                        FamilyTreeProjectionChange.FamilyStructure);
                },
                (sequence, outcome) =>
                {
                    if (pending.TryComplete(601L, sequence,
                            out FamilyTreeProjectionChange change) &&
                        change == FamilyTreeProjectionChange.FamilyStructure)
                        Interlocked.Increment(ref revisionAdvanceCount);
                }, null, out _),
            "failure-first test accepts birth sequence");
        Equal(0L, birthReplacedSequence,
            "birth sequence replaces no queued owner");

        worker.Start();
        True(sink.FirstExecuteEntered.Wait(TimeSpan.FromSeconds(2)),
            "birth sequence is in flight before identity enqueue");
        True(callbacks.TryEnqueue(worker, identity,
                (sequence, replacedSequence) =>
                {
                    identityReplacedSequence = replacedSequence;
                    pending.TransferOwnership(601L, replacedSequence,
                        sequence);
                    pending.Publish(601L, sequence,
                        FamilyTreeProjectionChange.IdentityOrTitle);
                },
                (sequence, outcome) =>
                {
                    if (pending.TryComplete(601L, sequence, out _))
                        Interlocked.Increment(ref revisionAdvanceCount);
                }, (sequence, error) => pending.Fail(601L, sequence),
                out _),
            "failure-first test accepts identity sequence");
        Equal(0L, identityReplacedSequence,
            "in-flight birth is not reported as a queued replacement");

        sink.AllowFirstExecute.Set();
        True(SpinWait.SpinUntil(() => worker.PendingCompletionCount >= 2,
                TimeSpan.FromSeconds(2)),
            "worker publishes birth success and identity failure completions");

        var birthDrain = new Thread(() =>
        {
            if (!worker.TryDequeueCompletion(
                    out HistoricalWriteCompletion completion)) return;
            callbacks.Take(completion.Sequences[0],
                out Action<long, object> committed,
                out Action<long, string> failed);
            birthCallbackTaken.Set();
            allowBirthCallback.Wait(TimeSpan.FromSeconds(2));
            committed?.Invoke(completion.Sequences[0],
                completion.Outcomes[0]);
        });
        var failureDrain = new Thread(() =>
        {
            birthCallbackTaken.Wait(TimeSpan.FromSeconds(2));
            if (!worker.TryDequeueCompletion(
                    out HistoricalWriteCompletion completion)) return;
            callbacks.Take(completion.Sequences[0],
                out Action<long, object> committed,
                out Action<long, string> failed);
            failed?.Invoke(completion.Sequences[0], completion.Error);
            identityFailureFinished.Set();
        });
        birthDrain.Start();
        failureDrain.Start();
        True(identityFailureFinished.Wait(TimeSpan.FromSeconds(2)),
            "newer identity failure runs before taken birth callback");
        Equal(0L, Volatile.Read(ref revisionAdvanceCount),
            "failure callback never advances projection revision");
        allowBirthCallback.Set();
        birthDrain.Join(TimeSpan.FromSeconds(2));
        failureDrain.Join(TimeSpan.FromSeconds(2));

        Equal(1L, Volatile.Read(ref revisionAdvanceCount),
            "taken committed birth source advances exactly once after failure");
    }

    private static void CompletionCannotRunBeforeAcceptedPendingPublication()
    {
        using var sink = new SuccessfulBatchSink();
        using var worker = new HistoricalWriteWorker(sink, batchSize: 1,
            retryDelayMilliseconds: 0, maxRetryAttempts: 1);
        var callbacks = new HistoricalWriteCallbackRegistry();
        var pending = new FamilyTreeProjectionPendingState();
        var acceptedEntered = new ManualResetEventSlim(false);
        var allowPendingPublication = new ManualResetEventSlim(false);
        var takeStarted = new ManualResetEventSlim(false);
        var takeReturned = new ManualResetEventSlim(false);
        bool revisionReleased = false;
        bool enqueueSucceeded = false;
        var envelope = new HistoricalWriteEnvelope(61L,
            "accepted-publication:61", string.Empty,
            Array.Empty<HistoricalSqlValue>(), HistoricalWriteKind.State,
            new AWAsyncStamp(1L, 61L, 61L));

        worker.Start();
        var producer = new Thread(() =>
        {
            enqueueSucceeded = callbacks.TryEnqueue(worker, envelope,
                sequence =>
                {
                    acceptedEntered.Set();
                    allowPendingPublication.Wait(TimeSpan.FromSeconds(2));
                    pending.Publish(501L, sequence,
                        FamilyTreeProjectionChange.FamilyStructure);
                }, (sequence, outcome) =>
                    revisionReleased = pending.TryComplete(501L, sequence,
                        out FamilyTreeProjectionChange change) &&
                        change == FamilyTreeProjectionChange.FamilyStructure,
                null, out _);
        });
        producer.Start();

        True(acceptedEntered.Wait(TimeSpan.FromSeconds(2)),
            "accepted callback enters before publishing pending metadata");
        True(SpinWait.SpinUntil(() => worker.PendingCompletionCount > 0,
                TimeSpan.FromSeconds(2)),
            "the worker completes before pending publication");
        var consumer = new Thread(() =>
        {
            if (!worker.TryDequeueCompletion(
                    out HistoricalWriteCompletion completion)) return;
            takeStarted.Set();
            callbacks.Take(completion.Sequences[0],
                out Action<long, object> committed,
                out Action<long, string> failed);
            takeReturned.Set();
            committed?.Invoke(completion.Sequences[0],
                completion.Outcomes[0]);
        });
        consumer.Start();
        True(takeStarted.Wait(TimeSpan.FromSeconds(2)),
            "completion drain attempts callback lookup during acceptance");
        True(SpinWait.SpinUntil(
                () => (consumer.ThreadState &
                       System.Threading.ThreadState.WaitSleepJoin) != 0,
                TimeSpan.FromSeconds(2)),
            "completion drain blocks on the acceptance registry lock");
        False(takeReturned.IsSet,
            "completion callback lookup cannot return before publication");
        allowPendingPublication.Set();
        producer.Join(TimeSpan.FromSeconds(2));
        consumer.Join(TimeSpan.FromSeconds(2));

        True(enqueueSucceeded, "interleaving test accepts the write");
        True(revisionReleased,
            "accepted pending publication must precede completion callback");
    }

    private static void TerminalBatchProducesFailureCompletionForEverySequence()
    {
        using var sink = new TerminalBatchSink("injected terminal failure");
        using var worker = new HistoricalWriteWorker(sink, batchSize: 1,
            retryDelayMilliseconds: 0, maxRetryAttempts: 1);
        for (long sequence = 1L; sequence <= 3L; sequence++)
        {
            string key = "terminal-test:" + sequence;
            var envelope = new HistoricalWriteEnvelope(sequence, key,
                string.Empty, Array.Empty<HistoricalSqlValue>(),
                HistoricalWriteKind.Append,
                new AWAsyncStamp(1L, sequence, sequence));
            True(worker.TryEnqueue(envelope, out _),
                "terminal test accepts sequence " + sequence);
        }

        worker.Start();
        True(SpinWait.SpinUntil(() => worker.TerminalFaulted,
                TimeSpan.FromSeconds(2)),
            "terminal sink faults the worker");
        True(worker.TryDequeueCompletion(out HistoricalWriteCompletion result),
            "terminal batch publishes a completion");
        False(ReadProperty<bool>(result, "IsCommitted"),
            "terminal batch completion is not committed");
        Equal("injected terminal failure",
            ReadProperty<string>(result, "Error"),
            "terminal batch completion preserves its error");
        Equal(3, result.Sequences.Count,
            "terminal batch completion covers every accepted sequence");
        Equal(1L, result.Sequences[0], "terminal completion sequence 1");
        Equal(2L, result.Sequences[1], "terminal completion sequence 2");
        Equal(3L, result.Sequences[2], "terminal completion sequence 3");
        Equal(0, worker.PendingCount,
            "terminal failure leaves no accepted operation pending");
        False(worker.TryDequeueCompletion(out _),
            "terminal batch publishes exactly one completion");
    }

    private static void OpenFailureProducesFailureCompletionForEverySequence()
    {
        using var sink = new OpenFailureSink("injected open failure");
        using var worker = new HistoricalWriteWorker(sink, batchSize: 1,
            retryDelayMilliseconds: 0, maxRetryAttempts: 1);
        for (long sequence = 31L; sequence <= 33L; sequence++)
        {
            string key = "open-failure-test:" + sequence;
            var envelope = new HistoricalWriteEnvelope(sequence, key,
                string.Empty, Array.Empty<HistoricalSqlValue>(),
                HistoricalWriteKind.Append,
                new AWAsyncStamp(1L, sequence, sequence));
            True(worker.TryEnqueue(envelope, out _),
                "open-failure test accepts sequence " + sequence);
        }

        worker.Start();
        True(SpinWait.SpinUntil(() => worker.TerminalFaulted,
                TimeSpan.FromSeconds(2)),
            "open failure faults the worker");
        True(worker.TryDequeueCompletion(out HistoricalWriteCompletion result),
            "open failure publishes a completion");
        False(ReadProperty<bool>(result, "IsCommitted"),
            "open-failure completion is not committed");
        Equal("injected open failure", ReadProperty<string>(result, "Error"),
            "open-failure completion preserves its error");
        Equal(3, result.Sequences.Count,
            "open-failure completion covers every accepted sequence");
        Equal(31L, result.Sequences[0], "open completion sequence 31");
        Equal(32L, result.Sequences[1], "open completion sequence 32");
        Equal(33L, result.Sequences[2], "open completion sequence 33");
        Equal(0, worker.PendingCount,
            "open failure leaves no accepted operation pending");
        False(worker.TryDequeueCompletion(out _),
            "open failure publishes exactly one completion");
    }

    private static void Contains(string pSource, string pExpected,
        string pName)
    {
        True(pSource.Contains(pExpected, StringComparison.Ordinal), pName);
    }

    private static void Matches(string pSource, string pPattern,
        string pName)
    {
        True(Regex.IsMatch(pSource, pPattern, RegexOptions.Singleline), pName);
    }

    private static T ReadProperty<T>(object pTarget, string pName)
    {
        System.Reflection.PropertyInfo property = pTarget.GetType()
            .GetProperty(pName);
        if (property == null)
            throw new InvalidOperationException(
                "completion property is missing: " + pName);
        return (T)property.GetValue(pTarget);
    }

    private static void True(bool pValue, string pName)
    {
        if (!pValue)
            throw new InvalidOperationException(pName + ": expected true");
    }

    private static void False(bool pValue, string pName)
    {
        if (pValue)
            throw new InvalidOperationException(pName + ": expected false");
    }

    private static void Equal(long pExpected, long pActual, string pName)
    {
        if (pExpected != pActual)
            throw new InvalidOperationException(pName + ": expected " +
                                                pExpected + ", got " +
                                                pActual);
    }

    private static void Equal(string pExpected, string pActual, string pName)
    {
        if (!string.Equals(pExpected, pActual, StringComparison.Ordinal))
            throw new InvalidOperationException(pName + ": expected '" +
                                                pExpected + "', got '" +
                                                pActual + "'");
    }

        }

        public void Dispose()
        {
        }
    }

    private sealed class SuccessfulBatchSink : IHistoricalWriteBatchSink
    {
        public void Open()
        {
        }

        public HistoricalWriteBatchResult Execute(
            IReadOnlyList<HistoricalWriteEnvelope> pOperations)
        {
            return HistoricalWriteBatchResult.Committed(
                pOperations.Select(_ => (object)null).ToArray());
        }

        public void Dispose()
        {
        }
    }

    private sealed class SuccessfulThenTerminalSink :
        IHistoricalWriteBatchSink
    {
        private readonly string _error;
        private int _executeCount;
        public readonly ManualResetEventSlim FirstExecuteEntered =
            new ManualResetEventSlim(false);
        public readonly ManualResetEventSlim AllowFirstExecute =
            new ManualResetEventSlim(false);

        public SuccessfulThenTerminalSink(string pError)
        {
            _error = pError;
        }

        public void Open()
        {
        }

        public HistoricalWriteBatchResult Execute(
            IReadOnlyList<HistoricalWriteEnvelope> pOperations)
        {
            if (Interlocked.Increment(ref _executeCount) == 1)
            {
                FirstExecuteEntered.Set();
                True(AllowFirstExecute.Wait(ConcurrentDrainTimeout),
                    "blocked first batch receives its release signal");
                return HistoricalWriteBatchResult.Committed(
                    pOperations.Select(_ => (object)null).ToArray());
            }
            return HistoricalWriteBatchResult.Terminal(_error);
        }

        public void Dispose()
        {
        }
    }

    private sealed class RetryableBatchSink : IHistoricalWriteBatchSink
    {
        private readonly string _error;

        public RetryableBatchSink(string pError)
        {
            _error = pError;
        }

        public int ExecuteCount { get; private set; }

        public void Open()
        {
        }

        public HistoricalWriteBatchResult Execute(
            IReadOnlyList<HistoricalWriteEnvelope> pOperations)
        {
            ExecuteCount++;
            return HistoricalWriteBatchResult.Retryable(_error);
        }

        public void Dispose()
        {
        }
    }

    private sealed class OpenFailureSink : IHistoricalWriteBatchSink
    {
        private readonly string _error;

        public OpenFailureSink(string pError)
        {
            _error = pError;
        }

        public void Open()
        {
            throw new InvalidOperationException(_error);
        }

        public HistoricalWriteBatchResult Execute(
            IReadOnlyList<HistoricalWriteEnvelope> pOperations)
        {
            throw new InvalidOperationException(
                "open-failure sink must not execute a batch");
        }

        public void Dispose()
        {
        }
    }
