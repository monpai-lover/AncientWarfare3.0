using System.Reflection;
using AncientWarfare3.core.lineage;

internal static class ArmyRtsRouteChoiceRulesTests
{
    public static void Run()
    {
        Assert(ArmyRtsRouteChoiceRules.Resolve(
                landCost: 120f, transportAvailable: true,
                pickupCost: 4f, queueCost: 8f, seaCost: 42f,
                landingCost: 0f) == ArmyRtsTravelChoice.Transport,
            "a substantially shorter sea route must beat a long land route");
        Assert(ArmyRtsRouteChoiceRules.Resolve(
                landCost: 58f, transportAvailable: true,
                pickupCost: 4f, queueCost: 8f, seaCost: 42f,
                landingCost: 0f) == ArmyRtsTravelChoice.Land,
            "transport hysteresis must retain land when sea travel has no clear gain");
        Assert(!ArmyRtsTransportRules.ShouldUseTransportBeforeLandRoute(
                authoritative: true, strategicMovementReady: true,
                actorTileValid: true, targetTileValid: true,
                sameIsland: false, transportRouteConfirmed: false),
            "an unconfirmed island boundary must allow the bounded swim or land-route attempt first");
        Assert(ArmyRtsTransportRules.ShouldInitiateTransportImmediately(
                routeRequiresTransport: true, voyageAlreadyActive: false,
                captainCanBeginTransport: true),
            "a selected cross-island route must open transport immediately instead of waiting for a later behaviour tick");
        Assert(!ArmyRtsTransportRules.ShouldInitiateTransportImmediately(
                routeRequiresTransport: true, voyageAlreadyActive: true,
                captainCanBeginTransport: true),
            "an active voyage must not receive a duplicate immediate transport request");
        Assert(!ArmyRtsTransportRules.ShouldInitiateTransportImmediately(
                routeRequiresTransport: true, voyageAlreadyActive: false,
                captainCanBeginTransport: false),
            "an invalid captain cannot manufacture a transport request");
        Assert(!ArmyRtsTransportRules.CanCreateVoyageState(
                actorInsideBoat: false, physicalRouteAvailable: false),
            "a cross-island objective without a real dock route must not create transport ownership");
        Assert(ArmyRtsTransportRules.CanCreateVoyageState(
                actorInsideBoat: false, physicalRouteAvailable: true),
            "a confirmed dock route may create transport ownership");
        Assert(ArmyRtsTransportRules.CanCreateVoyageState(
                actorInsideBoat: true, physicalRouteAvailable: false),
            "an already embarked actor must retain transport ownership while its route is active");
        Assert(!ArmyRtsTransportRules.ShouldCountAsLanded(
                sameTargetIsland: true, actorOnStableLand: false,
                targetOnStableLand: true),
            "a water tile must never complete a landing even when island identity matches");
        Assert(!ArmyRtsTransportRules.ShouldCountAsLanded(
                sameTargetIsland: true, actorOnStableLand: true,
                targetOnStableLand: false),
            "an invalid target tile must never complete a landing");
        Assert(ArmyRtsTransportRules.ShouldCountAsLanded(
                sameTargetIsland: true, actorOnStableLand: true,
                targetOnStableLand: true),
            "only stable land on the target island completes a landing");
        Assert(!ArmyRtsTransportRules.ShouldReplaceActiveVoyageTarget(
                activeTargetMatches: false, callerMayBegin: true,
                callerIsCaptain: false, forceTransport: false,
                hasEmbarkedMembers: false),
            "a non-captain return transport must not replace an active army voyage target");
        Assert(ArmyRtsTransportRules.ShouldReplaceActiveVoyageTarget(
                activeTargetMatches: false, callerMayBegin: true,
                callerIsCaptain: true, forceTransport: true,
                hasEmbarkedMembers: false),
            "the RTS captain may replace an unembarked voyage for a newly forced route");
        Assert(!ArmyRtsTransportRules.ShouldReplaceActiveVoyageTarget(
                activeTargetMatches: false, callerMayBegin: true,
                callerIsCaptain: true, forceTransport: true,
                hasEmbarkedMembers: true),
            "an embarked army voyage must retain its destination");
        Assert(ArmyRtsTransportRules.ShouldEscalateCrossIslandRouteFailure(
                routeFailed: true, sameIsland: false,
                transportAlreadyActive: false),
            "a rejected cross-island route must immediately fall back to transport instead of changing its war target");
        Assert(!ArmyRtsTransportRules.ShouldEscalateCrossIslandRouteFailure(
                routeFailed: true, sameIsland: true,
                transportAlreadyActive: false),
            "a land-route failure on the same island must retain normal route recovery");
        Assert(!ArmyRtsTransportRules.ShouldEscalateCrossIslandRouteFailure(
                routeFailed: true, sameIsland: false,
                transportAlreadyActive: true),
            "an already active voyage must not be restarted by a route failure");
        Assert(ArmyRtsRules.ShouldRetryMissingStrategicRoute(
                currentState: ArmyRtsState.March,
                nextState: ArmyRtsState.March,
                routeSubmitted: false, routeArrived: false,
                transportActive: false),
            "a March route cleared by recovery must be submitted again");
        Assert(ArmyRtsRules.ResolveStableStrategicEndpoint(
                lockedTileId: 41, lockedEndpointLive: true,
                candidateTileId: 77) == 41,
            "a mission must retain its selected endpoint when a closer zone tile appears");
        Assert(ArmyRtsRules.ResolveStableStrategicEndpoint(
                lockedTileId: 41, lockedEndpointLive: false,
                candidateTileId: 77) == 77,
            "an invalidated endpoint must fall back to the current city candidate");
        TransportRecoveryUsesLockedStrategicEndpoint();
        Assert(ArmyRtsControllerRules.HasReachedStrategicDestination(
                captainTileId: 41, destinationTileId: 41,
                distanceSquared: 0, sameTargetZone: false,
                endpointValidated: false, arrivalRadius: 2),
            "the exact strategic tile always completes the route");
        Assert(ArmyRtsControllerRules.HasReachedStrategicDestination(
                captainTileId: 40, destinationTileId: 41,
                distanceSquared: 1, sameTargetZone: true,
                endpointValidated: true, arrivalRadius: 2),
            "a validated nearby endpoint in the target zone completes");
        Assert(!ArmyRtsControllerRules.HasReachedStrategicDestination(
                40, 41, distanceSquared: 1, sameTargetZone: false,
                endpointValidated: true, arrivalRadius: 2),
            "nearby tiles across a zone boundary do not complete");
        Assert(!ArmyRtsControllerRules.HasReachedStrategicDestination(
                40, 41, distanceSquared: 1, sameTargetZone: true,
                endpointValidated: false, arrivalRadius: 2),
            "an unvalidated nearby tile does not complete");
        Assert(!ArmyRtsControllerRules.HasReachedStrategicDestination(
                30, 41, distanceSquared: 9, sameTargetZone: true,
                endpointValidated: true, arrivalRadius: 2),
            "a validated endpoint outside arrival radius does not complete");
        RetainForwardFrontHoldAfterAllAttackObjectivesClose();
        Assert(!ArmyFormationRules.ShouldRestartObservationForAnchorUpdate(
                allowObservationRestart: true,
                deploymentEligibilityChanged: false,
                deploymentEligible: true,
                formationGeometryChanged: true),
            "a moving deployment anchor must not starve an in-progress member observation");
        Assert(ArmyFormationRules.ShouldRestartObservationForAnchorUpdate(
                allowObservationRestart: true,
                deploymentEligibilityChanged: true,
                deploymentEligible: true,
                formationGeometryChanged: false),
            "entering deployment must still refresh member readiness");
        ArmyFormationObservationProgress progress =
            ArmyFormationRules.DescribeObservationProgress(
                memberCount: 37, cursor: 16, observationComplete: false,
                restartCount: 2);
        Assert(progress.MemberCount == 37 && progress.Cursor == 16 &&
               !progress.Complete && progress.RestartCount == 2,
            "formation diagnostics must preserve the incomplete scan cursor and reset count");
        Assert(ArmyRtsTaskOwnershipRules.ResolveWatchdogTaskId(
                ArmyWatchdogPositionSource.Captain,
                ArmyRtsState.Retreat, ArmyRtsProposalKind.Retreat,
                ArmyRtsTransportPhase.None) ==
               "aw_army_rts_retreat",
            "a retreating captain must be checked against the retreat task instead of the generic mission task");
        Assert(ArmyRtsTaskOwnershipRules.ResolveWatchdogTaskId(
                ArmyWatchdogPositionSource.FormationMember,
                ArmyRtsState.Deploy, ArmyRtsProposalKind.Attack,
                ArmyRtsTransportPhase.None) ==
               "aw_army_rts_formation",
            "a sampled formation member must be checked against the formation task");
        Assert(ArmyStallWatchdogRules.HasPhysicalMovementProgress(
                movementTiles: 0.1d, fixedRetreatTransit: true),
            "slow but real retreat motion must reset the stall watchdog");
        Assert(!ArmyStallWatchdogRules.HasPhysicalMovementProgress(
                movementTiles: 0.1d, fixedRetreatTransit: false),
            "ordinary strategic movement retains the existing progress threshold");
        Assert(ArmyStallWatchdogRules.NormalizeRecoveryForFixedRetreat(
                fixedRetreatTransit: true,
                ArmyStallRecoveryAction.AlternateEndpoint) ==
               ArmyStallRecoveryAction.ReassertCommand,
            "retreat recovery must keep its selected safe city instead of choosing another endpoint");
        Assert(ArmyStallWatchdogRules.NormalizeRecoveryForFixedRetreat(
                fixedRetreatTransit: true,
                ArmyStallRecoveryAction.RebuildRoute) ==
               ArmyStallRecoveryAction.ReassertCommand,
            "retreat route recovery reasserts the same fixed-city command");
        Assert(ArmyStallWatchdogRules.NormalizeRecoveryForFixedRetreat(
                fixedRetreatTransit: true,
                ArmyStallRecoveryAction.EnterTransport) ==
               ArmyStallRecoveryAction.EnterTransport,
            "a cross-island retreat may still hand movement to transport");
    }

    private static void TransportRecoveryUsesLockedStrategicEndpoint()
    {
        string root = Directory.GetCurrentDirectory();
        while (!File.Exists(Path.Combine(root, "AncientWarfare3.csproj")))
        {
            DirectoryInfo parent = Directory.GetParent(root);
            if (parent == null)
                throw new InvalidOperationException("project root unavailable");
            root = parent.FullName;
        }
        string controller = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsControllerService.cs"));
        int start = controller.IndexOf(
            "public static bool RequestTransportRecovery(long pArmyId)",
            StringComparison.Ordinal);
        int end = controller.IndexOf(
            "public static bool TryBeginCrossIslandTransportAfterRouteFailure(",
            start, StringComparison.Ordinal);
        Assert(start >= 0 && end > start,
            "transport recovery source must be discoverable");
        string body = controller.Substring(start, end - start);
        Assert(body.Contains("ResolveStableStrategicEndpoint(",
                StringComparison.Ordinal) &&
               !body.Contains("target?.getTile()", StringComparison.Ordinal),
            "transport recovery must reuse the mission-locked strategic endpoint instead of switching to the city center");
        Assert(body.Contains("TryGetRouteEstimate(",
                StringComparison.Ordinal),
            "transport recovery must verify a physical boat or provisionable dock route before taking movement ownership");
    }

    private static void RetainForwardFrontHoldAfterAllAttackObjectivesClose()
    {
        MethodInfo method = typeof(ArmyRtsObjectiveRules).GetMethod(
            "ShouldRetainForwardFrontHoldTarget",
            new[] { typeof(bool), typeof(bool), typeof(bool) });
        if (method == null)
            throw new InvalidOperationException(
                "RTS must retain a completed forward objective instead of recalling a marching army to its anchor city");

        Assert((bool)method.Invoke(null, new object[]
            { true, true, true }),
            "when no attack objective remains, a live city secured by this war remains the front-hold target");
        Assert(!(bool)method.Invoke(null, new object[]
            { false, true, true }),
            "an open attack objective must continue through ordinary attack allocation");
        Assert(!(bool)method.Invoke(null, new object[]
            { true, true, false }),
            "a city not secured by this war cannot become a forward front-hold target");
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
}
