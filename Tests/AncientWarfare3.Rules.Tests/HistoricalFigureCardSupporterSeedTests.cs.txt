using System;
using System.Linq;
using AncientWarfare3.content.figures;
using AncientWarfare3.content.supporters;

internal static class HistoricalFigureCardSupporterSeedTests
{
    public static void Run()
    {
        var entries = SupporterRosterData.Parse(new[]
        {
            "rank,name,amount,date,description",
            "1,威行,50,2026-08-16,",
            "2,Another,,,提供测试",
            "3,威行,100,2026-08-27,提供贴图"
        });
        var cards = HistoricalFigureCardSupporterSeeds.Build(entries);
        var reordered = HistoricalFigureCardSupporterSeeds.Build(
            entries.AsEnumerable().Reverse());

        Equal(2, cards.Count, "duplicate supporter names merge into one card");
        var wei = cards.Single(p => p.DisplayName == "威行");
        Equal(wei.CardId,
            reordered.Single(p => p.DisplayName == "威行").CardId,
            "card id is independent of rank and source order");
        True(wei.CardId.StartsWith("supporter_", StringComparison.Ordinal),
            "card id uses the supporter namespace");
        Equal(26, wei.CardId.Length, "card id uses a stable truncated hash");
        Equal(HistoricalFigureCardRarity.Pink, wei.Rarity,
            "supporter cards are pink");
        Equal(HistoricalFigureCardRole.Minister, wei.Role,
            "supporter cards deploy as ministers");
        Equal(HistoricalFigureCardMinisterType.CivilOfficial, wei.MinisterType,
            "supporter cards use the civil-official path");
        True(!string.IsNullOrWhiteSpace(wei.FamilyName) &&
             !string.IsNullOrWhiteSpace(wei.ClanName),
            "supporter ministers have complete lineage identity fields");
        Equal("supporters", wei.CollectionId,
            "supporter cards belong only to the supporter crate");
        Equal(HistoricalFigureCardCatalog.UnknownYear, wei.BirthYear,
            "supporter cards use the catalogue unknown-year sentinel");
        True(wei.DetailedBiography.Contains("50") &&
             wei.DetailedBiography.Contains("100") &&
             wei.DetailedBiography.Contains("提供贴图"),
            "all duplicate records are aggregated into the biography");

        var currentRoster = SupporterRosterData.Read();
        var currentCards = HistoricalFigureCardSupporterSeeds.Build(currentRoster);
        Equal(currentRoster.Select(p => p.Name)
                .Distinct(StringComparer.OrdinalIgnoreCase).Count(),
            currentCards.Count,
            "current supporters.csv contributes one card per distinct name");
        True(currentCards.All(p => p.Rarity.Equals(
                HistoricalFigureCardRarity.Pink)),
            "the current supporter pool contains only pink local cards");
    }

    private static void True(bool value, string message)
    {
        if (!value) throw new InvalidOperationException(message);
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(message + ": expected=" +
                                                expected + ", actual=" + actual);
    }
}
