using AncientWarfare3.core.schools;

internal static class HistoricalSchoolDeathRuntimeRulesTests
{
    public static void Run()
    {
        Equal(HistoricalSchoolMembershipLoadAction.KeepActive,
            HistoricalSchoolDeathRuntimeRules.ResolveLoadAction(
                activeMembership: true, actorLookupReady: true,
                actorExists: true, actorAlive: true, actorWrecked: false),
            "a living actor keeps its active school membership");
        Equal(HistoricalSchoolMembershipLoadAction.QueueDeath,
            HistoricalSchoolDeathRuntimeRules.ResolveLoadAction(
                activeMembership: true, actorLookupReady: true,
                actorExists: true, actorAlive: false, actorWrecked: false),
            "a loaded corpse must resume the school death transaction");
        Equal(HistoricalSchoolMembershipLoadAction.QueueDeath,
            HistoricalSchoolDeathRuntimeRules.ResolveLoadAction(
                activeMembership: true, actorLookupReady: true,
                actorExists: true, actorAlive: true, actorWrecked: true),
            "a wrecked actor must resume the school death transaction");
        Equal(HistoricalSchoolMembershipLoadAction.KeepActive,
            HistoricalSchoolDeathRuntimeRules.ResolveLoadAction(
                activeMembership: true, actorLookupReady: true,
                actorExists: false, actorAlive: false, actorWrecked: true),
            "an actor missing during staged loading must not be persisted as dead");
        Equal(HistoricalSchoolMembershipLoadAction.KeepActive,
            HistoricalSchoolDeathRuntimeRules.ResolveLoadAction(
                activeMembership: true, actorLookupReady: false,
                actorExists: false, actorAlive: false, actorWrecked: true),
            "an unavailable actor manager must not destroy persisted membership state");
        Equal(HistoricalSchoolMembershipLoadAction.Ignore,
            HistoricalSchoolDeathRuntimeRules.ResolveLoadAction(
                activeMembership: false, actorLookupReady: true,
                actorExists: true, actorAlive: true, actorWrecked: false),
            "an inactive membership is ignored");

        True(HistoricalSchoolDeathRuntimeRules.ShouldExposeMembership(
                activeMembership: true, actorExists: true,
                actorAlive: true, actorWrecked: false, deathPending: false),
            "only a living non-pending member is exposed to school candidates");
        False(HistoricalSchoolDeathRuntimeRules.ShouldExposeMembership(
                activeMembership: true, actorExists: true,
                actorAlive: false, actorWrecked: false, deathPending: true),
            "a dead member is quarantined before persistence completes");
        False(HistoricalSchoolDeathRuntimeRules.ShouldExposeMembership(
                activeMembership: true, actorExists: false,
                actorAlive: false, actorWrecked: true, deathPending: false),
            "a missing actor is never exposed as a school candidate");
    }

    private static void Equal<T>(T pExpected, T pActual, string pMessage)
    {
        if (!EqualityComparer<T>.Default.Equals(pExpected, pActual))
            throw new InvalidOperationException(pMessage +
                $" (expected={pExpected}, actual={pActual})");
    }

    private static void True(bool pValue, string pMessage)
    {
        if (!pValue) throw new InvalidOperationException(pMessage);
    }

    private static void False(bool pValue, string pMessage)
    {
        if (pValue) throw new InvalidOperationException(pMessage);
    }
}
