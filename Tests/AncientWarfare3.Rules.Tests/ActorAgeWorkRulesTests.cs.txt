using AncientWarfare3.core.lineage;

internal static class ActorAgeWorkRulesTests
{
    public static void Run()
    {
        Equal(false, ActorAgeWorkRules.ShouldTrack(
                wasTracked: false, dirty: false, dynasticEligible: false,
                warrior: false),
            "ordinary civilians do not enter AW3 annual Actor tracking");
        Equal(true, ActorAgeWorkRules.ShouldTrack(
                wasTracked: false, dirty: false, dynasticEligible: true,
                warrior: false),
            "dynastic civilians remain eligible for annual work");
        Equal(true, ActorAgeWorkRules.ShouldTrack(
                wasTracked: true, dirty: false, dynasticEligible: false,
                warrior: false),
            "previously tracked Actors receive one cleanup transition");

        var same = new ActorAgeWorkState(isAdult: true, profession: 2,
            inPermanentArmy: true, atWar: true, dynasticEligible: false,
            shouldUsePeacetimeJob: false,
            shouldReleaseMilitaryRole: false,
            yearBucket: 12);
        Equal(ActorAgeWorkStage.None,
            ActorAgeWorkRules.Resolve(same, same, force: false),
            "unchanged Actor state performs no AW3 age work");

        var child = new ActorAgeWorkState(isAdult: false, profession: 2,
            inPermanentArmy: false, atWar: false, dynasticEligible: true,
            shouldUsePeacetimeJob: false,
            shouldReleaseMilitaryRole: false,
            yearBucket: 12);
        var adult = new ActorAgeWorkState(isAdult: true, profession: 2,
            inPermanentArmy: false, atWar: false, dynasticEligible: true,
            shouldUsePeacetimeJob: false,
            shouldReleaseMilitaryRole: false,
            yearBucket: 12);
        Has(ActorAgeWorkRules.Resolve(child, adult, force: false),
            ActorAgeWorkStage.DynasticTitle,
            "adulthood triggers dynastic-title work");

        var jobChanged = new ActorAgeWorkState(isAdult: true,
            profession: 2, inPermanentArmy: true, atWar: false,
            dynasticEligible: false, shouldUsePeacetimeJob: true,
            shouldReleaseMilitaryRole: false,
            yearBucket: 12);
        Has(ActorAgeWorkRules.Resolve(same, jobChanged, force: false),
            ActorAgeWorkStage.StandingArmyJob,
            "peacetime job decision changes trigger job refresh");

        var releaseChanged = new ActorAgeWorkState(isAdult: true,
            profession: 2, inPermanentArmy: false, atWar: false,
            dynasticEligible: true, shouldUsePeacetimeJob: false,
            shouldReleaseMilitaryRole: true,
            yearBucket: 12);
        Has(ActorAgeWorkRules.Resolve(adult, releaseChanged, force: false),
            ActorAgeWorkStage.MilitaryRoleRelease,
            "reproduction protection changes trigger military release");

        var nextYear = new ActorAgeWorkState(isAdult: true, profession: 2,
            inPermanentArmy: true, atWar: true, dynasticEligible: false,
            shouldUsePeacetimeJob: false,
            shouldReleaseMilitaryRole: false,
            yearBucket: 13);
        ActorAgeWorkStage annual = ActorAgeWorkRules.Resolve(same, nextYear,
            force: false);
        Equal(ActorAgeWorkStage.None, annual,
            "ordinary year rollover performs no AW3 age work");

        var releasePreviousYear = new ActorAgeWorkState(isAdult: true,
            profession: 2, inPermanentArmy: false, atWar: false,
            dynasticEligible: false, shouldUsePeacetimeJob: false,
            shouldReleaseMilitaryRole: true,
            yearBucket: 12);
        var releaseNextYear = new ActorAgeWorkState(isAdult: true,
            profession: 2, inPermanentArmy: false, atWar: false,
            dynasticEligible: false, shouldUsePeacetimeJob: false,
            shouldReleaseMilitaryRole: true,
            yearBucket: 13);
        Equal(ActorAgeWorkStage.MilitaryRoleRelease,
            ActorAgeWorkRules.Resolve(releasePreviousYear, releaseNextYear,
                force: false),
            "annual fallback retries only a required military release");

        var dynasticNextYear = new ActorAgeWorkState(isAdult: true,
            profession: 2, inPermanentArmy: false, atWar: false,
            dynasticEligible: true, shouldUsePeacetimeJob: false,
            shouldReleaseMilitaryRole: false,
            yearBucket: 13);
        Equal(ActorAgeWorkStage.DynasticTitle,
            ActorAgeWorkRules.Resolve(adult, dynasticNextYear,
                force: false),
            "annual fallback retries only a dynastic title candidate");

        var ordinary = new ActorAgeWorkState(isAdult: true, profession: 1,
            inPermanentArmy: false, atWar: false,
            dynasticEligible: false, shouldUsePeacetimeJob: false,
            shouldReleaseMilitaryRole: false,
            yearBucket: 12);
        Equal(ActorAgeWorkStage.None,
            ActorAgeWorkRules.Resolve(ordinary, ordinary, force: true),
            "forced refresh skips unrelated ordinary Actors");

        var relevant = new ActorAgeWorkState(isAdult: true, profession: 2,
            inPermanentArmy: true, atWar: false, dynasticEligible: true,
            shouldUsePeacetimeJob: true,
            shouldReleaseMilitaryRole: true,
            yearBucket: 12);
        Equal(ActorAgeWorkStage.All,
            ActorAgeWorkRules.Resolve(relevant, relevant, force: true),
            "forced refresh selects every currently relevant age stage");
    }

    private static void Has(ActorAgeWorkStage pActual,
        ActorAgeWorkStage pExpected, string pMessage)
    {
        if ((pActual & pExpected) == 0)
            throw new InvalidOperationException(pMessage +
                ": actual=" + pActual);
    }

    private static void Equal<T>(T pExpected, T pActual, string pMessage)
    {
        if (!EqualityComparer<T>.Default.Equals(pExpected, pActual))
            throw new InvalidOperationException(pMessage +
                ": expected=" + pExpected + ", actual=" + pActual);
    }
}
