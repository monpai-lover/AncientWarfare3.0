using System;
using System.Collections.Generic;
using System.Linq;
using AncientWarfare3.core.court;

internal static class RegionalGovernmentRulesTests
{
    public static void Run()
    {
        var facts = new[]
        {
            Fact(10, 1, "临淄州", 0.80f, 100, 11, 12),
            Fact(11, 1, "即墨州", 0.50f, 80, 10),
            Fact(12, 1, "琅琊州", 0.40f, 70, 10),
            Fact(20, 1, "安邑城", 0.90f, 100, 21),
            Fact(21, 1, "平阳府", 0.30f, 60, 20)
        };
        IReadOnlyList<RegionalGovernmentFact> regions =
            RegionalGovernmentRules.Build(facts, "郡");
        Equal(2, regions.Count, "disconnected groups produce two regions");
        Equal(10L, regions.Single(region => region.SeatCityId == 10).SeatCityId,
            "seat city is stable");
        Equal(3, regions.Single(region => region.SeatCityId == 10)
            .MemberCityIds.Count, "seat absorbs adjacent cities");
        Equal("临淄", RegionalGovernmentRules.RegionName("临淄州", "郡"),
            "region name is the seat stem without an appended level");
        Equal("安邑", RegionalGovernmentRules.RegionName("安邑城", "道"),
            "custom regional levels do not mutate the place name");
        Equal("即墨", RegionalGovernmentRules.CityName("即墨"),
            "the WorldBox city name is already the complete prefecture name");
        Equal("临淄 · 郡", RegionalGovernmentRules.AdministrativeLabel(
                "临淄", "郡"),
            "court labels visually separate place and administrative level");
        Equal("即墨 · 州", RegionalGovernmentRules.AdministrativeLabel(
                RegionalGovernmentRules.CityName("即墨"), "州"),
            "city labels visually separate the unchanged city name and level");

        IReadOnlyList<RegionalGovernmentFact> tie =
            RegionalGovernmentRules.Build(new[]
            {
                Fact(4, 2, "甲", .5f, 10), Fact(3, 2, "乙", .5f, 10)
            }, "郡");
        Equal(3L, tie[0].SeatCityId, "city ID breaks a development tie");
    }

    private static RegionalGovernmentCityFact Fact(long pCityId,
        long pKingdomId, string pName, float pDevelopment, int pPopulation,
        params long[] pNeighbors)
    {
        return new RegionalGovernmentCityFact
        {
            CityId = pCityId, KingdomId = pKingdomId, CityName = pName,
            Development = pDevelopment, Population = pPopulation,
            NeighborCityIds = pNeighbors ?? Array.Empty<long>()
        };
    }

    private static void Equal<T>(T pExpected, T pActual, string pMessage)
    {
        if (!Equals(pExpected, pActual))
            throw new InvalidOperationException(pMessage + ": expected " +
                pExpected + ", got " + pActual);
    }
}
