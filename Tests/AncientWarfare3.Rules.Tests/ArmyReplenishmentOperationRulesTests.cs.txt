using AncientWarfare3.core.lineage;
using System;
using System.IO;

internal static class ArmyReplenishmentOperationRulesTests
{
    internal static void Run()
    {
        Equal(20d, ArmyReplenishmentOperationRules.DurationWorldSeconds,
            "a replenishment operation has one bounded twenty-second window");
        Equal(9, ArmyReplenishmentOperationRules.ResolveApprovedShortage(
            existingApproved: 9, requestedShortage: 20),
            "a repeated request cannot enlarge approval");
        Equal(20, ArmyReplenishmentOperationRules.ResolveApprovedShortage(
            existingApproved: 0, requestedShortage: 20),
            "the first request records its full shortage");
        Equal(2, ArmyReplenishmentOperationRules.AllowedCumulative(
            approved: 9, start: 100d, now: 105d),
            "the legacy progress projection follows the bounded window");
        Equal(4, ArmyReplenishmentOperationRules.AllowedCumulative(
            approved: 9, start: 100d, now: 110d),
            "the legacy progress projection remains bounded");
        Equal(9, ArmyReplenishmentOperationRules.AllowedCumulative(
            approved: 9, start: 100d, now: 120d),
            "the bounded window eventually exposes the full approval");
        Equal(5, ArmyReplenishmentOperationRules.BatchRequest(
            approved: 9, enlisted: 4, liveShortage: 7,
            start: 100d, now: 110d),
            "the first authority cycle consumes every approved missing slot");
        Equal(true, ArmyReplenishmentOperationRules.ShouldFinishEarly(
            liveShortage: 0), "a filled shortage finishes immediately");
        Equal(false,
            ArmyRtsRules.ShouldContinueRequestedReplenishment(
                replenishmentRequested: true, currentStrength: 8,
                targetStrength: 10, reserveAvailable: false),
            "an exhausted source reserve cannot keep an army in replenishment");
        Equal(true, ArmyReplenishmentOperationRules.
            HasConsumableSourceReserve(ArmyMobilizationPhase.Notice,
                availableCount: 20),
            "notice preparation and formal war share the reserve ledger");
        Equal(false, ArmyReplenishmentOperationRules.
            HasConsumableSourceReserve(ArmyMobilizationPhase.War,
                availableCount: 0),
            "an empty wartime reserve pool cannot keep an army replenishing");
        Equal(true, ArmyReplenishmentOperationRules.
            HasConsumableSourceReserve(ArmyMobilizationPhase.War,
                availableCount: 1),
            "a wartime city with a reserve actor keeps replenishment available");
        Equal(false, ArmyReplenishmentOperationRules.
            HasConsumableSourceReserve(ArmyMobilizationPhase.Peace,
                availableCount: 20),
            "peace cannot consume indexed reserves");
        Equal(false, ArmyReplenishmentOperationRules.ShouldReleaseDeparture(
            living: 79, target: 100),
            "an army below quorum remains in preparation");
        Equal(true, ArmyReplenishmentOperationRules.ShouldReleaseDeparture(
            living: 80, target: 100),
            "an army at quorum may deploy while later replacements continue");
        Equal(true,
            ArmyRtsRules.ShouldContinueRequestedReplenishment(
                replenishmentRequested: true, currentStrength: 8,
                targetStrength: 10, reserveAvailable: true),
            "a live source reserve keeps an outstanding replenishment request active");
        Equal(9, ArmyReplenishmentOperationRules.ClampEnlisted(
            approved: 9, persistedEnlisted: 40),
            "restore cannot overstate progress");
        Equal(120d, ArmyReplenishmentOperationRules.ResolveDeadline(
            start: 100d, persistedDeadline: 120d),
            "restore cannot extend the twenty-second deadline");
        Equal(true, ArmyReplenishmentOperationRules.ShouldResumeAttack(
            living: 2, minimum: 2),
            "a viable army leaves at the deadline");
        Equal(true, ArmyReplenishmentOperationRules.ShouldMergeSecondary(
            living: 1, minimum: 2, ordinary: true, primaryExists: true),
            "a weak ordinary secondary merges");
        Equal(false, ArmyReplenishmentOperationRules.ShouldMergeSecondary(
            living: 1, minimum: 2, ordinary: false, primaryExists: true),
            "special forces never merge");
        Equal(true, ArmyReplenishmentOperationRules.MustMaintainAttack(
            totalOrdinary: 2, minimum: 2, validEnemyTarget: true),
            "viable national strength preserves one attack");
        Equal(false, ArmyReplenishmentOperationRules.MustMaintainAttack(
            totalOrdinary: 1, minimum: 2, validEnemyTarget: true),
            "one actor is not sent alone");
        Equal(true, ArmyReplenishmentOperationRules.CanUseReservePool(
                ordinaryArmy: true, royalGuardArmy: false),
            "an ordinary city army can use its city reserve pool");
        Equal(false, ArmyReplenishmentOperationRules.CanUseReservePool(
                ordinaryArmy: true, royalGuardArmy: true),
            "a royal guard never uses the ordinary reserve pool even with stale ordinary identity");
        Equal(false, ArmyReplenishmentOperationRules.CanUseReservePool(
                ordinaryArmy: false, royalGuardArmy: false),
            "other special armies cannot enter ordinary replenishment");
        Equal(true,
            ArmyReplenishmentOperationRules.ShouldClearIneligibleOperation(
                hasPersistedOperation: true, canUseReservePool: false),
            "a stale guard replenishment operation is repaired lazily");
        Equal(false,
            ArmyReplenishmentOperationRules.ShouldClearIneligibleOperation(
                hasPersistedOperation: false, canUseReservePool: false),
            "an ineligible army without stale state requires no persistence write");
        Equal(true, ArmyReplenishmentOperationRules.ShouldStartOperation(
                liveShortage: true, reserveAvailable: true),
            "a replenishment operation requires a live shortage and reserve");
        Equal(false, ArmyReplenishmentOperationRules.ShouldStartOperation(
            liveShortage: true, reserveAvailable: false),
            "zero reserve cannot start a replenishment loop");
        Equal(false,
            ArmyReplenishmentOperationRules.ShouldRestartAfterConfirmedExhaustion(
                reserveAvailable: false, exhaustionConfirmed: true),
            "confirmed reserve exhaustion blocks an empty restart");
        Equal(true,
            ArmyReplenishmentOperationRules.ShouldRestartAfterConfirmedExhaustion(
                reserveAvailable: true, exhaustionConfirmed: true),
            "observed reserve availability permits a later retry");

        string root = Directory.GetCurrentDirectory();
        string controller = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsControllerService.cs"));
        int start = controller.IndexOf(
            "internal static bool CanGenerateWartimeReplacements(",
            StringComparison.Ordinal);
        int end = controller.IndexOf(
            "private static void ClearIneligibleReplenishmentState(", start,
            StringComparison.Ordinal);
        string body = start >= 0 && end > start
            ? controller.Substring(start, end - start)
            : string.Empty;
        Equal(false, body.Contains("HasActiveVoyage(",
                StringComparison.Ordinal) ||
            body.Contains("HasMilitaryTransportOwnership(",
                StringComparison.Ordinal),
            "direct wartime spawning cannot wait for an active voyage to end");
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(name + ": expected " +
                expected + ", got " + actual);
    }
}
