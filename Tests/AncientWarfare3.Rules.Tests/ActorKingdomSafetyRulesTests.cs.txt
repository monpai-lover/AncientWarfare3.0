using AncientWarfare3.core.lineage;

internal static class ActorKingdomSafetyRulesTests
{
    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(
                $"{name}: expected {expected}, got {actual}");
    }

    public static void Run()
    {
        Equal(false, ActorKingdomSafetyRules.ShouldRepairActor(
                actorExists: true, actorAlive: false, actorWrecked: false),
            "a loaded corpse must never be attached to a kingdom again");
        Equal(false, ActorKingdomSafetyRules.ShouldRepairActor(
                actorExists: true, actorAlive: true, actorWrecked: true),
            "a wrecked actor must never be attached to a kingdom again");
        Equal(true, ActorKingdomSafetyRules.ShouldRepairActor(
                actorExists: true, actorAlive: true, actorWrecked: false),
            "a living actor remains eligible for kingdom pointer repair");

        Equal(false, ActorKingdomSafetyRules.IsCityKingdomRepairable(
                cityKingdomAssetExists: true,
                cityKingdomIsRekt: true),
            "a destroyed city kingdom must not trap a kingdomless actor");
        Equal(true, ActorKingdomSafetyRules.IsCityKingdomRepairable(
                cityKingdomAssetExists: true,
                cityKingdomIsRekt: false),
            "a living city kingdom remains the authoritative repair source");
        Equal(true, ActorKingdomSafetyRules.ShouldQueueDeferredRepair(
                immediateRepairSucceeded: false),
            "a load-time repair failure must retry after world initialization");
        Equal(false, ActorKingdomSafetyRules.ShouldQueueDeferredRepair(
                immediateRepairSucceeded: true),
            "a repaired actor does not require deferred repair");

        Equal(false, ActorKingdomSafetyRules.CanRunEnemyCheck(
                actorExists: true,
                actorAssetExists: true,
                kingdomDataExists: false,
                kingdomAssetExists: false),
            "an actor with a null kingdom cannot enter vanilla enemy checks");
        Equal(false, ActorKingdomSafetyRules.CanRunEnemyCheck(
                actorExists: true,
                actorAssetExists: true,
                kingdomDataExists: false,
                kingdomAssetExists: true),
            "an actor with disposed kingdom data cannot enter vanilla enemy checks");
        Equal(true, ActorKingdomSafetyRules.CanRunEnemyCheck(
                actorExists: true,
                actorAssetExists: true,
                kingdomDataExists: true,
                kingdomAssetExists: true),
            "a fully restored actor keeps vanilla enemy checks");

        Equal(true,
            ActorKingdomSafetyRules.ShouldDetachInvalidKingdomBeforeRepair(
                kingdomObjectExists: true,
                kingdomDataExists: false,
                kingdomAssetExists: false),
            "a stale destroyed kingdom must be detached before vanilla joinKingdom");
        Equal(true,
            ActorKingdomSafetyRules.ShouldDetachInvalidKingdomBeforeRepair(
                kingdomObjectExists: true,
                kingdomDataExists: false,
                kingdomAssetExists: true),
            "a kingdom with disposed data must be detached before repair");
        Equal(false,
            ActorKingdomSafetyRules.ShouldDetachInvalidKingdomBeforeRepair(
                kingdomObjectExists: false,
                kingdomDataExists: false,
                kingdomAssetExists: false),
            "a genuinely kingdomless actor needs no stale-reference detach");
        Equal(false,
            ActorKingdomSafetyRules.ShouldDetachInvalidKingdomBeforeRepair(
                kingdomObjectExists: true,
                kingdomDataExists: true,
                kingdomAssetExists: true),
            "a valid kingdom reference must never be detached by repair");

        Equal(ActorKingdomRepairSource.City,
            ActorKingdomSafetyRules.SelectRepairSource(
                actorExists: true,
                actorAssetExists: true,
                kingdomDataExists: false,
                kingdomAssetExists: false,
                cityKingdomAssetExists: true,
                cityKingdomIsRekt: false,
                wildKingdomIdExists: true),
            "the loaded city kingdom is the authoritative repair source");
        Equal(ActorKingdomRepairSource.Wild,
            ActorKingdomSafetyRules.SelectRepairSource(
                actorExists: true,
                actorAssetExists: true,
                kingdomDataExists: false,
                kingdomAssetExists: false,
                cityKingdomAssetExists: false,
                cityKingdomIsRekt: false,
                wildKingdomIdExists: true),
            "ownerless actors fall back to their species wild kingdom");
        Equal(ActorKingdomRepairSource.None,
            ActorKingdomSafetyRules.SelectRepairSource(
                actorExists: true,
                actorAssetExists: true,
                kingdomDataExists: true,
                kingdomAssetExists: true,
                cityKingdomAssetExists: true,
                cityKingdomIsRekt: false,
                wildKingdomIdExists: true),
            "valid kingdom references are never overwritten");
        Equal(ActorKingdomRepairSource.None,
            ActorKingdomSafetyRules.SelectRepairSource(
                actorExists: true,
                actorAssetExists: true,
                kingdomDataExists: false,
                kingdomAssetExists: false,
                cityKingdomAssetExists: false,
                cityKingdomIsRekt: false,
                wildKingdomIdExists: false),
            "actors without a valid repair source remain quarantined");
        Equal(ActorKingdomRepairSource.City,
            ActorKingdomSafetyRules.SelectRepairSource(
                actorExists: true,
                actorAssetExists: true,
                kingdomDataExists: false,
                kingdomAssetExists: true,
                cityKingdomAssetExists: true,
                cityKingdomIsRekt: false,
                wildKingdomIdExists: true),
            "disposed kingdom data is repaired from the actor city");

        Equal(true, ActorKingdomSafetyRules.CanEnterVanillaZoneProcessing(
                actorExists: true,
                actorAssetExists: true,
                tileExists: true,
                professionAssetExists: true,
                kingdomDataExists: true,
                kingdomAssetExists: true),
            "only fully initialized actors may enter vanilla zone processing");
        Equal(false, ActorKingdomSafetyRules.CanEnterVanillaZoneProcessing(
                actorExists: true,
                actorAssetExists: true,
                tileExists: true,
                professionAssetExists: true,
                kingdomDataExists: false,
                kingdomAssetExists: false),
            "a kingdomless actor must be quarantined before conquest scanning");
        Equal(false, ActorKingdomSafetyRules.CanEnterVanillaZoneProcessing(
                actorExists: true,
                actorAssetExists: true,
                tileExists: true,
                professionAssetExists: true,
                kingdomDataExists: false,
                kingdomAssetExists: true),
            "disposed kingdom data must be quarantined before zone insertion");
        Equal(false, ActorKingdomSafetyRules.CanRenderUnit(
                actorExists: true,
                actorAssetExists: true,
                tileExists: false),
            "a unit without a current tile must be skipped by minimap rendering");

        Equal(true, ActorKingdomSafetyRules.ShouldUseFallbackKingdomColor(
                kingdomObjectExists: true,
                kingdomDataExists: true,
                kingdomAssetExists: false),
            "a destroyed kingdom must use a stable fallback color instead of " +
            "throwing from the parallel renderer");
        Equal(false, ActorKingdomSafetyRules.ShouldUseFallbackKingdomColor(
                kingdomObjectExists: true,
                kingdomDataExists: true,
                kingdomAssetExists: true),
            "a live kingdom must retain its normal color path");
        Equal(false, ActorKingdomSafetyRules.ShouldUseFallbackKingdomColor(
                kingdomObjectExists: false,
                kingdomDataExists: false,
                kingdomAssetExists: false),
            "a null receiver is not a recoverable kingdom color request");
        Equal(true, ActorKingdomSafetyRules.ShouldUseFallbackKingdomColor(
                kingdomObjectExists: true,
                kingdomDataExists: false,
                kingdomAssetExists: true),
            "a kingdom with missing data must not enter vanilla color lookup " +
            "from a parallel renderer");
        Equal(true,
            ActorKingdomSafetyRules.ShouldSuppressKingdomDependentPresentation(
                actorExists: true,
                actorAssetExists: true,
                kingdomObjectExists: false),
            "a kingdomless actor must skip presentation paths that would " +
            "dereference its kingdom in a render worker");
        Equal(false,
            ActorKingdomSafetyRules.ShouldSuppressKingdomDependentPresentation(
                actorExists: true,
                actorAssetExists: true,
                kingdomObjectExists: true),
            "a fully initialized actor keeps normal presentation");
    }
}
