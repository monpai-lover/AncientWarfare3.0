using AncientWarfare3.core.lineage;
using System;
using System.IO;

internal static class CityReservePoolPersistenceRulesTests
{
    internal static void Run()
    {
        Equal(2, CityReservePoolPersistenceRules.CurrentVersion,
            "count ledgers use snapshot version two");
        Equal(false, CityReservePoolPersistenceRules.CanUseSnapshotVersion(1),
            "actor-id snapshots rebuild instead of restoring stale members");
        Equal(true, CityReservePoolPersistenceRules.CanUseSnapshotVersion(2),
            "count-ledger snapshots restore directly");
        Equal("aw3_city_reserve_pools.json",
            CityReservePoolPersistenceRules.SnapshotFileName,
            "reserve pool persistence uses a stable save sidecar name");
        Equal(true, CityReservePoolPersistenceRules.ShouldWriteSnapshot(
            worldReady: true, directoryValid: true),
            "a ready world writes the reserve pool snapshot");
        Equal(false, CityReservePoolPersistenceRules.ShouldWriteSnapshot(
            worldReady: false, directoryValid: true),
            "an unavailable world cannot write a reserve pool snapshot");
        Equal(false, CityReservePoolPersistenceRules.ShouldWriteSnapshot(
            worldReady: true, directoryValid: false),
            "an invalid save directory cannot write a reserve pool snapshot");
        Equal(true, CityReservePoolPersistenceRules.ShouldRestoreSnapshot(
            snapshotExists: true, worldLoaded: true),
            "a loaded world applies an existing reserve pool snapshot");
        Equal(false, CityReservePoolPersistenceRules.ShouldRestoreSnapshot(
            snapshotExists: false, worldLoaded: true),
            "a missing reserve pool snapshot falls back to actor data rebuild");
        Equal(true, CityReservePoolPersistenceRules.ShouldRunFinalRebuild(
            worldLoaded: true, actorCallbacksComplete: true),
            "actor callbacks require a final reserve pool rebuild");
        Equal(false, CityReservePoolPersistenceRules.ShouldRunFinalRebuild(
            worldLoaded: true, actorCallbacksComplete: false),
            "reserve pool rebuild waits until actor callbacks complete");
        MissingSnapshotFinalizesActorPoolOnTheRestoreFrame();

        string savePatch = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "patch",
            "AW_SavePatch.cs"));
        Equal(false, savePatch.Contains(
                "CityReservePoolService.TryWriteSnapshot",
                StringComparison.Ordinal),
            "saves no longer publish the retired resident pool");
        Equal(true, savePatch.Contains(
                "SyntheticMobilizationLedgerService.TryWriteSnapshot",
                StringComparison.Ordinal),
            "saves publish the unified synthetic mobilization ledger");
        string restorePipeline = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "multiplayer",
            "AW3RuntimeRestorePipeline.cs"));
        Equal(false, restorePipeline.Contains(
                "CityReservePoolService.TryRestoreSnapshot",
                StringComparison.Ordinal),
            "world loads do not activate the retired resident pool");
        Equal(true, restorePipeline.Contains(
                "SyntheticMobilizationLedgerService.",
                StringComparison.Ordinal) &&
                    restorePipeline.Contains("TryRestoreSnapshot(",
                        StringComparison.Ordinal),
            "world loads restore the unified ledger and migrate legacy data");
        Equal(true, restorePipeline.IndexOf(
                        "new AW3RestoreStage(\"city_reserve_pools\"",
                        StringComparison.Ordinal) <
                    restorePipeline.IndexOf(
                        "new AW3RestoreStage(\"synthetic_mobilization_ledger\"",
                        StringComparison.Ordinal),
            "load clears legacy runtime state before restoring the new ledger");
    }

    private static void MissingSnapshotFinalizesActorPoolOnTheRestoreFrame()
    {
        int Actions(bool snapshotRestored, bool callbacksComplete,
            bool formalWarActive) => Convert.ToInt32(
                CityReservePoolPersistenceRules.ResolveFinalRestoreActions(
                    snapshotRestored, callbacksComplete, formalWarActive));

        Equal(7, Actions(snapshotRestored: false, callbacksComplete: true,
                formalWarActive: true),
            "a missing wartime sidecar validates actors, populates the pool, " +
            "and restores the emergency immediately");
        Equal(3, Actions(snapshotRestored: false, callbacksComplete: true,
                formalWarActive: false),
            "a missing peacetime sidecar validates and populates immediately");
        Equal(0, Actions(snapshotRestored: true, callbacksComplete: true,
                formalWarActive: true),
            "a restored count ledger is not overwritten by fallback rebuild");
        Equal(0, Actions(snapshotRestored: false, callbacksComplete: false,
                formalWarActive: true),
            "fallback rebuilding waits only while actor callbacks are incomplete");
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(name + ": expected " +
                expected + ", got " + actual);
    }
}
