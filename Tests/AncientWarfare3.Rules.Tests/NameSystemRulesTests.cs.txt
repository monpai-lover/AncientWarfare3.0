internal static class NameSystemRulesTests
{
    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException($"{name}: expected {expected}, got {actual}");
    }

    private static void SequenceEqual<T>(IEnumerable<T> expected, IEnumerable<T> actual, string name)
    {
        string expectedText = string.Join(",", expected);
        string actualText = string.Join(",", actual);
        if (expectedText != actualText)
            throw new InvalidOperationException($"{name}: expected {expectedText}, got {actualText}");
    }

            new HashSet<string>(statePool), 9, 41, 12);

            currentDisplay: "燕姬氏", parentDisplay: "新秦姬氏", rootDisplay: "新秦姬氏");
        Equal("燕姬氏", cadetProjection.BranchDisplay,
            "a real cadet Shi retains its branch display");
        Equal("新秦姬氏", cadetProjection.ParentDisplay,
            "a real cadet Shi retains its direct parent display");
        Equal("新秦姬氏", cadetProjection.RootDisplay,
            "a real cadet Shi retains its ancestral house display");

        var parents = new Dictionary<long, long>
        {
            [30] = 20,
            [20] = 10,
            [10] = -1
        };
        SequenceEqual(new long[] { 20, 10 }, ShiBranchRules.TraceParents(
                30, id => parents.TryGetValue(id, out long parent) ? parent : -1, 64),
            "parent tracing reaches the root in order");

        parents[10] = 30;
        SequenceEqual(new long[] { 20, 10 }, ShiBranchRules.TraceParents(
                30, id => parents.TryGetValue(id, out long parent) ? parent : -1, 64),
            "parent tracing stops before a corrupt cycle");

        AssertTitleLocales();
    }

    private static void AssertTitleLocales()
    {
        string root = Directory.GetCurrentDirectory();
        while (!Directory.Exists(Path.Combine(root, "Locales")))
        {
            DirectoryInfo parent = Directory.GetParent(root);
            if (parent == null)
                throw new InvalidOperationException("cannot locate repository Locales directory");
            root = parent.FullName;
        }
        string path = Path.Combine(root, "Locales", "aw3_titles.csv");
        if (!File.Exists(path))
            throw new InvalidOperationException("title locale file is missing");

        var rows = new Dictionary<string, string[]>(StringComparer.Ordinal);
        foreach (string line in File.ReadAllLines(path).Skip(1))
        {
            if (string.IsNullOrWhiteSpace(line)) continue;
            string[] columns = line.Split(',');
            if (columns.Length < 4)
                throw new InvalidOperationException("title locale row has fewer than four columns: " + line);
            rows[columns[0]] = columns;
        }
        string[] required =
        {
            "aw_title_era_window", "aw_title_preview", "aw_title_era_candidates",
            "aw_title_era_input", "aw_title_era_cost", "aw_title_confirm",
            "aw_title_cancel", "aw_title_committing", "aw_ruler_appellation",
            "aw_ruler_real_name", "aw_title_error_not_hereditary_emperor",
            "aw_title_error_below_empire_rank", "aw_title_error_not_independent",
            "aw_title_error_at_war", "aw_title_error_cooldown",
            "aw_title_error_insufficient_points", "aw_title_error_invalid_name",
            "aw_title_error_duplicate_name", "aw_title_error_archive_unavailable",
            "aw_title_error_missing_shi", "aw_title_error_missing_reign",
            "aw_title_error_persistence_failed", "aw_hist_title_awarded",
            "aw_hist_title_accession_era", "aw_hist_title_voluntary_era",
            "aw_hist_title_state_name", "aw_hist_title_retrospective",
            "aw_hist_title_shi_branch"
        };
        foreach (string key in required)
        {
            if (!rows.TryGetValue(key, out string[] columns))
                throw new InvalidOperationException("missing title locale key " + key);
            for (int i = 1; i <= 3; i++)
                if (string.IsNullOrWhiteSpace(columns[i]) || columns[i] == key)
                    throw new InvalidOperationException(
                        $"title locale {key} has no value in column {i}");
        }
    }
