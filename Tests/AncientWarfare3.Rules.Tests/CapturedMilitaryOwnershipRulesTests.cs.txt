using AncientWarfare3.core.lineage;

internal static class CapturedMilitaryOwnershipRulesTests
{
    public static void Run()
    {
        Equal(true, CapturedMilitaryOwnershipRules.ShouldReleaseFormerArmy(
                captureSucceeded: true, hasFormerArmy: true,
                formerKingdomId: 2, currentKingdomId: 5),
            "a successful cross-kingdom capture releases the former army");
        Equal(false, CapturedMilitaryOwnershipRules.ShouldReleaseFormerArmy(
                captureSucceeded: false, hasFormerArmy: true,
                formerKingdomId: 2, currentKingdomId: 5),
            "a failed capture preserves the former army");
        Equal(false, CapturedMilitaryOwnershipRules.ShouldReleaseFormerArmy(
                captureSucceeded: true, hasFormerArmy: true,
                formerKingdomId: 2, currentKingdomId: 2),
            "same-kingdom capture does not disturb army membership");
        Equal(false, CapturedMilitaryOwnershipRules.ShouldReleaseFormerArmy(
                captureSucceeded: true, hasFormerArmy: false,
                formerKingdomId: 2, currentKingdomId: 5),
            "a civilian capture has no former army to release");
    }

    private static void Equal<T>(T pExpected, T pActual, string pMessage)
    {
        if (!EqualityComparer<T>.Default.Equals(pExpected, pActual))
            throw new InvalidOperationException(pMessage +
                ": expected=" + pExpected + ", actual=" + pActual);
    }
}
