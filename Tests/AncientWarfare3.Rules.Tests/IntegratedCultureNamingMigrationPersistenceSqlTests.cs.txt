using System.Data.SQLite;
using AncientWarfare3.core.lineage;

internal static class IntegratedCultureNamingMigrationPersistenceSqlTests
{
    public static void Run()
    {
        using SQLiteConnection db = OpenDatabase();
        IntegratedCultureNamingMigrationStatePersistence.Request(
            db, cultureId: 9L, now: 10d);

        IntegratedCultureNamingMigrationState state =
            IntegratedCultureNamingMigrationStatePersistence.Load(db, 9L);
        Equal("pending", state.Phase,
            "a culture request starts in the pending phase");
        Equal(-1L, state.CursorActorId,
            "a new culture migration starts before the first stable actor id");

        IntegratedCultureNamingMigrationStatePersistence.AdvanceCursor(
            db, 9L, actorId: 101L, now: 11d);
        IntegratedCultureNamingMigrationStatePersistence.AdvanceCursor(
            db, 9L, actorId: 99L, now: 12d);
        state = IntegratedCultureNamingMigrationStatePersistence.Load(db, 9L);
        Equal(101L, state.CursorActorId,
            "a retry cannot move a stable actor cursor backwards");

        IntegratedCultureNamingMigrationStatePersistence.RecordFailure(
            db, 9L, "temporary_write_failure", now: 13d);
        state = IntegratedCultureNamingMigrationStatePersistence.Load(db, 9L);
        Equal("pending", state.Phase,
            "a failed batch remains resumable");
        Equal(1, state.FailureCount,
            "a failed batch increments its retry count");
        Equal("temporary_write_failure", state.LastError,
            "the last migration failure is persisted for diagnosis");

        IntegratedCultureNamingMigrationStatePersistence.MarkComplete(
            db, 9L, now: 14d);
        state = IntegratedCultureNamingMigrationStatePersistence.Load(db, 9L);
        Equal("complete", state.Phase,
            "a drained culture migration becomes complete");

        IntegratedCultureNamingMigrationStatePersistence.Request(
            db, cultureId: 9L, now: 15d);
        state = IntegratedCultureNamingMigrationStatePersistence.Load(db, 9L);
        Equal("pending", state.Phase,
            "a late arrival reopens a completed culture migration");
        Equal(101L, state.CursorActorId,
            "a late arrival resumes after the committed actor cursor");
        Equal(0, state.FailureCount,
            "a new request clears stale failure state");
    }

    private static SQLiteConnection OpenDatabase()
    {
        var db = new SQLiteConnection("Data Source=:memory:");
        db.Open();
        using var command = new SQLiteCommand(
            "CREATE TABLE CultureNamingMigrationState (" +
            "CULTURE_ID INTEGER PRIMARY KEY,VERSION INTEGER,PHASE TEXT," +
            "CURSOR_ACTOR_ID INTEGER,FAILURE_COUNT INTEGER," +
            "LAST_ERROR TEXT,REQUESTED_TIME REAL,UPDATED_TIME REAL)", db);
        command.ExecuteNonQuery();
        return db;
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(
                $"{message}: expected {expected}, got {actual}");
    }
}
