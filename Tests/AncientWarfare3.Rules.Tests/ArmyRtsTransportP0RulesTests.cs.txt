using System;
using System.IO;
using AncientWarfare3.core.lineage;
using AncientWarfare3.core.pathfinding;

internal static class ArmyRtsTransportP0RulesTests
{
    internal static void Run()
    {
        Assert(ArmyRtsTransportRules.ResolveP0Stage(true, false, false,
                false, false, false) ==
               ArmyRtsTransportP0Stage.AssembleAtEntry,
            "a selected route first assembles the captain");
        Assert(ArmyRtsTransportRules.ResolveP0Stage(true, true, false,
                false, false, false) ==
               ArmyRtsTransportP0Stage.BoatToPickup,
            "the boat must reach the selected pickup sea tile");
        Assert(ArmyRtsTransportRules.ResolveP0Stage(true, true, true,
                false, false, false) ==
               ArmyRtsTransportP0Stage.Boarding,
            "P0 boards the manifest when captain and boat are ready");
        Assert(ArmyRtsTransportRules.ResolveP0Stage(true, true, true,
                true, false, false) ==
               ArmyRtsTransportP0Stage.Sailing,
            "a fully embarked roster begins sailing");
        Assert(ArmyRtsTransportRules.ResolveP0Stage(true, true, true,
                true, true, false) ==
               ArmyRtsTransportP0Stage.Landing,
            "arrival begins landing");
        Assert(ArmyRtsTransportRules.ResolveP0Stage(true, true, true,
                true, true, true) ==
               ArmyRtsTransportP0Stage.Complete,
            "all landed members complete the voyage");
        Assert(ArmyRtsTransportRules.ShouldActivateStrategicTransport(
                authoritative: true, strategicMovementReady: true,
                captainTileValid: true, targetTileValid: true,
                sameIsland: false, physicalRouteAvailable: true,
                voyageAlreadyActive: false),
            "a confirmed cross-island strategic route immediately starts P0 transport");
        Assert(!ArmyRtsTransportRules.ShouldActivateStrategicTransport(
                authoritative: true, strategicMovementReady: true,
                captainTileValid: true, targetTileValid: true,
                sameIsland: true, physicalRouteAvailable: true,
                voyageAlreadyActive: false),
            "a connected land objective remains under the land march path");
        Assert(ArmyRtsTransportRules.ShouldAdmitMemberP0(
                pValidMember: true, pInsideBoat: false),
            "a valid unembarked member remains P0 work");
        Assert(ArmyRtsTransportRules.ShouldAdmitMemberP0(
                pValidMember: true, pInsideBoat: true),
            "an embarked passenger remains in P0 so vanilla AI cannot steal transport ownership");
        Assert(!ArmyRtsTransportRules.ShouldAdmitMemberP0(
                pValidMember: false, pInsideBoat: false),
            "an invalid member is never admitted to P0");
        Assert(ArmyRtsTransportRules.ShouldKeepExistingP0Path(
                exactTarget: true, customPathOwned: true,
                nativeTransportBoat: false),
            "a captain keeps its pending AW path when P0 revisits the same shore target");
        Assert(ArmyRtsTransportRules.ShouldKeepExistingP0Path(
                exactTarget: true, customPathOwned: false,
                nativeTransportBoat: true),
            "a temporary boat keeps its native path when P0 revisits the same sea target");
        Assert(!ArmyRtsTransportRules.ShouldKeepExistingP0Path(
                exactTarget: false, customPathOwned: true,
                nativeTransportBoat: false),
            "a changed transport waypoint replaces the previous path");
        Assert(ArmyRtsTransportRules.SelectEmergencyLanding(
                   entryValid: true, entryDistanceSquared: 25f,
                   exitValid: true, exitDistanceSquared: 9f) ==
               ArmyRtsEmergencyLandingChoice.Exit,
            "transport cancellation lands at the nearest live coast");
        Assert(ArmyRtsTransportRules.SelectEmergencyLanding(
                   entryValid: true, entryDistanceSquared: 25f,
                   exitValid: false, exitDistanceSquared: 0f) ==
               ArmyRtsEmergencyLandingChoice.Entry,
            "transport cancellation falls back to the live entry coast");

        string root = Directory.GetCurrentDirectory();
        string transport = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsTransportService.cs"));
        string production = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsTransportProductionService.cs"));
        string controller = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsControllerService.cs"));
        string movementBridge = File.ReadAllText(Path.Combine(root, "Code",
            "core", "pathfinding", "AWPathMovementBridge.cs"));
        string dockTransport = File.ReadAllText(Path.Combine(root, "Code",
            "core", "pathfinding", "AWDockTransportService.cs"));
        string runner = File.ReadAllText(Path.Combine(root, "Code", "core",
            "performance", "AWCooperativeActorPostRunner.cs"));
        string boatLifecycle = File.ReadAllText(Path.Combine(root, "Code",
            "patch", "AW_ActorBoatLifecyclePatch.cs"));

        int dockLookupStart = dockTransport.IndexOf(

        int censusStart = transport.IndexOf(

        int p0DriveStart = transport.IndexOf(

            StringComparison.Ordinal);

        Assert(p0DriveBody.Contains(

        Assert(p0DriveBody.Contains("CaptureTransportPathDiagnostic(",

        int runtimeStateStart = movementBridge.IndexOf(

        string runtimeStateBody = runtimeStateStart >= 0 &&

        Assert(runtimeStateBody.Contains("poll=",

               !runtimeStateBody.Contains(".Poll(",

               !runtimeStateBody.Contains("OpenReadyCursor(",

        int lockedSubmitStart = transport.IndexOf(
            "private static void SubmitLockedPathIfNeeded(",

        int lockedSubmitEnd = transport.IndexOf(

        string lockedSubmitBody = lockedSubmitStart >= 0 &&
            lockedSubmitEnd > lockedSubmitStart
            ? transport.Substring(lockedSubmitStart,

            : string.Empty;
        Assert(lockedSubmitBody.Contains("pActor.goTo(pTarget",
                   StringComparison.Ordinal) &&
               !lockedSubmitBody.Contains("AWPathMovementBridge.Submit(",
                   StringComparison.Ordinal),

        int validMemberStart = transport.IndexOf(
            "private static bool IsValidMember(",
            StringComparison.Ordinal);
        int validMemberEnd = transport.IndexOf(
            "private static bool IsCaptain(", validMemberStart,
            StringComparison.Ordinal);
        string validMemberBody = validMemberStart >= 0 &&
            validMemberEnd > validMemberStart

        Assert(!validMemberBody.Contains("is_profession_warrior",
                   StringComparison.Ordinal) &&

        int advanceRouteStart = controller.IndexOf(
            "private static void AdvanceRoute(", StringComparison.Ordinal);
        int advanceRouteEnd = controller.IndexOf(
            "private static void LogStrategicRouteFailure(",
            advanceRouteStart, StringComparison.Ordinal);
        string advanceRouteBody = advanceRouteStart >= 0 &&
            advanceRouteEnd > advanceRouteStart
            ? controller.Substring(advanceRouteStart,
                advanceRouteEnd - advanceRouteStart)

                   "TryActivateStrategicTransport(",
                   StringComparison.Ordinal),
            "strategic route publication activates a selected portal voyage before actor movement");
        Assert(advanceRouteBody.IndexOf(
                   "TryActivateStrategicTransport(",
                   StringComparison.Ordinal) <
               advanceRouteBody.IndexOf(
                   "TryIssueVanillaCityAttackOrder(",
                   StringComparison.Ordinal),
            "cross-island transport ownership is decided before any vanilla land attack order");

        Assert(transport.Contains("AWDockTransportService.TryResolveRoute(",
                StringComparison.Ordinal),
            "a voyage resolves one physical portal route when it begins");
        Assert(transport.Contains("ArmyRtsTransportP0Stage.AssembleAtEntry",
                StringComparison.Ordinal) &&
               transport.Contains("ArmyRtsTransportP0Stage.BoatToPickup",
                   StringComparison.Ordinal) &&
               transport.Contains("ArmyRtsTransportP0Stage.Boarding",
                   StringComparison.Ordinal) &&
               transport.Contains("ArmyRtsTransportP0Stage.Sailing",
                   StringComparison.Ordinal) &&
               transport.Contains("ArmyRtsTransportP0Stage.Landing",
                   StringComparison.Ordinal),
            "the voyage is executed as an explicit military P0 state machine");
        Assert(production.Contains("TryProvisionAtRoute(",
                StringComparison.Ordinal),
            "temporary transport production accepts the locked physical route");
        Assert(transport.Contains("DriveActorToTileP0(",
                StringComparison.Ordinal),
            "P0 drives the captain to the selected entry land tile");
        Assert(transport.Contains("DriveBoatToTileP0(",
                StringComparison.Ordinal),
            "P0 drives the temporary boat to pickup and destination sea tiles");
        Assert(transport.Contains(
                   "member.data.transportID = boat.actor.data.id",
                   StringComparison.Ordinal) &&
               transport.Contains("member.is_inside_boat = true",
                   StringComparison.Ordinal) &&
               transport.Contains("member.inside_boat = boat",
                   StringComparison.Ordinal) &&
               transport.Contains("boat.addPassenger(member)",
                   StringComparison.Ordinal),
            "P0 directly attaches every valid army member to the boat");
        Assert(transport.Contains("member.disembarkTo(boat, landingTile)",
                StringComparison.Ordinal),
            "P0 directly lands every valid army member on the selected coast");

        Assert(!transport.Contains("TaxiManager.newRequest(",
                   StringComparison.Ordinal) &&
               !transport.Contains("TaxiRequestState.Loading",
                   StringComparison.Ordinal) &&
               !transport.Contains("VanillaForceEmbarkTaskId",
                   StringComparison.Ordinal) &&
               !transport.Contains("boat_transport_go_load",
                   StringComparison.Ordinal) &&
               !transport.Contains("boat_transport_go_unload",
                   StringComparison.Ordinal),
            "RTS P0 voyages cannot wait for vanilla taxi or boat task state");

        Assert(transport.Contains("ProcessMilitaryP0(",
                StringComparison.Ordinal),
            "active RTS voyages expose a dedicated military P0 driver");
        Assert(transport.Contains("RefreshMilitaryP0Priority(",
                StringComparison.Ordinal),
            "every active voyage re-admits its roster to military P0");
        Assert(!controller.Contains(
                "ArmyRtsTransportService.ProcessFrame();",
                StringComparison.Ordinal),
            "the ordinary RTS controller cannot advance a large-step voyage");

        int p0Stage = runner.IndexOf("case PostStage.MilitaryP0:",
            StringComparison.Ordinal);
        int p0Chunk = runner.IndexOf("RunMilitaryP0Chunk(elapsed);",
            p0Stage, StringComparison.Ordinal);
        int transportP0 = runner.IndexOf(
            "ArmyRtsTransportService.ProcessMilitaryP0(elapsed);", p0Stage,
            StringComparison.Ordinal);
        Assert(p0Stage >= 0 && transportP0 > p0Stage &&
               transportP0 < p0Chunk,
            "transport state and boat movement run before passenger P0 work");
        Assert(runner.Contains(
                "ArmyRtsTransportService.TryDriveMemberP0(",
                StringComparison.Ordinal),
            "the actor runner lets transport advance captain assembly before suppression");
        Assert(transport.Contains("AWDockTransportService.TryResolveDestination(",
                   StringComparison.Ordinal) &&
               transport.Contains("census.HasAnyEmbarked",
                   StringComparison.Ordinal),
            "an embarked voyage reselects only its destination portal when the exit changes");
        Assert(transport.Contains(
                   "ArmyRtsTransportP0Stage.BoatToPickup",
                   StringComparison.Ordinal) &&
               transport.Contains("if (!IsLiveBoat(state.Boat))",
                   StringComparison.Ordinal),
            "a destroyed temporary boat returns the voyage to pickup provisioning");
        Assert(production.Contains("if (boat?.hasPassengers() == true)",
                   StringComparison.Ordinal) &&
               production.Contains("PendingTemporaryBoatRemovalIds.Add(",
                   StringComparison.Ordinal),
            "temporary boats remain alive until every live passenger has landed");
        Assert(transport.Contains("ProcessVoyageFrame(Time.deltaTime,",
                   StringComparison.Ordinal),
            "ordinary simulation advances direct transport movement with a real elapsed time");
        Assert(transport.Contains("reason=route_unavailable",
                   StringComparison.Ordinal) &&
               transport.Contains("reason=boat_provision_failed",
                   StringComparison.Ordinal) &&
               transport.Contains("reason=boarding_incomplete",
                   StringComparison.Ordinal) &&
               transport.Contains("reason=sailing_path_pending",
                   StringComparison.Ordinal) &&
               transport.Contains("reason=landing_incomplete",
                   StringComparison.Ordinal),
            "every P0 voyage boundary emits a stable failure or pending reason");

            "private static bool TryMove(Actor pActor, AWPathStep pStep)",

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
