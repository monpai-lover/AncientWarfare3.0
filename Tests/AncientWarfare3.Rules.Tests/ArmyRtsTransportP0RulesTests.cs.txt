using System;
using System.IO;

internal static class ArmyRtsTransportP0RulesTests
{
    internal static void Run()
    {
        string root = Directory.GetCurrentDirectory();
        string transport = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsTransportService.cs"));
        string controller = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsControllerService.cs"));
        string runner = File.ReadAllText(Path.Combine(root, "Code", "core",
            "performance", "AWCooperativeActorPostRunner.cs"));

        Assert(transport.Contains("ProcessMilitaryP0(",
                StringComparison.Ordinal),
            "active RTS voyages expose a dedicated military P0 driver");
        Assert(transport.Contains("RefreshMilitaryP0Priority(",
                StringComparison.Ordinal),
            "every active voyage re-admits every expected passenger to P0");
        Assert(!controller.Contains(
                "ArmyRtsTransportService.ProcessFrame();",
                StringComparison.Ordinal),
            "the ordinary RTS controller cannot advance a voyage in large-step mode");

        int p0Stage = runner.IndexOf("case PostStage.MilitaryP0:",
            StringComparison.Ordinal);
        int p0Chunk = runner.IndexOf("RunMilitaryP0Chunk(elapsed);",
            p0Stage, StringComparison.Ordinal);
        int transportP0 = runner.IndexOf(
            "ArmyRtsTransportService.ProcessMilitaryP0(elapsed);", p0Stage,
            StringComparison.Ordinal);
        Assert(p0Stage >= 0 && transportP0 > p0Stage &&
               transportP0 < p0Chunk,
            "transport state and boat movement run before passenger P0 work");

        Assert(transport.Contains("DriveBoundBoatP0(",
                StringComparison.Ordinal),
            "the transport-owned boat is driven in the same P0 transaction");
        Assert(transport.Contains("SuppressCombatForVoyage(",
                StringComparison.Ordinal),
            "transport ownership suppresses combat until the whole roster lands");
        Assert(transport.Contains("landedExpectedMembers >= validExpectedMembers",
                StringComparison.Ordinal),
            "voyage completion waits for every valid expected member to land");
        Assert(transport.Contains("ForceEmbarkRoster(",
                StringComparison.Ordinal),
            "a loading transport explicitly transfers the full RTS roster " +
            "to the native embark task");
        Assert(transport.Contains("TeleportRosterToBoat(",
                StringComparison.Ordinal),
            "a bound RTS transport has a deterministic full-roster embark " +
            "fallback");
        Assert(transport.Contains("if (hasEmbarkedMember)",
                StringComparison.Ordinal),
            "the full-roster fallback waits until the captain has boarded");
        Assert(transport.Contains("member.embarkInto(boat)",
                StringComparison.Ordinal),
            "the fallback places every valid member into the assigned boat");
        Assert(transport.Contains("TeleportRosterFromBoat(",
                StringComparison.Ordinal) &&
               transport.Contains("if (captainLanded)",
                   StringComparison.Ordinal) &&
               transport.Contains("member.disembarkTo(boat, pLandingTile)",
                   StringComparison.Ordinal),
            "after the captain reaches stable land, any residual passengers " +
            "are moved to the same landing tile");
        string production = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsTransportProductionService.cs"));
        Assert(production.Contains("OceanTile", StringComparison.Ordinal),
            "temporary transport production retains the validated route " +
            "ocean tile");
        Assert(production.Contains("routeDock.OceanTile",
                StringComparison.Ordinal),
            "temporary transport spawning uses the validated route ocean");
        Assert(transport.Contains("ForceEmbarkRoster(state, anyRequestLoading, assignedBoat)",
                StringComparison.Ordinal),
            "the embark handoff is driven by a live assigned boat after " +
            "the native request enters Loading");
        Assert(transport.Contains("ClearMovementForTransport(",
                StringComparison.Ordinal),
            "switching from a failed land route clears stale RTS movement " +
            "before the voyage can take ownership");

        int captainRouteStart = controller.IndexOf(
            "private static void TrySubmitCaptainObjectiveRoute(",
            StringComparison.Ordinal);
        int captainRouteEnd = controller.IndexOf(
            "private static long ResolveCaptainMissionTargetCityId(",
            captainRouteStart, StringComparison.Ordinal);
        string captainRouteBody = captainRouteStart >= 0 &&
            captainRouteEnd > captainRouteStart
            ? controller.Substring(captainRouteStart,
                captainRouteEnd - captainRouteStart)
            : string.Empty;
        Assert(captainRouteBody.Contains("HasActiveVoyage(army)",
                StringComparison.Ordinal),
            "the captain cannot resubmit a land route while transport owns " +
            "the army");

        int followerRouteStart = controller.IndexOf(
            "private static void TrySubmitMemberObjectiveRoute(",
            StringComparison.Ordinal);
        int followerRouteEnd = controller.IndexOf(
            "private static void TrySubmitCaptainObjectiveRoute(",
            followerRouteStart, StringComparison.Ordinal);
        string followerRouteBody = followerRouteStart >= 0 &&
            followerRouteEnd > followerRouteStart
            ? controller.Substring(followerRouteStart,
                followerRouteEnd - followerRouteStart)
            : string.Empty;
        Assert(followerRouteBody.Contains("HasActiveVoyage(pActor?.army)",
                StringComparison.Ordinal),
            "followers cannot resubmit formation paths while transport owns " +
            "the army");

        int transportP0Start = transport.IndexOf(
            "internal static void ProcessMilitaryP0(float pCycleElapsed)",
            StringComparison.Ordinal);
        int transportP0End = transport.IndexOf(
            "internal static bool SuppressCombatForVoyage", transportP0Start,
            StringComparison.Ordinal);
        string transportP0Body = transportP0Start >= 0 &&
            transportP0End > transportP0Start
            ? transport.Substring(transportP0Start,
                transportP0End - transportP0Start)
            : string.Empty;
        int disposal = transportP0Body.IndexOf(
            "ProcessTemporaryBoatDisposals();", StringComparison.Ordinal);
        int voyageGate = transportP0Body.IndexOf(
            "ShouldProcessInMilitaryP0(", StringComparison.Ordinal);
        Assert(disposal >= 0 && voyageGate > disposal,
            "P0 drains temporary-boat disposals even after the final voyage completed");
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
}
