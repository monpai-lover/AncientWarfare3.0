using AncientWarfare3.core.court;
using AncientWarfare3.core.policy;

internal static class WesternCourtProfileRulesTests
{
    public static void Run()
    {
        ICourtProfile western = CourtProfileRegistry.For(
            KingdomPolicyProfileId.WesternGeneral);
        Equal(CourtProfileId.Western, western.Id,
            "Western policy profile selects Western court");

        CourtOfficeDefinition executive = Require(western,
            CourtOfficeId.WestExecutive);
        Equal(CourtOfficeLayer.Central, executive.Layer,
            "executive is central");
        Equal(10, executive.Grade, "executive grade");
        Equal("aw_court_office_west_executive", executive.LocalizationKey,
            "executive localization");
        Equal(CourtSchoolId.Diplomat, executive.PreferredSchoolId,
            "executive school");

        CourtOfficeDefinition mayor = Require(western,
            CourtOfficeId.WestMayor);
        Equal(CourtOfficeLayer.City, mayor.Layer, "mayor is city office");
        Equal(30, mayor.Grade, "mayor grade");

        CourtOfficeDefinition count = Require(western,
            CourtOfficeId.WestCount);
        Equal(CourtOfficeLayer.City, count.Layer, "count is city office");
        Equal(true, count.AvailableIn(
                CourtInstitutionId.WesternFeudalBureaucratic),
            "count unlocks under feudal bureaucratic institution");
        Equal(false, count.AvailableIn(
                CourtInstitutionId.WesternBureaucratic),
            "count is not a base office");

        CourtOfficeDefinition fieldGeneral = Require(western,
            CourtOfficeId.WestFieldGeneral);
        Equal(CourtOfficeLayer.Military, fieldGeneral.Layer,
            "field general is military");
        Equal(true, fieldGeneral.MilitaryCapable,
            "field general supports military careers");

        CourtOfficeDefinition marshal = Require(western,
            CourtOfficeId.WestMarshal);
        Equal(CourtOfficeLayer.Military, marshal.Layer,
            "marshal is military");
        Equal(true, marshal.MilitaryCapable,
            "marshal supports military careers");
        Equal(false, marshal.AvailableIn(
                CourtInstitutionId.WesternBureaucratic),
            "marshal requires the feudal stage");
        CourtOfficeDefinition constable = Require(western,
            CourtOfficeId.WestRoyalConstable);
        Equal(CourtOfficeLayer.Central, constable.Layer,
            "royal constable is a central military office");
        Equal(true, constable.MilitaryCapable,
            "royal constable uses military lifecycle");
        Equal(CourtOfficeId.WestRoyalConstable,
            WesternCourtMigrationRules.NormalizeOfficeId(
                CourtOfficeId.WestRoyalChamberlain),
            "legacy chamberlain migrates to royal constable");
        Equal(CourtOfficeId.WestTreasurer,
            WesternCourtMigrationRules.NormalizeOfficeId(
                CourtOfficeId.WestTreasurer),
            "canonical office ids remain unchanged");

        Equal(CourtInstitutionId.WesternBureaucratic,
            western.ResolveInstitution(false, false),
            "Western kingdoms begin with the basic bureaucratic court");
        Equal(CourtInstitutionId.WesternBureaucratic,
            western.ResolveInstitution(true, false),
            "office system opens the bureaucratic court");

        Equal(CourtInstitutionId.WesternBureaucratic,
            western.DefaultInstitutionId,
            "the Western default institution has formal offices");
        Equal(CourtInstitutionId.WesternFeudalBureaucratic,
            western.ResolveInstitution(true, true),
            "feudal retainers open the cumulative advanced court");

        Sequence(new[]
            {
                CourtOfficeId.WestExecutive,
                CourtOfficeId.WestSenateElder,
                CourtOfficeId.WestHighPriest,
                CourtOfficeId.WestFieldGeneral,
                CourtOfficeId.WestMayor
            }, western.OfficeIdsForInstitution(
                CourtInstitutionId.WesternBureaucratic),
            "bureaucratic office catalog");
        Equal(12, western.OfficeIdsForInstitution(
                CourtInstitutionId.WesternFeudalBureaucratic).Count,
            "advanced catalog retains basic offices and adds seven");

        Equal(CourtInstitutionId.WesternBureaucratic,
            CourtInstitutionRules.MigrateWesternLegacy(
                CourtInstitutionId.WesternElective,
                feudalRetainersUnlocked: false),
            "elective government migrates to basic bureaucracy");
        Equal(CourtInstitutionId.WesternBureaucratic,
            CourtInstitutionRules.MigrateWesternLegacy(
                CourtInstitutionId.WesternPrimitive,
                feudalRetainersUnlocked: false),
            "legacy primitive courts migrate to basic bureaucracy");
        Equal(CourtInstitutionId.WesternFeudalBureaucratic,
            CourtInstitutionRules.MigrateWesternLegacy(
                CourtInstitutionId.WesternFeudal,
                feudalRetainersUnlocked: false),
            "legacy feudal institution migrates to advanced bureaucracy");
        Equal(CourtInstitutionId.WesternBureaucratic,
            CourtInstitutionRules.MigrateWesternLegacy(
                CourtInstitutionId.WesternRoyalDirect,
                feudalRetainersUnlocked: false),
            "royal government without advanced tech migrates to basic");
        Equal(CourtInstitutionId.WesternFeudalBureaucratic,
            CourtInstitutionRules.MigrateWesternLegacy(
                CourtInstitutionId.WesternRoyalDirect,
                feudalRetainersUnlocked: true),
            "royal government with advanced tech migrates to advanced");
        Equal(CourtOfficeId.Governor, CourtCityOfficeRules.Resolve(
                CourtProfileId.Xia, CourtInstitutionId.Zhou, false),
            "Xia keeps the native governor office");
        Equal(CourtOfficeId.WestMayor, CourtCityOfficeRules.Resolve(
                CourtProfileId.Western,
                CourtInstitutionId.WesternBureaucratic, false),
            "Western bureaucracy projects normal cities as Mayors");
        Equal(CourtOfficeId.WestCount, CourtCityOfficeRules.Resolve(
                CourtProfileId.Western,
                CourtInstitutionId.WesternFeudalBureaucratic, true),
            "advanced feudatory seats project as Counts");
        Equal("", CourtCityOfficeRules.Resolve(
                CourtProfileId.Western, CourtInstitutionId.WesternPrimitive,
                false), "primitive Western courts have no formal city office");

        ICourtProfile xia = CourtProfileRegistry.For(
            KingdomPolicyProfileId.Xia);
        Equal(CourtProfileId.Xia, xia.Id,
            "Xia profile keeps Xia court");
        Sequence(new[]
            {
                CourtOfficeId.TaiZai, CourtOfficeId.SiTu,
                CourtOfficeId.ZongBo, CourtOfficeId.SiMa,
                CourtOfficeId.SiKou, CourtOfficeId.SiKong
            }, xia.OfficeIdsForInstitution(CourtInstitutionId.Zhou),
            "Zhou offices stay unchanged");
        Sequence(CourtTierRules.CentralOfficesForTier(
                CourtTier.SanGongJiuQing),
            xia.OfficeIdsForInstitution(CourtInstitutionId.Han),
            "Han offices stay unchanged");
        Sequence(CourtTierRules.CentralOfficesForTier(
                CourtTier.SanShengLiuBu),
            xia.OfficeIdsForInstitution(CourtInstitutionId.Tang),
            "Tang offices stay unchanged");

        Equal(false, CourtProfileRules.CanHoldTogether(
                CourtProfileId.Xia, CourtOfficeLayer.Central,
                CourtProfileId.Western, CourtOfficeLayer.Central),
            "one actor cannot hold central offices from both profiles");
        Equal(true, CourtProfileRules.CanHoldTogether(
                CourtProfileId.Xia, CourtOfficeLayer.Central,
                CourtProfileId.Xia, CourtOfficeLayer.City),
            "same profile can combine central and city layers");
    }

    private static CourtOfficeDefinition Require(ICourtProfile profile,
        string id)
    {
        CourtOfficeDefinition result = profile.FindOffice(id);
        if (result == null)
            throw new InvalidOperationException("missing office: " + id);
        return result;
    }

    private static void Sequence(IReadOnlyList<string> expected,
        IReadOnlyList<string> actual, string message)
    {
        if (expected.Count != actual.Count)
            throw new InvalidOperationException(message + ": count mismatch");
        for (int index = 0; index < expected.Count; index++)
            Equal(expected[index], actual[index], message + "[" + index + "]");
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(
                $"{message}: expected {expected}, got {actual}");
    }
}
