using AncientWarfare3.core.lineage;

internal static class ActorDeathArchiveRulesTests
{
    public static void Run()
    {
        Equal("death:7:42:3:lineage_archive",
            ActorDeathArchiveRules.Key(7, 42, 3, "lineage_archive"),
            "death archive keys are stable and stage-specific");
        Equal(true, ActorDeathArchiveRules.AcceptCompletion(
                resultWorldGeneration: 7, currentWorldGeneration: 7,
                resultDeathRevision: 3, currentDeathRevision: 3),
            "matching world and death revisions commit");
        Equal(false, ActorDeathArchiveRules.AcceptCompletion(
                resultWorldGeneration: 7, currentWorldGeneration: 8,
                resultDeathRevision: 3, currentDeathRevision: 3),
            "old-world completion is stale");
        Equal(false, ActorDeathArchiveRules.AcceptCompletion(
                resultWorldGeneration: 7, currentWorldGeneration: 7,
                resultDeathRevision: 2, currentDeathRevision: 3),
            "older death revisions are stale");
        Equal(1, ActorDeathArchiveRules.RetryDelayFrames(1),
            "the first retry waits one frame");
        Equal(8, ActorDeathArchiveRules.RetryDelayFrames(4),
            "retry delay grows exponentially");
        Equal(256, ActorDeathArchiveRules.RetryDelayFrames(20),
            "retry delay has a hard cap");
        Equal(true, ActorDeathArchiveRules.ShouldUseSynchronousFallback(
                writerReady: false, queueAccepted: false),
            "an unavailable writer uses the bounded main-thread fallback");
        Equal(true, ActorDeathArchiveRules.ShouldUseSynchronousFallback(
                writerReady: true, queueAccepted: false),
            "a full writer queue uses the bounded main-thread fallback");
        Equal(false, ActorDeathArchiveRules.ShouldUseSynchronousFallback(
                writerReady: true, queueAccepted: true),
            "accepted asynchronous work needs no fallback");
        Equal(true, ActorDeathArchiveRules.ShouldAttemptSynchronousWrite(
                queueAccepted: false),
            "rejected asynchronous death work uses authority fallback");
        Equal(false, ActorDeathArchiveRules.ShouldAttemptSynchronousWrite(
                queueAccepted: true),
            "accepted asynchronous death work is not written twice");
        Equal(true, ActorDeathArchiveRules.ShouldQueueDeathInMemory(
                alive: false, forceSynchronous: false,
                allowSynchronousFallback: false),
            "runtime death callbacks publish to the memory authority queue first");
        Equal(false, ActorDeathArchiveRules.ShouldQueueDeathInMemory(
                alive: true, forceSynchronous: false,
                allowSynchronousFallback: false),
            "live archive updates keep their normal persistence path");
        Equal(false, ActorDeathArchiveRules.ShouldQueueDeathInMemory(
                alive: false, forceSynchronous: true,
                allowSynchronousFallback: false),
            "explicit synchronous death writes bypass the runtime capture queue");
        Equal(true, ActorDeathArchiveRules.ReadyForSave(
                captured: 0, running: 0, retries: 0, completions: 0),
            "save requires every death queue to be empty");
        Equal(false, ActorDeathArchiveRules.ReadyForSave(
                captured: 0, running: 1, retries: 0, completions: 0),
            "running death work blocks save");
        Equal(64, ActorDeathArchiveRules.ResolveAuthorityItemLimit(32),
            "small death backlogs keep the normal authority budget");
        Equal(256, ActorDeathArchiveRules.ResolveAuthorityItemLimit(821),
            "large death bursts receive a bounded catch-up budget");
        Equal(1d, ActorDeathArchiveRules.ResolveAuthorityMilliseconds(32),
            "small death backlogs keep the normal time slice");
        Equal(4d, ActorDeathArchiveRules.ResolveAuthorityMilliseconds(821),
            "large death bursts receive a bounded catch-up time slice");
        Equal(5, ActorDeathArchiveRules.ResolveSaveTimeoutSeconds(
                baseSeconds: 5, pendingCount: 0),
            "an empty death queue keeps the base save timeout");
        Equal(30, ActorDeathArchiveRules.ResolveSaveTimeoutSeconds(
                baseSeconds: 5, pendingCount: 821),
            "a mass-death save gets a bounded backlog-aware timeout");
        Equal(60, ActorDeathArchiveRules.ResolveSaveTimeoutSeconds(
                baseSeconds: 60, pendingCount: 821),
            "backlog scaling never shortens a caller's base timeout");
        string pendingDiagnostic = ActorDeathArchiveRules.
            DescribePendingForSave(92, 734L, 3);
        Equal(true, pendingDiagnostic.Contains(
                "pending actor death archives=92",
                StringComparison.Ordinal),
            "save diagnostics include the pending death count");
        Equal(true, pendingDiagnostic.Contains("first_actor_id=734",
                StringComparison.Ordinal),
            "save diagnostics identify the first pending actor");
        Equal(true, pendingDiagnostic.Contains("first_attempts=3",
                StringComparison.Ordinal),
            "save diagnostics include the first retry count");
    }

    private static void Equal<T>(T pExpected, T pActual, string pMessage)
    {
        if (!EqualityComparer<T>.Default.Equals(pExpected, pActual))
            throw new InvalidOperationException(pMessage +
                ": expected=" + pExpected + ", actual=" + pActual);
    }
}
