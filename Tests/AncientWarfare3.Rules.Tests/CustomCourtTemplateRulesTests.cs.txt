using System;
using System.IO;
using AncientWarfare3.core.court;

internal static class CustomCourtTemplateRulesTests
{
    public static void Run()
    {
        True(CustomCourtTemplateRules.IsValidTemplateId("xia_custom_1"),
            "stable lower-case template identifiers are accepted");
        False(CustomCourtTemplateRules.IsValidTemplateId("Xia Custom"),
            "template identifiers reject display text");

        Equal(CustomCourtTemplateValidationError.Cycle,
            CustomCourtTemplateRules.ValidateGraph(new[]
            {
                Edge("a", "b", CustomCourtEdgeKind.Management),
                Edge("b", "a", CustomCourtEdgeKind.Management)
            }), "cycles are rejected across management edges");

        Equal(CustomCourtTemplateValidationError.DanglingOffice,
            CustomCourtTemplateRules.ValidateGraph(new[] { "a" }, new[]
            {
                Edge("a", "missing",
                    CustomCourtEdgeKind.AppointmentPrerequisite)
            }), "edges must reference defined offices");

        False(CustomCourtTemplateRules.IsEffectValueValid(
                CustomCourtEffectId.TaxIncome,
                CustomCourtEffectMode.AddPercent, 999),
            "effects are bounded by their preset module");
        False(CustomCourtTemplateJsonCodec.TryImport(
                "{\"Id\":\"court_broken\",\"Name\":{\"Chinese\":\"坏\",\"English\":\"Broken\"}," +
                "\"Offices\":[{\"Id\":\"office_a\",\"Name\":{\"Chinese\":\"坏官\",\"English\":\"Broken Office\"}," +
                "\"Layer\":\"central\",\"Grade\":10,\"Slots\":1,\"Effects\":[null]}],\"Edges\":[]}",
                out _, out CustomCourtTemplateValidationError malformedError),
            "structurally malformed effect entries are rejected");
        Equal(CustomCourtTemplateValidationError.InvalidEffect,
            malformedError,
            "null effect entries report the invalid-effect category");

        string modRoot = Path.Combine(Path.GetTempPath(), "aw3-courtjson-" +
            Guid.NewGuid().ToString("N"));
        string courtJson = CustomCourtTemplatePathService.ResolveCourtJsonRoot(
            modRoot);
        Equal(Path.Combine(Path.GetFullPath(modRoot), "Courtjson"), courtJson,
            "court templates resolve inside the mod folder");
        Directory.CreateDirectory(modRoot);
        try
        {
            var store = new CustomCourtTemplateStore(courtJson);
            var template = new CustomCourtTemplate
            {
                Id = "court_alpha",
                Name = new CustomCourtLocalizedText
                {
                    Chinese = "甲朝廷",
                    English = "Alpha Court"
                },
                Offices =
                {
                    new CustomCourtOffice
                    {
                        Id = "office_a",
                        Name = new CustomCourtLocalizedText
                        {
                            Chinese = "甲官",
                            English = "Office A"
                        },
                        Layer = CourtOfficeLayer.Central,
                        Grade = 10,
                        Slots = 1
                    }
                }
            };
            True(store.TrySave(template, out CustomCourtTemplateValidationError
                    saveError, out string savedPath),
                "valid template exports to Courtjson");
            Equal(CustomCourtTemplateValidationError.None, saveError,
                "export reports no validation error");
            Equal(Path.Combine(courtJson, "court_alpha.json"), savedPath,
                "export reports its full path");
            File.WriteAllText(Path.Combine(courtJson, "zeta.json"), "{}");
            File.WriteAllText(Path.Combine(courtJson, "ignored.txt"), "{}");
            string[] files = store.ListFileNames();
            Equal(2, files.Length, "only json files are listed");
            Equal("court_alpha.json", files[0],
                "json files are sorted by file name");
            Equal("zeta.json", files[1],
                "later json file follows sorted order");
            True(store.TryLoadFile("court_alpha.json",
                    out CustomCourtTemplate loaded,
                    out CustomCourtTemplateValidationError loadError),
                "selected json file imports successfully");
            Equal("court_alpha", loaded.Id,
                "selected import returns the stored template");
            False(store.TryLoadFile("..\\court_alpha.json", out _, out _),
                "selected import rejects path traversal");
        }
        finally
        {
            string fullTemp = Path.GetFullPath(modRoot);
            if (fullTemp.StartsWith(Path.GetFullPath(Path.GetTempPath()),
                    StringComparison.OrdinalIgnoreCase) &&
                Directory.Exists(fullTemp))
                Directory.Delete(fullTemp, true);
        }
    }

    private static CustomCourtEdge Edge(string fromOfficeId, string toOfficeId,
        CustomCourtEdgeKind kind)
    {
        return new CustomCourtEdge
        {
            FromOfficeId = fromOfficeId,
            ToOfficeId = toOfficeId,
            Kind = kind
        };
    }

    private static void True(bool actual, string message)
    {
        if (!actual)
            throw new InvalidOperationException(message);
    }

    private static void False(bool actual, string message)
    {
        if (actual)
            throw new InvalidOperationException(message);
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(
                $"{message}: expected {expected}, got {actual}");
    }
}
