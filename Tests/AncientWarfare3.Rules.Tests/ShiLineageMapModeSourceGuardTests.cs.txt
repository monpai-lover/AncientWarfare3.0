    private static string Read(string pRelative)
    {
        return File.ReadAllText(Path.Combine(Directory.GetCurrentDirectory(),
                pRelative.Replace('/', Path.DirectorySeparatorChar)))
            .Replace("\r\n", "\n");
    }

    private static void Contains(string pText, string pNeedle, string pMessage)
    {
        if (!pText.Contains(pNeedle, StringComparison.Ordinal))
            throw new InvalidOperationException(pMessage);
    }

    private static void NotContains(string pText, string pNeedle, string pMessage)
    {
        if (pText.Contains(pNeedle, StringComparison.Ordinal))
            throw new InvalidOperationException(pMessage);
    }

    private static void Before(string pText, string pFirst, string pSecond,
        string pMessage)
    {
        int first = pText.IndexOf(pFirst, StringComparison.Ordinal);
        int second = pText.IndexOf(pSecond, StringComparison.Ordinal);
        if (first < 0 || second < 0 || first >= second)
            throw new InvalidOperationException(pMessage);
    }
