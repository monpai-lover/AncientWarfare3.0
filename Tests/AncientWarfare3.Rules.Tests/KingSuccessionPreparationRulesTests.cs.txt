using System;
using AncientWarfare3.core.lineage;

internal static class KingSuccessionPreparationRulesTests
{
    public static void Run()
    {
        SameLineageKeepsFlagUntilLastKingLeaves();
        DynastyChangeMovesOnlyThatKingdom();
        WorldLoadRebuildStartsFromEmptyState();
        OrdinaryActorCannotRequestSuccessionMaintenance();
        ForeignReigningBranchCannotDirtyThisKingdom();
        ValidRegisteredHeirSkipsFallback();
        InvalidRegisteredHeirRunsFallbackOncePerDeath();
        WorldGenerationSeparatesFallbackAttempts();
        UnchangedHeirSignatureSkipsMaintenance();
    }

    private static void SameLineageKeepsFlagUntilLastKingLeaves()
    {
        var index = new ReigningRoyalLineageIndexState();
        index.Register(10L, 100L);
        index.Register(10L, 200L);

        index.RemoveKingdom(100L);

        True(index.HasReigningKing(10L),
            "one surviving king keeps the lineage active");
        True(index.Contains(10L, 200L),
            "the second kingdom remains registered");

        index.RemoveKingdom(200L);

        Equal(false, index.HasReigningKing(10L),
            "the last king clears the lineage gate");
    }

    private static void DynastyChangeMovesOnlyThatKingdom()
    {
        var index = new ReigningRoyalLineageIndexState();
        index.Register(10L, 100L);
        index.Register(10L, 200L);

        index.Register(20L, 100L);

        Equal(false, index.Contains(10L, 100L),
            "dynasty change removes the old membership");
        True(index.Contains(10L, 200L),
            "another king of the old lineage remains");
        True(index.Contains(20L, 100L),
            "the changed kingdom joins the new lineage");
    }

    private static void WorldLoadRebuildStartsFromEmptyState()
    {
        var index = new ReigningRoyalLineageIndexState();
        index.Register(10L, 100L);
        index.CompleteRebuild();

        index.BeginRebuild();

        Equal(false, index.IsReady,
            "a world rebuild is closed until live kingdoms are indexed");
        Equal(false, index.HasReigningKing(10L),
            "stale save-derived memberships are not trusted");

        index.Register(20L, 200L);
        index.CompleteRebuild();

        True(index.IsReady, "the rebuilt live index becomes queryable");
        True(index.Contains(20L, 200L),
            "live kingdoms populate the rebuilt state");
    }

    private static void OrdinaryActorCannotRequestSuccessionMaintenance()
    {
        Equal(false, RoyalSuccessionEventRules.ShouldMarkSelectionDirty(
                pActorLineageId: -1L, pKingdomRoyalLineageId: 10L,
                pLineageReignsInThisKingdom: true,
                pRegisteredHeir: false, pDirectRoyalChild: true),
            "an ordinary actor without LINEAGE_ID is filtered");
        Equal(false, RoyalSuccessionEventRules.ShouldMarkSelectionDirty(
                pActorLineageId: 10L, pKingdomRoyalLineageId: 10L,
                pLineageReignsInThisKingdom: true,
                pRegisteredHeir: false, pDirectRoyalChild: false),
            "an unrelated lineage member is filtered");
    }

    private static void ForeignReigningBranchCannotDirtyThisKingdom()
    {
        Equal(false, RoyalSuccessionEventRules.ShouldMarkSelectionDirty(
                pActorLineageId: 10L, pKingdomRoyalLineageId: 10L,
                pLineageReignsInThisKingdom: false,
                pRegisteredHeir: true, pDirectRoyalChild: false),
            "a king of the lineage in another country cannot dirty this country");
        True(RoyalSuccessionEventRules.ShouldMarkSelectionDirty(
                pActorLineageId: 10L, pKingdomRoyalLineageId: 10L,
                pLineageReignsInThisKingdom: true,
                pRegisteredHeir: true, pDirectRoyalChild: false),
            "this kingdom's registered royal heir may request maintenance");
    }

    private static void ValidRegisteredHeirSkipsFallback()
    {
        Equal(false, AuthoritativeSuccessionRules.ShouldRunFallback(
                pHasValidRegisteredHeir: true,
                pFallbackAlreadyAttempted: false),
            "valid registered heir never recomputes");
    }

    private static void InvalidRegisteredHeirRunsFallbackOncePerDeath()
    {
        True(AuthoritativeSuccessionRules.ShouldRunFallback(
                pHasValidRegisteredHeir: false,
                pFallbackAlreadyAttempted: false),
            "missing registered heir runs the existing fallback once");
        Equal(false, AuthoritativeSuccessionRules.ShouldRunFallback(
                pHasValidRegisteredHeir: false,
                pFallbackAlreadyAttempted: true),
            "an attempted death does not repeat fallback selection");

        var attempts = new SuccessionFallbackAttemptState();
        var key = new KingSuccessionKey(7L, 100L, 900L);
        True(attempts.TryBegin(key), "the first invalid-heir fallback runs");
        Equal(false, attempts.TryBegin(key),
            "the same death cannot run fallback twice");
        attempts.Complete(key);
        True(attempts.TryBegin(key),
            "successful installation releases the completed death key");
    }

    private static void WorldGenerationSeparatesFallbackAttempts()
    {
        var attempts = new SuccessionFallbackAttemptState();
        True(attempts.TryBegin(new KingSuccessionKey(7L, 100L, 900L)),
            "the first world accepts the death key");
        True(attempts.TryBegin(new KingSuccessionKey(8L, 100L, 900L)),
            "a new world generation is independent");
    }

    private static void UnchangedHeirSignatureSkipsMaintenance()
    {
        True(HeirSelectionSignatureRules.IsUnchanged(
                202L, "registered", 101L, false,
                202L, "registered", 101L),
            "an identical clean heir signature is a no-op");
        Equal(false, HeirSelectionSignatureRules.IsUnchanged(
                202L, "registered", 101L, true,
                202L, "registered", 101L),
            "a dirty selection is recomputed");
    }

    private static void True(bool value, string name)
    {
        if (!value) throw new InvalidOperationException(name);
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(
                name + ": expected " + expected + ", got " + actual);
    }
}
