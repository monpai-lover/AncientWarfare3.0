    private static void FemaleSuccessionLawInvalidatesCachedHeir()
    {
        string heirSource = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "HeirService.cs"));
        int reconcileStart = heirSource.IndexOf(
            "public static Actor ReconcileHeir(", StringComparison.Ordinal);
        int peekStart = heirSource.IndexOf(
            "public static Actor PeekRegisteredHeir(", StringComparison.Ordinal);
        int peekEnd = heirSource.IndexOf(
            "public static Actor PeekStoredHeirForMinimap(", peekStart,
            StringComparison.Ordinal);
        int minimapEnd = heirSource.IndexOf(
            "public static Actor FindHeirReadOnly(", peekEnd,
            StringComparison.Ordinal);
        True(reconcileStart >= 0 && peekStart >= 0 && peekEnd > peekStart &&
             minimapEnd > peekEnd,
            "female succession cache validation source is present");

        string reconcile = heirSource.Substring(reconcileStart,
            peekStart - reconcileStart);
        True(reconcile.Contains(
                "IsSuccessionSexEligible(cached, pKingdom)"),
            "a cached female heir is not eligible after the law is disabled");

        string peek = heirSource.Substring(peekStart, peekEnd - peekStart);
        True(peek.Contains("IsSuccessionSexEligible(heir, pKingdom)"),
            "fast registered-heir reads reject a female heir under a male-only law");

        string minimap = heirSource.Substring(peekEnd,
            minimapEnd - peekEnd);
        True(minimap.Contains("IsSuccessionSexEligible(heir, pKingdom)"),
            "map and window heir reads reject a female heir under a male-only law");

        string courtSource = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "court",
            "CourtAuxiliaryLawService.cs"));
        int changeStart = courtSource.IndexOf(
            "public static CourtAuxiliaryLawChangeResult TryChangeLaw(",
            StringComparison.Ordinal);
        int changeEnd = courtSource.IndexOf(
            "public static int ResolveTermEndYear(", changeStart,
            StringComparison.Ordinal);
        True(changeStart >= 0 && changeEnd > changeStart,
            "auxiliary law change source is present");
        string change = courtSource.Substring(changeStart,
            changeEnd - changeStart);
        True(change.Contains(
                "pKind == CourtAuxiliaryLawKind.FemaleSuccession") &&
             change.Contains("HeirService.RefreshHeir(pKingdom)"),
            "changing female succession law immediately refreshes the heir");

        int storeStart = heirSource.IndexOf(
            "public static void StoreSelectedHeir(", StringComparison.Ordinal);
        int storeEnd = heirSource.IndexOf(
            "public static string ResolveSuccessionModeForCandidate(",
            storeStart, StringComparison.Ordinal);
        True(storeStart >= 0 && storeEnd > storeStart,
            "explicit heir registration source is present");
        string store = heirSource.Substring(storeStart, storeEnd - storeStart);
        True(store.Contains("IsSuccessionSexEligible(pHeir, pKingdom)"),
            "explicit hereditary heir registration applies the female succession law");
    }

            "HeirService.RememberPreSuccessionKing(__state.DyingKingdom, __instance);",

        True(snapshot > dyingKing,
            "the dying king stores its AW identity snapshot before death mutation");

            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "AuthoritativeSuccessionService.cs"));

            "private static bool TryResolveLegitimateContext(",

            "TryReadPreSuccessionSnapshot(", context,
            StringComparison.Ordinal);

            "ResolveIdentity(pPredecessorId)", context,

            "private static bool TryReadPreSuccessionSnapshot(",
            StringComparison.Ordinal);
        string snapshotBody = authoritative.Substring(snapshotMethod,
            authoritative.IndexOf(
                "private static bool TryResolvePredecessorGeneration(",
                snapshotMethod, StringComparison.Ordinal) - snapshotMethod);
        True(snapshotBody.Contains("int.MinValue") &&
             snapshotBody.Contains("pGeneration == int.MinValue"),

    private static void AuthoritativeFallbackConsumesArchiveEvidenceWithoutFalseUsurpation()
    {
        string source = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "AuthoritativeSuccessionService.cs"));
        int archiveResolve = source.IndexOf(
            "SuccessionArchiveFallbackService.Resolve(",
            StringComparison.Ordinal);
        True(archiveResolve >= 0,
            "authoritative succession calls the bounded AW archive fallback");
        Equal(false, source.Contains(
                "else if (SuccessionRelationshipIndex.IsReady)"),
            "a ready relationship index cannot short-circuit bounded archive candidate resolution");
        int found = source.IndexOf(
            "SuccessionEvidenceStatus.Found", archiveResolve,
            StringComparison.Ordinal);
        int store = source.IndexOf(
            "SuccessionMode.COLLATERAL_RESTORE", found,
            StringComparison.Ordinal);
        int restart = source.IndexOf(
            "SuccessionArchiveFallbackService.Restart(key)", store,
            StringComparison.Ordinal);
        int refresh = source.IndexOf("HeirService.RefreshHeir(pKingdom)",
            restart, StringComparison.Ordinal);
        True(found > archiveResolve && store > found,
            "unready relationship indexes consume the AW archive result");
        True(source.Contains("TryResolvePredecessorGeneration("),
            "archive fallback anchors its generation window to the live or archived predecessor");
        True(source.Contains("SuccessionRelationshipIndex.IsReady") &&
             source.Contains("SuccessionEvidenceStatus.PendingEvidence") &&
             source.Contains("SuccessionRelationshipIndex.HasLivingLineageMembers("),
            "terminal archive evidence waits for the live relationship index and reconciles wider living kin");
        int scanProgress = source.IndexOf("archive.ScanInProgress",
            archiveResolve, StringComparison.Ordinal);
        True(scanProgress > archiveResolve,
            "archive result exposes normal page progress");
        int scheduleNextFrame = source.IndexOf(
            "FallbackAttempts.ScheduleRetry(key, Time.frameCount, 1L)",
            scanProgress, StringComparison.Ordinal);
        True(scheduleNextFrame > scanProgress,
            "successful archive page progress resumes on the next frame");
        True(restart > store && refresh > restart,
            "failed archive registration resets evidence and reruns heir search");
    }

    private static void AssertOrdered(string source, params string[] tokens)
    {
        int previous = -1;
        foreach (string token in tokens)
        {
            int current = source.IndexOf(token, previous + 1,
                StringComparison.Ordinal);
            True(current > previous, "expected ordered source token: " + token);
            previous = current;
        }
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(
                name + ": expected " + expected + ", got " + actual);
    }
