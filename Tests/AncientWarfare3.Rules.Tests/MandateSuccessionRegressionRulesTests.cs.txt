    private static void ActiveMandateSuccessionRefreshesInstalledRuler()
    {
        Equal(true, MandateSuccessionRules.ShouldRefreshRulerProjection(
                active: true, trackedKingdomId: 17L, installedKingdomId: 17L,
                installedActorId: 42L, liveKingActorId: 42L),
            "the installed ruler immediately refreshes the active Mandate state");
        Equal(false, MandateSuccessionRules.ShouldRefreshRulerProjection(
                active: true, trackedKingdomId: 17L, installedKingdomId: 18L,
                installedActorId: 42L, liveKingActorId: 42L),
            "a foreign accession cannot rewrite the active Mandate state");
    }

    private static void MandateTraitMovesToTheInstalledRuler()
    {
        Equal(true, MandateSuccessionRules.ShouldTransferRulerTrait(
                projectionRefreshAccepted: true, previousActorId: 41L,
                installedActorId: 42L),
            "a real succession transfers the Mandate trait");
        Equal(false, MandateSuccessionRules.ShouldTransferRulerTrait(
                projectionRefreshAccepted: true, previousActorId: 42L,
                installedActorId: 42L),
            "an idempotent refresh does not transfer the trait from the ruler to itself");
    }

    private static void FailedPersistenceRetriesBeforeRuntimeProjection()
    {
        using var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        Execute(db,
            "CREATE TABLE MandateState(" +
            "STATE_ID INTEGER PRIMARY KEY,ACTIVE INTEGER," +
            "KINGDOM_ID INTEGER,PERIOD_ID INTEGER," +
            "EMPEROR_ACTOR_ID INTEGER,EMPEROR_NAME TEXT," +
            "DYNASTY_NAME TEXT," +
            "UPDATED_TIME REAL);" +
            "INSERT INTO MandateState VALUES" +
            "(1,1,17,91,41,'Old','Old Dynasty',10);" +
            "CREATE TRIGGER fail_mandate_succession BEFORE UPDATE ON " +
            "MandateState BEGIN SELECT RAISE(ABORT," +
            "'injected mandate succession failure'); END;");

        var calls = new List<string>();
        string error = "";
        bool first = MandateSuccessionRules.TryCommitRulerProjection(
            () =>
            {
                calls.Add("persist");
                return MandateSuccessionPersistence.TryRefreshRuler(
                    db, "MandateState", stateId: 1L, kingdomId: 17L,
                    periodId: 91L, rulerActorId: 42L,
                    rulerName: "New", dynastyName: "New Dynasty",
                    updatedTime: 11d, pError: out error);
            },
            () => calls.Add("runtime"));
        Equal(false, first,
            "a failed MandateState refresh rejects runtime projection");
        Equal(41L, Scalar(db,
                "SELECT EMPEROR_ACTOR_ID FROM MandateState WHERE STATE_ID=1"),
            "a failed refresh leaves the persisted emperor unchanged");
        Equal("Old Dynasty", ScalarText(db,
                "SELECT DYNASTY_NAME FROM MandateState WHERE STATE_ID=1"),
            "a failed refresh leaves the persisted dynasty unchanged");
        Equal("persist", string.Join(",", calls),
            "failure stops before title trait and display projection");

        Execute(db, "DROP TRIGGER fail_mandate_succession;");
        bool second = MandateSuccessionRules.TryCommitRulerProjection(
            () =>
            {
                calls.Add("persist");
                return MandateSuccessionPersistence.TryRefreshRuler(
                    db, "MandateState", stateId: 1L, kingdomId: 17L,
                    periodId: 91L, rulerActorId: 42L,
                    rulerName: "New", dynastyName: "New Dynasty",
                    updatedTime: 12d, pError: out error);
            },
            () => calls.Add("runtime"));
        Equal(true, second,
            "the same installed ruler can retry after persistence recovers");
        Equal(42L, Scalar(db,
                "SELECT EMPEROR_ACTOR_ID FROM MandateState WHERE STATE_ID=1"),
            "the successful retry persists the installed emperor");
        Equal("New Dynasty", ScalarText(db,
                "SELECT DYNASTY_NAME FROM MandateState WHERE STATE_ID=1"),
            "the successful retry persists the installed dynasty");
        Equal("persist,persist,runtime", string.Join(",", calls),
            "runtime projection commits once and only after persistence");
    }

    private static void
        AccessionSettlementRequiresOpenReignAndInstalledKing()
    {
        int phase = 0;
        bool missingReign = MandateAccessionCoordinator.TrySettle(
            () => false,
            () => throw new InvalidOperationException(
                "king identity checked before an open reign exists"),
            () => throw new InvalidOperationException(
                "Mandate committed without an open reign"));
        Equal(false, missingReign,
            "a missing open reign blocks Mandate succession");

        bool wrongKing = MandateAccessionCoordinator.TrySettle(
            () =>
            {
                Equal(0, phase, "open reign is ensured first");
                phase = 1;
                return true;
            },
            () =>
            {
                Equal(1, phase, "installed king is checked second");
                phase = 2;
                return false;
            },
            () => throw new InvalidOperationException(
                "Mandate committed for a stale king"));
        Equal(false, wrongKing,
            "a stale king blocks Mandate succession after reign creation");
        Equal(2, phase, "a rejected king never reaches Mandate commit");

        phase = 0;
        bool settled = MandateAccessionCoordinator.TrySettle(
            () =>
            {
                Equal(0, phase, "open reign is ensured first on success");
                phase = 1;
                return true;
            },
            () =>
            {
                Equal(1, phase, "installed king is confirmed second");
                phase = 2;
                return true;
            },
            () =>
            {
                Equal(2, phase, "Mandate commits only after both gates");
                phase = 3;
                return true;
            });
        Equal(true, settled, "a valid accession settles successfully");
        Equal(3, phase, "all accession phases execute in order");
    }

    private static void
        DynastyTransitionStatusSeparatesContinuityFromFailure()
    {
        Equal(true, DynastyTransitionRules.TryResolve(
                DynastyTransitionStatus.NoChange,
                out bool continuityCreated),
            "dynastic continuity is a successful no-change result");
        Equal(false, continuityCreated,
            "dynastic continuity does not claim a new dynasty");
        Equal(true, DynastyTransitionRules.TryResolve(
                DynastyTransitionStatus.Created,
                out bool created),
            "a created dynasty is a successful transition result");
        Equal(true, created,
            "a created transition reports the new dynasty");
        Equal(true, DynastyTransitionRules.ShouldProjectCreatedDynasty(
                DynastyTransitionStatus.Created),
            "a created dynasty must be projected onto the new reign");
        Equal(false, DynastyTransitionRules.ShouldProjectCreatedDynasty(
                DynastyTransitionStatus.NoChange),
            "no-change accession does not require a dynasty id");

        long noChangeLastKingId = 41L;
        bool noChangeSettled = MandateAccessionCoordinator.TrySettle(
            () => true,
            () =>
            {
                DynastyTransitionStatus status =
                    DynastyTransitionStatus.NoChange;
                Equal(true, DynastyTransitionRules.TryResolve(
                        status,
                        out bool noChangeCreated),
                    "no-change dynasty status is successful");
                Equal(false, noChangeCreated,
                    "no-change status reports no created dynasty");
                return !DynastyTransitionRules.
                    ShouldProjectCreatedDynasty(status);
            },
            () => true,
            () => true,
            () => noChangeLastKingId = 42L);
        Equal(true, noChangeSettled,
            "no-change accession with dynasty id -1 still settles");
        Equal(42L, noChangeLastKingId,
            "no-change accession advances the last-king projection");

        long lastKingId = 41L;
        int downstreamCalls = 0;
        bool settled = MandateAccessionCoordinator.TrySettle(
            () => true,
            () => DynastyTransitionRules.TryResolve(
                DynastyTransitionStatus.Failure, out _),
            () =>
            {
                downstreamCalls++;
                return true;
            },
            () =>
            {
                downstreamCalls++;
                return true;
            },
            () => lastKingId = 42L);
        Equal(false, settled,
            "a dynasty persistence failure blocks accession settlement");
        Equal(0, downstreamCalls,
            "dynasty failure stops before installed-king and Mandate work");
        Equal(41L, lastKingId,
            "dynasty failure cannot advance the last-king projection");
    }

    private static void
        FailedReignTransitionRetriesBeforeProjectionAdvances()
    {
        using var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        Execute(db,
            "CREATE TABLE KingdomReign(" +
            "REIGN_ID INTEGER PRIMARY KEY,KINGDOM_ID INTEGER," +
            "KINGDOM_COLOR TEXT,KING_ACTOR_ID INTEGER,SHI_ID INTEGER," +
            "DYNASTY_ID INTEGER,MANDATE_PERIOD_ID INTEGER," +
            "HIGHEST_TITLE INTEGER,STATE_NAME_SNAPSHOT TEXT," +
            "KING_NAME TEXT,KING_COLOR TEXT,REIGN_INDEX INTEGER," +
            "START_TIME REAL,END_TIME REAL,YEAR_NAME_STEM TEXT," +
            "YEAR_NAME_COLOR TEXT,END_REASON TEXT," +
            "START_POPULATION INTEGER,START_CITY_COUNT INTEGER," +
            "START_ARMY_COUNT INTEGER,END_POPULATION INTEGER," +
            "END_CITY_COUNT INTEGER,END_ARMY_COUNT INTEGER," +
            "IS_FOUNDER INTEGER,WAR_WINS INTEGER,WAR_LOSSES INTEGER," +
            "LOST_CAPITAL INTEGER,DEATH_CAUSE TEXT);" +
            "INSERT INTO KingdomReign VALUES" +
            "(1,17,'old',41,1,1,-1,2,'Old State','Old King','old',1," +
            "10,-1,'','','',100,2,3,0,0,0,1,0,0,0,'');" +
            "CREATE TRIGGER fail_old_reign_close BEFORE UPDATE ON " +
            "KingdomReign WHEN OLD.REIGN_ID=1 BEGIN SELECT RAISE(ABORT," +
            "'injected old reign close failure'); END;");

        var request = new ReignAccessionPersistence.Request
        {
            KingdomId = 17L,
            NewReignId = 2L,
            NewRulerActorId = 42L,
            NewKingdomColor = "new",
            NewShiId = 2L,
            NewDynastyId = 2L,
            NewMandatePeriodId = -1L,
            NewHighestTitle = 2,
            NewStateName = "New State",
            NewRulerName = "New King",
            NewRulerColor = "new",
            NewReignIndex = 2,
            NewStartTime = 11d,
            NewYearNameStem = "",
            NewYearNameColor = "new",
            NewStartPopulation = 110,
            NewStartCityCount = 3,
            NewStartArmyCount = 4,
            NewIsFounder = 0,
            OldEndTime = 11d,
            OldEndReason = "replaced",
            OldEndPopulation = 110,
            OldEndCityCount = 3,
            OldEndArmyCount = 4,
            OldWarWins = 0,
            OldWarLosses = 0,
            OldLostCapital = 0,
            OldHighestTitle = 2,
            OldDeathCause = ""
        };
        long lastKingId = 41L;
        long mandateActorId = 41L;
        string error = "";
        bool first = MandateAccessionCoordinator.TrySettle(
            () => ReignAccessionPersistence.TryTransition(
                db, "KingdomReign", request, out error),
            () => true,
            () =>
            {
                mandateActorId = 42L;
                return true;
            },
            () => lastKingId = 42L);
        Equal(false, first,
            "a failed old-reign close aborts accession settlement");
        Equal(41L, lastKingId,
            "a failed reign transition does not advance last king id");
        Equal(41L, mandateActorId,
            "a failed reign transition does not commit Mandate succession");
        Equal(-1L, Scalar(db,
                "SELECT END_TIME FROM KingdomReign WHERE REIGN_ID=1"),
            "the old reign remains open after rollback");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM KingdomReign WHERE REIGN_ID=2"),
            "the new reign is not half-created after rollback");

        Execute(db, "DROP TRIGGER fail_old_reign_close;");
        bool second = MandateAccessionCoordinator.TrySettle(
            () => ReignAccessionPersistence.TryTransition(
                db, "KingdomReign", request, out error),
            () => false,
            () => true,
            () =>
            {
                mandateActorId = 42L;
                return true;
            },
            () => lastKingId = 42L);
        Equal(false, second,
            "a failed dynasty projection blocks the committed reign transition");
        Equal(41L, lastKingId,
            "a failed dynasty projection does not advance last king id");
        Equal(41L, mandateActorId,
            "a failed dynasty projection does not commit Mandate succession");

        int dynastyPublishes = 0;
        int stateNamePublishes = 0;
        int historyPublishes = 0;
        bool third = MandateAccessionCoordinator.TrySettle(
            () => ReignAccessionPersistence.TryTransition(
                db, "KingdomReign", request, out error),
            () =>
            {
                dynastyPublishes++;
                stateNamePublishes++;
                historyPublishes++;
                return true;
            },
            () => true,
            () =>
            {
                mandateActorId = 42L;
                return true;
            },
            () => lastKingId = 42L);
        Equal(true, third,
            "the same accession succeeds after dynasty projection recovers");
        Equal(42L, lastKingId,
            "last king id advances only after the full settlement");
        Equal(42L, mandateActorId,
            "Mandate succession commits to the installed new king");
        Equal(11L, Scalar(db,
                "SELECT END_TIME FROM KingdomReign WHERE REIGN_ID=1"),
            "the successful retry closes the old reign");
        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM KingdomReign WHERE KINGDOM_ID=17 " +
                "AND KING_ACTOR_ID=42 AND END_TIME=-1"),
            "the successful retry creates one new-ruler open reign");
        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM KingdomReign WHERE KINGDOM_ID=17 " +
                "AND END_TIME=-1"),
            "the successful retry leaves a unique open reign");
        Equal(1, dynastyPublishes,
            "dynasty projection publishes exactly once after the DB gate");
        Equal(1, stateNamePublishes,
            "state-name projection publishes exactly once after the DB gate");
        Equal(1, historyPublishes,
            "accession history publishes exactly once after the DB gate");
    }

    private static void
        MandateContextTargetsOnlyTheInstalledRulersLatestOpenReign()
    {
        using var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        Execute(db,
            "CREATE TABLE KingdomReign(" +
            "REIGN_ID INTEGER PRIMARY KEY,KINGDOM_ID INTEGER," +
            "KING_ACTOR_ID INTEGER,DYNASTY_ID INTEGER," +
            "MANDATE_PERIOD_ID INTEGER," +
            "HIGHEST_TITLE INTEGER,STATE_NAME_SNAPSHOT TEXT," +
            "START_TIME REAL,END_TIME REAL);" +
            "INSERT INTO KingdomReign VALUES" +
            "(1,17,41,7,-1,2,'Old ruler',10,-1)," +
            "(2,17,42,7,-1,2,'Current ruler',11,-1)," +
            "(3,17,42,7,-1,2,'Older duplicate',9,-1);");

        bool projected = ReignMandateProjectionPersistence.TryProject(
            db, "KingdomReign", kingdomId: 17L, rulerActorId: 42L,
            mandatePeriodId: 91L, emperorTitle: 4,
            stateName: "Imperial State", out string error);
        Equal(true, projected,
            "the installed ruler's deterministic open reign is projected");
        Equal(91L, Scalar(db,
                "SELECT MANDATE_PERIOD_ID FROM KingdomReign WHERE REIGN_ID=2"),
            "the latest installed-ruler reign receives the Mandate period");
        Equal(4L, Scalar(db,
                "SELECT HIGHEST_TITLE FROM KingdomReign WHERE REIGN_ID=2"),
            "the latest installed-ruler reign receives imperial rank");
        Equal("Imperial State", ScalarText(db,
                "SELECT STATE_NAME_SNAPSHOT FROM KingdomReign " +
                "WHERE REIGN_ID=2"),
            "the latest installed-ruler reign receives the state snapshot");
        Equal(-1L, Scalar(db,
                "SELECT MANDATE_PERIOD_ID FROM KingdomReign WHERE REIGN_ID=1"),
            "the previous ruler's ordinary open reign remains unchanged");
        Equal(2L, Scalar(db,
                "SELECT HIGHEST_TITLE FROM KingdomReign WHERE REIGN_ID=1"),
            "the previous ruler keeps its ordinary historical rank");
        Equal("Old ruler", ScalarText(db,
                "SELECT STATE_NAME_SNAPSHOT FROM KingdomReign " +
                "WHERE REIGN_ID=1"),
            "the previous ruler keeps its historical state snapshot");
        Equal(-1L, Scalar(db,
                "SELECT MANDATE_PERIOD_ID FROM KingdomReign WHERE REIGN_ID=3"),
            "an older duplicate open reign remains unchanged");

        Equal(true, ReignMandateProjectionPersistence.TryProjectDynasty(
                db, "KingdomReign", kingdomId: 17L, rulerActorId: 42L,
                dynastyId: 9L, out error),
            "the installed ruler's reign accepts the created dynasty id");
        Equal(9L, Scalar(db,
                "SELECT DYNASTY_ID FROM KingdomReign WHERE REIGN_ID=2"),
            "post-dynasty projection fixes the new reign attribution");
        Equal(7L, Scalar(db,
                "SELECT DYNASTY_ID FROM KingdomReign WHERE REIGN_ID=1"),
            "post-dynasty projection leaves the prior ruler unchanged");
        Equal(7L, Scalar(db,
                "SELECT DYNASTY_ID FROM KingdomReign WHERE REIGN_ID=3"),
            "post-dynasty projection leaves older duplicate rows unchanged");

        bool missing = ReignMandateProjectionPersistence.TryProject(
            db, "KingdomReign", kingdomId: 17L, rulerActorId: 99L,
            mandatePeriodId: 91L, emperorTitle: 4,
            stateName: "Imperial State", out error);
        Equal(false, missing,
            "a missing installed-ruler open reign reports failure");
        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM KingdomReign " +
                "WHERE MANDATE_PERIOD_ID=91"),
            "a failed projection changes no unrelated reign");
    }

    private static void FailedMandateDeclarationRollsBackAndRetriesOnce()
    {
        using var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        Execute(db,
            "CREATE TABLE MandatePeriod(" +
            "PERIOD_ID INTEGER PRIMARY KEY,KINGDOM_ID INTEGER," +
            "KINGDOM_NAME TEXT,KINGDOM_COLOR TEXT,DYNASTY_NAME TEXT," +
            "FOUNDER_ACTOR_ID INTEGER,FOUNDER_NAME TEXT,START_TIME REAL," +

            "END_MANDATE INTEGER,LEGAL_CORE_COUNT INTEGER," +
            "ORIGIN_TYPE TEXT,REBEL_ORIGIN_KINGDOM_ID INTEGER," +
            "REBEL_ORIGIN_KINGDOM_NAME TEXT,CLAIMANT_KIND TEXT);" +

            "STATE_ID INTEGER PRIMARY KEY,ACTIVE INTEGER," +
            "KINGDOM_ID INTEGER,KINGDOM_NAME TEXT,KINGDOM_COLOR TEXT," +
            "DYNASTY_NAME TEXT,EMPEROR_ACTOR_ID INTEGER," +
            "EMPEROR_NAME TEXT,PERIOD_ID INTEGER,MANDATE_VALUE INTEGER," +
            "IMPERIAL_AUTHORITY INTEGER,DYNASTY_PRESTIGE INTEGER," +
            "CORE_CONTROL REAL,VASSAL_LOYALTY REAL,CRISIS_LEVEL TEXT," +
            "START_TIME REAL,UPDATED_TIME REAL,LAST_YEAR INTEGER," +
            "ORIGIN_TYPE TEXT,ORIGINAL_CORE_COUNT INTEGER," +
            "REBEL_ORIGIN_KINGDOM_ID INTEGER," +
            "REBEL_ORIGIN_KINGDOM_NAME TEXT,CLAIMANT_KIND TEXT," +
            "MAP_MARKER_KIND TEXT);" +
            "INSERT INTO MandateState(STATE_ID,ACTIVE,KINGDOM_ID," +
            "PERIOD_ID) VALUES(1,0,-1,-1);" +
            "CREATE TABLE KingdomReign(" +
            "REIGN_ID INTEGER PRIMARY KEY,KINGDOM_ID INTEGER," +
            "KING_ACTOR_ID INTEGER,MANDATE_PERIOD_ID INTEGER," +
            "HIGHEST_TITLE INTEGER,STATE_NAME_SNAPSHOT TEXT," +
            "START_TIME REAL,END_TIME REAL);" +
            "CREATE TRIGGER fail_declaration_reign BEFORE UPDATE ON " +
            "KingdomReign BEGIN SELECT RAISE(ABORT," +
            "'injected declaration reign failure'); END;");

        var request = new MandateDeclarationPersistence.Request
        {
            StateId = 1L,
            PeriodId = 91L,
            OperationKey = "mandate-declare:91",
            KingdomId = 17L,
            KingdomName = "Imperial State",
            KingdomColor = "#abcdef",
            DynastyName = "New Dynasty",

            RulerName = "New Emperor",

            CurrentYear = 101,

            ImperialAuthority = 45,

            OriginType = "native",
            RebelOriginKingdomId = -1L,
            RebelOriginKingdomName = "",

            MapMarkerKind = "moh",

                CityName = "Founding City",
                OriginalKingdomId = 17L,
                OriginalKingdomName = "Imperial State",
                OriginalKingdomColor = "#abcdef",
                CoreType = "founding"
            });
        int historyPublishes = 0;
        int traitPublishes = 0;
        int runtimePublishes = 0;
        string error = "";

                runtimePublishes++;

        Equal(0L, Scalar(db,
                "SELECT ACTIVE FROM MandateState WHERE STATE_ID=1"),
            "failed declaration leaves no active Mandate state");
        Equal(0L, Scalar(db, "SELECT COUNT(*) FROM KingdomReign"),
            "zero-row failure does not invent a reign");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM sqlite_master WHERE type='table' " +
                "AND name='MandateProjectionCoreSnapshot'"),
            "failed declaration creates no core snapshot schema or rows");
        Equal(0, historyPublishes,
            "failed declaration publishes no history");
        Equal(0, traitPublishes,
            "failed declaration publishes no trait");
        Equal(0, runtimePublishes,
            "failed declaration publishes no runtime projection");

        Execute(db,
            "INSERT INTO KingdomReign VALUES" +
            "(7,17,42,-1,2,'Ordinary State',11,-1);");

            MandateDeclarationRules.TryCommitProjection(
                () => MandateDeclarationPersistence.TryCommit(

        Equal(0L, Scalar(db,
                "SELECT ACTIVE FROM MandateState WHERE STATE_ID=1"),

        Equal(-1L, Scalar(db,
                "SELECT MANDATE_PERIOD_ID FROM KingdomReign WHERE REIGN_ID=7"),
            "exceptional declaration leaves the ordinary reign unchanged");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM sqlite_master WHERE type='table' " +
                "AND name='MandateProjectionCoreSnapshot'"),
            "exceptional declaration leaves no core snapshot rows");
        Equal(0, historyPublishes,
            "exceptional declaration publishes no history");
        Equal(0, traitPublishes,
            "exceptional declaration publishes no trait");
        Equal(0, runtimePublishes,
            "exceptional declaration publishes no runtime projection");

        Execute(db, "DROP TRIGGER fail_declaration_reign;");
        bool second = MandateDeclarationRules.TryCommitProjection(
            () => MandateDeclarationPersistence.TryCommit(
                db, "MandatePeriod", "MandateState", "KingdomReign",
                request, out error),
            () =>
            {
                historyPublishes++;
                traitPublishes++;
                runtimePublishes++;
            });
        Equal(true, second,
            "the declaration safely retries after persistence recovers");
        Equal(1L, Scalar(db, "SELECT COUNT(*) FROM MandatePeriod"),
            "successful retry creates exactly one period");
        Equal(1L, Scalar(db,
                "SELECT ACTIVE FROM MandateState WHERE STATE_ID=1"),
            "successful retry activates Mandate state");
        Equal(91L, Scalar(db,
                "SELECT MANDATE_PERIOD_ID FROM KingdomReign WHERE REIGN_ID=7"),
            "successful retry projects the ruler's reign");
        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM MandateProjectionCoreSnapshot"),
            "successful declaration commits its core snapshot once");
        Equal(1, historyPublishes,
            "successful retry publishes history once");
        Equal(1, traitPublishes,
            "successful retry publishes the trait once");
        Equal(1, runtimePublishes,
            "successful retry publishes runtime state once");

        bool duplicate = MandateDeclarationRules.TryCommitProjection(
            () => MandateDeclarationPersistence.TryCommit(

    private static void VerifyOutboxRetry(string failedEffect)
    {
        using var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        var pending = new MandateProjectionOutboxPersistence.PendingProjection
        {
            OperationKey = "mandate-declare:91:" + failedEffect,
            PeriodId = 91L,
            KingdomId = 17L,
            RulerActorId = 42L,
            PreviousPeriodId = 80L,
            PreviousKingdomId = 9L,
            PreviousMandateValue = 44,
            PreviousEndReason = "replaced",
            OldEndRequired = true,
            CurrentYear = 101,
            WasAlreadyEmperor = false,
            CreatedTime = 12d
        };
        using (SQLiteTransaction transaction = db.BeginTransaction())
        {
            bool enqueued = MandateProjectionOutboxPersistence.TryEnqueue(
                db, transaction, pending, out string enqueueError);
            Equal(true, enqueued, "declaration outbox is transactionally queued");
            transaction.Commit();
        }

        var successes = new Dictionary<string, int>();
        bool failedOnce = false;
        bool first = MandateProjectionOutboxPersistence.TryDrain(
            db, pending.OperationKey,
            effect =>
            {
                if (effect == failedEffect && !failedOnce)
                {
                    failedOnce = true;
                    return false;
                }
                successes[effect] = successes.TryGetValue(effect,
                    out int count) ? count + 1 : 1;
                return true;
            }, out bool firstComplete, out string error);
        Equal(false, first,
            failedEffect + " failure remains retryable");
        Equal(false, firstComplete,
            failedEffect + " failure keeps the operation pending");
        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM MandateProjectionOutbox " +
                "WHERE OPERATION_KEY='" + pending.OperationKey + "'"),
            failedEffect + " failure keeps its durable pending row");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM MandateProjectionEffect " +
                "WHERE OPERATION_KEY='" + pending.OperationKey +
                "' AND EFFECT_KEY='" + failedEffect + "'"),
            failedEffect + " failure does not mark history/core complete");

        bool second = MandateProjectionOutboxPersistence.TryDrain(
            db, pending.OperationKey,
            effect =>
            {
                successes[effect] = successes.TryGetValue(effect,
                    out int count) ? count + 1 : 1;
                return true;
            }, out bool secondComplete, out error);
        Equal(true, second, failedEffect + " retry drains successfully");
        Equal(true, secondComplete,
            failedEffect + " retry clears the pending operation");
        foreach (string effect in
                 MandateProjectionOutboxPersistence.RequiredEffects(true))
            Equal(1, successes.TryGetValue(effect, out int count) ? count : 0,
                effect + " publishes exactly once");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM MandateProjectionOutbox " +
                "WHERE OPERATION_KEY='" + pending.OperationKey + "'"),
            "completed outbox is no longer pending");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM MandateProjectionEffect " +
                "WHERE OPERATION_KEY='" + pending.OperationKey + "'"),
            "completed outbox removes its effect ledger rows");

        bool third = MandateProjectionOutboxPersistence.TryDrain(
            db, pending.OperationKey,
            effect => throw new InvalidOperationException(
                "completed effect reran: " + effect),
            out bool thirdComplete, out error);
        Equal(true, third, "completed outbox drain is a no-op");
        Equal(true, thirdComplete, "completed outbox remains complete");
    }

    private static void
        OutboxReplayAfterBusinessCommitDoesNotDuplicateEffects()
    {
        using var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        Execute(db,
            "CREATE TABLE KingdomHistory(" +
            "ID INTEGER PRIMARY KEY,PROJECTION_KEY TEXT UNIQUE);" +
            "CREATE TABLE PersonBiography(" +
            "ID INTEGER PRIMARY KEY,PROJECTION_KEY TEXT UNIQUE);" +
            "CREATE TABLE MandateEvent(" +
            "ID INTEGER PRIMARY KEY,PROJECTION_KEY TEXT UNIQUE);" +
            "CREATE TABLE MandateCoreCity(" +
            "ID INTEGER PRIMARY KEY,PROJECTION_KEY TEXT UNIQUE);");
        var pending = new MandateProjectionOutboxPersistence.PendingProjection
        {
            OperationKey = "mandate-declare:91:crash-window",
            PeriodId = 91L,
            KingdomId = 17L,
            RulerActorId = 42L,
            PreviousPeriodId = 80L,
            PreviousKingdomId = 9L,
            OldEndRequired = true,
            CurrentYear = 101,
            CreatedTime = 12d
        };
        using (SQLiteTransaction transaction = db.BeginTransaction())
        {
            Equal(true, MandateProjectionOutboxPersistence.TryEnqueue(
                    db, transaction, pending, out string enqueueError),
                "crash-window outbox is queued");
            transaction.Commit();
        }

        var tables = new Dictionary<string, string>
        {
            ["old_kingdom_history"] = "KingdomHistory",
            ["old_mandate_event"] = "MandateEvent",
            ["new_mandate_event"] = "MandateEvent",
            ["new_kingdom_history"] = "KingdomHistory",
            ["new_person_history"] = "PersonBiography",
            ["legal_cores"] = "MandateCoreCity"
        };
        int businessWrites = 0;
        foreach ((string effect, string table) in tables)
            Equal(true, ApplyProjectionRecord(db, table,
                    pending.OperationKey + ":" + effect,
                    ref businessWrites),
                effect + " business effect commits before its marker");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM MandateProjectionEffect WHERE " +
                "OPERATION_KEY='" + pending.OperationKey + "'"),
            "simulated crash leaves every outbox marker missing");

        bool drained = MandateProjectionOutboxPersistence.TryDrain(
            db, pending.OperationKey,
            effect => effect != "new_maps" &&
                      (!tables.TryGetValue(effect, out string table) ||
                      ApplyProjectionRecord(db, table,
                          pending.OperationKey + ":" + effect,
                          ref businessWrites)),
            out bool complete, out string error);

        Equal(false, drained,
            "the injected final effect failure keeps repaired markers visible");
        Equal(false, complete,
            "the final effect failure keeps the outbox pending");
        Equal(6, businessWrites,
            "replay does not execute any durable business insert twice");
        Equal(2L, Scalar(db, "SELECT COUNT(*) FROM KingdomHistory"),
            "kingdom history remains exactly once per effect");
        Equal(1L, Scalar(db, "SELECT COUNT(*) FROM PersonBiography"),
            "person history remains exactly once");
        Equal(2L, Scalar(db, "SELECT COUNT(*) FROM MandateEvent"),
            "Mandate events remain exactly once per effect");
        Equal(1L, Scalar(db, "SELECT COUNT(*) FROM MandateCoreCity"),
            "legal core remains exactly once");
        foreach (string effect in tables.Keys)
            Equal(1L, Scalar(db,
                    "SELECT COUNT(*) FROM MandateProjectionEffect WHERE " +
                    "OPERATION_KEY='" + pending.OperationKey +
                    "' AND EFFECT_KEY='" + effect + "'"),
                effect + " marker is repaired after idempotent replay");

        bool finished = MandateProjectionOutboxPersistence.TryDrain(
            db, pending.OperationKey, effect => effect == "new_maps",
            out bool finishedComplete, out error);
        Equal(true, finished,
            "repaired durable effects let the final retry converge");
        Equal(true, finishedComplete,
            "the final retry completes the outbox");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM MandateProjectionOutbox WHERE " +
                "OPERATION_KEY='" + pending.OperationKey + "'"),
            "converged crash-window outbox removes its pending row");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM MandateProjectionEffect WHERE " +
                "OPERATION_KEY='" + pending.OperationKey + "'"),
            "converged crash-window outbox removes its marker rows");
    }

    private static bool ApplyProjectionRecord(SQLiteConnection db,
        string table, string projectionKey, ref int businessWrites)
    {
        int writes = businessWrites;
        bool applied = MandateProjectionOutboxPersistence.
            TryApplyIdempotentRecord(db, table, projectionKey,
                transaction =>
                {
                    writes++;
                    using var command = new SQLiteCommand(db)
                    {
                        Transaction = transaction,
                        CommandText = "INSERT INTO " + table +
                                      "(PROJECTION_KEY) VALUES(@key)"
                    };
                    command.Parameters.AddWithValue("@key", projectionKey);
                    return command.ExecuteNonQuery() == 1;
                }, out string error);
        businessWrites = writes;
        return applied && string.IsNullOrEmpty(error);
    }

    private static bool ApplyCoreProjectionRecord(SQLiteConnection db,
        long cityId, string projectionKey, ref int businessWrites)
    {
        int writes = businessWrites;
        bool applied = MandateProjectionOutboxPersistence.
            TryApplyIdempotentRecord(db, "MandateCoreCity", projectionKey,
                transaction =>
                {
                    writes++;
                    using var command = new SQLiteCommand(db)
                    {
                        Transaction = transaction,
                        CommandText = "INSERT INTO MandateCoreCity" +
                                      "(CITY_ID,PROJECTION_KEY) " +
                                      "VALUES(@city,@key)"
                    };
                    command.Parameters.AddWithValue("@city", cityId);
                    command.Parameters.AddWithValue("@key", projectionKey);
                    return command.ExecuteNonQuery() == 1;
                }, out string error);
        businessWrites = writes;
        return applied && string.IsNullOrEmpty(error);
    }

    private static void
        PendingProjectionResumeIsStableBoundedAndSnapshotBased()
    {
        using var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        EnqueuePending(db, Pending("resume-c", 93L, 2d));
        EnqueuePending(db, Pending("resume-b", 92L, 1d));
        EnqueuePending(db, Pending("resume-a", 91L, 1d));

        Equal(true, MandateProjectionOutboxPersistence.TryReadPendingBatch(
                db, 2, out List<MandateProjectionOutboxPersistence.
                    PendingProjection> firstBatch, out string error),
            "pending projection batch can be read");
        Equal(2, firstBatch.Count,
            "pending projection batch obeys its hard limit");
        Equal("resume-a", firstBatch[0].OperationKey,
            "pending projections sort equal times by period");
        Equal("resume-b", firstBatch[1].OperationKey,
            "pending projections use a stable period order");
        Equal("Founder 91", firstBatch[0].RulerName,
            "declaration ruler name survives in the outbox");
        Equal("Realm 91", firstBatch[0].KingdomName,
            "declaration kingdom name survives in the outbox");
        Equal("Dynasty 91", firstBatch[0].DynastyName,
            "declaration dynasty name survives in the outbox");
        Equal("Old Realm 91", firstBatch[0].PreviousKingdomName,
            "replaced kingdom snapshot survives in the outbox");
        Equal("native", firstBatch[0].OriginType,
            "declaration origin survives in the outbox");

        var resumed = new List<string>();
        Equal(true, MandateProjectionOutboxPersistence.
                TryResumePendingBatch(db, 2,
                    (pending, effect) =>
                    {
                        if (!resumed.Contains(pending.OperationKey))
                            resumed.Add(pending.OperationKey);
                        return true;
                    }, out int attempted, out int completed, out error),
            "load-style resume completes its bounded batch");
        Equal(2, attempted,
            "one resume cycle attempts no more than two operations");
        Equal(2, completed,
            "one resume cycle completes its first two operations");
        Equal("resume-a", resumed[0],
            "resume starts with the oldest stable operation");
        Equal("resume-b", resumed[1],
            "resume preserves stable operation order");
        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM MandateProjectionOutbox"),
            "bounded resume leaves later work for another cycle");
        Equal("resume-c", ScalarText(db,
                "SELECT OPERATION_KEY FROM MandateProjectionOutbox"),
            "bounded resume leaves the newest operation pending");
    }

    private static void SuppressedKeyedHistoryRemainsPendingUntilRetry()
    {
        using var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        Execute(db, "CREATE TABLE KingdomHistory(" +
                    "ID INTEGER PRIMARY KEY AUTOINCREMENT," +
                    "PROJECTION_KEY TEXT UNIQUE);");
        EnqueuePending(db, Pending("suppressed-history", 91L, 1d));

        bool timingAllowsRecord = false;
        int historyWrites = 0;
        bool Publish(MandateProjectionOutboxPersistence.PendingProjection
            pending, string effect)
        {
            if (effect != "new_kingdom_history") return true;
            string key = pending.OperationKey + ":new_kingdom_history";
            if (!timingAllowsRecord)
                return HistoryWriteTimingRules.
                    ResolveSuppressedWriteResult(key);
            return ApplyProjectionRecord(db, "KingdomHistory", key,
                ref historyWrites);
        }

        Equal(false, MandateProjectionOutboxPersistence.
                TryResumePendingBatch(db, 1, Publish,
                    out int attempted, out int completed,
                    out string error),
            "suppressed keyed history defers its outbox operation");
        Equal(1, attempted,
            "the suppressed operation is attempted once");
        Equal(0, completed,
            "suppression cannot complete the pending operation");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM KingdomHistory"),
            "suppression writes no history business row");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM MandateProjectionEffect " +
                "WHERE OPERATION_KEY='suppressed-history' " +
                "AND EFFECT_KEY='new_kingdom_history'"),
            "suppression writes no history effect marker");
        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM MandateProjectionOutbox"),
            "suppression retains the pending operation");
        Equal(true, !string.IsNullOrEmpty(error),
            "suppression remains an observable deferred result");

        timingAllowsRecord = true;
        Equal(true, MandateProjectionOutboxPersistence.
                TryResumePendingBatch(db, 1, Publish,
                    out attempted, out completed, out error),
            "keyed history resumes after suppression ends");
        Equal(1, completed,
            "the resumed operation completes exactly once");
        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM KingdomHistory"),
            "the resumed history writes one business row");
        Equal(1, historyWrites,
            "the resumed history publisher executes once");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM MandateProjectionOutbox"),
            "the resumed operation clears its pending row");
        Equal(0L, Scalar(db,
                "SELECT COUNT(*) FROM MandateProjectionEffect"),
            "completed operation cleanup removes its effect markers");
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(
                $"{name}: expected {expected}, got {actual}");
    }

    private static void Execute(SQLiteConnection db, string sql)
    {
        using var command = new SQLiteCommand(sql, db);
        command.ExecuteNonQuery();
    }

    private static long Scalar(SQLiteConnection db, string sql)
    {
        using var command = new SQLiteCommand(sql, db);
        return Convert.ToInt64(command.ExecuteScalar());
    }

    private static string ScalarText(SQLiteConnection db, string sql)
    {
        using var command = new SQLiteCommand(sql, db);
        return Convert.ToString(command.ExecuteScalar()) ?? "";
    }
