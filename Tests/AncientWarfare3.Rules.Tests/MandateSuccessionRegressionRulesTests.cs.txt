using System.Data.SQLite;
using AncientWarfare3.core.lineage;

internal static class MandateSuccessionRegressionRulesTests
{
    public static void Run()
    {
        ActiveMandateSuccessionRefreshesInstalledRuler();
        MandateTraitMovesToTheInstalledRuler();
        AccessionSettlementRequiresOpenReignAndInstalledKing();
        FailedPersistenceRetriesBeforeRuntimeProjection();
        MandateContextTargetsOnlyTheInstalledRulersLatestOpenReign();
        FailedMandateDeclarationRollsBackAndRetriesOnce();
        ActiveMandateKeepsItsImperialStateName();
        SavedMandateReignUsesImperialPosthumousRank();
    }

    private static void ActiveMandateSuccessionRefreshesInstalledRuler()
    {
        Equal(true, MandateSuccessionRules.ShouldRefreshRulerProjection(
                active: true, trackedKingdomId: 17L, installedKingdomId: 17L,
                installedActorId: 42L, liveKingActorId: 42L),
            "the installed ruler immediately refreshes the active Mandate state");
        Equal(false, MandateSuccessionRules.ShouldRefreshRulerProjection(
                active: true, trackedKingdomId: 17L, installedKingdomId: 18L,
                installedActorId: 42L, liveKingActorId: 42L),
            "a foreign accession cannot rewrite the active Mandate state");
    }

    private static void MandateTraitMovesToTheInstalledRuler()
    {
        Equal(true, MandateSuccessionRules.ShouldTransferRulerTrait(
                projectionRefreshAccepted: true, previousActorId: 41L,
                installedActorId: 42L),
            "a real succession transfers the Mandate trait");
        Equal(false, MandateSuccessionRules.ShouldTransferRulerTrait(
                projectionRefreshAccepted: true, previousActorId: 42L,
                installedActorId: 42L),
            "an idempotent refresh does not transfer the trait from the ruler to itself");
    }

    private static void FailedPersistenceRetriesBeforeRuntimeProjection()
    {
        using var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        Execute(db,
            "CREATE TABLE MandateState(" +
            "STATE_ID INTEGER PRIMARY KEY,ACTIVE INTEGER," +
            "KINGDOM_ID INTEGER,PERIOD_ID INTEGER," +
            "EMPEROR_ACTOR_ID INTEGER,EMPEROR_NAME TEXT," +
            "DYNASTY_NAME TEXT," +
            "UPDATED_TIME REAL);" +
            "INSERT INTO MandateState VALUES" +
            "(1,1,17,91,41,'Old','Old Dynasty',10);" +
            "CREATE TRIGGER fail_mandate_succession BEFORE UPDATE ON " +
            "MandateState BEGIN SELECT RAISE(ABORT," +
            "'injected mandate succession failure'); END;");

        var calls = new List<string>();
        string error = "";
        bool first = MandateSuccessionRules.TryCommitRulerProjection(
            () =>
            {
                calls.Add("persist");
                return MandateSuccessionPersistence.TryRefreshRuler(
                    db, "MandateState", stateId: 1L, kingdomId: 17L,
                    periodId: 91L, rulerActorId: 42L,
                    rulerName: "New", dynastyName: "New Dynasty",
                    updatedTime: 11d, pError: out error);
            },
            () => calls.Add("runtime"));
        Equal(false, first,
            "a failed MandateState refresh rejects runtime projection");
        Equal(41L, Scalar(db,
                "SELECT EMPEROR_ACTOR_ID FROM MandateState WHERE STATE_ID=1"),
            "a failed refresh leaves the persisted emperor unchanged");
        Equal("Old Dynasty", ScalarText(db,
                "SELECT DYNASTY_NAME FROM MandateState WHERE STATE_ID=1"),
            "a failed refresh leaves the persisted dynasty unchanged");
        Equal("persist", string.Join(",", calls),
            "failure stops before title trait and display projection");

        Execute(db, "DROP TRIGGER fail_mandate_succession;");
        bool second = MandateSuccessionRules.TryCommitRulerProjection(
            () =>
            {
                calls.Add("persist");
                return MandateSuccessionPersistence.TryRefreshRuler(
                    db, "MandateState", stateId: 1L, kingdomId: 17L,
                    periodId: 91L, rulerActorId: 42L,
                    rulerName: "New", dynastyName: "New Dynasty",
                    updatedTime: 12d, pError: out error);
            },
            () => calls.Add("runtime"));
        Equal(true, second,
            "the same installed ruler can retry after persistence recovers");
        Equal(42L, Scalar(db,
                "SELECT EMPEROR_ACTOR_ID FROM MandateState WHERE STATE_ID=1"),
            "the successful retry persists the installed emperor");
        Equal("New Dynasty", ScalarText(db,
                "SELECT DYNASTY_NAME FROM MandateState WHERE STATE_ID=1"),
            "the successful retry persists the installed dynasty");
        Equal("persist,persist,runtime", string.Join(",", calls),
            "runtime projection commits once and only after persistence");
    }

    private static void
        AccessionSettlementRequiresOpenReignAndInstalledKing()
    {
        int phase = 0;
        bool missingReign = MandateAccessionCoordinator.TrySettle(
            () => false,
            () => throw new InvalidOperationException(
                "king identity checked before an open reign exists"),
            () => throw new InvalidOperationException(
                "Mandate committed without an open reign"));
        Equal(false, missingReign,
            "a missing open reign blocks Mandate succession");

        bool wrongKing = MandateAccessionCoordinator.TrySettle(
            () =>
            {
                Equal(0, phase, "open reign is ensured first");
                phase = 1;
                return true;
            },
            () =>
            {
                Equal(1, phase, "installed king is checked second");
                phase = 2;
                return false;
            },
            () => throw new InvalidOperationException(
                "Mandate committed for a stale king"));
        Equal(false, wrongKing,
            "a stale king blocks Mandate succession after reign creation");
        Equal(2, phase, "a rejected king never reaches Mandate commit");

        phase = 0;
        bool settled = MandateAccessionCoordinator.TrySettle(
            () =>
            {
                Equal(0, phase, "open reign is ensured first on success");
                phase = 1;
                return true;
            },
            () =>
            {
                Equal(1, phase, "installed king is confirmed second");
                phase = 2;
                return true;
            },
            () =>
            {
                Equal(2, phase, "Mandate commits only after both gates");
                phase = 3;
                return true;
            });
        Equal(true, settled, "a valid accession settles successfully");
        Equal(3, phase, "all accession phases execute in order");
    }

    private static void
        MandateContextTargetsOnlyTheInstalledRulersLatestOpenReign()
    {
        using var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        Execute(db,
            "CREATE TABLE KingdomReign(" +
            "REIGN_ID INTEGER PRIMARY KEY,KINGDOM_ID INTEGER," +
            "KING_ACTOR_ID INTEGER,MANDATE_PERIOD_ID INTEGER," +
            "HIGHEST_TITLE INTEGER,STATE_NAME_SNAPSHOT TEXT," +
            "START_TIME REAL,END_TIME REAL);" +
            "INSERT INTO KingdomReign VALUES" +
            "(1,17,41,-1,2,'Old ruler',10,-1)," +
            "(2,17,42,-1,2,'Current ruler',11,-1)," +
            "(3,17,42,-1,2,'Older duplicate',9,-1);");

        bool projected = ReignMandateProjectionPersistence.TryProject(
            db, "KingdomReign", kingdomId: 17L, rulerActorId: 42L,
            mandatePeriodId: 91L, emperorTitle: 4,
            stateName: "Imperial State", out string error);
        Equal(true, projected,
            "the installed ruler's deterministic open reign is projected");
        Equal(91L, Scalar(db,
                "SELECT MANDATE_PERIOD_ID FROM KingdomReign WHERE REIGN_ID=2"),
            "the latest installed-ruler reign receives the Mandate period");
        Equal(4L, Scalar(db,
                "SELECT HIGHEST_TITLE FROM KingdomReign WHERE REIGN_ID=2"),
            "the latest installed-ruler reign receives imperial rank");
        Equal("Imperial State", ScalarText(db,
                "SELECT STATE_NAME_SNAPSHOT FROM KingdomReign " +
                "WHERE REIGN_ID=2"),
            "the latest installed-ruler reign receives the state snapshot");
        Equal(-1L, Scalar(db,
                "SELECT MANDATE_PERIOD_ID FROM KingdomReign WHERE REIGN_ID=1"),
            "the previous ruler's ordinary open reign remains unchanged");
        Equal(2L, Scalar(db,
                "SELECT HIGHEST_TITLE FROM KingdomReign WHERE REIGN_ID=1"),
            "the previous ruler keeps its ordinary historical rank");
        Equal("Old ruler", ScalarText(db,
                "SELECT STATE_NAME_SNAPSHOT FROM KingdomReign " +
                "WHERE REIGN_ID=1"),
            "the previous ruler keeps its historical state snapshot");
        Equal(-1L, Scalar(db,
                "SELECT MANDATE_PERIOD_ID FROM KingdomReign WHERE REIGN_ID=3"),
            "an older duplicate open reign remains unchanged");

        bool missing = ReignMandateProjectionPersistence.TryProject(
            db, "KingdomReign", kingdomId: 17L, rulerActorId: 99L,
            mandatePeriodId: 91L, emperorTitle: 4,
            stateName: "Imperial State", out error);
        Equal(false, missing,
            "a missing installed-ruler open reign reports failure");
        Equal(1L, Scalar(db,
                "SELECT COUNT(*) FROM KingdomReign " +
                "WHERE MANDATE_PERIOD_ID=91"),
            "a failed projection changes no unrelated reign");
    }

    private static void FailedMandateDeclarationRollsBackAndRetriesOnce()
    {
        using var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        Execute(db,
            "CREATE TABLE MandatePeriod(" +
            "PERIOD_ID INTEGER PRIMARY KEY,KINGDOM_ID INTEGER," +
            "KINGDOM_NAME TEXT,KINGDOM_COLOR TEXT,DYNASTY_NAME TEXT," +
            "FOUNDER_ACTOR_ID INTEGER,FOUNDER_NAME TEXT,START_TIME REAL," +
            "END_TIME REAL,END_REASON TEXT,START_MANDATE INTEGER," +
            "END_MANDATE INTEGER,LEGAL_CORE_COUNT INTEGER," +
            "ORIGIN_TYPE TEXT,REBEL_ORIGIN_KINGDOM_ID INTEGER," +
            "REBEL_ORIGIN_KINGDOM_NAME TEXT,CLAIMANT_KIND TEXT);" +
            "CREATE TABLE MandateState(" +
            "STATE_ID INTEGER PRIMARY KEY,ACTIVE INTEGER," +
            "KINGDOM_ID INTEGER,KINGDOM_NAME TEXT,KINGDOM_COLOR TEXT," +
            "DYNASTY_NAME TEXT,EMPEROR_ACTOR_ID INTEGER," +
            "EMPEROR_NAME TEXT,PERIOD_ID INTEGER,MANDATE_VALUE INTEGER," +
            "IMPERIAL_AUTHORITY INTEGER,DYNASTY_PRESTIGE INTEGER," +
            "CORE_CONTROL REAL,VASSAL_LOYALTY REAL,CRISIS_LEVEL TEXT," +
            "START_TIME REAL,UPDATED_TIME REAL,LAST_YEAR INTEGER," +
            "ORIGIN_TYPE TEXT,ORIGINAL_CORE_COUNT INTEGER," +
            "REBEL_ORIGIN_KINGDOM_ID INTEGER," +
            "REBEL_ORIGIN_KINGDOM_NAME TEXT,CLAIMANT_KIND TEXT," +
            "MAP_MARKER_KIND TEXT);" +
            "INSERT INTO MandateState(STATE_ID,ACTIVE,KINGDOM_ID," +
            "PERIOD_ID) VALUES(1,0,-1,-1);" +
            "CREATE TABLE KingdomReign(" +
            "REIGN_ID INTEGER PRIMARY KEY,KINGDOM_ID INTEGER," +
            "KING_ACTOR_ID INTEGER,MANDATE_PERIOD_ID INTEGER," +
            "HIGHEST_TITLE INTEGER,STATE_NAME_SNAPSHOT TEXT," +
            "START_TIME REAL,END_TIME REAL);" +
            "CREATE TRIGGER fail_declaration_reign BEFORE UPDATE ON " +
            "KingdomReign BEGIN SELECT RAISE(ABORT," +
            "'injected declaration reign failure'); END;");

        var request = new MandateDeclarationPersistence.Request
        {
            StateId = 1L,
            PeriodId = 91L,
            KingdomId = 17L,
            KingdomName = "Imperial State",
            KingdomColor = "#abcdef",
            DynastyName = "New Dynasty",
            RulerActorId = 42L,
            RulerName = "New Emperor",
            StartTime = 12d,
            CurrentYear = 101,
            StartMandate = 30,
            ImperialAuthority = 45,
            DynastyPrestige = 0,
            CoreControl = 1d,
            VassalLoyalty = 1d,
            CrisisLevel = "stable",
            OriginType = "native",
            RebelOriginKingdomId = -1L,
            RebelOriginKingdomName = "",
            ClaimantKind = "orthodox",
            MapMarkerKind = "moh",
            EmperorTitle = 4
        };
        int historyPublishes = 0;
        int traitPublishes = 0;
        int runtimePublishes = 0;
        string error = "";
        bool first = MandateDeclarationRules.TryCommitProjection(
            () => MandateDeclarationPersistence.TryCommit(
                db, "MandatePeriod", "MandateState", "KingdomReign",
                request, out error),
            () =>
            {
                historyPublishes++;
                traitPublishes++;
                runtimePublishes++;
            });
        Equal(false, first,
            "a failed reign projection rejects Mandate declaration");
        Equal(0L, Scalar(db, "SELECT COUNT(*) FROM MandatePeriod"),
            "failed declaration rolls back its period");
        Equal(0L, Scalar(db,
                "SELECT ACTIVE FROM MandateState WHERE STATE_ID=1"),
            "failed declaration leaves no active Mandate state");
        Equal(0L, Scalar(db, "SELECT COUNT(*) FROM KingdomReign"),
            "zero-row failure does not invent a reign");
        Equal(0, historyPublishes,
            "failed declaration publishes no history");
        Equal(0, traitPublishes,
            "failed declaration publishes no trait");
        Equal(0, runtimePublishes,
            "failed declaration publishes no runtime projection");

        Execute(db,
            "INSERT INTO KingdomReign VALUES" +
            "(7,17,42,-1,2,'Ordinary State',11,-1);");
        bool exceptionFailure =
            MandateDeclarationRules.TryCommitProjection(
                () => MandateDeclarationPersistence.TryCommit(
                    db, "MandatePeriod", "MandateState", "KingdomReign",
                    request, out error),
                () =>
                {
                    historyPublishes++;
                    traitPublishes++;
                    runtimePublishes++;
                });
        Equal(false, exceptionFailure,
            "a reign projection exception rejects Mandate declaration");
        Equal(0L, Scalar(db, "SELECT COUNT(*) FROM MandatePeriod"),
            "exceptional declaration rolls back its period");
        Equal(0L, Scalar(db,
                "SELECT ACTIVE FROM MandateState WHERE STATE_ID=1"),
            "exceptional declaration leaves no active Mandate state");
        Equal(-1L, Scalar(db,
                "SELECT MANDATE_PERIOD_ID FROM KingdomReign WHERE REIGN_ID=7"),
            "exceptional declaration leaves the ordinary reign unchanged");
        Equal(0, historyPublishes,
            "exceptional declaration publishes no history");
        Equal(0, traitPublishes,
            "exceptional declaration publishes no trait");
        Equal(0, runtimePublishes,
            "exceptional declaration publishes no runtime projection");

        Execute(db, "DROP TRIGGER fail_declaration_reign;");
        bool second = MandateDeclarationRules.TryCommitProjection(
            () => MandateDeclarationPersistence.TryCommit(
                db, "MandatePeriod", "MandateState", "KingdomReign",
                request, out error),
            () =>
            {
                historyPublishes++;
                traitPublishes++;
                runtimePublishes++;
            });
        Equal(true, second,
            "the declaration safely retries after persistence recovers");
        Equal(1L, Scalar(db, "SELECT COUNT(*) FROM MandatePeriod"),
            "successful retry creates exactly one period");
        Equal(1L, Scalar(db,
                "SELECT ACTIVE FROM MandateState WHERE STATE_ID=1"),
            "successful retry activates Mandate state");
        Equal(91L, Scalar(db,
                "SELECT MANDATE_PERIOD_ID FROM KingdomReign WHERE REIGN_ID=7"),
            "successful retry projects the ruler's reign");
        Equal(1, historyPublishes,
            "successful retry publishes history once");
        Equal(1, traitPublishes,
            "successful retry publishes the trait once");
        Equal(1, runtimePublishes,
            "successful retry publishes runtime state once");

        bool duplicate = MandateDeclarationRules.TryCommitProjection(
            () => MandateDeclarationPersistence.TryCommit(
                db, "MandatePeriod", "MandateState", "KingdomReign",
                request, out error),
            () =>
            {
                historyPublishes++;
                traitPublishes++;
                runtimePublishes++;
            });
        Equal(false, duplicate,
            "the committed period cannot be published a second time");
        Equal(1, historyPublishes,
            "duplicate retry does not duplicate history");
        Equal(1, traitPublishes,
            "duplicate retry does not duplicate the trait projection");
        Equal(1, runtimePublishes,
            "duplicate retry does not duplicate runtime projection");
    }

    private static void ActiveMandateKeepsItsImperialStateName()
    {
        Equal(false, StateNameRules.ShouldProjectDynasticStateName(
                newDynastyCreated: true, isEmpireRank: true,
                changedRulingShi: true, hasExistingBoundStateName: true,
                isActiveMandate: true),
            "a Mandate succession cannot replace the imperial state name with a branch state");
    }

    private static void SavedMandateReignUsesImperialPosthumousRank()
    {
        Equal(4, RulerTitleFactRules.ResolveSavedHighestTitle(
                savedHighestTitle: 2, savedMandatePeriodId: 91L),
            "a saved Mandate reign remains imperial after the live realm is downgraded");
        Equal(2, RulerTitleFactRules.ResolveSavedHighestTitle(
                savedHighestTitle: 2, savedMandatePeriodId: -1L),
            "an ordinary saved ducal reign remains ducal");
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(
                $"{name}: expected {expected}, got {actual}");
    }

    private static void Execute(SQLiteConnection db, string sql)
    {
        using var command = new SQLiteCommand(sql, db);
        command.ExecuteNonQuery();
    }

    private static long Scalar(SQLiteConnection db, string sql)
    {
        using var command = new SQLiteCommand(sql, db);
        return Convert.ToInt64(command.ExecuteScalar());
    }

    private static string ScalarText(SQLiteConnection db, string sql)
    {
        using var command = new SQLiteCommand(sql, db);
        return Convert.ToString(command.ExecuteScalar()) ?? "";
    }
}
