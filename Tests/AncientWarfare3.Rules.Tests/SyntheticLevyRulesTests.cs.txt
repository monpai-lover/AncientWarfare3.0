using AncientWarfare3.core.lineage;
using System;
using System.IO;

internal static class SyntheticLevyRulesTests
{
    internal static void RunPopulationOnly()
    {
        Equal(80, SyntheticLevyRules.AuthenticPopulation(
            reportedPopulation: 100, syntheticPopulation: 20),
            "synthetic soldiers are excluded from city population");
        Equal(0, SyntheticLevyRules.AuthenticPopulation(
            reportedPopulation: 3, syntheticPopulation: 9),
            "a stale synthetic counter cannot produce negative population");
        SyntheticSoldiersAreExcludedFromCityPopulation();
    }

    internal static void Run()
    {
        Equal("aw_synthetic_levy", LineageKeys.SYNTHETIC_LEVY,
            "synthetic provenance uses a stable actor key");
        Equal("aw_synthetic_levy_source_city_id",
            LineageKeys.SYNTHETIC_LEVY_SOURCE_CITY_ID,
            "synthetic soldiers retain their source city");
        Equal("aw_synthetic_levy_emergency_id",
            LineageKeys.SYNTHETIC_LEVY_EMERGENCY_ID,
            "synthetic soldiers retain their formal war emergency");
        Equal("aw_synthetic_levy_ledger_released",
            LineageKeys.SYNTHETIC_LEVY_LEDGER_RELEASED,
            "synthetic manpower is released from the live ledger once");
        Equal(SyntheticLevyDisposition.RestoreCivilian,
            SyntheticLevyRules.ResolveDemobilization(
                synthetic: false, alive: true, militaryMerit: 0),
            "an authentic levy returns to civilian life");
        Equal(SyntheticLevyDisposition.RemoveActor,
            SyntheticLevyRules.ResolveDemobilization(
                synthetic: true, alive: true, militaryMerit: 0),
            "an unmerited synthetic levy is removed after war");
        Equal(SyntheticLevyDisposition.RemoveActor,
            SyntheticLevyRules.ResolveDemobilization(
                synthetic: true, alive: true, militaryMerit: 1),
            "military merit never makes a synthetic levy permanent");
        Equal(SyntheticLevyDisposition.Ignore,
            SyntheticLevyRules.ResolveDemobilization(
                synthetic: true, alive: false, militaryMerit: 10),
            "dead levies are not demobilized twice");
        Equal(true, SyntheticLevyRules.SuppressPersonalHistory(
            synthetic: true, promoted: false),
            "unpromoted synthetic actors have no person history");
        Equal(false, SyntheticLevyRules.SuppressPersonalHistory(
            synthetic: true, promoted: true),
            "promoted actors enter normal history");
        RunPopulationOnly();
        Equal(true, SyntheticLevyRules.AllowTask(
            synthetic: true, SyntheticLevyTask.Military),
            "synthetic soldiers can execute military tasks");
        Equal(true, SyntheticLevyRules.AllowTask(
            synthetic: true, SyntheticLevyTask.Healing),
            "synthetic soldiers may heal");
        Equal(true, SyntheticLevyRules.AllowTask(
            synthetic: true, SyntheticLevyTask.Sleep),
            "synthetic soldiers may rest");
        Equal(false, SyntheticLevyRules.AllowTask(
            synthetic: true, SyntheticLevyTask.Social),
            "synthetic soldiers cannot socialize");
        Equal(false, SyntheticLevyRules.AllowTask(
            synthetic: true, SyntheticLevyTask.CivilianWork),
            "synthetic soldiers cannot enter civilian work");
        foreach (SyntheticLevyTask task in Enum.GetValues<
                     SyntheticLevyTask>())
        {
            bool expected = task == SyntheticLevyTask.Military ||
                            task == SyntheticLevyTask.Food ||
                            task == SyntheticLevyTask.Healing ||
                            task == SyntheticLevyTask.Transport ||
                            task == SyntheticLevyTask.Retreat ||
                            task == SyntheticLevyTask.Formation ||
                            task == SyntheticLevyTask.Sleep;
            Equal(expected, SyntheticLevyRules.AllowTask(true, task),
                "synthetic task whitelist is exhaustive for " + task);
        }
        Equal(true, SyntheticLevyRules.AllowTaskId(true,
                "aw_army_rts_march"),
            "RTS marching remains available");
        Equal(true, SyntheticLevyRules.AllowTaskId(true, "sleep_inside"),
            "basic rest remains available");
        Equal(false, SyntheticLevyRules.AllowTaskId(true,
                "aw_historical_school_lecture"),
            "unknown civil work is denied by default");
        Equal(true, SyntheticLevyRules.ShouldClearSyntheticFields(
            SyntheticLevyDisposition.PromotePermanent),
            "promotion clears synthetic provenance");
        Equal(true, SyntheticLevyRules.ShouldPromoteForCaptain(
                synthetic: true, selectedAsCaptain: true),
            "a selected synthetic levy is made permanent before receiving captain authority");
        Equal(false, SyntheticLevyRules.ShouldPromoteForCaptain(
                synthetic: true, selectedAsCaptain: false),
            "an unselected synthetic levy remains subject to wartime cleanup");
        Equal(false, SyntheticLevyRules.ShouldPromoteForCaptain(
                synthetic: false, selectedAsCaptain: true),
            "normal soldiers do not receive synthetic cleanup work");
        Equal(true, SyntheticLevyRules.ShouldRemoveActor(
            SyntheticLevyDisposition.RemoveActor),
            "remove disposition destroys the temporary actor");
        Equal(false, SyntheticLevyRules.ShouldRemoveActor(
            SyntheticLevyDisposition.RestoreCivilian),
            "authentic levies are not destroyed");
        Equal(false, SyntheticLevySpawnScope.IsActive,
            "synthetic creation scope starts inactive");
        using (SyntheticLevySpawnScope.Open())
        {
            Equal(true, SyntheticLevySpawnScope.IsActive,
                "synthetic creation scope suppresses intermediate callbacks");
            using (SyntheticLevySpawnScope.Open())
                Equal(true, SyntheticLevySpawnScope.IsActive,
                    "nested synthetic creation remains active");
            Equal(true, SyntheticLevySpawnScope.IsActive,
                "disposing a nested scope preserves its parent");
        }
        Equal(false, SyntheticLevySpawnScope.IsActive,
            "disposing the outer scope restores normal callbacks");
        RuntimeCreationIsAtomicAndImmediatelyMilitary();
        FormalWarReplenishmentUsesSyntheticReserveOnly();
        SyntheticLifecycleBoundariesArePatched();
    }

    private static void SyntheticLifecycleBoundariesArePatched()
    {
        string root = RepositoryRoot();
        string tasks = File.ReadAllText(Path.Combine(root, "Code", "patch",
            "AW_WartimeMilitaryTaskPatch.cs"));
        Equal(false, tasks.Contains(
                "AccessTools.Method(typeof(AiSystemActor), \"setJob\")",
                StringComparison.Ordinal),
            "synthetic levies do not add a global setJob hot-path poll");
        Equal(true, tasks.Contains(
                "AccessTools.Method(typeof(AiSystemActor), \"setTask\")",
                StringComparison.Ordinal),
            "direct task assignment is intercepted");
        Equal(true, tasks.Contains(
                "AccessTools.Method(typeof(AiSystemActor), \"update\")",
                StringComparison.Ordinal),
            "an already active forbidden task is intercepted");

        string reproduction = File.ReadAllText(Path.Combine(root, "Code",
            "patch", "AW_DynasticReproductionPatch.cs"));
        int reproductionPrefix = reproduction.IndexOf(
            "private static bool MakeBaby_Prefix(", StringComparison.Ordinal);
        int reproductionPostfix = reproduction.IndexOf(
            "[HarmonyPostfix]", reproductionPrefix,
            StringComparison.Ordinal);
        string reproductionBody = reproduction.Substring(reproductionPrefix,
            reproductionPostfix - reproductionPrefix);
        Equal(true, reproductionBody.Contains(
                "SyntheticLevyService.IsSynthetic(pParent1)",
                StringComparison.Ordinal),
            "the first direct parent cannot be synthetic");
        Equal(true, reproductionBody.Contains(
                "SyntheticLevyService.IsSynthetic(pParent2)",
                StringComparison.Ordinal),
            "the second direct parent cannot be synthetic");
        Equal(true, reproductionBody.Contains("return false;",
                StringComparison.Ordinal),
            "synthetic two-parent reproduction is rejected at the vanilla entry point");
        Equal(true, reproductionBody.Contains("ref Actor __result",
                StringComparison.Ordinal),
            "blocked reproduction returns no child");

        string lovers = File.ReadAllText(Path.Combine(root, "Code", "patch",
            "AW_LoversPatch.cs"));
        int loverPrefix = lovers.IndexOf(
            "SetLover_Prefix", StringComparison.Ordinal);
        int loverPostfix = lovers.IndexOf(
            "[HarmonyPostfix]", loverPrefix, StringComparison.Ordinal);
        string loverBody = lovers.Substring(loverPrefix,
            loverPostfix - loverPrefix);
        Equal(true, loverBody.Contains("public static bool",
                StringComparison.Ordinal) && loverBody.Contains(
                "return false;", StringComparison.Ordinal),
            "synthetic actors are rejected before vanilla lover assignment");
        Equal(true, lovers.Contains(
                "BecomeLoversWith_Prefix", StringComparison.Ordinal),
            "direct vanilla lover formation is rejected before execution");

        string ledger = File.ReadAllText(Path.Combine(root, "Code", "core",
            "lineage", "SyntheticMobilizationLedgerService.cs"));
        Equal(true, ledger.Contains(
                "private static void ProcessDemobilization(",
                StringComparison.Ordinal) &&
            ledger.Contains(
                "SyntheticLevyService.RemoveWithoutPersonalHistory(actor)",
                StringComparison.Ordinal),
            "synthetic demobilization is owned by the bounded war ledger");
    }

    private static void SyntheticSoldiersAreExcludedFromCityPopulation()
    {
        string source = File.ReadAllText(Path.Combine(RepositoryRoot(),
            "Code", "patch", "AW_SyntheticPopulationPatch.cs"));
        Equal(true, source.Contains(
                "nameof(City.getPopulationPeople)",
                StringComparison.Ordinal),
            "the normal city population API excludes synthetic soldiers");
        Equal(true, source.Contains(
                "\"getPopulationTotal\", new[] { typeof(bool) }",
                StringComparison.Ordinal),
            "compatible game versions also exclude synthetic soldiers " +
            "from the total population API");
        Equal(true, source.Contains(
                "SyntheticMobilizationLedgerService.LiveSyntheticForCity(",
                StringComparison.Ordinal),
            "population queries use the existing O(1) synthetic ledger");
        Equal(true, source.Contains(
                "SyntheticLevyRules.AuthenticPopulation(",
                StringComparison.Ordinal),
            "both runtime APIs share the clamped population rule");
    }

    private static void RuntimeCreationIsAtomicAndImmediatelyMilitary()
    {
        string root = RepositoryRoot();
        string path = Path.Combine(root, "Code", "core", "lineage",
            "SyntheticLevyService.cs");
        Equal(true, File.Exists(path),
            "the synthetic levy runtime boundary exists");
        string source = File.ReadAllText(path);
        int scope = source.IndexOf("SyntheticLevySpawnScope.Open()",
            StringComparison.Ordinal);
        int create = source.IndexOf("createNewUnit(",
            StringComparison.Ordinal);
        int mark = source.IndexOf("Mark(actor, city, kingdom, emergencyId)",
            StringComparison.Ordinal);
        int join = source.IndexOf("actor.joinCity(city)",
            StringComparison.Ordinal);
        int warrior = source.IndexOf("city.makeWarrior(actor)",
            StringComparison.Ordinal);
        int army = source.IndexOf("AWArmyService.AddToArmy(actor, army)",
            StringComparison.Ordinal);
        Equal(true, scope >= 0 && scope < create && create < mark &&
                    mark < join && join < warrior && warrior < army,
            "a generated actor is marked, joined, made Warrior, and assigned " +
            "before leaving the synthetic spawn scope");
        Equal(true, source.Contains(
                "SyntheticMobilizationLedgerService.OnSyntheticMaterialized(",
                StringComparison.Ordinal),
            "generated soldiers enter the unified mobilization ledger");
        Equal(true, source.Contains(
                "SyntheticMobilizationLedgerService.OnSyntheticRemoved(",
                StringComparison.Ordinal),
            "synthetic removal releases unified ledger membership");
        Equal(true, source.Contains(
                "WorldTile spawnTile = pSpawnTile ?? city?.getTile();",
                StringComparison.Ordinal),
            "frontline synthetic creation falls back to the source city tile when the captain has no tile");
    }

    private static void FormalWarReplenishmentUsesSyntheticReserveOnly()
    {
        string root = RepositoryRoot();
        string source = File.ReadAllText(Path.Combine(root, "Code", "core",
            "lineage", "ArmyReplenishmentOperationService.cs"));
        string controller = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsControllerService.cs"));
        Equal(true, source.Contains(
                "SyntheticLevyService.CreateBatchAtTile(",
                StringComparison.Ordinal),
            "formal-war replenishment materializes the city war reserve at the frontline");
        Equal(true, source.Contains(
                "CreateBatchAtTile(", StringComparison.Ordinal),
            "formal-war replenishment materializes synthetic soldiers at the field army");
        Equal(false, controller.Contains(
                "currentCity == pSourceCity", StringComparison.Ordinal),
            "frontline replenishment must not require the army to return to its source city");
        Equal(true, source.Contains(
                "SyntheticMobilizationLedgerService.TryReserveReplacement(",
                StringComparison.Ordinal),
            "formal-war replenishment reserves the unified numeric ledger");
        string reservePool = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "CityReservePoolService.cs"));
        Equal(false, reservePool.Contains(
                "if (phase != ArmyMobilizationPhase.Notice) return 0;",
                StringComparison.Ordinal),
            "notice preparation no longer converts real residents");
        Equal(false, reservePool.Contains(
                "TryTakeNextActorId(pool.ActorIds",
                StringComparison.Ordinal),
            "the old pool no longer extracts real residents");
        string levy = File.ReadAllText(Path.Combine(root, "Code", "core",
            "lineage", "TemporaryLevyService.cs"));
        Equal(false, levy.Contains(
                "bool mobilizationActive = ArmyMobilizationRules.CanConsume(",
                StringComparison.Ordinal),
            "formal War cannot keep the real-actor preparation plan alive");
        Equal(false, levy.Contains("SyntheticLevyService.Promote(",
                StringComparison.Ordinal),
            "generated wartime soldiers can never become permanent actors");
        string syntheticLevy = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "SyntheticLevyService.cs"));
        Equal(false, syntheticLevy.Contains(
                "internal static void Promote(Actor",
                StringComparison.Ordinal),
            "the generated-actor service exposes no promotion boundary");
        int casualty = levy.IndexOf(
            "public static void OnMilitaryCasualty(Actor pActor)",
            StringComparison.Ordinal);
        int casualtyEnd = levy.IndexOf(
            "public static void OnKingdomDestroying", casualty,
            StringComparison.Ordinal);
        string casualtyBody = levy.Substring(casualty,
            casualtyEnd - casualty);
        Equal(true, casualtyBody.Contains(
                "SyntheticLevySpawnScope.IsActive",
                StringComparison.Ordinal),
            "synthetic creation rollback cannot create casualty demand");
        int disposition = levy.IndexOf(
            "SyntheticLevyRules.ResolveDemobilization(",
            StringComparison.Ordinal);
        int clear = levy.IndexOf("ClearFields(pActor)", disposition,
            StringComparison.Ordinal);
        Equal(true, disposition >= 0 && clear > disposition,
            "demobilization resolves synthetic disposition before clearing provenance");

        string birth = File.ReadAllText(Path.Combine(root, "Code", "patch",
            "AW_BirthPatch.cs"));
        string death = File.ReadAllText(Path.Combine(root, "Code", "patch",
            "AW_ActorDeathPatch.cs"));
        string enlist = File.ReadAllText(Path.Combine(root, "Code", "patch",
            "AW_EnlistPatch.cs")).Replace("\r\n", "\n");
        Equal(true, enlist.Contains(
                "SyntheticLevySpawnScope.IsActive &&\n" +
                "                SyntheticLevyService.IsSynthetic(pActor)",
                StringComparison.Ordinal),
            "synthetic creation bypasses resident age and office enlistment guards");
        Equal(true, birth.Contains(
                "SyntheticLevyService.SuppressPersonalHistory(",
                StringComparison.Ordinal),
            "birth persistence ignores unpromoted synthetic soldiers");
        Equal(true, death.Contains(
                "SyntheticLevyService.SuppressPersonalHistory(",
                StringComparison.Ordinal),
            "death persistence ignores unpromoted synthetic soldiers");
    }

    private static string RepositoryRoot()
    {
        return Path.GetFullPath(Path.Combine(AppContext.BaseDirectory,
            "..", "..", "..", "..", ".."));
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(name + ": expected " +
                expected + ", got " + actual);
    }
}
