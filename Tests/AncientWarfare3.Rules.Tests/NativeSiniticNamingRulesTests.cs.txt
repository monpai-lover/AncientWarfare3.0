using System;
using System.IO;
using AncientWarfare3.core.lineage;
using AncientWarfare3.core.naming;

internal static class NativeSiniticNamingRulesTests
{
    public static void Run()
    {
        string[] included =
        {
            "civ_dog", "civ_fox", "civ_lemon_man", "civ_rabbit",
            "civ_turtle"
        };
        foreach (string assetId in included)
            Equal(true,
                AWNativeSiniticSpeciesRules.IsNativeSiniticSpecies(assetId),
                assetId + " uses native Sinitic naming");

        string[] excluded =
        {
            "dog", "fox", "rabbit", "turtle", "lemon_snail",
            "miniciv_lemon_snail", "civ_bear", "civ_monkey", "", null
        };
        foreach (string assetId in excluded)
            Equal(false,
                AWNativeSiniticSpeciesRules.IsNativeSiniticSpecies(assetId),
                (assetId ?? "null") + " stays outside the exact species set");

        // Read through NormalizeNewlines: git checks these sources out with
        // CRLF on Windows, so a guard that spans two lines cannot embed a
        // bare LF and still match. Without this the assertion below fails on
        // every default Windows clone while passing on Linux and macOS.
        string lineageService = ReadNormalized("Code", "core", "lineage",
            "LineageService.cs");
        Equal(true, lineageService.Contains(
                "bool UsesCommonGenealogy(Actor pActor)"),
            "native Sinitic species have a genealogy gate distinct from Xia naming");
        Equal(true, lineageService.Contains(
                "return IsNativeXiaCultureActor(pActor) ||\n                   AWNativeSiniticSpeciesRules.IsNativeSiniticSpecies"),
            "native Sinitic species enter the common genealogy gate");
        string promotionPatch = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "patch",
            "AW_PromotionPatch.cs"));
        Equal(true, promotionPatch.Contains(
                "LineageService.UsesCommonGenealogy(pActor)"),
            "native Sinitic kings use the common promotion genealogy path");
        string localizedNames = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "naming",
            "AWLocalizedNameService.cs"));
        Equal(true, localizedNames.Contains(
                "pActor.data.set(LineageKeys.CLAN_NAME, pParts.FamilyName);"),
            "native Sinitic generation stores its single Shi as the family-tree branch");
        Equal(true, localizedNames.Contains(
                "pActor.data.set(LineageKeys.NAME_INTEGRATED, true);"),
            "native Sinitic generation keeps live and archived name projection aligned");
        string legacyMigration = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "WesternLineageMigrationService.cs"));
        Equal(true, legacyMigration.Contains(
                "LineageService.RepairNativeSiniticIdentity(pActor)"),
            "existing native Sinitic actors are repaired through the bounded load migration");

        Equal(NamingProfileId.NativeSinitic,
            AWNamingProfileRules.Resolve(false, false,
                nativeSinitic: true, orc: false, civilized: true, valid: true),
            "the exact species set receives its own natural profile");
        Equal("native_sinitic",
            AWCultureNamingTraditionRules.SerializeProfile(
                NamingProfileId.NativeSinitic),
            "the profile has a stable persisted id");
        Equal(NamingProfileId.NativeSinitic,
            AWCultureNamingTraditionRules.ParseProfile("native_sinitic"),
            "the persisted profile is restored");
        Equal(false,
            AWCultureNamingTraditionRules.ShouldPersistWesternTradition(
                NamingProfileId.NativeSinitic),
            "native Sinitic cultures never persist a Western tradition");
        Equal(NamingProfileId.NativeSinitic,
            AWCultureNamingTraditionRules.ResolveEffectiveProfile(
                NamingProfileId.NativeSinitic, NamingProfileId.Western,
                fullyXiaized: false),
            "a legacy Western culture resolves to its native profile");
        Equal(NamingProfileId.NativeSinitic,
            AWCultureNamingTraditionRules.ResolveInheritedProfile(
                NamingProfileId.NativeSinitic, NamingProfileId.Western,
                fullyXiaized: false),
            "a culture split cannot inherit a Western profile");
        Equal(NamingProfileId.NativeSinitic,
            AWCultureNamingTraditionRules.ResolveActorProfile(
                NamingProfileId.NativeSinitic, NamingProfileId.Xia),
            "the species keeps its native personal-name profile in Xia culture");

        foreach (string assetId in included)
        {
            Equal(assetId + "_name",
                AWCultureNamingTraditionRules.ResolveGeneratorId(
                    NamingProfileId.NativeSinitic,
                    WesternNamingTradition.Von, AWNamingObjectKind.Actor,
                    assetId, "western_von_name"),
                assetId + " uses its current actor generator");
            Equal(assetId + "_city",
                AWCultureNamingTraditionRules.ResolveGeneratorId(
                    NamingProfileId.NativeSinitic,
                    WesternNamingTradition.Von, AWNamingObjectKind.City,
                    assetId, "western_city"),
                assetId + " uses its current city generator");
            Equal(assetId + "_kingdom",
                AWCultureNamingTraditionRules.ResolveGeneratorId(
                    NamingProfileId.NativeSinitic,
                    WesternNamingTradition.Von, AWNamingObjectKind.Kingdom,
                    assetId, "western_kingdom"),
                assetId + " uses its current kingdom generator");
            Equal(assetId + "_name",
                AWCultureNamingTraditionRules.ResolveFallbackGeneratorId(
                    NamingProfileId.NativeSinitic,
                    AWNamingObjectKind.Actor, assetId, "human_name"),
                assetId + " fallback cannot cross into another profile");
        }

        NativeSiniticNameParts japanese =
            AWNativeSiniticNamePartsRules.Resolve(
                "山田太郎", "山田", "太");
        Equal(true, japanese.Valid,
            "a current Japanese-library template produces valid parts");
        Equal("山田", japanese.FamilyName,
            "the tagged current-library surname is retained");
        Equal("太郎", japanese.GivenName,
            "untagged suffix words remain in the complete given name");

        NativeSiniticNameParts inherited =
            AWNativeSiniticNamePartsRules.Resolve(
                "山田太郎", "山田", "太", "赵");
        Equal(true, inherited.Valid,
            "an inherited family can replace the generated family");
        Equal("赵", inherited.FamilyName,
            "the selected parent family remains authoritative");
        Equal("太郎", inherited.GivenName,
            "family inheritance preserves every generated given-name word");
        Equal("赵太郎", inherited.DisplayName,
            "native Sinitic projection stays surname first without a space");

        NativeSiniticNameParts shanhai =
            AWNativeSiniticNamePartsRules.Resolve("狐甲子", "狐", "");
        Equal(true, shanhai.Valid,
            "a current Shanhai-library template produces valid parts");
        Equal("甲子", shanhai.GivenName,
            "a template without a given tag derives the complete suffix");

        Equal(false,
            AWNativeSiniticNamePartsRules.Resolve("甲子", "", "甲").Valid,
            "an empty generated surname rejects the identity");
        Equal(false,
            AWNativeSiniticNamePartsRules.Resolve("甲狐子", "狐", "甲").Valid,
            "a surname outside the visible prefix rejects the identity");
        Equal(false,
            AWNativeSiniticNamePartsRules.Resolve("狐", "狐", "").Valid,
            "a surname-only result rejects the identity");

        Equal(false,
            IntegratedCultureNamingMigrationRules.ShouldMigrateNaturalProfile(
                NamingProfileId.NativeSinitic, false),
            "native Sinitic cultures never enter live-name migration");
        Equal(false,
            IntegratedCultureNamingMigrationRules.ShouldMigrateNaturalProfile(
                NamingProfileId.Monkey, true),
            "civilized monkey cultures never enter live-name migration");
        Equal(true,
            IntegratedCultureNamingMigrationRules.ShouldMigrateNaturalProfile(
                NamingProfileId.Western, false),
            "western cultures may opt into future-only Xia naming");
        Equal(NamingProfileId.Western,
            AWCultureNamingTraditionRules.ResolveActorSnapshotProfile(
                NamingProfileId.Western, NamingProfileId.None,
                NamingProfileId.Xia, pCreationBoundary: false),
            "legacy living actors keep their natural profile");
        Equal(NamingProfileId.Xia,
            AWCultureNamingTraditionRules.ResolveActorSnapshotProfile(
                NamingProfileId.Western, NamingProfileId.None,
                NamingProfileId.Xia, pCreationBoundary: true),
            "new western actors adopt the current Xia culture profile");
        Equal(NamingProfileId.Western,
            AWCultureNamingTraditionRules.ResolveActorSnapshotProfile(
                NamingProfileId.Western, NamingProfileId.Western,
                NamingProfileId.Xia, pCreationBoundary: true),
            "an existing actor snapshot is never reinterpreted");
        Equal(true,
            AWCultureNamingTraditionRules
                .ShouldDeferActorProfileInitialization(
                    NamingProfileId.Western,
                    pHasCultureContext: false),
            "a western newborn waits for parent culture before snapshotting");
        Equal(true,
            AWCultureNamingTraditionRules
                .ShouldDeferActorProfileInitialization(
                    NamingProfileId.OrcNomadic,
                    pHasCultureContext: false),
            "an orc newborn waits for parent culture before snapshotting");
        Equal(false,
            AWCultureNamingTraditionRules
                .ShouldDeferActorProfileInitialization(
                    NamingProfileId.NativeSinitic,
                    pHasCultureContext: false),
            "native Sinitic newborn identity never depends on Xia culture");

    }

    // Source guards compare against literals written with LF. Normalize the
    // file so the guard describes the code rather than the checkout's
    // line-ending configuration.
    private static string ReadNormalized(params string[] pParts)
    {
        string path = Directory.GetCurrentDirectory();
        foreach (string part in pParts)
            path = Path.Combine(path, part);
        return File.ReadAllText(path)
            .Replace("\r\n", "\n")
            .Replace("\r", "\n");
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(
                $"{message}: expected {expected}, got {actual}");
    }
}
