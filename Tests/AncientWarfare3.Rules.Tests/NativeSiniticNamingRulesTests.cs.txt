using System;
using AncientWarfare3.core.naming;

internal static class NativeSiniticNamingRulesTests
{
    public static void Run()
    {
        string[] included =
        {
            "civ_dog", "civ_fox", "civ_lemon_man", "civ_rabbit",
            "civ_turtle"
        };
        foreach (string assetId in included)
            Equal(true,
                AWNativeSiniticSpeciesRules.IsNativeSiniticSpecies(assetId),
                assetId + " uses native Sinitic naming");

        string[] excluded =
        {
            "dog", "fox", "rabbit", "turtle", "lemon_snail",
            "miniciv_lemon_snail", "civ_bear", "civ_monkey", "", null
        };
        foreach (string assetId in excluded)
            Equal(false,
                AWNativeSiniticSpeciesRules.IsNativeSiniticSpecies(assetId),
                (assetId ?? "null") + " stays outside the exact species set");

        Equal(NamingProfileId.NativeSinitic,
            AWNamingProfileRules.Resolve(false, false,
                nativeSinitic: true, orc: false, civilized: true, valid: true),
            "the exact species set receives its own natural profile");
        Equal("native_sinitic",
            AWCultureNamingTraditionRules.SerializeProfile(
                NamingProfileId.NativeSinitic),
            "the profile has a stable persisted id");
        Equal(NamingProfileId.NativeSinitic,
            AWCultureNamingTraditionRules.ParseProfile("native_sinitic"),
            "the persisted profile is restored");
        Equal(false,
            AWCultureNamingTraditionRules.ShouldPersistWesternTradition(
                NamingProfileId.NativeSinitic),
            "native Sinitic cultures never persist a Western tradition");
        Equal(NamingProfileId.NativeSinitic,
            AWCultureNamingTraditionRules.ResolveEffectiveProfile(
                NamingProfileId.NativeSinitic, NamingProfileId.Western,
                fullyXiaized: false),
            "a legacy Western culture resolves to its native profile");
        Equal(NamingProfileId.NativeSinitic,
            AWCultureNamingTraditionRules.ResolveInheritedProfile(
                NamingProfileId.NativeSinitic, NamingProfileId.Western,
                fullyXiaized: false),
            "a culture split cannot inherit a Western profile");
        Equal(NamingProfileId.NativeSinitic,
            AWCultureNamingTraditionRules.ResolveActorProfile(
                NamingProfileId.NativeSinitic, NamingProfileId.Xia),
            "the species keeps its native personal-name profile in Xia culture");

        foreach (string assetId in included)
        {
            Equal(assetId + "_name",
                AWCultureNamingTraditionRules.ResolveGeneratorId(
                    NamingProfileId.NativeSinitic,
                    WesternNamingTradition.Von, AWNamingObjectKind.Actor,
                    assetId, "western_von_name"),
                assetId + " uses its current actor generator");
            Equal(assetId + "_city",
                AWCultureNamingTraditionRules.ResolveGeneratorId(
                    NamingProfileId.NativeSinitic,
                    WesternNamingTradition.Von, AWNamingObjectKind.City,
                    assetId, "western_city"),
                assetId + " uses its current city generator");
            Equal(assetId + "_kingdom",
                AWCultureNamingTraditionRules.ResolveGeneratorId(
                    NamingProfileId.NativeSinitic,
                    WesternNamingTradition.Von, AWNamingObjectKind.Kingdom,
                    assetId, "western_kingdom"),
                assetId + " uses its current kingdom generator");
            Equal(assetId + "_name",
                AWCultureNamingTraditionRules.ResolveFallbackGeneratorId(
                    NamingProfileId.NativeSinitic,
                    AWNamingObjectKind.Actor, assetId, "human_name"),
                assetId + " fallback cannot cross into another profile");
        }

        NativeSiniticNameParts japanese =
            AWNativeSiniticNamePartsRules.Resolve(
                "山田太郎", "山田", "太");
        Equal(true, japanese.Valid,
            "a current Japanese-library template produces valid parts");
        Equal("山田", japanese.FamilyName,
            "the tagged current-library surname is retained");
        Equal("太郎", japanese.GivenName,
            "untagged suffix words remain in the complete given name");

        NativeSiniticNameParts inherited =
            AWNativeSiniticNamePartsRules.Resolve(
                "山田太郎", "山田", "太", "赵");
        Equal(true, inherited.Valid,
            "an inherited family can replace the generated family");
        Equal("赵", inherited.FamilyName,
            "the selected parent family remains authoritative");
        Equal("太郎", inherited.GivenName,
            "family inheritance preserves every generated given-name word");
        Equal("赵太郎", inherited.DisplayName,
            "native Sinitic projection stays surname first without a space");

        NativeSiniticNameParts shanhai =
            AWNativeSiniticNamePartsRules.Resolve("狐甲子", "狐", "");
        Equal(true, shanhai.Valid,
            "a current Shanhai-library template produces valid parts");
        Equal("甲子", shanhai.GivenName,
            "a template without a given tag derives the complete suffix");

        Equal(false,
            AWNativeSiniticNamePartsRules.Resolve("甲子", "", "甲").Valid,
            "an empty generated surname rejects the identity");
        Equal(false,
            AWNativeSiniticNamePartsRules.Resolve("甲狐子", "狐", "甲").Valid,
            "a surname outside the visible prefix rejects the identity");
        Equal(false,
            AWNativeSiniticNamePartsRules.Resolve("狐", "狐", "").Valid,
            "a surname-only result rejects the identity");

        Equal("赵",
            AWNativeSiniticNamePartsRules.ResolveLineageFamily(
                hasExistingShi: true, clanName: "赵",
                chineseFamilyName: "钱", familyName: "孙"),
            "an established Shi remains authoritative");
        Equal("钱",
            AWNativeSiniticNamePartsRules.ResolveLineageFamily(
                hasExistingShi: false, clanName: "赵",
                chineseFamilyName: "钱", familyName: "孙"),
            "a new branch preserves the current-library Chinese family");
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(
                $"{message}: expected {expected}, got {actual}");
    }
}
