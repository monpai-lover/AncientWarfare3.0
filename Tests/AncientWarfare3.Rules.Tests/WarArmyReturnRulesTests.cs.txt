    private static void ScopedParticipantDepartureRules()
    {
        Equal(true, WarArmyReturnRules.MatchesDepartedParticipant(
                missionWarId: 17L, missionKingdomId: 3L,
                departedWarId: 17L, departedKingdomId: 3L),
            "separate peace selects the departing kingdom mission");

                departedWarId: 17L, departedKingdomId: 3L),
            "separate peace preserves allied missions in the same war");

                missionWarId: 18L, missionKingdomId: 3L,
                departedWarId: 17L, departedKingdomId: 3L),
            "separate peace preserves the kingdom's other war missions");
    }

    private static void InvalidMissionFallbackRules()

            "a departed participant returns when validation catches up");

                armyAlive: true, missionExists: true,
                missionWarActive: false,
                missionKingdomParticipating: false),
            "an ended war returns when its callback was missed");
        Equal(false, WarArmyReturnRules.ShouldReturnInvalidMission(
                armyAlive: true, missionExists: true,
                missionWarActive: true,
                missionKingdomParticipating: true),
            "an invalid target in an active war replans instead of returning");
        Equal(false, WarArmyReturnRules.ShouldReturnInvalidMission(
                armyAlive: false, missionExists: true,
                missionWarActive: false,
                missionKingdomParticipating: false),
            "a dead army cannot enter fallback return");
    }

    private static void StaleTokensDoNotConsumeActiveFrameBudget()
    {
        var queue = new WarArmyReturnQueueCore();
        for (long armyId = 1L; armyId <= 8L; armyId++)
            queue.RestoreLegacyToken(armyId);
        for (long armyId = 9L; armyId <= 12L; armyId++)
            queue.Begin(armyId, 1L, 11L);

        IReadOnlyList<WarArmyReturnQueueOrder> frame = queue.TakeFrame(
            maximumActive: 4, maximumScans: 64);

        Equal(4, frame.Count,
            "stale tokens do not consume the four active-order budget");
        Equal(9L, frame[0].ArmyId, "first active order is preserved");
        Equal(12L, frame[3].ArmyId, "fourth active order is preserved");
        Equal(0, queue.WorkCount,
            "fixed scan budget drains the examined compatibility tokens");
        Equal(12, queue.LastFrameScanCount,
            "compatibility cleanup scans each stale and active node once");

        for (long armyId = 20L; armyId <= 27L; armyId++)
            queue.RestoreLegacyToken(armyId);
        Equal(0, queue.TakeFrame(maximumActive: 4, maximumScans: 3).Count,
            "a finite scan ceiling may stop before an active order");
        Equal(5, queue.WorkCount,
            "the frame drain examines no more than its explicit scan ceiling");
    }

    private static void BatchRequeueWaitsForNextFrame()

        Equal(4, queue.WorkCount,
            "requeued orders wait in the next frame queue");
        Equal(4, frame.Count,
            "requeue does not mutate the current frame batch");
    }

    private static void Collect(List<string> failures, bool condition,
        string message)
    {
        if (!condition) failures.Add(message);
    }

    private static string Section(string source, string start, string end)
    {
        int startIndex = source.IndexOf(start, StringComparison.Ordinal);
        int endIndex = source.IndexOf(end, startIndex + start.Length,
            StringComparison.Ordinal);
        if (startIndex < 0 || endIndex < 0)
            throw new InvalidOperationException("source section not found");
        return source.Substring(startIndex, endIndex - startIndex);
    }

    private static void True(bool value, string message)
    {
        if (!value) throw new InvalidOperationException(message);
    }

    private static void False(bool value, string message)
    {
        if (value) throw new InvalidOperationException(message);
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(
                $"{message}: expected {expected}, got {actual}");
    }
