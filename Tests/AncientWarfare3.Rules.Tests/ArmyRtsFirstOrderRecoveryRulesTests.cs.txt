using System;
using System.Collections.Generic;
using System.IO;
using AncientWarfare3.core.lineage;

internal static class ArmyRtsFirstOrderRecoveryRulesTests
{
    internal static void Run()
    {
        RetryKeepsTheFirstOrderAlive();
        TerminalResultsReleaseTheDeduplicationKey();
        RemovingWarClearsInflightOrders();
        RetryDoesNotStarveRegularPlanning();
        FirstOrderSelectionSkipsOnlyBusyArmies();

        var queue = new KingdomWarFirstOrderQueue();
        queue.EnqueueWar(7L, new[] { 10L, 11L }, new[] { 20L, 21L });
        Equal(4, queue.Count,
            "one first-order item is queued for each participant");
        Equal(new WarFirstOrderAssignment(7L, 10L,
                WarFirstOrderSide.Attacker), TakeAndComplete(queue),
            "the main attacker receives the first pulse");
        Equal(new WarFirstOrderAssignment(7L, 20L,
                WarFirstOrderSide.Defender), TakeAndComplete(queue),
            "the main defender receives the second pulse");
        Equal(new WarFirstOrderAssignment(7L, 11L,
                WarFirstOrderSide.Attacker), TakeAndComplete(queue),
            "additional attackers remain in their lane");
        Equal(new WarFirstOrderAssignment(7L, 21L,
                WarFirstOrderSide.Defender), TakeAndComplete(queue),
            "additional defenders alternate fairly");
        queue.EnqueueWar(8L, new[] { 30L }, new[] { 40L });
        queue.EnqueueWar(8L, new[] { 30L }, new[] { 40L });
        Equal(2, queue.Count,
            "duplicate war callbacks coalesce first-order work");

        Equal(true, ArmyRtsAssignmentReconciliationRules.
                MustRehydrateRetainedMission(
                    handoffLatched: true, watchdogRegistered: false,
                    sameStrategicIntent: true,
                    replacementPublished: false),
            "a retained mission rehydrates the broken online handoff");
        Equal(false, ArmyRtsAssignmentReconciliationRules.
                MustRehydrateRetainedMission(
                    handoffLatched: false, watchdogRegistered: false,
                    sameStrategicIntent: true,
                    replacementPublished: false),
            "an ordinary retained mission does not reset its runtime");

        string root = Directory.GetCurrentDirectory();
        string director = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "KingdomWarDirectorService.cs"));
        Assert(director.Contains("FirstOrderQueue",
                StringComparison.Ordinal) &&
               director.Contains("CaptureFirstOrder(",
                   StringComparison.Ordinal) &&
               director.Contains("TryAssignFirstOrderMission(",
                   StringComparison.Ordinal),
            "the director publishes a bounded first mission before refinement");
        Assert(director.Contains(
                "EnqueueFirstOrderForKingdom(pKingdom)",
                StringComparison.Ordinal),
            "a newly created army wakes a previously consumed first order");
        int firstOrderStart = director.IndexOf(
            "private static WarFirstOrderResult TryAssignFirstOrderMission(",
            StringComparison.Ordinal);
        int firstOrderEnd = director.IndexOf(
            "private static bool TryTakeWarIdBatch(", firstOrderStart,
            StringComparison.Ordinal);
        string firstOrderBody = director.Substring(firstOrderStart,
            firstOrderEnd - firstOrderStart);
        Assert(!firstOrderBody.Contains("HasActiveMissionForKingdom",
                StringComparison.Ordinal),
            "one active army cannot suppress first-order assignment for the kingdom");

        string lifecycle = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsWarLifecycleService.cs"));
        int warStart = lifecycle.IndexOf("public static void OnWarStarted(",
            StringComparison.Ordinal);
        int missionAssigned = lifecycle.IndexOf(
            "public static ArmyRtsWarLifecycleRecord OnMissionAssigned(",
            warStart, StringComparison.Ordinal);
        string warStartBody = lifecycle.Substring(warStart,
            missionAssigned - warStart);
        Assert(!warStartBody.Contains("World.world.armies",
                StringComparison.Ordinal),
            "war start cannot scan every world army");
        Assert(lifecycle.Contains("ProcessAuthorityCycle(",
                StringComparison.Ordinal) &&
               lifecycle.Contains(
                   "ArmyStrategicIndexService.CreateSnapshotCursor(",
                   StringComparison.Ordinal),
            "war lifecycle discovery advances through a bounded index cursor");

        string controller = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsControllerService.cs"));
        Assert(controller.Contains("MustRehydrateRetainedMission(",
                StringComparison.Ordinal) &&
               controller.Contains("IsRegistered(",
                   StringComparison.Ordinal),
            "retained missions repair route and watchdog state online");
    }

    private static void RetryDoesNotStarveRegularPlanning()
    {
        Equal(false, KingdomWarDirectorWorkRules.
                ShouldYieldAfterFirstOrderBatch(assignedCount: 0),
            "a retried first order cannot block regular war planning");
        Equal(true, KingdomWarDirectorWorkRules.
                ShouldYieldAfterFirstOrderBatch(assignedCount: 1),
            "a published first order yields the remaining director slice");

        string director = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "KingdomWarDirectorService.cs"));
        Assert(director.Contains("assigned++", StringComparison.Ordinal),
            "first-order processing counts published missions, not attempts");
    }

    private static void FirstOrderSelectionSkipsOnlyBusyArmies()
    {
        Equal(true, KingdomWarDirectorRules.ShouldSelectFirstOrderArmy(
                eligible: true, hasMission: false),
            "an eligible army without a mission can receive a first order");
        Equal(false, KingdomWarDirectorRules.ShouldSelectFirstOrderArmy(
                eligible: true, hasMission: true),
            "one busy army must not be selected again for the kingdom");
        Equal(false, KingdomWarDirectorRules.ShouldSelectFirstOrderArmy(
                eligible: false, hasMission: false),
            "ineligible armies remain excluded from first-order selection");
    }

    private static void RetryKeepsTheFirstOrderAlive()
    {
        var queue = new KingdomWarFirstOrderQueue();
        var assignment = new WarFirstOrderAssignment(31L, 41L,
            WarFirstOrderSide.Attacker);
        queue.EnqueueWar(31L, new[] { 41L }, Array.Empty<long>());

        Equal(assignment, Take(queue),
            "the pending first order enters processing");
        Equal(0, queue.Count,
            "in-flight work is not also present in the ready queue");
        Equal(true, queue.Complete(assignment,
                WarFirstOrderResult.RetryLater),
            "temporary failure returns the first order to the queue");
        Equal(1, queue.Count,
            "retried first order remains pending");

        queue.EnqueueWar(31L, new[] { 41L }, Array.Empty<long>());
        Equal(1, queue.Count,
            "a duplicate war callback cannot duplicate retried work");
        Equal(assignment, Take(queue),
            "retried work is available on a later pass");
        Equal(true, queue.Complete(assignment,
                WarFirstOrderResult.Assigned),
            "successful assignment acknowledges the order");
        Equal(0, queue.Count, "acknowledged work leaves the queue");
    }

    private static void TerminalResultsReleaseTheDeduplicationKey()
    {
        var queue = new KingdomWarFirstOrderQueue();
        var assignment = new WarFirstOrderAssignment(51L, 61L,
            WarFirstOrderSide.Defender);
        queue.EnqueueWar(51L, Array.Empty<long>(), new[] { 61L });
        Equal(assignment, Take(queue), "terminal test takes the order");
        Equal(true, queue.Complete(assignment,
                WarFirstOrderResult.Discard),
            "invalid war work is discarded explicitly");

        queue.EnqueueWar(51L, Array.Empty<long>(), new[] { 61L });
        Equal(1, queue.Count,
            "terminal completion releases the deduplication key");
    }

    private static void RemovingWarClearsInflightOrders()
    {
        var queue = new KingdomWarFirstOrderQueue();
        var assignment = new WarFirstOrderAssignment(71L, 81L,
            WarFirstOrderSide.Attacker);
        queue.EnqueueWar(71L, new[] { 81L }, Array.Empty<long>());
        Equal(assignment, Take(queue),
            "war-removal test moves the order in flight");

        queue.RemoveWar(71L);
        Equal(false, queue.Complete(assignment,
                WarFirstOrderResult.RetryLater),
            "ended war cannot requeue its in-flight first order");

        queue.EnqueueWar(71L, new[] { 81L }, Array.Empty<long>());
        Equal(1, queue.Count,
            "war removal releases all first-order deduplication state");
    }

    private static WarFirstOrderAssignment Take(
        KingdomWarFirstOrderQueue queue)
    {
        if (!queue.TryTake(out WarFirstOrderAssignment item))
            throw new InvalidOperationException("expected queued work");
        return item;
    }

    private static WarFirstOrderAssignment TakeAndComplete(
        KingdomWarFirstOrderQueue queue)
    {
        WarFirstOrderAssignment item = Take(queue);
        if (!queue.Complete(item, WarFirstOrderResult.Assigned))
            throw new InvalidOperationException(
                "expected first-order acknowledgement");
        return item;
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(message +
                $"; expected={expected}, actual={actual}");
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
}
