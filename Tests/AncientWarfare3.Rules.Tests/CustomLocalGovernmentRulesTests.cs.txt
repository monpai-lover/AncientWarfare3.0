using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using AncientWarfare3.core.court;

internal static class CustomLocalGovernmentRulesTests
{
    public static void Run()
    {
        Equal(CustomLocalGovernmentDefaultKind.Military,
            CustomLocalGovernmentRules.SelectDefault(false, true, false),
            "foreign land border selects military government");
        Equal(CustomLocalGovernmentDefaultKind.Military,
            CustomLocalGovernmentRules.SelectDefault(false, false, true),
            "frontier military role selects military government");
        Equal(CustomLocalGovernmentDefaultKind.Civil,
            CustomLocalGovernmentRules.SelectDefault(false, false, false),
            "ordinary interior city selects civil government");
        Equal(CustomLocalGovernmentDefaultKind.Manual,
            CustomLocalGovernmentRules.SelectDefault(true, true, true),
            "manual template binding always wins");
        Equal(CustomLocalGovernmentDefaultKind.Civil,
            CustomLocalGovernmentRules.SelectDefault(false, true, true, true),
            "capital city cannot default to military government");

        CustomLocalCourtTemplate civil =
            CustomLocalGovernmentPresetRules.CreateCivil("minzhou");
        CustomLocalCourtTemplate military =
            CustomLocalGovernmentPresetRules.CreateMilitary("junfu");
        Equal(4, civil.Offices.Count, "civil preset office count");
        Equal(4, military.Offices.Count, "military preset office count");
        HasOffice(civil, "minzhou_governor");
        HasOffice(civil, "minzhou_changshi");
        HasOffice(civil, "minzhou_sihu");
        HasOffice(civil, "minzhou_sicang");
        HasOffice(military, "junfu_dudu");
        HasOffice(military, "junfu_changshi");
        HasOffice(military, "junfu_sima");
        HasOffice(military, "junfu_canjun");
        Equal(3, civil.Edges.Count, "civil preset management edge count");
        Equal(3, military.Edges.Count,
            "military preset management edge count");
        Equal(CustomLocalCourtDefaultKind.CivilDefault, civil.DefaultKind,
            "civil preset default role");
        Equal(CustomLocalCourtDefaultKind.MilitaryDefault,
            military.DefaultKind, "military preset default role");
        IReadOnlyList<CustomLocalCourtTemplate> builtIns =
            CustomLocalGovernmentPresetRules.CreateBuiltInCatalog();
        Equal(2, builtIns.Count, "built-in local government count");
        Equal(4, Find(builtIns, "minzhou").Offices.Count,
            "built-in civil government has all vacancies");
        Equal(4, Find(builtIns, "junfu").Offices.Count,
            "built-in military government has all vacancies");
        IReadOnlyList<CustomLocalCourtTemplate> freshBuiltIns =
            CustomLocalGovernmentPresetRules.CreateBuiltInCatalog();
        True(!ReferenceEquals(builtIns[0], freshBuiltIns[0]),
            "built-in local templates are fresh mutable instances");
        True(military.Offices.First(office =>
            office.Id == "junfu_dudu").MilitaryCapable,
            "military commander can lead troops");
        True(military.Offices.First(office =>
            office.Id == "junfu_sima").MilitaryCapable,
            "military marshal can lead troops");

        CustomCourtTemplate legacy = LegacyTemplate(includeMilitary: false);
        CustomCourtTemplate upgraded =
            CustomLocalCourtTemplateRules.UpgradeLegacy(legacy);
        Equal(CustomLocalCourtTemplateRules.CurrentSchemaVersion,
            upgraded.SchemaVersion, "migration advances schema");
        Equal(4, Find(upgraded, "minzhou").Offices.Count,
            "pristine legacy civil template expands");
        Equal(4, Find(upgraded, "junfu").Offices.Count,
            "missing military template is added");
        string once = CustomCourtTemplateJsonCodec.Export(upgraded);
        string twice = CustomCourtTemplateJsonCodec.Export(
            CustomCourtTemplateJsonCodec.Normalize(upgraded));
        Equal(once, twice, "migration is idempotent");

        CustomCourtTemplate customized = LegacyTemplate(includeMilitary: true);
        CustomLocalCourtTemplate customCivil = Find(customized, "minzhou");
        customCivil.Offices[0].Grade = 11;
        CustomLocalCourtTemplateRules.UpgradeLegacy(customized);
        Equal(1, Find(customized, "minzhou").Offices.Count,
            "modified legacy template is preserved");
        Equal(11, Find(customized, "minzhou").Offices[0].Grade,
            "modified office properties are not overwritten");

        CustomLocalCourtTemplate rebased =
            CustomLocalGovernmentPresetRules.CreateMilitary("junfu");
        rebased.Offices[3].Requirements.RequiredOfficeId = "junfu_sima";
        CustomLocalCourtTemplateRules.RebaseOfficeIds(rebased,
            "junfu_copy_1");
        True(rebased.Offices.All(office => office.Id.StartsWith(
                "junfu_copy_1_office_", StringComparison.Ordinal)),
            "duplicate office ids are namespaced to the new template");
        True(rebased.Edges.All(edge => rebased.Offices.Any(office =>
                office.Id == edge.FromOfficeId) && rebased.Offices.Any(office =>
                office.Id == edge.ToOfficeId)),
            "duplicate management edges follow rebased office ids");
        True(rebased.Offices.Any(office => office.Requirements
                .RequiredOfficeId == "junfu_copy_1_office_3"),
            "duplicate office prerequisites follow rebased office ids");

        RuntimeUsesCityFacts();
    }

    private static CustomCourtTemplate LegacyTemplate(bool includeMilitary)
    {
        var template = new CustomCourtTemplate
        {
            SchemaVersion = 2,
            Id = "legacy_court",
            Name = new CustomCourtLocalizedText
            {
                Chinese = "旧朝廷",
                English = "Legacy Court"
            }
        };
        template.LocalTemplates.Add(LegacyLocal("minzhou", "民州",
            CustomLocalCourtDefaultKind.CivilDefault));
        if (includeMilitary)
            template.LocalTemplates.Add(LegacyLocal("junfu", "军府",
                CustomLocalCourtDefaultKind.MilitaryDefault));
        return template;
    }

    private static CustomLocalCourtTemplate LegacyLocal(string id,
        string name, CustomLocalCourtDefaultKind kind)
    {
        var local = new CustomLocalCourtTemplate
        {
            Id = id,
            Name = new CustomCourtLocalizedText
            {
                Chinese = name,
                English = name
            },
            DefaultKind = kind
        };
        local.Offices.Add(new CustomCourtOffice
        {
            Id = id + "_governor",
            Name = new CustomCourtLocalizedText
            {
                Chinese = "地方长官",
                English = "地方长官"
            },
            Layer = CourtOfficeLayer.City,
            Grade = 10,
            Slots = 1,
            Layout = new CustomCourtOfficeLayout()
        });
        return local;
    }

    private static CustomLocalCourtTemplate Find(CustomCourtTemplate template,
        string id)
    {
        return template.LocalTemplates.First(item => item.Id == id);
    }

    private static CustomLocalCourtTemplate Find(
        IReadOnlyList<CustomLocalCourtTemplate> templates, string id)
    {
        return templates.First(item => item.Id == id);
    }

    private static void HasOffice(CustomLocalCourtTemplate template,
        string id)
    {
        True(template.Offices.Any(office => office.Id == id),
            "missing office " + id);
    }

    private static void RuntimeUsesCityFacts()
    {
        string root = Directory.GetCurrentDirectory();
        string runtime = File.ReadAllText(Path.Combine(root, "Code", "core",
            "court", "CustomCourtRuntime.cs"));
        string cityFacts = File.ReadAllText(Path.Combine(root, "Code", "core",
            "court", "CustomLocalGovernmentCityService.cs"));
        string economy = File.ReadAllText(Path.Combine(root, "Code", "core",
            "policy", "CityEconomyService.cs"));
        string effects = File.ReadAllText(Path.Combine(root, "Code", "core",
            "court", "CustomCourtRuntimeEffectService.cs"));
        string battle = File.ReadAllText(Path.Combine(root, "Code", "core",
            "lineage", "ArmyAbstractBattleService.cs"));
        string workflow = File.ReadAllText(Path.Combine(root, "Code", "ui",
            "windows", "CustomCourtWorkflowWindow.cs"));
        True(runtime.Contains(
                "CustomLocalGovernmentCityService.HasForeignLandBorder("),
            "runtime reads actual city land borders");
        True(runtime.Contains("CityEconomyService.IsFrontierMilitary("),
            "runtime reads the city economy role");
        True(!runtime.Contains(
                "MilitaryGovernorateStore.TryGetRuntimeProjection("),
            "realm-wide military governorate state is not a city fact");
        True(cityFacts.Contains("pCity.neighbours_kingdoms"),
            "border adapter stays bounded to city neighbours");
        True(economy.Contains("public static bool IsFrontierMilitary("),
            "economy exposes the stored frontier role");
        True(effects.Contains("BuildCityModifiers("),
            "local office effects are aggregated in one kingdom batch");
        True(effects.Contains("row.city_id != pCity.id"),
            "local incumbents are scoped to their city");
        True(effects.Contains("snapshot.LocalTemplates"),
            "runtime effects inspect local templates inside the custom court");
        True(effects.Contains("BuildBoundedArmyModifier("),
            "local military effects use a bounded aggregation path");
        True(effects.Contains("activeLocalTemplateIds.Add("),
            "each active local template contributes to army effects once");
        True(effects.Contains("IsActiveLocalTemplateRow("),
            "stale local office rows do not grant army effects");
        True(effects.Contains("FindOfficeForIncumbent("),
            "duplicate local office ids resolve through the incumbent city");
        True(effects.Contains("row.city_id"),
            "local influence lookup retains city context");
        True(battle.Contains(
                "MoraleModifier = ResolveMoraleModifier(kingdom, moraleByKingdom)"),
            "attacking abstract armies receive custom court morale");
        True(battle.Contains(
                "MoraleModifier = ResolveMoraleModifier(defenderKingdom, moraleByKingdom)"),
            "defending abstract armies receive custom court morale");
        True(battle.Contains(
                "CustomCourtRuntimeEffectService.GetArmyModifier(pKingdom)"),
            "abstract battle morale uses the custom court effect service");
        True(workflow.Contains(
                "CustomLocalCourtTemplateRules.RebaseOfficeIds(clone, id)"),
            "duplicated local templates receive unique office ids");
        True(workflow.Contains(
                "if (pDefaultKind == CustomLocalCourtDefaultKind.ManualOnly)"),
            "built-in local defaults preserve bilingual preset names");
        True(economy.Contains("cityEffectModifiers.TryGetValue("),
            "city economy consumes per-city local office modifiers");
    }

    private static void True(bool actual, string message)
    {
        if (!actual) throw new InvalidOperationException(message);
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(message + ": expected " +
                expected + ", got " + actual);
    }
}
