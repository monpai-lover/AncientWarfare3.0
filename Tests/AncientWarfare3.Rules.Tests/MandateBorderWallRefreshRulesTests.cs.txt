using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using AncientWarfare3.core.lineage;

internal static class MandateBorderWallRefreshRulesTests
{
    public static void Run()
    {
        UsesSingleTileWallWidth();
        StateStartsInactiveAndEmpty();
        RequiresDecisionActivation();
        SelectsOnlyTheChangedLocalNeighborhood();
        RestoresOnlyTheWallTypeOwnedByTheManifest();
        ExpiresOnlyAtTheWallLifespan();
        FiltersVerticalSegments();
    }

    private static void UsesSingleTileWallWidth()
    {
        string root = Environment.CurrentDirectory;
        string source = File.ReadAllText(Path.Combine(root, "Code", "core",
            "lineage", "MandateBorderWallRefreshService.cs"));
        True(source.Contains("private const int WallWidth = 1;",
                StringComparison.Ordinal),
            "mandate border walls use one tile of thickness");
    }

    private static void StateStartsInactiveAndEmpty()
    {
        var state = new MandateBorderWallState();
        Equal(2, state.SchemaVersion,
            "wall state starts at schema version two");
        False(state.Activated,
            "wall lifecycle is not active before the decision");
        Equal(0, state.Cities.Count,
            "wall state owns no city manifests initially");
        var point = new MandateBorderWallPointState
        {
            X = 3,
            Y = 4,
            OriginalTopTypeId = "road"
        };
        Equal("road", point.OriginalTopTypeId,
            "manifest point preserves the prior top tile id");
    }

    private static void RequiresDecisionActivation()
    {
        False(MandateBorderWallRefreshRules.ShouldRefresh(
            activated: false, cityEligible: true),
            "war and ownership events do not activate border walls");
        True(MandateBorderWallRefreshRules.ShouldRefresh(
            activated: true, cityEligible: true),
            "an activated eligible border city refreshes");
        False(MandateBorderWallRefreshRules.ShouldRefresh(
            activated: true, cityEligible: false),
            "an obsolete border city restores without rebuilding");
    }

    private static void SelectsOnlyTheChangedLocalNeighborhood()
    {
        SetEqual(new long[] { 2, 3, 4 },
            MandateBorderWallRefreshRules.AffectedCityIds(
                changedCityId: 2, previousCityId: 3,
                neighbourCityIds: new long[] { 2, 4, -1, 4 }),
            "refresh scope is the changed city, old city, and direct neighbor");
    }

    private static void RestoresOnlyTheWallTypeOwnedByTheManifest()
    {
        True(MandateBorderWallRefreshRules.ShouldRestore(
            currentTopTypeId: "wall_order",
            placedWallTypeId: "wall_order"),
            "an unchanged owned wall may be restored");
        False(MandateBorderWallRefreshRules.ShouldRestore(
            currentTopTypeId: "road",
            placedWallTypeId: "wall_order"),
            "a later player or system edit is preserved");
        False(MandateBorderWallRefreshRules.ShouldRestore(
            currentTopTypeId: "wall_order",
            placedWallTypeId: ""),
            "a corrupt manifest cannot restore a tile");
    }

    private static void ExpiresOnlyAtTheWallLifespan()
    {
        False(MandateBorderWallRefreshRules.HasExpired(2049, 2000),
            "a wall remains before fifty full years");
        True(MandateBorderWallRefreshRules.HasExpired(2050, 2000),
            "a wall expires at fifty full years");
        False(MandateBorderWallRefreshRules.HasExpired(2000,
                int.MinValue),
            "an uninitialized build year is not treated as expired");
    }

    private static void FiltersVerticalSegments()
    {
        var points = new HashSet<string>(StringComparer.Ordinal)
        {
            "1,1", "2,1", "2,2"
        };
        True(MandateBorderWallRefreshRules.ShouldKeepHorizontalSegment(
            1, 1, (x, y) => points.Contains(x + "," + y)),
            "a horizontal segment point is retained");
        True(MandateBorderWallRefreshRules.ShouldKeepHorizontalSegment(
            2, 1, (x, y) => points.Contains(x + "," + y)),
            "a horizontal segment endpoint is retained");
        False(MandateBorderWallRefreshRules.ShouldKeepHorizontalSegment(
            2, 2, (x, y) => points.Contains(x + "," + y)),
            "a vertical-only point is removed");
    }

    private static void SetEqual(IEnumerable<long> expected,
        IEnumerable<long> actual, string message)
    {
        long[] left = expected.OrderBy(value => value).ToArray();
        long[] right = actual.OrderBy(value => value).ToArray();
        if (left.SequenceEqual(right)) return;
        throw new InvalidOperationException(message + ": expected [" +
            string.Join(",", left) + "], got [" +
            string.Join(",", right) + "]");
    }

    private static void True(bool value, string message)
    {
        if (!value) throw new InvalidOperationException(message);
    }

    private static void False(bool value, string message)
    {
        if (value) throw new InvalidOperationException(message);
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(message +
                $": expected {expected}, got {actual}");
    }
}
