using AncientWarfare3.core.court;

internal static class WesternMayorTermRulesTests
{
    public static void Run()
    {
        Equal(10, WesternMayorTermRules.TermYears,
            "Western Mayor term is exactly ten years");
        Equal(110, WesternMayorTermRules.InitialCycleEndYear(100),
            "the first shared cycle lasts exactly ten years");

        Equal(110, WesternMayorTermRules.AppointmentTermEndYear(
                currentYear: 103, sharedCycleEndYear: 110),
            "all Mayors inherit the kingdom shared cycle");
        Equal(110, WesternMayorTermRules.AppointmentTermEndYear(
                currentYear: 109, sharedCycleEndYear: 110),
            "new cities and vacancy replacements inherit the live cycle");

        Equal(120, WesternMayorTermRules.AdvanceExpiredCycleEndYear(
                currentYear: 117, sharedCycleEndYear: 90),
            "an expired shared cycle advances in ten-year increments");
        Equal(120, WesternMayorTermRules.AdvanceExpiredCycleEndYear(
                currentYear: 110, sharedCycleEndYear: 110),
            "a completed cycle advances by exactly ten years");

        Equal(111, WesternMayorTermRules.RetryTermEndYear(110),
            "a one-city realm retries one year later");
        Equal(110, WesternMayorTermRules.SharedCycleAfterFailedRotation(110),
            "a failed or undersized rotation does not advance the shared cycle");

        RuntimeIntegrationUsesSharedCycleAndExplicitOffice();
    }

    private static void RuntimeIntegrationUsesSharedCycleAndExplicitOffice()
    {
        string root = Directory.GetCurrentDirectory();
        string keys = File.ReadAllText(Path.Combine(root, "Code", "core",
            "lineage", "LineageKeys.cs"));
        string service = File.ReadAllText(Path.Combine(root, "Code", "core",
            "court", "OfficialCareerStateService.cs"));
        Contains(keys, "WESTERN_MAYOR_CYCLE_END_YEAR",
            "kingdom data exposes the shared Mayor cycle key");
        Contains(service, "ResolveWesternMayorTermEndYear(pKingdom",
            "formal Mayor appointments read the kingdom shared cycle");
        Contains(service,
            "OfficialCirculationRules.IsRotatingCityOffice(",
            "annual rotation identifies Mayor independently of Xia ranks");
        Contains(service, "bool xiaCirculationUnlocked",
            "annual rotation reads the Nine-Rank circulation unlock");
        Contains(service, "customCityLeader && xiaCirculationUnlocked",
            "custom city leaders follow the circulation unlock");
        Contains(service, "LocalOfficialTermRules.TermLength(",
            "ordinary local rotation assigns a fresh finite term");
        Contains(service,
            "item.Mutation.State.OfficeId != CourtOfficeId.WestMayor",
            "ordinary local rotation does not overwrite the Mayor shared cycle");
        Contains(service, "dueLocalSubordinates = new List<Actor>()",
            "expired local subordinate officials enter replacement handling");
        Contains(service, "TryDismissOfficer(dueLocalSubordinate",
            "expired local subordinate officials are released for refilling");
        Contains(service, "RenewLocalTerm(mutation, year)",
            "locked local leaders receive a fresh finite term");
        Contains(service, "pYearAfter(year)",
            "single-city local leaders defer rotation without expiring permanently");
        Contains(service, "item.Mutation.State.OfficeId",
            "rotation persistence uses each assignment's explicit office id");
        Contains(service, "AdvanceExpiredCycleEndYear(",
            "successful Mayor circulation advances the shared cycle");
        int processStart = service.IndexOf(
            "private static void ProcessDueGovernorRotations",
            StringComparison.Ordinal);
        int persistenceCommit = service.IndexOf(
            "CommitGovernorRotationPersistence(pKingdom, pPlan)",
            processStart, StringComparison.Ordinal);
        int firstRuntimeMove = service.IndexOf(
            "item.FormerCity.removeLeader()", processStart,
            StringComparison.Ordinal);
        Equal(true, processStart >= 0 && persistenceCommit > processStart &&
                    (firstRuntimeMove < 0 || firstRuntimeMove > persistenceCommit),
            "rotation persistence commits before runtime projection");
        Equal(false, service.Contains(
                "Parameters.AddWithValue(\"@office\",\r\n                        CourtOfficeId.Governor)"),
            "rotation SQL no longer hard-codes Governor");

        PersonalBiographyRecordsOfficialTransfers(service);
    }

    private static void PersonalBiographyRecordsOfficialTransfers(string service)
    {
        string keys = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "ChronicleKeys.cs"));
        string biographyRules = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "court",
            "OfficialCareerBiographyRules.cs"));
        string chronicle = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "ChronicleEvents.cs"));
        string localization = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "lineage",
            "HistoryLocalizationRules.cs"));
        string appointmentWindow = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "ui", "windows",
            "CourtAppointmentWindow.cs"));
        string appointmentRules = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "court",
            "CourtManualAppointmentRules.cs"));

        Contains(keys, "OFFICIAL_TRANSFERRED = \"official_transferred\"",
            "official transfer has a dedicated person event");
        Contains(biographyRules, "case \"official_transferred\":",
            "official transfer is included in career biography events");
        Contains(chronicle, "OnOfficialTransferred(Actor pActor",
            "official transfer has a dedicated chronicle writer");
        Contains(chronicle, "HistoryWriter.TryRecordPerson(pActor.data.id",
            "official transfer is written to the person's biography");
        Contains(chronicle, "official_transfer:",
            "official transfer biography writes are idempotent");
        Contains(chronicle, "HistoryTarget.City(pDestinationCity)",
            "official transfer points to the destination city");
        Contains(localization, "aw_hist_event_official_transferred",
            "official transfer has an event label");
        Contains(localization, "aw_hist_official_transfer_from",
            "official transfer has localized movement text");
        Contains(appointmentRules, "ShouldRebaseVacancyTarget(",
            "manual appointment rules support stale vacancy rebasing");
        Contains(appointmentWindow,
            "ReadManualAppointmentIncumbentActorId(",
            "appointment window reads the current incumbent before scanning");
        Contains(appointmentWindow,
            "ShouldRebaseVacancyTarget(_expectedIncumbentActorId",
            "appointment window rebases a stale vacancy into replacement mode");

        int processStart = service.IndexOf(
            "private static void ProcessDueGovernorRotations",
            StringComparison.Ordinal);
        int commit = service.IndexOf(
            "if (!CommitGovernorRotationPersistence(pKingdom, pPlan))",
            processStart, StringComparison.Ordinal);
        int transfer = service.IndexOf(
            "ChronicleEvents.OnOfficialTransferred(", processStart,
            StringComparison.Ordinal);
        int publish = service.IndexOf(
            "PublishCommittedGovernorRotation(pPlan);", processStart,
            StringComparison.Ordinal);
        Equal(true, commit >= 0 && publish >= 0 && transfer > publish,
            "successful rotation commits before biography records are written");
        Contains(service, "item.FormerCity, item.DestinationCity",
            "transfer biography receives both former and destination cities");
    }

    private static void Contains(string source, string expected,
        string message)
    {
        if (!source.Contains(expected, StringComparison.Ordinal))
            throw new InvalidOperationException(message + ": missing " + expected);
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(
                $"{message}: expected {expected}, got {actual}");
    }
}
