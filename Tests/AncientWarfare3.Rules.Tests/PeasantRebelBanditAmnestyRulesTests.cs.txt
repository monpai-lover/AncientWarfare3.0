using System;
using System.IO;
using AncientWarfare3.core.lineage;

internal static class PeasantRebelBanditAmnestyRulesTests
{
    internal static void Run()
    {
        Assert(PeasantRebelBanditAmnestyRules.CanAccept(
                bandit: true, strongholdActive: true, originValid: true,
                offeringIsOrigin: true, authoritative: true, applying: false),
            "the origin may grant amnesty to an active bandit stronghold");
        Assert(!PeasantRebelBanditAmnestyRules.CanAccept(
                bandit: true, strongholdActive: true, originValid: true,
                offeringIsOrigin: false, authoritative: true, applying: false),
            "a third party cannot grant origin amnesty");
        Assert(PeasantRebelBanditAmnestyRules.ShouldEndWars(true),
            "accepted amnesty ends bandit wars");
        Assert(PeasantRebelBanditAmnestyRules.ResolveSettlementClass(true) ==
                "default",
            "accepted amnesty restores the ordinary government class");
        Assert(PeasantRebelBanditAmnestyRules.ResolveFailureKey(
                false, false, false, false) == "not_bandit_stronghold",
            "invalid target has a stable failure reason");
        Assert(PeasantRebelBanditAmnestyRules.CanSelectOffice(
                true, true, true),
            "a vacant eligible office may be promised");
        Assert(!PeasantRebelBanditAmnestyRules.CanSelectOffice(
                true, false, true),
            "an occupied office cannot be promised");
        Assert(!PeasantRebelBanditAmnestyRules.CanSelectOffice(
            true, true, false),
            "an ineligible leader cannot receive an office promise");
        Assert(PeasantRebelBanditAmnestyRules.IncludesOffice(
                BanditAmnestyRewardKind.OfficeAndFief) &&
               PeasantRebelBanditAmnestyRules.IncludesFief(
                BanditAmnestyRewardKind.OfficeAndFief),
            "an amnesty offer may promise an office and a fief together");
        Assert(!PeasantRebelBanditAmnestyRules.IncludesFief(
                BanditAmnestyRewardKind.Office) &&
               PeasantRebelBanditAmnestyRules.IncludesFief(
                BanditAmnestyRewardKind.Fief),
            "fief selection is independent from the office reward");
        Assert(PeasantRebelBanditAmnestyRules.IsRewardRecipientNaturalized(
                leaderBelongsToOrigin: true,
                leaderHasCity: true,
                leaderCityBelongsToOrigin: true),
            "a leader settled in an origin city may receive the reward");
        Assert(PeasantRebelBanditAmnestyRules.IsRewardRecipientNaturalized(
                leaderBelongsToOrigin: true,
                leaderHasCity: false,
                leaderCityBelongsToOrigin: false),
            "a naturalized leader without a city may still receive a reward");
        Assert(!PeasantRebelBanditAmnestyRules.IsRewardRecipientNaturalized(
                leaderBelongsToOrigin: false,
                leaderHasCity: true,
                leaderCityBelongsToOrigin: true),
            "a foreign leader cannot receive a domestic office or fief");
        Assert(!PeasantRebelBanditAmnestyRules.IsRewardRecipientNaturalized(
                leaderBelongsToOrigin: true,
                leaderHasCity: true,
                leaderCityBelongsToOrigin: false),
            "a stale foreign residence makes naturalization incomplete");
        Assert(PeasantRebelBanditAmnestyRules.ResolveAiDecision(
                banditStrength: 151, originStrength: 100) ==
                BanditAmnestyAiDecision.Amnesty,
            "a materially stronger bandit force should prefer amnesty");
        Assert(PeasantRebelBanditAmnestyRules.ResolveAiDecision(
                banditStrength: 100, originStrength: 151) ==
                BanditAmnestyAiDecision.Suppression,
            "a weaker bandit force should be suppressed");
        Assert(PeasantRebelBanditAmnestyRules.ResolveAiDecision(
                banditStrength: 1, originStrength: 0) ==
                BanditAmnestyAiDecision.Amnesty,
            "a live bandit with no measurable origin strength is not " +
            "forced into a pointless suppression war");
        Assert(PeasantRebelBanditAmnestyRules.CanAdvance(
                BanditAmnestySettlementPhase.Prepared,
                BanditAmnestySettlementPhase.TerritorialSettlement),
            "prepared settlement advances to territorial work");
        Assert(PeasantRebelBanditAmnestyRules.CanAdvance(
                BanditAmnestySettlementPhase.RewardPending,
                BanditAmnestySettlementPhase.Completed),
            "fulfilled reward completes settlement");
        Assert(!PeasantRebelBanditAmnestyRules.CanAdvance(
            BanditAmnestySettlementPhase.Completed,
            BanditAmnestySettlementPhase.RewardPending),
            "completed settlement never moves backwards");

        string locales = File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Locales", "others.csv"));
        Assert(locales.Contains("aw_bandit_amnesty_unavailable,"),
            "the amnesty god-power failure message has a localized key");
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
}
