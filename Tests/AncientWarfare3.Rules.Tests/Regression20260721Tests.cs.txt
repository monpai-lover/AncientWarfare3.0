    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException($"{name}: expected {expected}, got {actual}");
    }

    private static string ResolveAnnexProgressState(bool pSuzerainValid,
        bool pTargetValid, bool pTargetDirectVassal,
        bool pIndependenceSuspended, float pProgress, float pCost)
    {
        return VassalAnnexDecisionRules.ResolveProgressState(
            pSuzerainValid, pTargetValid, pTargetDirectVassal,
            pIndependenceSuspended, pProgress, pCost).ToString();
    }
