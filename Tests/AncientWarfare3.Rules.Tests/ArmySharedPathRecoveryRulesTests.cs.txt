using System;
using System.IO;
using AncientWarfare3.core.lineage;

internal static class ArmySharedPathRecoveryRulesTests
{
    public static void Run()
    {
        Assert(ArmySharedPathRules.ShouldReuseInstalledSharedRoute(
                installedRevision: 4, availableRevision: 4,
                localPathCount: 3, actorFollowingLocalPath: true,
                atInstalledEndpoint: false),
            "a matching non-empty followed path remains reusable");
        Assert(ArmySharedPathRules.ShouldReuseInstalledSharedRoute(
                installedRevision: 4, availableRevision: 4,
                localPathCount: 0, actorFollowingLocalPath: false,
                atInstalledEndpoint: true),
            "a matching route at its endpoint remains complete");
        Assert(!ArmySharedPathRules.ShouldReuseInstalledSharedRoute(
                installedRevision: 4, availableRevision: 4,
                localPathCount: 0, actorFollowingLocalPath: false,
                atInstalledEndpoint: false),
            "an empty non-following historical install is not reusable");

        ArmySharedRouteInstallStatus stale = ArmySharedPathRules.
            ResolveCurrentInstallStatus(providerAvailable: true,
                transportActive: false, hasMatchingRevision: true,
                atInstalledEndpoint: false, localPathCount: 0,
                actorFollowingLocalPath: false,
                recordedStatus: ArmySharedRouteInstallStatus.Installed);
        Assert(stale == ArmySharedRouteInstallStatus.StaleInstalled,
            "current empty path truth overrides the historical installed status");
        Assert(ArmySharedPathRules.ShouldRecoverStaleInstalledRoute(stale,
                combatActive: false, transportActive: false),
            "an idle stale install requests reset and reinstall");
        Assert(!ArmySharedPathRules.ShouldRecoverStaleInstalledRoute(stale,
                combatActive: true, transportActive: false),
            "combat blocks shared-route recovery");
        Assert(!ArmySharedPathRules.ShouldRecoverStaleInstalledRoute(stale,
                combatActive: false, transportActive: true),
            "transport blocks shared-route recovery");
        Assert(!ArmySharedPathRules.ShouldUseLongRangeFollowerRoute(
                targetValid: false, targetDistanceSquared: 2500f),
            "an invalid follower target cannot start long-range movement");
        Assert(!ArmySharedPathRules.ShouldUseLongRangeFollowerRoute(
                targetValid: true, targetDistanceSquared: 64f),
            "an eight-tile target remains eligible for local correction");
        Assert(ArmySharedPathRules.ShouldUseLongRangeFollowerRoute(
                targetValid: true, targetDistanceSquared: 81f),
            "a nine-tile target uses the long-range follower route");
        Assert(ArmySharedPathRules.ShouldUseLongRangeFollowerRoute(
                targetValid: true, targetDistanceSquared: 2500f),
            "a fifty-tile target uses the long-range follower route");

        string root = Directory.GetCurrentDirectory();
        string controller = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyRtsControllerService.cs"));
        int recoveryStart = controller.IndexOf(
            "internal static bool RecoverEmptySharedRoute(",
            StringComparison.Ordinal);
        int recoveryEnd = controller.IndexOf(
            "private static void TrySubmitIndependentFollowerRecoveryRoute(",
            recoveryStart, StringComparison.Ordinal);
        string recoveryBody = controller.Substring(recoveryStart,
            recoveryEnd - recoveryStart);
        Assert(recoveryBody.Contains("RequestRouteReplan(pArmyId",
                StringComparison.Ordinal),
            "a stale captain path clears the controller anchor and replans");

        int captainTargetStart = controller.IndexOf(
            "public static bool TryGetCaptainTarget(Actor pActor,",
            StringComparison.Ordinal);
        int captainTargetEnd = controller.IndexOf(
            "public static ArmyFollowerTargetResult ResolveFollowerTarget(",
            captainTargetStart, StringComparison.Ordinal);
        string captainTargetBody = controller.Substring(captainTargetStart,
            captainTargetEnd - captainTargetStart);
        Assert(captainTargetBody.Contains(
                "ArmyFormationService.GetObservationProgress(army)",
                StringComparison.Ordinal) &&
            captainTargetBody.Contains(
                "observationComplete: observation.Complete",
                StringComparison.Ordinal),
            "captain movement blocks departure until the bounded formation observation completes");
        Assert(controller.Contains("HasLandEscortQuorum(escortPopulation",
                StringComparison.Ordinal),
            "controller rally readiness uses the ninety-percent land escort quorum");

        string watchdog = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "ArmyStallWatchdogService.cs"));
        Assert(watchdog.Contains("formation_observation_pending",
                StringComparison.Ordinal),
            "incomplete formation observation has an explicit diagnostic instead of an ambiguous zero count");

        int escortHoldStart = controller.IndexOf(
            "private static void UpdateEscortHold(",
            StringComparison.Ordinal);
        Assert(escortHoldStart >= 0,
            "the controller tracks continuous escort loss per army");
        int escortHoldEnd = controller.IndexOf(
            "private static ArmyRtsTransitionFacts BuildFacts(",
            escortHoldStart, StringComparison.Ordinal);
        string escortHoldBody = controller.Substring(escortHoldStart,
            escortHoldEnd - escortHoldStart);
        Assert(escortHoldBody.Contains("ShouldHoldAfterEscortLoss",
                StringComparison.Ordinal) &&
            !escortHoldBody.Contains("Invalidate(",
                StringComparison.Ordinal) &&
            !escortHoldBody.Contains("OnArmyChanged(",
                StringComparison.Ordinal) &&
            !escortHoldBody.Contains("EnsureOffensiveContinuity",
                StringComparison.Ordinal),
            "escort hold is tactical and leaves the strategic mission intact");
        Assert(captainTargetBody.Contains("runtime.EscortHoldActive",
                StringComparison.Ordinal),
            "captain target publication consumes the tactical escort hold");

        Assert(controller.Contains("pPreferAlternateSlot",
                StringComparison.Ordinal) &&
            controller.Contains("RecoverFormationMember(pArmyId,",
                StringComparison.Ordinal),
            "follower recovery has a member-scoped alternate-slot stage");

        int processFrameStart = controller.IndexOf(
            "public static void ProcessFrame(int pControllerBudget,",
            StringComparison.Ordinal);
        int processFrameEnd = controller.IndexOf(
            "private static void ProcessOne(", processFrameStart,
            StringComparison.Ordinal);
        string processFrameBody = controller.Substring(processFrameStart,
            processFrameEnd - processFrameStart);
        Assert(!processFrameBody.Contains(
                "ArmyRtsRuntimeModeRules.ShouldPlan(mode)",
                StringComparison.Ordinal) &&
            processFrameBody.Contains("mode == ArmyRtsMode.Off",
                StringComparison.Ordinal),
            "controller recovery runs on admitted large-step logical passes while planning stays separately throttled");
        Assert(watchdog.Contains("follower_recovery_action",
                StringComparison.Ordinal),
            "follower recovery stages are available in sampled RTS diagnostics");
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }
}
