using System;
using System.IO;
using AncientWarfare3.core.performance;

internal static class CityManagerMutationScopeTests
{
    internal static void Run()
    {
        CityManagerMutationScope.Reset();
        False(CityManagerMutationScope.IsCityUpdateActive,
            "scope starts inactive");

        CityManagerMutationScope.EnterCityUpdate();
        CityManagerMutationScope.EnterCityUpdate();
        True(CityManagerMutationScope.IsCityUpdateActive,
            "nested city updates remain active");

        CityManagerMutationScope.ExitCityUpdate();
        True(CityManagerMutationScope.IsCityUpdateActive,
            "one nested city update remains active");

        CityManagerMutationScope.ExitCityUpdate();
        False(CityManagerMutationScope.IsCityUpdateActive,
            "scope closes when the final city update exits");

        CityManagerMutationScope.ExitCityUpdate();
        False(CityManagerMutationScope.IsCityUpdateActive,
            "an unmatched exit cannot underflow the scope");

        CityManagerMutationScope.EnterCityUpdate();
        CityManagerMutationScope.Reset();
        False(CityManagerMutationScope.IsCityUpdateActive,
            "world reset clears an interrupted city update scope");

        string root = Directory.GetCurrentDirectory();
        string patchPath = Path.Combine(root, "Code", "patch",
            "AW_CityManagerMutationBoundaryPatch.cs");
        True(File.Exists(patchPath),
            "city manager update has a mutation boundary patch");
        string patch = File.Exists(patchPath)
            ? File.ReadAllText(patchPath)
            : string.Empty;
        True(patch.Contains("HarmonyPrefix", StringComparison.Ordinal) &&
             patch.Contains("nameof(CityManager.update)",
                 StringComparison.Ordinal) &&
             patch.Contains("CityManagerMutationScope.EnterCityUpdate()",
                 StringComparison.Ordinal),
            "city manager update enters the mutation scope");
        True(patch.Contains("HarmonyFinalizer", StringComparison.Ordinal) &&
             patch.Contains("CityManagerMutationScope.ExitCityUpdate()",
                 StringComparison.Ordinal) &&
             patch.Contains("return __exception", StringComparison.Ordinal),
            "city manager update always exits without swallowing errors");

        string disposalPath = Path.Combine(root, "Code", "core", "lineage",
            "BanditStrongholdCityDisposalService.cs");
        True(File.Exists(disposalPath),
            "bandit stronghold city disposal has a dedicated service");
        string disposal = File.Exists(disposalPath)
            ? File.ReadAllText(disposalPath)
            : string.Empty;
        True(disposal.Contains("CityManagerMutationScope.IsCityUpdateActive",
                 StringComparison.Ordinal) &&
             disposal.Contains("ProcessAuthorityCycle()",
                 StringComparison.Ordinal) &&
             disposal.Contains("World.world.cities.removeObject(",
                 StringComparison.Ordinal),
            "physical stronghold removal is guarded and authority-owned");
        True(disposal.Contains(
                "CanDisposeStrongholdCity(",
                StringComparison.Ordinal) &&
             disposal.Contains("Pending.Remove(cityId)",
                 StringComparison.Ordinal),
            "disposal validates the live city and removes only completed queue items");

        string stronghold = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "PeasantRebelBanditStrongholdService.cs"));
        False(stronghold.Contains("World.world.cities.removeObject(",
                StringComparison.Ordinal),
            "stronghold lifecycle code cannot mutate the live city set directly");
        True(stronghold.Contains(
                "BanditStrongholdCityDisposalService.Schedule(",
                StringComparison.Ordinal),
            "stronghold lifecycle schedules physical city disposal");
        string towerRemoval = ExtractSection(stronghold,
            "private static bool RemoveStrongholdTowers(",
            "private static bool RestoreWalls(");
        False(towerRemoval.Contains(
                "World.world.buildings.removeObject(",
                StringComparison.Ordinal),
            "stronghold tower removal cannot bypass the native building cleanup lifecycle");
        True(towerRemoval.Contains("building.removeBuildingFinal()",
                 StringComparison.Ordinal),
            "stronghold tower removal clears tiles and zone membership before disposal");
        True(towerRemoval.Contains("building.isRekt()",
                 StringComparison.Ordinal) &&
             towerRemoval.Contains("building.isRemoved()",
                 StringComparison.Ordinal) &&
             towerRemoval.Contains("building.isOnRemove()",
                 StringComparison.Ordinal),
            "stronghold tower removal is idempotent for invalid or already-cleaned buildings");

        string authority = File.ReadAllText(Path.Combine(root, "Code",
            "core", "performance", "AWAuthorityCycleService.cs"));
        string authorityReset = ExtractSection(authority,
            "public static void Reset()",
            "private static void ProcessCycle(");
        True(CountOccurrences(authorityReset,
                 "PeasantRebelBanditStrongholdPopulationService.Clear()") == 1,
            "world reset clears pending stronghold population settlement exactly once");
        True(CountOccurrences(authorityReset,
                 "BanditStrongholdCityDisposalService.Clear()") == 1,
            "world reset clears pending stronghold city disposal exactly once");
        string deferredAuthority = ExtractSection(authority,
            "private static void DrainDeferredAuthorityWork()",
            "private static void FlushPendingWarParticipantSources()");
        string compactDeferredAuthority = RemoveWhitespace(
            deferredAuthority);
        True(CountOccurrences(compactDeferredAuthority,
                 "PeasantRebelBanditStrongholdPopulationService." +
                 "ProcessAuthorityCycle()") == 1,
            "the bounded authority stage settles empty strongholds exactly once");
        True(CountOccurrences(compactDeferredAuthority,
                 "BanditStrongholdCityDisposalService.ProcessAuthorityCycle()") == 1,
            "the bounded authority stage drains stronghold city disposal exactly once");
        True(compactDeferredAuthority.Contains(
                 "DeferredRuntimeWorkRules.ResolveItemsPerAuthorityFrame(",
                 StringComparison.Ordinal) &&
             compactDeferredAuthority.Contains("DeferredRuntimeWorkService.DrainFrame(",
                 StringComparison.Ordinal),
            "stronghold settlement remains inside the existing bounded deferred authority stage");
    }

    private static void True(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }

    private static void False(bool condition, string message)
    {
        True(!condition, message);
    }

    private static string ExtractSection(string source, string startMarker,
        string endMarker)
    {
        int start = source.IndexOf(startMarker, StringComparison.Ordinal);
        if (start < 0) return string.Empty;
        int end = source.IndexOf(endMarker, start + startMarker.Length,
            StringComparison.Ordinal);
        return end < 0 ? source.Substring(start) :
            source.Substring(start, end - start);
    }

    private static int CountOccurrences(string source, string value)
    {
        int count = 0;
        int index = 0;
        while ((index = source.IndexOf(value, index,
                   StringComparison.Ordinal)) >= 0)
        {
            count++;
            index += value.Length;
        }
        return count;
    }

    private static string RemoveWhitespace(string source)
    {
        return source.Replace("\r", string.Empty,
                StringComparison.Ordinal)
            .Replace("\n", string.Empty, StringComparison.Ordinal)
            .Replace(" ", string.Empty, StringComparison.Ordinal)
            .Replace("\t", string.Empty, StringComparison.Ordinal);
    }
}
