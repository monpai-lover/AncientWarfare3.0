using System;
using System.Linq;
using AncientWarfare3.core.court;

internal static class RegionalGovernmentTemplateRulesTests
{
    public static void Run()
    {
        string oldCentral = "{\"SchemaVersion\":3,\"Id\":\"old_court\",\"Revision\":1," +
            "\"Name\":{\"Chinese\":\"旧朝\",\"English\":\"Old Court\"}," +
            "\"Offices\":[{\"Id\":\"minister\",\"Layer\":\"central\",\"Grade\":10," +
            "\"Slots\":1,\"Name\":{\"Chinese\":\"官\",\"English\":\"Office\"}}]," +
            "\"Edges\":[],\"LocalTemplates\":[],\"ArchivedCrossLayerEdges\":[]}";
        True(CustomCourtTemplateJsonCodec.TryImport(oldCentral,
            out CustomCourtTemplate imported, out _), "old central imports");
        True(imported.RegionalGovernmentLayer != null,
            "old central receives dynamic layer");
        Equal("郡", imported.RegionalGovernmentLayer.RegionTitle.Chinese,
            "default region title");
        Equal("郡守", imported.RegionalGovernmentLayer.GovernorTitle.Chinese,
            "default governor title");
        Equal("州", imported.RegionalGovernmentLayer.LocalLevelTitle.Chinese,
            "legacy central templates receive the default local level");
        Equal("Prefecture",
            imported.RegionalGovernmentLayer.LocalLevelTitle.English,
            "legacy central templates receive the English local level");

        CustomCourtTemplate custom = CustomCourtTemplateJsonCodec.Normalize(
            imported);
        custom.RegionalGovernmentLayer.RegionTitle.Chinese = "道";
        custom.RegionalGovernmentLayer.RegionTitle.English = "Circuit";
        custom.RegionalGovernmentLayer.GovernorTitle.Chinese = "观察使";
        custom.RegionalGovernmentLayer.GovernorTitle.English = "Observer";
        custom.RegionalGovernmentLayer.LocalLevelTitle.Chinese = "州";
        custom.RegionalGovernmentLayer.LocalLevelTitle.English = "Prefecture";
        custom.RegionalGovernmentLayer.ManagementOfficeIds.Add("minister");
        custom.RegionalGovernmentLayer.ManagementOfficeIds.Add("missing");
        custom.RegionalGovernmentLayer.Layout.X = 321f;
        custom.RegionalGovernmentLayer.Layout.Y = 654f;
        string roundTrip = CustomCourtTemplateJsonCodec.Export(custom);
        True(CustomCourtTemplateJsonCodec.TryImport(roundTrip,
            out CustomCourtTemplate restored, out _), "custom layer round trips");
        Equal("道", restored.RegionalGovernmentLayer.RegionTitle.Chinese,
            "custom region title survives export");
        Equal("观察使", restored.RegionalGovernmentLayer.GovernorTitle.Chinese,
            "custom governor title survives export");
        Equal("Circuit", restored.RegionalGovernmentLayer.RegionTitle.English,
            "English regional title survives independently");
        Equal("Observer", restored.RegionalGovernmentLayer.GovernorTitle.English,
            "English governor title survives independently");
        Equal("州", restored.RegionalGovernmentLayer.LocalLevelTitle.Chinese,
            "custom local level survives export");
        Equal("Prefecture",
            restored.RegionalGovernmentLayer.LocalLevelTitle.English,
            "English local level survives independently");
        Equal("minister", restored.RegionalGovernmentLayer.ManagementOfficeIds
            .Single(), "management link survives export");
        Equal(321f, restored.RegionalGovernmentLayer.Layout.X,
            "regional card x survives export");
        Equal(654f, restored.RegionalGovernmentLayer.Layout.Y,
            "regional card y survives export");

        CustomLocalCourtTemplate local = CustomLocalGovernmentPresetRules
            .CreateCivil("local_only");
        CustomCourtTemplate localDocument =
            CustomCourtTemplateDocumentRules.CreateLocalDocument(local);
        False(localDocument.RegionalGovernmentLayer != null,
            "local document does not copy central layer");
    }

    private static void True(bool pActual, string pMessage)
    {
        if (!pActual) throw new InvalidOperationException(pMessage);
    }

    private static void False(bool pActual, string pMessage)
    {
        if (pActual) throw new InvalidOperationException(pMessage);
    }

    private static void Equal<T>(T pExpected, T pActual, string pMessage)
    {
        if (!Equals(pExpected, pActual))
            throw new InvalidOperationException(pMessage + ": expected " +
                pExpected + ", got " + pActual);
    }
}
