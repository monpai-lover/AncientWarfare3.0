using System;
using System.Collections.Generic;
using System.Linq;
using AncientWarfare3.core.lineage;

internal static class CultiwayStyleWallGeometryRulesTests
{
    public static void Run()
    {
        ClosedBoundaryExcludesDetachedLand();
        SecondLayerPeelsInwardWithoutBreakingClosure();
        DiagonalBoundaryReceivesFourWayBridge();
        RoadGateWinsOverDirectionalFallback();
        DockWithinEightTilesCarvesThreeTilePassage();
        CoreReachabilityCarvesMinimumThreeTilePassage();
    }

    private static void ClosedBoundaryExcludesDetachedLand()
    {
        HashSet<CultiwayWallPoint> cityLand = Rectangle(1, 1, 7, 7);
        cityLand.Add(Point(8, 8));

        HashSet<CultiwayWallPoint> actual = Compute(Input(9, 9,
            Point(4, 4), Bounds(4, 4, 3, 3), cityLand, cityLand,
            gates: false));
        HashSet<CultiwayWallPoint> expected = RectanglePerimeter(1, 1, 7, 7);

        SetEqual(expected, actual,
            "one layer follows the closed connected-land boundary");
        True(actual.Contains(Point(1, 1)) && actual.Contains(Point(7, 1)) &&
             actual.Contains(Point(1, 7)) && actual.Contains(Point(7, 7)),
            "closed boundary retains all four corners before gates");
        False(actual.Contains(Point(8, 8)),
            "detached city land is excluded");
    }

    private static void SecondLayerPeelsInwardWithoutBreakingClosure()
    {
        HashSet<CultiwayWallPoint> cityLand = Rectangle(1, 1, 7, 7);
        HashSet<CultiwayWallPoint> actual = Compute(Input(9, 9,
            Point(4, 4), Bounds(4, 4, 3, 3), cityLand, cityLand,
            wallWidth: 2, gates: false));
        HashSet<CultiwayWallPoint> expected = RectanglePerimeter(1, 1, 7, 7);
        expected.UnionWith(RectanglePerimeter(2, 2, 6, 6));

        SetEqual(expected, actual,
            "second layer peels inward and remains complete");
        True(actual.Contains(Point(2, 2)) && actual.Contains(Point(6, 2)) &&
             actual.Contains(Point(2, 6)) && actual.Contains(Point(6, 6)),
            "inner layer retains all four corners");
    }

    private static void DiagonalBoundaryReceivesFourWayBridge()
    {
        var cityLand = new HashSet<CultiwayWallPoint>
        {
            Point(3, 3), Point(3, 2), Point(3, 4),
            Point(2, 3), Point(4, 3)
        };
        HashSet<CultiwayWallPoint> actual = Compute(Input(7, 7,
            Point(3, 3), Bounds(3, 3, 3, 3), cityLand, cityLand,
            gates: false));

        SetEqual(cityLand, actual,
            "the center bridges every diagonal pair of cross-arm walls");
        AssertEveryDiagonalPairHasBridge(actual);
    }

    private static void RoadGateWinsOverDirectionalFallback()
    {
        HashSet<CultiwayWallPoint> cityLand = Rectangle(4, 4, 20, 20);
        HashSet<CultiwayWallPoint> passable = Rectangle(0, 0, 24, 24);
        HashSet<CultiwayWallPoint> roads = new() { Point(20, 20) };
        HashSet<CultiwayWallPoint> actual = Compute(Input(25, 25,
            Point(12, 12), Bounds(12, 12, 8, 8), cityLand, passable,
            roads: roads));

        True(actual.Contains(Point(20, 11)),
            "road-priority east gate does not use the centered fallback");
        False(actual.Contains(Point(20, 15)),
            "east gate moves toward the nearest qualifying road");
    }

    private static void DockWithinEightTilesCarvesThreeTilePassage()
    {
        HashSet<CultiwayWallPoint> cityLand = Rectangle(4, 4, 20, 20);
        HashSet<CultiwayWallPoint> passable = Rectangle(0, 0, 24, 24);
        CultiwayWallGeometryInput baselineInput = Input(25, 25,
            Point(12, 12), Bounds(12, 12, 8, 8), cityLand, passable);
        CultiwayWallGeometryInput dockInput = Input(25, 25,
            Point(12, 12), Bounds(12, 12, 8, 8), cityLand, passable,
            docks: new[] { Point(4, 6) });

        HashSet<CultiwayWallPoint> baseline = Compute(baselineInput);
        HashSet<CultiwayWallPoint> actual = Compute(dockInput);
        baseline.ExceptWith(actual);

        SetEqual(new HashSet<CultiwayWallPoint>
        {
            Point(4, 5), Point(4, 6), Point(4, 7)
        }, baseline, "dock passage removes exactly three wall tiles");
    }

    private static void CoreReachabilityCarvesMinimumThreeTilePassage()
    {
        HashSet<CultiwayWallPoint> cityLand = Rectangle(1, 1, 7, 7);
        HashSet<CultiwayWallPoint> passable = new();
        for (int x = 0; x <= 4; x++) passable.Add(Point(x, 4));

        HashSet<CultiwayWallPoint> actual = Compute(Input(9, 9,
            Point(4, 4), Bounds(4, 4, 3, 3), cityLand, passable,
            gates: false));

        False(actual.Contains(Point(1, 3)),
            "minimum crossing opens the lower side of a three-tile passage");
        False(actual.Contains(Point(1, 4)),
            "minimum crossing opens the crossed wall tile");
        False(actual.Contains(Point(1, 5)),
            "minimum crossing opens the upper side of a three-tile passage");
        True(actual.Contains(Point(1, 2)) && actual.Contains(Point(1, 6)),
            "core reachability does not remove the rest of the wall");
    }

    private static CultiwayWallGeometryInput Input(int width, int height,
        CultiwayWallPoint center, CultiwayWallBounds bounds,
        IEnumerable<CultiwayWallPoint> cityLand,
        IEnumerable<CultiwayWallPoint> passable,
        IEnumerable<CultiwayWallPoint> roads = null,
        IEnumerable<CultiwayWallPoint> docks = null,
        int wallWidth = 1, bool gates = true)
    {
        return new CultiwayWallGeometryInput(width, height, center, bounds,
            cityLand, passable, roads ?? Array.Empty<CultiwayWallPoint>(),
            docks ?? Array.Empty<CultiwayWallPoint>(), wallWidth, gates);
    }

    private static HashSet<CultiwayWallPoint> Compute(
        CultiwayWallGeometryInput input)
    {
        return CultiwayStyleWallGeometryRules.Compute(input).ToHashSet();
    }

    private static CultiwayWallPoint Point(int x, int y)
    {
        return new CultiwayWallPoint(x, y);
    }

    private static CultiwayWallBounds Bounds(int cx, int cy, int hx, int hy)
    {
        return new CultiwayWallBounds(cx, cy, hx, hy);
    }

    private static HashSet<CultiwayWallPoint> Rectangle(
        int minX, int minY, int maxX, int maxY)
    {
        var points = new HashSet<CultiwayWallPoint>();
        for (int y = minY; y <= maxY; y++)
        {
            for (int x = minX; x <= maxX; x++) points.Add(Point(x, y));
        }
        return points;
    }

    private static HashSet<CultiwayWallPoint> RectanglePerimeter(
        int minX, int minY, int maxX, int maxY)
    {
        var points = new HashSet<CultiwayWallPoint>();
        for (int x = minX; x <= maxX; x++)
        {
            points.Add(Point(x, minY));
            points.Add(Point(x, maxY));
        }
        for (int y = minY + 1; y < maxY; y++)
        {
            points.Add(Point(minX, y));
            points.Add(Point(maxX, y));
        }
        return points;
    }

    private static void AssertEveryDiagonalPairHasBridge(
        HashSet<CultiwayWallPoint> walls)
    {
        foreach (CultiwayWallPoint point in walls)
        {
            foreach ((int dx, int dy) in new[] { (1, 1), (1, -1) })
            {
                if (!walls.Contains(Point(point.X + dx, point.Y + dy)))
                    continue;
                bool horizontal = walls.Contains(Point(point.X + dx, point.Y));
                bool vertical = walls.Contains(Point(point.X, point.Y + dy));
                True(horizontal || vertical,
                    "diagonal wall pair must have an orthogonal bridge");
            }
        }
    }

    private static void SetEqual(HashSet<CultiwayWallPoint> expected,
        HashSet<CultiwayWallPoint> actual, string message)
    {
        if (!expected.SetEquals(actual))
        {
            string missing = string.Join(", ", expected.Except(actual));
            string extra = string.Join(", ", actual.Except(expected));
            throw new InvalidOperationException(message +
                $": missing [{missing}], extra [{extra}]");
        }
    }

    private static void True(bool value, string message)
    {
        if (!value) throw new InvalidOperationException(message);
    }

    private static void False(bool value, string message)
    {
        if (value) throw new InvalidOperationException(message);
    }
}
