using System;
using System.Collections.Generic;
using System.Linq;
using AncientWarfare3.core.lineage;

internal static class PeasantRebelBanditStrongholdRulesTests
{
    public static void Run()
    {
        RanksTwoByTwoContainingCore();
        RejectsIncompleteTwoByTwoBlock();
        RanksOffsetTwoByTwoContainingCore();
        RanksExpandableStrongholdCandidates();
        ResolvesStrongholdSizeProgression();
        RequiresExactlyFourStrongholdZones();
        LocksBanditGrowthToPersistedZones();
        RestoresOnlyUnchangedStrongholdWalls();
        ResolvesSuppressionVictorByConfirmedPrecedence();
        AttributesOnlyLiveCombatAttackers();
        DefersStrongholdFallUntilTheCorrectBoundary();
        ResolvesLeaderlessSuppressionSettlement();
        AllowsStrongholdCreationWhenGateTowersCannotBePlaced();
        DisposesLogicallyEmptyStrongholdsAfterOwnershipChanges();
        CountsOnlyLivingStrongholdResidents();
        QueuesEveryTrulyEmptyActiveStronghold();
        RelocatesOnlyOrdinaryResidents();
        PrefersOrdinaryResidentsForRuler();
        AllowsEligibleCityLeaderFallbackRuler();
        RejectsRulerCandidatesThatCannotPassKingGate();
        TransfersFallenSurvivorsToTheMotherRealm();
        ComposesStrongholdNamesAndTitles();
    }

    private static void RanksTwoByTwoContainingCore()
    {
        IReadOnlyList<BanditZoneFact> zones = Grid(0, 0, 2, 2);

        IReadOnlyList<IReadOnlyList<string>> ranked =
            PeasantRebelBanditStrongholdRules.RankFourZoneCandidates(
                zones, "1:1");

        True(ranked.Count > 0,
            "a complete two-by-two candidate exists");
        True(new HashSet<string>(ranked[0]).SetEquals(new[]
        {
            "0:0", "1:0", "0:1", "1:1"
        }), "the stable two-by-two block containing the civic core ranks first");
        True(ranked.All(candidate => candidate.Count == 4),
            "every ranked candidate contains exactly four zones");
    }

    private static void RejectsIncompleteTwoByTwoBlock()
    {
        List<BanditZoneFact> zones = Grid(0, 0, 1, 1).ToList();
        zones.RemoveAll(zone => zone.Key == "1:1");

        IReadOnlyList<IReadOnlyList<string>> ranked =
            PeasantRebelBanditStrongholdRules.RankFourZoneCandidates(
                zones, "0:0");

        Equal(0, ranked.Count,
            "a connected but incomplete block cannot form a stronghold");
    }

    private static void RanksOffsetTwoByTwoContainingCore()
    {
        IReadOnlyList<BanditZoneFact> zones = Grid(5, 7, 6, 8);

        IReadOnlyList<IReadOnlyList<string>> ranked =
            PeasantRebelBanditStrongholdRules.RankFourZoneCandidates(
                zones, "5:7");

        Equal(1, ranked.Count,
            "a corner core may belong to an offset two-by-two block");
        Equal(4, ranked[0].Count,
            "the offset candidate still contains exactly four zones");
    }

    private static void RanksExpandableStrongholdCandidates()
    {
        IReadOnlyList<BanditZoneFact> zones = Grid(0, 0, 6, 6);
        Equal(9, PeasantRebelBanditStrongholdRules.RankZoneCandidates(
            zones, "3:3", 3).Count,
            "a centered three-by-three grid has nine candidate origins");
        Equal(16, PeasantRebelBanditStrongholdRules.RankZoneCandidates(
            zones, "3:3", 4).Count,
            "a four-by-four grid has sixteen candidate origins");
        Equal(0, PeasantRebelBanditStrongholdRules.RankZoneCandidates(
            zones, "3:3", 5).Count,
            "sizes above four-by-four are rejected");
    }

    private static void ResolvesStrongholdSizeProgression()
    {
        Equal(BanditStrongholdSize.Medium3x3,
            PeasantRebelBanditStrongholdRules.ResolveNextSize(
                BanditStrongholdSize.Small2x2, 60, false, true),
            "small expands to medium under pressure");
        Equal(BanditStrongholdSize.Large4x4,
            PeasantRebelBanditStrongholdRules.ResolveNextSize(
                BanditStrongholdSize.Medium3x3, 100, true, true),
            "medium expands to large under pressure");
        Equal(BanditStrongholdSize.Large4x4,
            PeasantRebelBanditStrongholdRules.ResolveNextSize(
                BanditStrongholdSize.Large4x4, 1000, true, true),
            "large is capped");
        Equal(BanditStrongholdSize.Small2x2,
            PeasantRebelBanditStrongholdRules.NormalizeSize(0),
            "legacy state defaults to small size");
    }

    private static void RequiresExactlyFourStrongholdZones()
    {
        True(PeasantRebelBanditStrongholdRules.IsViableSplit(4, 1),
            "four stronghold zones and one mother zone are viable");
        True(PeasantRebelBanditStrongholdRules.IsViableSplit(4, 0),
            "the protected mother city may temporarily retain zero zones");
        False(PeasantRebelBanditStrongholdRules.IsViableSplit(0, 3),
            "stronghold needs exactly four interior zones");
        False(PeasantRebelBanditStrongholdRules.IsViableSplit(3, 2),
            "three stronghold zones are below the fixed size");
        False(PeasantRebelBanditStrongholdRules.IsViableSplit(5, 1),
            "five stronghold zones exceed the fixed size");
    }

    private static void LocksBanditGrowthToPersistedZones()
    {
        var fixedZones = new HashSet<string> { "center", "east" };
        True(PeasantRebelBanditStrongholdRules.CanAcquireZone(
                true, "east", fixedZones),
            "bandit stronghold may restore a fixed zone");
        False(PeasantRebelBanditStrongholdRules.CanAcquireZone(
                true, "outside", fixedZones),
            "bandit stronghold cannot grow beyond its walls");
        True(PeasantRebelBanditStrongholdRules.CanAcquireZone(
                false, "outside", fixedZones),
            "non-bandit city growth remains unrestricted");
    }

    private static void RestoresOnlyUnchangedStrongholdWalls()
    {
        True(PeasantRebelBanditStrongholdRules.ShouldRestoreWall(
                "wall_wild"),
            "an unchanged stronghold wall is restored during suppression");
        False(PeasantRebelBanditStrongholdRules.ShouldRestoreWall("road"),
            "later terrain changes are preserved");
        False(PeasantRebelBanditStrongholdRules.ShouldRestoreWall(null),
            "an already removed wall is ignored");
    }

    private static void ResolvesSuppressionVictorByConfirmedPrecedence()
    {
        Equal(31L,
            PeasantRebelBanditStrongholdRules.ResolveSuppressorKingdomId(
                31L, 20L, true),
            "last hostile killer wins attribution");
        Equal(20L,
            PeasantRebelBanditStrongholdRules.ResolveSuppressorKingdomId(
                -1L, 20L, true),
            "warring mother kingdom is the fallback");
        Equal(-1L,
            PeasantRebelBanditStrongholdRules.ResolveSuppressorKingdomId(
                -1L, 20L, false),
            "starvation without a war has no suppressing kingdom");
    }

    private static void AttributesOnlyLiveCombatAttackers()
    {
        True(PeasantRebelBanditStrongholdRules.CanAttributeHostileKiller(
                hostileAttacker: true, deathClearsAttacker: false),
            "a combat death may retain its hostile killer");
        False(PeasantRebelBanditStrongholdRules.CanAttributeHostileKiller(
                hostileAttacker: true, deathClearsAttacker: true),
            "starvation and other native clear-attacker deaths have no killer");
        False(PeasantRebelBanditStrongholdRules.CanAttributeHostileKiller(
                hostileAttacker: false, deathClearsAttacker: false),
            "a non-hostile attacker is never credited");
    }

    private static void DefersStrongholdFallUntilTheCorrectBoundary()
    {
        Equal(BanditStrongholdFallAction.RecordSuppressorOnly,
            PeasantRebelBanditStrongholdRules.ResolveFallAction(
                population: 0, hostileKillerKingdomId: 42,
                captureFinished: false),
            "hostile population loss records attribution until capture finishes");
        Equal(BanditStrongholdFallAction.QueueFall,
            PeasantRebelBanditStrongholdRules.ResolveFallAction(
                population: 0, hostileKillerKingdomId: -1,
                captureFinished: false),
            "environmental population loss queues immediate fall");
        Equal(BanditStrongholdFallAction.QueueFall,
            PeasantRebelBanditStrongholdRules.ResolveFallAction(
                population: 12, hostileKillerKingdomId: 42,
                captureFinished: true),
            "completed capture queues fall regardless of remaining population");
        Equal(BanditStrongholdFallAction.None,
            PeasantRebelBanditStrongholdRules.ResolveFallAction(
                population: 12, hostileKillerKingdomId: -1,
                captureFinished: false),
            "a populated uncaptured stronghold remains active");
    }

    private static void ResolvesLeaderlessSuppressionSettlement()
    {
        True(PeasantRebelBanditStrongholdRules.
                ShouldCompleteLeaderlessSuppression(
                    isSuppressionWar: true, isBanditKingdom: true,
                    hasLivingKing: false, hasLivingRegisteredHeir: false),
            "a suppression war ends when the bandit has neither king nor heir");
        False(PeasantRebelBanditStrongholdRules.
                ShouldCompleteLeaderlessSuppression(true, true, true, false),
            "a living bandit king keeps the suppression war active");
        False(PeasantRebelBanditStrongholdRules.
                ShouldCompleteLeaderlessSuppression(true, true, false, true),
            "a living registered heir keeps the suppression war active");
        False(PeasantRebelBanditStrongholdRules.
                ShouldCompleteLeaderlessSuppression(false, true, false, false),
            "ordinary wars do not use bandit decapitation settlement");
        False(PeasantRebelBanditStrongholdRules.
                ShouldCompleteLeaderlessSuppression(true, false, false, false),
            "a non-bandit participant cannot trigger bandit collapse");
    }

    private static void RelocatesOnlyOrdinaryResidents()
    {
        True(PeasantRebelBanditStrongholdRules.
                CanRelocateOrdinaryResident(true, true, false, false,
                    false),
            "an adult civilian without a ruling role may join the stronghold");
        False(PeasantRebelBanditStrongholdRules.
                CanRelocateOrdinaryResident(true, true, true, false,
                    false),
            "the origin kingdom's king cannot be transferred");
        False(PeasantRebelBanditStrongholdRules.
                CanRelocateOrdinaryResident(true, true, false, true,
                    false),
            "an origin city leader cannot be transferred");
        False(PeasantRebelBanditStrongholdRules.
                CanRelocateOrdinaryResident(true, true, false, false,
                    true),
            "the origin kingdom's heir cannot be transferred");
        False(PeasantRebelBanditStrongholdRules.
                CanRelocateOrdinaryResident(true, false, false, false,
                    false),
            "a non-civilian profession is not an ordinary resident");
        False(PeasantRebelBanditStrongholdRules.
                CanRelocateOrdinaryResident(false, true, false, false,
                    false),
            "a child cannot found or populate the new stronghold");
    }

    private static void PrefersOrdinaryResidentsForRuler()
    {
        True(PeasantRebelBanditStrongholdRules.
                ShouldPreferOrdinaryRuler(true),
            "an eligible ordinary resident takes precedence over the city leader");
        False(PeasantRebelBanditStrongholdRules.
                ShouldPreferOrdinaryRuler(false),
            "the city leader fallback is only considered when no ordinary resident exists");
    }

    private static void AllowsEligibleCityLeaderFallbackRuler()
    {
        True(PeasantRebelBanditStrongholdRules.CanUseCityLeaderAsRuler(
                cityLeaderAlive: true, cityLeaderAdult: true,
                cityLeaderBelongsToMotherCity: true),
            "a living adult leader of the mother city may found the bandit realm");
        False(PeasantRebelBanditStrongholdRules.CanUseCityLeaderAsRuler(
                cityLeaderAlive: false, cityLeaderAdult: true,
                cityLeaderBelongsToMotherCity: true),
            "a dead city leader cannot found the bandit realm");
        False(PeasantRebelBanditStrongholdRules.CanUseCityLeaderAsRuler(
                cityLeaderAlive: true, cityLeaderAdult: false,
                cityLeaderBelongsToMotherCity: true),
            "a child city leader cannot found the bandit realm");
        False(PeasantRebelBanditStrongholdRules.CanUseCityLeaderAsRuler(
                cityLeaderAlive: true, cityLeaderAdult: true,
                cityLeaderBelongsToMotherCity: false),
            "an unrelated city leader cannot be used as the fallback ruler");
    }

    private static void RejectsRulerCandidatesThatCannotPassKingGate()
    {
        True(PeasantRebelBanditStrongholdRules.CanUseDirectRuler(
                candidateIsMale: true, femaleSuccessionAllowed: false),
            "a male direct ruler passes the native kingship gate");
        True(PeasantRebelBanditStrongholdRules.CanUseDirectRuler(
                candidateIsMale: false, femaleSuccessionAllowed: true),
            "a female direct ruler is allowed when the court law permits it");
        True(PeasantRebelBanditStrongholdRules.CanUseDirectRuler(
                candidateIsMale: false, femaleSuccessionAllowed: false),
            "a city-leader fallback is promoted directly even when the native court law is male-only");
    }

    private static void AllowsStrongholdCreationWhenGateTowersCannotBePlaced()
    {
        True(PeasantRebelBanditStrongholdRules.CanUseWallCandidate(
                wallPlanExists: true, gateTowerCount: 0),
            "a valid wall plan is enough to create the stronghold");
        True(PeasantRebelBanditStrongholdRules.CanUseWallCandidate(
                wallPlanExists: true, gateTowerCount: 4),
            "a complete four-tower plan remains preferred");
        False(PeasantRebelBanditStrongholdRules.CanUseWallCandidate(
                wallPlanExists: false, gateTowerCount: 4),
            "gate towers cannot replace a missing wall plan");
    }

    private static void DisposesLogicallyEmptyStrongholdsAfterOwnershipChanges()
    {
        True(PeasantRebelBanditStrongholdRules.
                CanDisposeStrongholdCity(true, false, 0, false),
            "an emptied stronghold is disposable after its owner changes");
        True(PeasantRebelBanditStrongholdRules.
                CanDisposeStrongholdCity(true, false, 4, true),
            "the originally scheduled stronghold remains disposable");
        False(PeasantRebelBanditStrongholdRules.
                CanDisposeStrongholdCity(true, false, 4, false),
            "a populated replacement city with another owner is preserved");
        False(PeasantRebelBanditStrongholdRules.
                CanDisposeStrongholdCity(false, false, 0, true),
            "an absent city needs no disposal call");
    }

    private static void CountsOnlyLivingStrongholdResidents()
    {
        True(PeasantRebelBanditStrongholdPopulationRules.
                IsLivingResident(true, true, false, false, true),
            "a living non-boat resident counts toward stronghold population");
        False(PeasantRebelBanditStrongholdPopulationRules.
                IsLivingResident(true, false, false, false, true),
            "a dead actor retained by the native city list does not count");
        False(PeasantRebelBanditStrongholdPopulationRules.
                IsLivingResident(true, true, false, true, true),
            "a boat does not keep a stronghold alive");
        False(PeasantRebelBanditStrongholdPopulationRules.
                IsLivingResident(true, true, false, false, false),
            "an actor already moved to another city does not count");
    }

    private static void QueuesEveryTrulyEmptyActiveStronghold()
    {
        True(PeasantRebelBanditStrongholdPopulationRules.
                ShouldQueueFall(true, 0),
            "an active stronghold with no living residents must fall");
        True(PeasantRebelBanditStrongholdPopulationRules.
                ShouldQueueFall(true, -1),
            "a defensive negative count is treated as empty");
        False(PeasantRebelBanditStrongholdPopulationRules.
                ShouldQueueFall(true, 1),
            "one living resident keeps the stronghold active");
        False(PeasantRebelBanditStrongholdPopulationRules.
                ShouldQueueFall(false, 0),
            "an inactive state is not replayed by the population observer");
    }

    private static void ComposesStrongholdNamesAndTitles()
    {
        Equal("\u864e\u8e1e\u5be8",
            PeasantRebelBanditStrongholdRules.ComposeStrongholdName(
                "\u864e\u8e1e\u8d3c\u4e49\u519b"),
            "stronghold city uses canonical root plus stronghold suffix");
        Equal("\u864e\u8e1e\u5be8\u5927\u5f53\u5bb6",
            PeasantRebelBanditStrongholdRules.ComposeCeremonialTitle(
                "\u864e\u8e1e\u8d3c", false),
            "ruler title includes stronghold name");
        Equal("\u864e\u8e1e\u5be8\u5c11\u5f53\u5bb6",
            PeasantRebelBanditStrongholdRules.ComposeCeremonialTitle(
                "\u864e\u8e1e\u4e49\u519b", true),
            "heir title includes stronghold name");
        Equal("\u864e\u8e1e\u5be8Chief",
            PeasantRebelBanditStrongholdRules.ComposeCeremonialTitle(
                "\u864e\u8e1e", "Chief"),
            "localized role text follows the canonical stronghold name");
    }

    private static void TransfersFallenSurvivorsToTheMotherRealm()
    {
        True(PeasantRebelBanditStrongholdRules.
                ShouldTransferFallenSurvivor(true, true, true, false),
            "a living bandit survivor moves to the mother realm");
        False(PeasantRebelBanditStrongholdRules.
                ShouldTransferFallenSurvivor(true, false, true, false),
            "an actor already outside the bandit realm is not reprocessed");
        False(PeasantRebelBanditStrongholdRules.
                ShouldTransferFallenSurvivor(false, true, true, false),
            "dead actors are never transferred");
        False(PeasantRebelBanditStrongholdRules.
                ShouldTransferFallenSurvivor(true, true, false, false),
            "a missing destination realm blocks completion");
        False(PeasantRebelBanditStrongholdRules.
                ShouldTransferFallenSurvivor(true, true, true, true),
            "the bandit realm cannot be its own suppression destination");
    }

    private static BanditZoneFact Zone(string key, int x, int y,
        params string[] neighbours)
    {
        return BanditZoneFact.At(key, x, y, neighbours);
    }

    private static IReadOnlyList<BanditZoneFact> Grid(
        int minX, int minY, int maxX, int maxY)
    {
        var zones = new List<BanditZoneFact>();
        for (int y = minY; y <= maxY; y++)
        for (int x = minX; x <= maxX; x++)
        {
            var neighbours = new List<string>();
            if (x > minX) neighbours.Add((x - 1) + ":" + y);
            if (x < maxX) neighbours.Add((x + 1) + ":" + y);
            if (y > minY) neighbours.Add(x + ":" + (y - 1));
            if (y < maxY) neighbours.Add(x + ":" + (y + 1));
            zones.Add(Zone(x + ":" + y, x, y,
                neighbours.ToArray()));
        }
        return zones;
    }

    private static void True(bool value, string message)
    {
        if (!value) throw new InvalidOperationException(message);
    }

    private static void False(bool value, string message)
    {
        if (value) throw new InvalidOperationException(message);
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(
                message + $": expected {expected}, got {actual}");
    }
}
