using AncientWarfare3.core.lineage;

internal static class FeudatoryIdentityRulesTests
{
    public static void Run()
    {
        False(FeudatoryIdentityRules.CanEnslave(
                ordinaryEligible: true, activePrince: true),
            "an active feudatory prince cannot be enslaved");
        True(FeudatoryIdentityRules.CanEnslave(
                ordinaryEligible: true, activePrince: false),
            "a former prince returns to ordinary eligibility rules");
        False(FeudatoryIdentityRules.CanEnslave(
                ordinaryEligible: false, activePrince: false),
            "feudatory rules do not bypass ordinary exclusions");
        Equal(10L, FeudatorySuccessionRules.SelectSuccessor(new[]
        {
            new FeudatorySuccessionCandidate(10, eligible: true,
                directSon: true, sameShiBranch: true, kinDistance: 0,
                birthTime: 10, directTreeDescendant: true,
                legitimateBirth: true, adult: false),
            new FeudatorySuccessionCandidate(11, eligible: true,
                directSon: true, sameShiBranch: true, kinDistance: 0,
                birthTime: 20, directTreeDescendant: true,
                legitimateBirth: true, adult: true)
        }), "the legitimate eldest son remains the feudatory successor");
    }

    private static void True(bool value, string name)
    {
        if (!value) throw new InvalidOperationException(name);
    }

    private static void False(bool value, string name)
    {
        if (value) throw new InvalidOperationException(name);
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(name);
    }
}
