            "monthly processing consumes the threatened-city workset");

            "monthly departure work does not rebuild the world destination index");

            "assimilation cannot become a monthly actor scan");
        Require(authority, "WarRefugeeService.ProcessAuthorityCycle(",

            "LoadActiveJourneys", "WarRefugeeJourneyService.ProcessMonthly(",
            "TryStepFollowerDirect(", "ProcessEveryFrame" })

        string departures = Section(service, "private static void CaptureThreatenedCities(",
            "private static int SettleHousehold(");
        Require(service, "CityBudgetPerMonth", "monthly city budget remains explicit");
        Require(departures, "processed++ < pBudget && remainingWorldBudget > 0",

        Require(departures, "WarRefugeeRules.DeparturePermille(",
            "departure probabilities retain the shared threat rules");

            "departure quotas retain minimum population and capacity rules");
        Require(departures, "remainingWorldBudget -= Math.Max(0, settled)",

        Require(departures, "AddMonthlyReservation(destination.id, settled)",

        Require(departures, "new List<long>(ThreatenedCityIds)",

        Reject(service, "ArmyRtsControllerService.SnapshotMissions()",

        Require(resolution, "IsLivingKingdom(originCity.kingdom)",
            "return requires a surviving origin kingdom");
        Require(resolution, "!ThreatenedCityIds.Contains(originCityId)",

        Before(resolution, "pActor.joinCity(originCity)", "ClearRefugeeKeys(pActor)",

        Require(resolution, "if (host?.culture != null) pActor.setCulture(host.culture)",

        Require(service, "ReturnBasePermille - years * ReturnDecayPermillePerYear",

        Require(warPatch.Substring(0, regularWar), "WarRefugeeService.OnWarEnded(pWar)",

        Require(patch, "MapBox.clearWorld", "world reset clears refugee runtime indexes");

        Require(restore, "WarRefugeeService.RebuildRuntime", "world load rebuilds destination indexes");

            "children of refugees adopt the host city's culture");

    private static string Section(string source, string start, string end)
    {
        int first = source.IndexOf(start, StringComparison.Ordinal);
        int last = first < 0 ? -1 : source.IndexOf(end, first, StringComparison.Ordinal);
        if (first < 0 || last <= first)
            throw new InvalidOperationException("missing lifecycle section: " + start);
        return source.Substring(first, last - first);
    }

    private static void Before(string source, string first, string second, string message)
    {
        int left = source.IndexOf(first, StringComparison.Ordinal);
        int right = source.IndexOf(second, StringComparison.Ordinal);
        if (left < 0 || right <= left) throw new InvalidOperationException(message);
    }

    private static string FindRoot()
    {
        string path = AppContext.BaseDirectory;
        while (!string.IsNullOrEmpty(path))
        {
            if (File.Exists(Path.Combine(path, "AncientWarfare3.csproj")))
                return path;
            path = Directory.GetParent(path)?.FullName;
        }
        throw new InvalidOperationException("repository root not found");
    }

    private static string Read(string root, string relative) =>
        File.ReadAllText(Path.Combine(root, relative)).Replace("\r\n", "\n");

    private static void Require(string source, string text, string message)
    {
        if (!source.Contains(text, StringComparison.Ordinal))
            throw new InvalidOperationException(message);
    }

    private static void Reject(string source, string text, string message)
    {
        if (source.Contains(text, StringComparison.Ordinal))
            throw new InvalidOperationException(message);
    }
}
