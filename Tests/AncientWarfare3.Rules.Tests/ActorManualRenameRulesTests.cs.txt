using AncientWarfare3.core.naming;

internal static class ActorManualRenameRulesTests
{
    public static void Run()
    {
        Equal(ActorManualNameMode.SiniticMerged,
            ActorManualRenameRules.ResolveMode(isXia: false,
                NamingProfileId.NativeSinitic),
            "native Sinitic species use one merged Shi field");
        Equal(ActorManualNameMode.SiniticMerged,
            ActorManualRenameRules.ResolveMode(isXia: false,
                NamingProfileId.Monkey),
            "monkeys use the same merged Shi field");
        Equal(ActorManualNameMode.Xia,
            ActorManualRenameRules.ResolveMode(isXia: true,
                NamingProfileId.NativeSinitic),
            "Xia identity retains separate surname and Shi editing");
        Equal("LeJia", ActorManualNameRules.CreateDraft(
                ActorManualNameMode.SiniticMerged, "Le", "Jia")
                .DisplayName,
            "merged Shi names remain family-first without a separator");

        ActorManualBranchPlan plan = ActorManualRenameRules.PlanBranchChange(
            10L, "Ji", "Jiang", new long[] { 10L, 11L, 12L });
        Sequence(new long[] { 10L, 11L, 12L }, plan.ActorIds,
            "surname change targets root and patrilineal descendants");
        Equal(true, plan.RequiresBranchFork,
            "changed branch identity forks");

        Equal(false, ActorManualRenameRules.PlanBranchChange(
            10L, "Jiang", "Jiang", new long[] { 10L, 11L })
            .RequiresBranchFork, "resubmitting surname is idempotent");

        Equal("Lu", ActorManualRenameRules.ResolveFamilyIdentity(
                ActorManualNameMode.NonXia, clanName: "",
                familyName: "", chineseFamilyName: " Lu ",
                localizedFamilyComponent: "Fallback"),
            "the editor falls back to the saved Chinese family identity");
        Equal("Fallback", ActorManualRenameRules.ResolveFamilyIdentity(
                ActorManualNameMode.NonXia, clanName: "",
                familyName: "", chineseFamilyName: "",
                localizedFamilyComponent: " Fallback "),
            "the editor falls back to the localized family component");

        ActorManualFamilyWritePlan integrated =
            ActorManualRenameRules.PlanIntegratedFamilyWrite("  Lu  ");
        Equal("Lu", integrated.FamilyName,
            "manual surname writes the family name");
        Equal("Lu", integrated.ChineseFamilyName,
            "manual surname writes the merged surname");
        Equal("Lu", integrated.ClanName,
            "manual surname writes the merged clan name");
        Equal("Lu", integrated.LocalizedFamilyComponent,
            "manual surname writes the localized family component");
        Equal(true, integrated.NameIntegrated,
            "manual surname directly marks surname-clan integration");
    }

    private static void Sequence(IReadOnlyList<long> expected,
        IReadOnlyList<long> actual, string name)
    {
        if (expected.Count != actual.Count)
            throw new InvalidOperationException(name + " count mismatch");
        for (int i = 0; i < expected.Count; i++)
            if (expected[i] != actual[i])
                throw new InvalidOperationException(name + " order mismatch");
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(
                $"{name}: expected {expected}, got {actual}");
    }
}
