using System;
using AncientWarfare3.core.pathfinding;

internal static class AWDockRouteRegistryTests
{
    internal static void Run()
    {
        var registry = new AWDockRouteRegistry();
        Assert(!registry.Register(new AWDockEndpoint(0, 1, 2)),
            "invalid dock identifiers are rejected");
        Assert(registry.Register(new AWDockEndpoint(1, 10, 4)),
            "physical docks can be registered");
        Assert(registry.Snapshot().Length == 1,
            "registered physical dock appears in the immutable lookup snapshot");
        Assert(registry.Remove(1) && registry.Snapshot().Length == 0,
            "destroyed docks are removed from the physical topology");
    }

    private static void Assert(bool pCondition, string pMessage)
    {
        if (!pCondition) throw new InvalidOperationException(pMessage);
    }
}
