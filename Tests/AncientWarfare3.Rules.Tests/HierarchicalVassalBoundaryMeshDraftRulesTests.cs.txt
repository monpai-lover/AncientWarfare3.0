using AncientWarfare3.core.policy;

internal static class HierarchicalVassalBoundaryMeshDraftRulesTests
{
    public static void Run()
    {
        CoherentFillIsConsolidated();
        CityFillUsesDisplayedOwnersAndHierarchyColors();
        FillColorsAvoidActualAdjacentOwnerCollisions();
        FillColorsAreChunkContextIndependentAndSeamStable();
        RibbonsCarryTwoSidedPrimitiveAttributesAndTierWidths();
        FullFootprintNarrowsAndRiverWaterIsTransparent();
        CoastFlagsAndTransparencyAreSideSpecific();
        CanonicalSeamsAreBitIdentical();
        UnsafeAcceptedRibbonRebuildsFromRawContour();
        UnsafeRawRibbonReturnsBoundedFailure();
        HierarchyColorsAreStableAndDistinguishable();
    }

    private static void CoherentFillIsConsolidated()
    {
        BoundaryCellRaster grid = Raster(32, 32,
            (x, y) => Land(x, y, 7, 70, 700, 0x336699FF));
        BoundaryMeshDraft fill =
            HierarchicalVassalBoundaryMeshDraftRules.BuildFill(
                grid, BoundaryDisplayLayer.Countries, Interior(grid));
        Equal(true, fill.VertexCount < 128,
            "a coherent 32x32 owner uses fewer than 128 vertices");
        Equal(true, fill.VertexCount < 32 * 32,
            "fill is not one quad per tile");
        Equal(true, fill.VassalRealmIndices.Length > 0,
            "coherent fill remains renderable");
    }

    private static void RibbonsCarryTwoSidedPrimitiveAttributesAndTierWidths()
    {
        BoundaryCellRaster grid = Raster(8, 8, (x, y) =>
            Land(x, y, x < 4 ? 1 : 2, x < 4 ? 10 : 20,
                x < 4 ? 100 : 200, x < 4 ? 0xAA2211FF : 0x1144CCFF));
        BoundaryRibbonInput[] inputs =
        {
            Input(BoundaryTier.SuzerainSystem, 1, 2, 0xAA2211FF, 0x1144CCFF, 1),
            Input(BoundaryTier.VassalRealm, 10, 20, 0xAA2211FF, 0x1144CCFF, 3),
            Input(BoundaryTier.City, 100, 200, 0xAA2211FF, 0x1144CCFF, 5)
        };
        BoundaryMeshDraft draft =
            HierarchicalVassalBoundaryMeshDraftRules.BuildRibbons(inputs, grid);

        NearlyEqual(0.12f,
            HierarchicalVassalBoundaryMeshDraftRules.TargetWidth(BoundaryTier.City),
            "city target width");
        NearlyEqual(0.20f,
            HierarchicalVassalBoundaryMeshDraftRules.TargetWidth(BoundaryTier.VassalRealm),
            "vassal target width");
        NearlyEqual(0.32f,
            HierarchicalVassalBoundaryMeshDraftRules.TargetWidth(BoundaryTier.SuzerainSystem),
            "suzerain target width");
        Equal(true, draft.CityIndices.Length > 0, "city submesh indices");
        Equal(true, draft.VassalRealmIndices.Length > 0, "vassal submesh indices");
        Equal(true, draft.SuzerainSystemIndices.Length > 0, "system submesh indices");
        Equal(draft.VertexCount, draft.PositionX.Length, "position x array");
        Equal(draft.VertexCount, draft.PositionY.Length, "position y array");
        Equal(draft.VertexCount, draft.NormalX.Length, "normal x array");
        Equal(draft.VertexCount, draft.NormalY.Length, "normal y array");
        Equal(draft.VertexCount, draft.SignedDistance.Length, "signed distance array");
        Equal(draft.VertexCount, draft.Tiers.Length, "tier array");
        Equal(draft.VertexCount, draft.LeftOwnerIds.Length, "left owner array");
        Equal(draft.VertexCount, draft.RightOwnerIds.Length, "right owner array");
        Equal(draft.VertexCount, draft.LeftRgba.Length, "left rgba array");
        Equal(draft.VertexCount, draft.RightRgba.Length, "right rgba array");
        Equal(draft.VertexCount, draft.LocalHalfWidths.Length, "half-width array");
        Equal(draft.VertexCount, draft.Flags.Length, "river/coast flags array");
        Equal(0xAA2211FFu, draft.LeftRgba[0], "left color is preserved");
        Equal(0x1144CCFFu, draft.RightRgba[0], "right color is preserved");
        Equal(1L, draft.LeftOwnerIds[0], "left owner is preserved");
        Equal(2L, draft.RightOwnerIds[0], "right owner is preserved");
        Equal(true, draft.SignedDistance.Any(value => value > 0f) &&
                    draft.SignedDistance.Any(value => value < 0f),
            "two visual faces have opposite signed distances");
    }

    private static void FillColorsAvoidActualAdjacentOwnerCollisions()
    {
        long[,] owners =
        {
            { 1, 8, 2, 3 },
            { 4, 5, 6, 7 }
        };
        BoundaryCellRaster grid = Raster(4, 2, (x, y) =>
            Land(x, y, 1, owners[y, x], owners[y, x], 0x336699FF));
        BoundaryMeshDraft fill =
            HierarchicalVassalBoundaryMeshDraftRules.BuildFill(
                grid, BoundaryDisplayLayer.Cities, Interior(grid));
        uint color1 = fill.LeftRgba[
            Array.IndexOf(fill.LeftOwnerIds, 1L)];
        uint color8 = fill.LeftRgba[
            Array.IndexOf(fill.LeftOwnerIds, 8L)];
        Equal(true, color1 != color8,
            "spatially adjacent owners avoid a hash collision even when IDs 2..7 exist");
        BoundaryMeshDraft rebuilt =
            HierarchicalVassalBoundaryMeshDraftRules.BuildFill(
                grid, BoundaryDisplayLayer.Cities, Interior(grid));
        Equal(color1, rebuilt.LeftRgba[
                Array.IndexOf(rebuilt.LeftOwnerIds, 1L)],
            "adjacency-aware owner color is rebuild-stable");
        Equal(color8, rebuilt.LeftRgba[
                Array.IndexOf(rebuilt.LeftOwnerIds, 8L)],
            "adjacency fallback choice is rebuild-stable");
    }

    private static void FillColorsAreChunkContextIndependentAndSeamStable()
    {
        BoundaryCellRaster contextGrid = Raster(64, 2, (x, y) =>
            Land(x, y, 1, x == 0 ? 1 : 8,
                x == 0 ? 1 : 8, 0x336699FF));
        BoundaryMeshDraft leftContext =
            HierarchicalVassalBoundaryMeshDraftRules.BuildFill(
                contextGrid, BoundaryDisplayLayer.Cities,
                Bounds(0, 0, 32, 2, contextGrid));
        BoundaryMeshDraft rightContext =
            HierarchicalVassalBoundaryMeshDraftRules.BuildFill(
                contextGrid, BoundaryDisplayLayer.Cities,
                Bounds(32, 0, 64, 2, contextGrid));
        uint leftColor8 = leftContext.LeftRgba[
            Array.IndexOf(leftContext.LeftOwnerIds, 8L)];
        uint rightColor8 = rightContext.LeftRgba[
            Array.IndexOf(rightContext.LeftOwnerIds, 8L)];
        Equal(leftColor8, rightColor8,
            "same displayed owner color is bit-identical across local adjacency contexts");

        BoundaryCellRaster seamGrid = Raster(64, 2, (x, y) =>
            Land(x, y, 1, x < 32 ? 1 : 8,
                x < 32 ? 1 : 8, 0x336699FF));
        BoundaryMeshDraft seamLeft =
            HierarchicalVassalBoundaryMeshDraftRules.BuildFill(
                seamGrid, BoundaryDisplayLayer.Cities,
                Bounds(0, 0, 32, 2, seamGrid));
        BoundaryMeshDraft seamRight =
            HierarchicalVassalBoundaryMeshDraftRules.BuildFill(
                seamGrid, BoundaryDisplayLayer.Cities,
                Bounds(32, 0, 64, 2, seamGrid));
        uint seamColor1 = seamLeft.LeftRgba[0];
        uint seamColor8 = seamRight.LeftRgba[0];
        Equal(true, seamColor1 != seamColor8,
            "owners adjacent exactly across a chunk seam remain distinguishable");
        Equal(0, seamLeft.FailureCount,
            "left seam chunk accepts canonical adjacent colors");
        Equal(0, seamRight.FailureCount,
            "right seam chunk accepts canonical adjacent colors");
    }

    private static void CityFillUsesDisplayedOwnersAndHierarchyColors()
    {
        BoundaryCellRaster grid = Raster(4, 2, (x, y) =>
            Land(x, y, 1, 10, x < 2 ? 100 : 101, 0x336699FF));
        BoundaryMeshDraft fill =
            HierarchicalVassalBoundaryMeshDraftRules.BuildFill(
                grid, BoundaryDisplayLayer.Cities, Interior(grid));
        Equal(true, fill.LeftOwnerIds.Contains(100L) &&
                    fill.LeftOwnerIds.Contains(101L),
            "city fill is grouped by displayed city owner");
        Equal(true, fill.Tiers.All(value => value == (byte)BoundaryTier.City),
            "city fill vertices carry the city tier");
        Equal(true, fill.CityIndices.Length > 0,
            "city fill uses the city submesh indices");
        Equal(true, fill.LeftRgba.Distinct().Count() == 2,
            "adjacent cities receive distinguishable hierarchy colors");
    }

    private static void FullFootprintNarrowsAndRiverWaterIsTransparent()
    {
        BoundaryCellRaster thirdOwner = Raster(4, 4, (x, y) =>
            Land(x, y, y < 2 ? 1 : x == 1 ? 3 : 2,
                y < 2 ? 10 : x == 1 ? 30 : 20,
                y < 2 ? 100 : x == 1 ? 300 : 200, 0xFFFFFFFF));
        var unsafeCurve = new BoundaryCurveDraft(new[]
        {
            new BoundaryFloatPoint(0.2f, 1.90f),
            new BoundaryFloatPoint(1.8f, 1.90f)
        }, false, false, 1f,
            new BoundaryFloatPoint(1, 0), new BoundaryFloatPoint(1, 0));
        var unsafeInput = new BoundaryRibbonInput(
            unsafeCurve,
            new[] { new BoundaryGridPoint(0, 2), new BoundaryGridPoint(2, 2) },
            BoundaryTier.SuzerainSystem, 1, 2,
            0x336699FF, 0x993333FF,
            isRiver: false, coastSide: BoundaryRibbonCoastSide.None);
        BoundaryMeshDraft narrowed =
            HierarchicalVassalBoundaryMeshDraftRules.BuildRibbons(
                new[] { unsafeInput }, thirdOwner);
        Equal(true, narrowed.LocalHalfWidths.Min() < 0.16f,
            "full footprint validation locally narrows before a third owner");
        Equal(true, narrowed.SegmentFallbacks.Any(value => value != 0) ||
                    narrowed.LocalHalfWidths.Min() <= 0.10f,
            "unsafe corridor uses local narrowing or raw fallback");

        var spanningCurve = new BoundaryCurveDraft(new[]
        {
            new BoundaryFloatPoint(0.2f, 1.90f),
            new BoundaryFloatPoint(2.8f, 1.90f)
        }, false, false, 1f,
            new BoundaryFloatPoint(1, 0), new BoundaryFloatPoint(1, 0));
        var spanningInput = new BoundaryRibbonInput(
            spanningCurve,
            new[] { new BoundaryGridPoint(0, 2), new BoundaryGridPoint(3, 2) },
            BoundaryTier.SuzerainSystem, 1, 2,
            0x336699FF, 0x993333FF,
            isRiver: false, coastSide: BoundaryRibbonCoastSide.None);
        BoundaryMeshDraft spanning =
            HierarchicalVassalBoundaryMeshDraftRules.BuildRibbons(
                new[] { spanningInput }, thirdOwner);
        Equal(true, spanning.VertexCount > 8,
            "long ribbon spans are deterministically resampled for footprint safety");
        Equal(true, spanning.LocalHalfWidths.Min() < 0.16f,
            "a forbidden mid-span footprint cannot hide between safe endpoints");

        BoundaryCellRaster river = Raster(3, 6, (x, y) =>
            x == 1 ? Water(x, y) :
            Land(x, y, x == 0 ? 1 : 2, x == 0 ? 10 : 20,
                x == 0 ? 100 : 200, 0xFFFFFFFF));
        var riverCurve = new BoundaryCurveDraft(new[]
        {
            new BoundaryFloatPoint(1.5f, 0.5f),
            new BoundaryFloatPoint(1.5f, 5.5f)
        }, false, false, 1f,
            new BoundaryFloatPoint(0, 1), new BoundaryFloatPoint(0, 1));
        var riverInput = new BoundaryRibbonInput(
            riverCurve,
            new[] { new BoundaryGridPoint(1, 0), new BoundaryGridPoint(1, 6) },
            BoundaryTier.SuzerainSystem, 1, 2,
            0x336699FF, 0x993333FF,
            isRiver: true, coastSide: BoundaryRibbonCoastSide.None);
        BoundaryMeshDraft riverDraft =
            HierarchicalVassalBoundaryMeshDraftRules.BuildRibbons(
                new[] { riverInput }, river);
        Equal(true, riverDraft.PoliticalAlpha.All(alpha => alpha == 0),
            "both political-color halves are transparent over river water");
        Equal(true, riverDraft.CenterLineIndices.Length > 0,
            "river water retains only the political center line");
        Equal(true, riverDraft.Flags.All(flags =>
                (flags & (byte)BoundaryRibbonFlags.River) != 0),
            "river flag is carried per vertex");
    }

    private static void CanonicalSeamsAreBitIdentical()
    {
        BoundaryCellRaster grid = Raster(64, 8, (x, y) =>
            Land(x, y, y < 4 ? 1 : 2, y < 4 ? 10 : 20,
                y < 4 ? 100 : 200, 0xFFFFFFFF));
        var canonical = new[]
        {
            new BoundaryGridPoint(30, 4), new BoundaryGridPoint(31, 4),
            new BoundaryGridPoint(32, 4), new BoundaryGridPoint(33, 4),
            new BoundaryGridPoint(34, 4)
        };
        var options = new BoundaryCurveOptions(
            BoundaryTier.SuzerainSystem, 1, 2, 0.45f, false, false);
        BoundaryCurveDraft leftCurve =
            HierarchicalVassalBoundaryCurveRules.FitCanonicalWindow(
                canonical, 0, 2, new HashSet<BoundaryGridPoint>(canonical),
                grid, options);
        BoundaryCurveDraft rightCurve =
            HierarchicalVassalBoundaryCurveRules.FitCanonicalWindow(
                canonical, 2, 4, new HashSet<BoundaryGridPoint>(canonical),
                grid, options);
        BoundaryMeshDraft left =
            HierarchicalVassalBoundaryMeshDraftRules.BuildRibbons(
                new[] { new BoundaryRibbonInput(leftCurve, canonical,
                    BoundaryTier.SuzerainSystem, 1, 2,
                    0xFFFFFFFF, 0x888888FF, false,
                    BoundaryRibbonCoastSide.None) }, grid);
        BoundaryMeshDraft right =
            HierarchicalVassalBoundaryMeshDraftRules.BuildRibbons(
                new[] { new BoundaryRibbonInput(rightCurve, canonical,
                    BoundaryTier.SuzerainSystem, 1, 2,
                    0xFFFFFFFF, 0x888888FF, false,
                    BoundaryRibbonCoastSide.None) }, grid);

        BitEqual(leftCurve.Points[^1], rightCurve.Points[0], "seam endpoint");
        BitEqual(leftCurve.EndTangent, rightCurve.StartTangent, "seam tangent");
        BitEqual(left.LocalHalfWidths[^1], right.LocalHalfWidths[0],
            "seam local half-width");
    }

    private static void CoastFlagsAndTransparencyAreSideSpecific()
    {
        BoundaryCellRaster coast = Raster(2, 3, (x, y) =>
            Land(x, y, 1, 10, 100, 0x336699FF));
        var curve = new BoundaryCurveDraft(new[]
        {
            new BoundaryFloatPoint(2, 0.2f),
            new BoundaryFloatPoint(2, 2.8f)
        }, false, false, 1f,
            new BoundaryFloatPoint(0, 1), new BoundaryFloatPoint(0, 1));
        var input = new BoundaryRibbonInput(
            curve,
            new[] { new BoundaryGridPoint(2, 0),
                new BoundaryGridPoint(2, 3) },
            BoundaryTier.SuzerainSystem, 1, -1,
            0x336699FF, 0, false, BoundaryRibbonCoastSide.Right);
        BoundaryMeshDraft draft =
            HierarchicalVassalBoundaryMeshDraftRules.BuildRibbons(
                new[] { input }, coast);
        Equal(true, draft.Flags.All(value =>
                (value & (byte)BoundaryRibbonFlags.Coast) != 0),
            "coast flag is carried per vertex");
        Equal((byte)255, draft.PoliticalAlpha[0],
            "land-facing coast half remains colored");
        Equal((byte)0, draft.PoliticalAlpha[3],
            "water-facing coast half is transparent");
    }

    private static void HierarchyColorsAreStableAndDistinguishable()
    {
        uint root = 0x336699FF;
        uint realmA = HierarchicalVassalBoundaryMeshDraftRules.HierarchyColor(
            root, rootSystemId: 1, displayedOwnerId: 10,
            BoundaryTier.VassalRealm, adjacentRgba: 0);
        uint realmARebuild = HierarchicalVassalBoundaryMeshDraftRules.HierarchyColor(
            root, 1, 10, BoundaryTier.VassalRealm, 0);
        uint realmB = HierarchicalVassalBoundaryMeshDraftRules.HierarchyColor(
            root, 1, 11, BoundaryTier.VassalRealm, realmA);
        uint city = HierarchicalVassalBoundaryMeshDraftRules.HierarchyColor(
            root, 1, 100, BoundaryTier.City, realmA);
        Equal(realmA, realmARebuild, "same hierarchy color is rebuild-stable");
        Equal(true, realmA != realmB, "adjacent realms are distinguishable");
        Equal(true, city != realmA, "city variation remains distinguishable");
        Equal(0xFFu, root & 0xFFu, "root hierarchy alpha identity is stable");
        Equal(0xFFu, realmA & 0xFFu, "derived color preserves root alpha");
    }

    private static void UnsafeAcceptedRibbonRebuildsFromRawContour()
    {
        BoundaryCellRaster grid = Raster(4, 4, (x, y) =>
            Land(x, y, x < 2 ? 1 : 2, x < 2 ? 10 : 20,
                x < 2 ? 100 : 200, 0xFFFFFFFF));
        var accepted = new BoundaryCurveDraft(new[]
        {
            new BoundaryFloatPoint(2.60f, 0.2f),
            new BoundaryFloatPoint(2.60f, 3.8f)
        }, false, false, 1f,
            new BoundaryFloatPoint(0, 1), new BoundaryFloatPoint(0, 1));
        var input = new BoundaryRibbonInput(
            accepted,
            new[] { new BoundaryGridPoint(2, 0),
                new BoundaryGridPoint(2, 4) },
            BoundaryTier.SuzerainSystem, 1, 2,
            0x336699FF, 0x993333FF, false,
            BoundaryRibbonCoastSide.None);
        BoundaryMeshDraft draft =
            HierarchicalVassalBoundaryMeshDraftRules.BuildRibbons(
                new[] { input }, grid);
        Equal(true, draft.SegmentFallbacks.All(value => value != 0),
            "unsafe accepted ribbon is marked as raw fallback");
        for (int vertex = 0; vertex < draft.VertexCount; vertex += 4)
        {
            float centerX = (draft.PositionX[vertex] +
                             draft.PositionX[vertex + 3]) * 0.5f;
            NearlyEqual(2f, centerX,
                "raw fallback ribbon center uses the raw contour");
        }
        Equal(true, draft.LocalHalfWidths.All(value => value > 0f),
            "raw fallback emits the widest safe non-degenerate ribbon");
    }

    private static void UnsafeRawRibbonReturnsBoundedFailure()
    {
        BoundaryCellRaster water = Raster(3, 3, (x, y) => Water(x, y));
        var curve = new BoundaryCurveDraft(new[]
        {
            new BoundaryFloatPoint(1.5f, 0.2f),
            new BoundaryFloatPoint(1.5f, 2.8f)
        }, false, false, 1f,
            new BoundaryFloatPoint(0, 1), new BoundaryFloatPoint(0, 1));
        var input = new BoundaryRibbonInput(
            curve,
            new[] { new BoundaryGridPoint(1, 0),
                new BoundaryGridPoint(1, 3) },
            BoundaryTier.SuzerainSystem, 1, 2,
            0xFFFFFFFF, 0x888888FF, false,
            BoundaryRibbonCoastSide.None);
        BoundaryMeshDraft draft =
            HierarchicalVassalBoundaryMeshDraftRules.BuildRibbons(
                new[] { input }, water);
        Equal(1, draft.FailureCount,
            "unsafe accepted and raw footprints report one bounded failure");
        Equal(0, draft.VertexCount,
            "bounded failure never emits zero-width ribbon vertices");
        Equal(0, draft.SuzerainSystemIndices.Length,
            "bounded failure never emits degenerate triangles");
    }

    private static BoundaryRibbonInput Input(
        BoundaryTier tier, long left, long right,
        uint leftColor, uint rightColor, float y)
    {
        var curve = new BoundaryCurveDraft(new[]
        {
            new BoundaryFloatPoint(4, y),
            new BoundaryFloatPoint(4, y + 1)
        }, false, false, 1f,
            new BoundaryFloatPoint(0, 1), new BoundaryFloatPoint(0, 1));
        return new BoundaryRibbonInput(curve,
            new[] { new BoundaryGridPoint(4, (int)y),
                new BoundaryGridPoint(4, (int)y + 1) },
            tier, left, right, leftColor, rightColor,
            false, BoundaryRibbonCoastSide.None);
    }

    private static BoundaryChunkBounds Interior(BoundaryCellRaster grid)
    {
        return new BoundaryChunkBounds(
            grid.OriginX, grid.OriginY, grid.MaxXExclusive, grid.MaxYExclusive,
            grid.OriginX, grid.OriginY, grid.MaxXExclusive, grid.MaxYExclusive);
    }

    private static BoundaryChunkBounds Bounds(
        int minimumX, int minimumY, int maximumX, int maximumY,
        BoundaryCellRaster grid)
    {
        return new BoundaryChunkBounds(
            grid.OriginX, grid.OriginY,
            grid.MaxXExclusive, grid.MaxYExclusive,
            minimumX, minimumY, maximumX, maximumY);
    }

    private static BoundaryCellRaster Raster(
        int width, int height, Func<int, int, BoundaryCellFacts> factory)
    {
        var cells = new BoundaryCellFacts[width * height];
        for (int y = 0; y < height; y++)
        for (int x = 0; x < width; x++)
            cells[y * width + x] = factory(x, y);
        return new BoundaryCellRaster(0, 0, width, height, cells);
    }

    private static BoundaryCellFacts Land(
        int x, int y, long system, long realm, long city, uint rgba)
    {
        return new BoundaryCellFacts(x, y, true, BoundaryWaterKind.Land,
            128, system, realm, city, rgba);
    }

    private static BoundaryCellFacts Water(int x, int y)
    {
        return new BoundaryCellFacts(x, y, true,
            BoundaryWaterKind.InlandWater, 40, -1, -1, -1, 0);
    }

    private static void NearlyEqual(float expected, float actual, string name)
    {
        if (Math.Abs(expected - actual) > 0.0001f)
            throw new InvalidOperationException(
                name + ": expected " + expected + ", got " + actual);
    }

    private static void BitEqual(float expected, float actual, string name)
    {
        Equal(BitConverter.SingleToInt32Bits(expected),
            BitConverter.SingleToInt32Bits(actual), name);
    }

    private static void BitEqual(
        BoundaryFloatPoint expected, BoundaryFloatPoint actual, string name)
    {
        BitEqual(expected.X, actual.X, name + " x");
        BitEqual(expected.Y, actual.Y, name + " y");
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(
                name + ": expected " + expected + ", got " + actual);
    }
}
