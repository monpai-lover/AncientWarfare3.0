using System.Data.SQLite;
using AncientWarfare3.core.court;

internal static class OfficialCareerHistorySqlTests
{
    public static void Run()
    {
        using var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        using (var create = new SQLiteCommand(
            "CREATE TABLE CourtOfficer (OFFICER_ID INTEGER PRIMARY KEY," +
            "KINGDOM_ID INTEGER,ACTOR_ID INTEGER,ACTOR_NAME TEXT," +
            "CITY_ID INTEGER,COUNTY_ID INTEGER,LAYER TEXT,OFFICE_ID TEXT," +
            "APPOINTED_YEAR INTEGER,APPOINTED_TIME REAL," +
            "ENDED_YEAR INTEGER,ACTIVE INTEGER,END_REASON TEXT," +
            "RANK_AT_APPOINTMENT INTEGER," +
            "LOCAL_GRADE_AT_APPOINTMENT INTEGER)", db))
            create.ExecuteNonQuery();

        Insert(db, 1, "前任", 110, 0, 118, "term_expired");
        Insert(db, 2, "张三", 130, 1, -1, "");
        using (var unrelated = new SQLiteCommand(
            "INSERT INTO CourtOfficer (OFFICER_ID,KINGDOM_ID,ACTOR_ID," +
            "ACTOR_NAME,CITY_ID,LAYER,OFFICE_ID,APPOINTED_YEAR," +
            "APPOINTED_TIME,ENDED_YEAR,ACTIVE,END_REASON," +
            "RANK_AT_APPOINTMENT,LOCAL_GRADE_AT_APPOINTMENT) VALUES " +
            "(3,7,99,'无关',3,'city','constable',131,7860,-1,1,'',6,5)",
            db))
            unrelated.ExecuteNonQuery();

        IReadOnlyList<OfficialCareerHistoryRow> rows =
            OfficialCareerHistoryQuery.Read(db,
                new OfficialCareerHistoryScope(7, 3, "city",
                    "granary_officer"), 32);
        Equal(2, rows.Count, "office history scope");
        Equal("张三", rows[0].ActorName, "stored actor name");
        True(rows[0].IsCurrent, "active history row");

        InsertCounty(db, 4, "county-a", 41);
        InsertCounty(db, 5, "county-b", 42);
        IReadOnlyList<OfficialCareerHistoryRow> countyRows =
            OfficialCareerHistoryQuery.Read(db,
                new OfficialCareerHistoryScope(7, 3, "county",
                    "county_magistrate", 41), 32);
        Equal(1, countyRows.Count,
            "county office history excludes sibling county seats");
        Equal("county-a", countyRows[0].ActorName,
            "county history keeps the selected incumbent");
        Equal(41L, countyRows[0].CountyId,
            "county history row preserves county identity");

        var polluted = new[]
        {
            Row(30, 8, 3, "city", "governor", 130, -1, true, ""),
            Row(29, 8, 3, "city", "governor", 129, 130, false,
                "promoted_city_leader"),
            Row(28, 8, 3, "city", "governor", 128, 129, false,
                "city_leader_mismatch"),
            Row(27, 9, 3, "city", "governor", 120, 128, false,
                "term_expired")
        };
        IReadOnlyList<OfficialCareerHistoryRow> collapsed =
            OfficialCareerHistoryRules.CollapseTechnicalTransitions(polluted);
        Equal(2, collapsed.Count, "technical leader churn is one visible term");
        Equal(8L, collapsed[0].ActorId, "current incumbent is preserved");
        Equal(128, collapsed[0].StartYear,
            "collapsed term starts at the first technical appointment");
        Equal(9L, collapsed[1].ActorId,
            "a real incumbent change remains visible");

        IReadOnlyList<OfficialCareerHistoryRow> separate =
            OfficialCareerHistoryRules.CollapseTechnicalTransitions(new[]
            {
                Row(4, 8, 3, "city", "governor", 130, -1, true, ""),
                Row(3, 8, 4, "city", "governor", 129, 130, false,
                    "city_leader_mismatch")
            });
        Equal(2, separate.Count,
            "different city scopes never merge technical transitions");
    }

    private static void Insert(SQLiteConnection pDb, long pId,
        string pName, int pYear, int pActive, int pEndYear,
        string pReason)
    {
        using var command = new SQLiteCommand(
            "INSERT INTO CourtOfficer (OFFICER_ID,KINGDOM_ID,ACTOR_ID," +
            "ACTOR_NAME,CITY_ID,LAYER,OFFICE_ID,APPOINTED_YEAR," +
            "APPOINTED_TIME,ENDED_YEAR,ACTIVE,END_REASON," +
            "RANK_AT_APPOINTMENT,LOCAL_GRADE_AT_APPOINTMENT) VALUES " +
            "(@id,7,@id,@name,3,'city','granary_officer',@year," +
            "@time,@end,@active,@reason,6,5)", pDb);
        command.Parameters.AddWithValue("@id", pId);
        command.Parameters.AddWithValue("@name", pName);
        command.Parameters.AddWithValue("@year", pYear);
        command.Parameters.AddWithValue("@time", pYear * 60d);
        command.Parameters.AddWithValue("@end", pEndYear);
        command.Parameters.AddWithValue("@active", pActive);
        command.Parameters.AddWithValue("@reason", pReason);
        command.ExecuteNonQuery();
    }

    private static void InsertCounty(SQLiteConnection pDb, long pId,
        string pName, long pCountyId)
    {
        using var command = new SQLiteCommand(
            "INSERT INTO CourtOfficer (OFFICER_ID,KINGDOM_ID,ACTOR_ID," +
            "ACTOR_NAME,CITY_ID,COUNTY_ID,LAYER,OFFICE_ID," +
            "APPOINTED_YEAR,APPOINTED_TIME,ENDED_YEAR,ACTIVE,END_REASON," +
            "RANK_AT_APPOINTMENT,LOCAL_GRADE_AT_APPOINTMENT) VALUES " +
            "(@id,7,@id,@name,3,@county,'county','county_magistrate'," +
            "131,7860,-1,1,'',6,5)", pDb);
        command.Parameters.AddWithValue("@id", pId);
        command.Parameters.AddWithValue("@name", pName);
        command.Parameters.AddWithValue("@county", pCountyId);
        command.ExecuteNonQuery();
    }

    private static void True(bool pValue, string pMessage)
    {
        if (!pValue) throw new InvalidOperationException(pMessage);
    }

    private static OfficialCareerHistoryRow Row(long pOfficerId,
        long pActorId, long pCityId, string pLayer, string pOfficeId,
        int pStart, int pEnd, bool pCurrent, string pReason)
    {
        return new OfficialCareerHistoryRow(7, pOfficerId, pActorId,
            pCityId, pLayer, pOfficeId, "actor", pStart, pEnd,
            pCurrent, pReason, pStart * 60d);
    }

    private static void Equal<T>(T pExpected, T pActual, string pMessage)
    {
        if (!EqualityComparer<T>.Default.Equals(pExpected, pActual))
            throw new InvalidOperationException(pMessage + ": expected " +
                pExpected + ", got " + pActual);
    }
}
