using System;
using System.Linq;
using AncientWarfare3.content.figures;

internal static class HistoricalFigureCardCatalogRulesTests
{
    public static void Run()
    {
        Equal(0.0026f, HistoricalFigureCardRarity.Gold.Probability,
            "gold probability is stable");
        Equal(0.0064f, HistoricalFigureCardRarity.Red.Probability,
            "red probability is stable");
        Equal(1f, HistoricalFigureCardRarity.TotalProbability,
            "rarity probabilities total one");

        var all = HistoricalFigureCardCatalog.All;
        True(HistoricalFigureCardCatalog.IsValid,
            "catalogue validation does not reject explicitly assigned minister collections");
        Equal(all.Count, all.Select(p => p.CardId).Distinct(StringComparer.Ordinal).Count(),
            "card ids are unique");
        True(all.All(p => p.FameScore >= 0 && p.FameScore <= 100),
            "fame scores are bounded");
        True(all.All(p => p.ParentReferencesAreValid(all)),
            "parent references point at cards or are empty");
        True(all.All(p => p.BirthYear != -1 && p.DeathYear != -1),
            "unknown years use an explicit sentinel instead of minus one");
        True(all.All(p => p.DisplayName != p.FatherDisplayName &&
                          p.DisplayName != p.MotherDisplayName),
            "no historical figure is recorded as their own parent");
        True(all.All(p => !string.IsNullOrWhiteSpace(p.BackgroundSummary) &&
                          p.BackgroundSummary.Length >= 8),
            "every historical figure has a background summary");
        True(all.All(p => !string.IsNullOrWhiteSpace(p.DetailedBiography) &&
                          p.DetailedBiography.Length >= 20),
            "every historical figure has a detailed biography");
        True(all.All(p => !p.DetailedBiography.Contains("该人物卡以现有历史目录")),
            "detailed biographies do not use the old placeholder suffix");
        True(all.All(p => p.DetailedBiography.Length >= 60),
            "every card exposes paragraph-length historical detail");
        True(all.All(p => !p.DetailedBiography.Contains(
            "是所属政权历史进程中的重要人物。其经历集中体现")),
            "detailed biographies do not use the generic card fallback");
        True(all.Count(p => p.DetailedBiography.Length >= 60) > all.Count / 2,
            "most cards expose paragraph-length historical detail");

        var historicalCards = all.Where(p => !string.Equals(p.CollectionId,
            HistoricalFigureCardSupporterSeeds.CollectionId,
            StringComparison.Ordinal)).ToArray();
        True(historicalCards.All(card =>
                HistoricalFigureCardNarratives.HasDetailed(card.CardId)),
            "every historical card has an exact CardId biography; remaining=" +
            string.Join(",", historicalCards.Where(card =>
                    !HistoricalFigureCardNarratives.HasDetailed(card.CardId))
                .Take(12).Select(card => card.CardId)));
        string[] genericBiographyFragments =
        {
            "本条资料按人物所属时代、政权和传记材料整理",
            "具体事迹需结合不同史料互证",
            "卡片部署后保留其历史国号",
            "具体评价应以史料背景而非游戏数值为准",
            "重要活动年代约为",
            "历史国号为"
        };
        var genericBiographyCards = historicalCards.Where(card =>
            genericBiographyFragments.Any(fragment =>
                card.DetailedBiography.Contains(fragment))).ToArray();
        Equal(0, genericBiographyCards.Length,
            "every historical card has specific detail; remaining=" +
            string.Join(",", genericBiographyCards.Take(12)
                .Select(p => p.CardId)));
        True(historicalCards.All(card => card.DetailedBiography.Length >= 60 &&
                                         SentenceCount(card.DetailedBiography) >= 3),
            "every historical card has at least three detailed sentences");
        True(historicalCards.All(card =>
                !card.DetailedBiography.Contains("?") &&
                !card.DetailedBiography.Contains("-{")),
            "historical biographies contain no broken glyph or wiki conversion markup");
        True(historicalCards.All(card =>
                !card.BackgroundSummary.Contains("其活动与") &&
                !card.BackgroundSummary.Contains("密切相关")),
            "every historical card has a person-specific background summary");
        True(historicalCards.All(card => card.BackgroundSummary.Length >= 12),
            "every historical card has a substantial background summary");
        True(!string.Equals(HistoricalFigureCardCatalog.Get("three_liu_yu")
                    .DetailedBiography,
                HistoricalFigureCardCatalog.Get("song_wudi").DetailedBiography,
                StringComparison.Ordinal),
            "Liu Yu minister and monarch cards have identity-specific detail");
        True(!string.Equals(HistoricalFigureCardCatalog.Get("sui_an_lushan")
                    .DetailedBiography,
                HistoricalFigureCardCatalog.Get("an_lushan").DetailedBiography,
                StringComparison.Ordinal),
            "An Lushan minister and monarch cards have identity-specific detail");
        True(historicalCards.GroupBy(card => card.DisplayName,
                StringComparer.Ordinal).Where(group => group.Count() > 1)
            .All(group => group.Select(card => card.DetailedBiography)
                .Distinct(StringComparer.Ordinal).Count() == group.Count()),
            "every same-name card identity has distinct biography text");
        Equal(historicalCards.Length, historicalCards
                .Select(card => card.DetailedBiography)
                .Distinct(StringComparer.Ordinal).Count(),
            "every historical card identity has globally distinct biography text");
        True(historicalCards.All(card => string.Equals(
                card.BackgroundSummary,
                HistoricalFigureCardCuratedBiographies.Summary(
                    card.DetailedBiography), StringComparison.Ordinal)),
            "every card background is the first sentence of its own biography");
        Equal("方孝孺", HistoricalFigureCardCatalog.Get("ming_fang_xiaoru")
                .DisplayName,
            "Fang Xiaoru uses the historically attested name");
        Equal("揭傒斯", HistoricalFigureCardCatalog.Get("yuan_jiexisi")
                .DisplayName,
            "Jie Xisi uses the historically attested name");
        Equal("赵孟頫", HistoricalFigureCardCatalog.Get("yuan_zhao_mengfu_blue")
                .DisplayName,
            "Zhao Mengfu uses the historically attested name");
        True(HistoricalFigureCardCatalog.Get("qin_hanfeizi")
                .DetailedBiography.StartsWith("韩非", StringComparison.Ordinal) &&
             !HistoricalFigureCardCatalog.Get("qin_hanfeizi")
                 .DetailedBiography.StartsWith("《韩非子》", StringComparison.Ordinal),
            "Han Feizi card describes the person rather than the book");
        True(all.All(p => !HistoricalFigureCardCatalog.HasGeographicPrefix(
            p.HistoricalKingdomName)), "historical names are short kingdom names");
        True(all.Count(p => p.LegacyFigureId != null) >= HistoricalFigureDef.All.Length,
            "every registered historical figure still has a card after catalogue expansion");
        True(all.Any(p => p.CardId == "qin_shihuang") &&
             all.Any(p => p.CardId == "han_wudi") &&
             all.Any(p => p.CardId == "tang_taizong") &&
             all.Any(p => p.CardId == "ming_taizu") &&
             all.Any(p => p.CardId == "qing_xuantong"),
            "emperor catalogue contains snapshot anchors");

        var gold = all.Where(p => p.Rarity.Equals(HistoricalFigureCardRarity.Gold))
            .ToArray();
        True(gold.Length >= 21,
            "shared gold pool has room for the added author avatar card");
        True(new[] { "ji_fa", "han_gaozu", "qin_shihuang", "three_cao_cao",
                     "tang_taizong", "yuan_taizu" }
            .All(id => gold.Any(card => card.CardId == id)),
            "shared gold pool contains the requested famous figures");
        HistoricalFigureCardDefinition authorAvatar =
            HistoricalFigureCardCatalog.Get("mengpai");
        True(authorAvatar != null, "author avatar card is registered");
        Equal("\u8499\u6d3e", authorAvatar.DisplayName,
            "author avatar keeps the requested display name");
        Equal("\u7334\u6c49", authorAvatar.HistoricalKingdomName,
            "author avatar uses the Monkey Han historical kingdom");
        Equal(HistoricalFigureCardRarity.Gold, authorAvatar.Rarity,
            "author avatar belongs to the gold pool");
        Equal(HistoricalFigureCardRole.Monarch, authorAvatar.Role,
            "author avatar is a monarch card");
        Equal("ui/historical_cards/mengpai", authorAvatar.PortraitPath,
            "author avatar uses its bounded card portrait resource");
        True(gold.Any(card => card.CardId == "mengpai"),
            "author avatar is included in the shared gold pool");
        HistoricalFigureCardDefinition caoCao =
            HistoricalFigureCardCatalog.Get("three_cao_cao");
        Equal(HistoricalFigureCardRole.Minister, caoCao.Role,
            "Cao Cao is a deployable minister card");
        Equal("曹嵩", caoCao.FatherDisplayName,
            "Cao Cao records his historical father");

        string[] portraitCardIds =
        {
            "shu_han_zhaolie", "three_guan_yu", "three_zhang_fei",
            "three_zhuge_liang", "wu_dadi", "three_yuan_shao"
        };
        True(portraitCardIds.All(id => HistoricalFigureCardCatalog.Get(id) != null),
            "all six supplied historical portraits have matching cards");
        True(portraitCardIds.All(id => string.Equals(
                HistoricalFigureCardCatalog.Get(id).PortraitPath,
                "ui/historical_cards/" + id, StringComparison.Ordinal)),
            "all six supplied historical portraits use card-specific resources");
        True(new[] { "three_guan_yu", "three_zhang_fei", "three_yuan_shao" }
            .Select(HistoricalFigureCardCatalog.Get)
            .All(card => card.Role == HistoricalFigureCardRole.Minister &&
                         card.IsMilitaryGeneral &&
                         card.CollectionId == "three_six_dynasties"),
            "Guan Yu, Zhang Fei and Yuan Shao are deployable Three Kingdoms generals");

        var sorted = HistoricalFigureCardCatalog.SortForDisplay(all);
        True(sorted[0].FameScore >= sorted[1].FameScore,
            "catalogue display order starts with highest fame");
        Equal("史料不详", HistoricalFigureCardCatalog.ParentDisplayName(""),
            "unknown parent has explicit display text");
        HistoricalFigureCardDefinition yuanShu =
            HistoricalFigureCardCatalog.Get("zhong_hou_shu");
        Equal(155, yuanShu.BirthYear, "Yuan Shu birth year is recorded");
        Equal(199, yuanShu.DeathYear, "Yuan Shu death year is recorded");
        Equal("东汉", yuanShu.DynastyName, "Yuan Shu belongs to late Han");
        Equal("仲", yuanShu.HistoricalKingdomName,
            "Yuan Shu founds Zhong");
        HistoricalFigureCardDefinition ziying =
            HistoricalFigureCardCatalog.Get("qin_ziying");
        Equal("嬴", ziying.FamilyName, "Ziying uses the Ying surname");
        Equal("赵", ziying.ClanName, "Ziying uses the Zhao clan");
    }

    private static void True(bool value, string message)
    {
        if (!value) throw new InvalidOperationException(message);
    }

    private static int SentenceCount(string pText)
    {
        return (pText ?? "").Count(p => p == '。');
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(message + ": expected=" + expected +
                                                ", actual=" + actual);
    }
}
