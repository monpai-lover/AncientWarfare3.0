using System;
using System.Linq;
using AncientWarfare3.content.figures;

internal static class HistoricalFigureCardCatalogRulesTests
{
    public static void Run()
    {
        Equal(0.0026f, HistoricalFigureCardRarity.Gold.Probability,
            "gold probability is stable");
        Equal(0.0064f, HistoricalFigureCardRarity.Red.Probability,
            "red probability is stable");
        Equal(1f, HistoricalFigureCardRarity.TotalProbability,
            "rarity probabilities total one");

        var all = HistoricalFigureCardCatalog.All;
        True(HistoricalFigureCardCatalog.IsValid,
            "catalogue validation does not reject explicitly assigned minister collections");
        Equal(all.Count, all.Select(p => p.CardId).Distinct(StringComparer.Ordinal).Count(),
            "card ids are unique");
        True(all.All(p => p.FameScore >= 0 && p.FameScore <= 100),
            "fame scores are bounded");
        True(all.All(p => p.ParentReferencesAreValid(all)),
            "parent references point at cards or are empty");
        True(all.All(p => p.BirthYear != -1 && p.DeathYear != -1),
            "unknown years use an explicit sentinel instead of minus one");
        True(all.All(p => p.DisplayName != p.FatherDisplayName &&
                          p.DisplayName != p.MotherDisplayName),
            "no historical figure is recorded as their own parent");
        True(all.All(p => !string.IsNullOrWhiteSpace(p.BackgroundSummary) &&
                          p.BackgroundSummary.Length >= 8),
            "every historical figure has a background summary");
        True(all.All(p => !string.IsNullOrWhiteSpace(p.DetailedBiography) &&
                          p.DetailedBiography.Length >= 20),
            "every historical figure has a detailed biography");
        True(all.All(p => !HistoricalFigureCardCatalog.HasGeographicPrefix(
            p.HistoricalKingdomName)), "historical names are short kingdom names");
        True(all.Count(p => p.LegacyFigureId != null) >= HistoricalFigureDef.All.Length,
            "every registered historical figure still has a card after catalogue expansion");
        True(all.Any(p => p.CardId == "qin_shihuang") &&
             all.Any(p => p.CardId == "han_wudi") &&
             all.Any(p => p.CardId == "tang_taizong") &&
             all.Any(p => p.CardId == "ming_taizu") &&
             all.Any(p => p.CardId == "qing_xuantong"),
            "emperor catalogue contains snapshot anchors");

        var gold = all.Where(p => p.Rarity.Equals(HistoricalFigureCardRarity.Gold))
            .ToArray();
        True(gold.Length >= 20,
            "shared gold pool contains a broad historical roster");
        True(new[] { "ji_fa", "han_gaozu", "qin_shihuang", "three_cao_cao",
                     "tang_taizong", "yuan_taizu" }
            .All(id => gold.Any(card => card.CardId == id)),
            "shared gold pool contains the requested famous figures");
        HistoricalFigureCardDefinition authorAvatar =
            HistoricalFigureCardCatalog.Get("mengpai");
        True(authorAvatar != null, "author avatar card is registered");
        Equal("\u8499\u6d3e", authorAvatar.DisplayName,
            "author avatar keeps the requested display name");
        Equal("\u7334\u6c49", authorAvatar.HistoricalKingdomName,
            "author avatar uses the Monkey Han historical kingdom");
        Equal(HistoricalFigureCardRarity.Gold, authorAvatar.Rarity,
            "author avatar belongs to the gold pool");
        Equal(HistoricalFigureCardRole.Monarch, authorAvatar.Role,
            "author avatar is a monarch card");
        True(gold.Any(card => card.CardId == "mengpai"),
            "author avatar is included in the shared gold pool");
        HistoricalFigureCardDefinition caoCao =
            HistoricalFigureCardCatalog.Get("three_cao_cao");
        Equal(HistoricalFigureCardRole.Minister, caoCao.Role,
            "Cao Cao is a deployable minister card");
        Equal("曹嵩", caoCao.FatherDisplayName,
            "Cao Cao records his historical father");

        var sorted = HistoricalFigureCardCatalog.SortForDisplay(all);
        True(sorted[0].FameScore >= sorted[1].FameScore,
            "catalogue display order starts with highest fame");
        Equal("史料不详", HistoricalFigureCardCatalog.ParentDisplayName(""),
            "unknown parent has explicit display text");
        HistoricalFigureCardDefinition yuanShu =
            HistoricalFigureCardCatalog.Get("zhong_hou_shu");
        Equal(155, yuanShu.BirthYear, "Yuan Shu birth year is recorded");
        Equal(199, yuanShu.DeathYear, "Yuan Shu death year is recorded");
        Equal("东汉", yuanShu.DynastyName, "Yuan Shu belongs to late Han");
        Equal("仲", yuanShu.HistoricalKingdomName,
            "Yuan Shu founds Zhong");
        HistoricalFigureCardDefinition ziying =
            HistoricalFigureCardCatalog.Get("qin_ziying");
        Equal("嬴", ziying.FamilyName, "Ziying uses the Ying surname");
        Equal("赵", ziying.ClanName, "Ziying uses the Zhao clan");
    }

    private static void True(bool value, string message)
    {
        if (!value) throw new InvalidOperationException(message);
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(message + ": expected=" + expected +
                                                ", actual=" + actual);
    }
}
