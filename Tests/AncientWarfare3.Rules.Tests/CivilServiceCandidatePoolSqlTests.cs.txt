using System;
using System.Collections.Generic;
using System.Data.SQLite;
using AncientWarfare3.core.court;

internal static class CivilServiceCandidatePoolSqlTests
{
    public static void Run()
    {
        RecentFailuresCannotStarveNewLocalCandidates();
        SameYearStudentsCannotConsumeTheCandidateWindow();
        ArchivedSlavesCannotStarveNewLocalCandidates();
        SeniorCentralOfficialsCannotConsumeTheCandidateWindow();
        HostGraduatesCannotConsumeTheForeignCandidateLimit();
        WomenCannotEnterLocalOrForeignCandidatePools();
    }

    private static void RecentFailuresCannotStarveNewLocalCandidates()
    {
        using var db = OpenDatabase();
        for (int index = 0; index < 96; index++)
        {
            long actorId = 1000 + index;
            InsertMembership(db, actorId, startYear: 1);
            InsertArchive(db, actorId, kingdomId: 7,
                status: "none", everNoble: 0, lineageId: -1);
            Execute(db, "INSERT INTO Session VALUES (" + actorId +
                ",7,20,'completed')");
            Execute(db, "INSERT INTO Candidate VALUES (" + actorId +
                ",7," + actorId + ",'none')");
        }
        InsertMembership(db, actorId: 2000, startYear: 30);
        InsertArchive(db, actorId: 2000, kingdomId: 7,
            status: "none", everNoble: 0, lineageId: -1);

        List<long> result = CivilServiceExamCandidatePoolQuery.LoadLocal(
            db, "Membership", "Archive", "Officer", "Career",
            "Institution", "Candidate", "Session", pKingdomId: 7,
            pSocialOrigin: "commoner", pYear: 31,
            pTributeMode: true, pLimit: 96);

        Equal(96, result.Count, "the local source remains bounded");
        True(result.Contains(2000L),
            "a newly educated candidate is considered before recent failures retry");
    }

    private static void SameYearStudentsCannotConsumeTheCandidateWindow()
    {
        using var db = OpenDatabase();
        for (int index = 0; index < 96; index++)
        {
            long actorId = 5000 + index;
            InsertMembership(db, actorId, startYear: 30);
            InsertArchive(db, actorId, kingdomId: 7,
                status: "none", everNoble: 0, lineageId: -1);
        }
        InsertMembership(db, actorId: 6000, startYear: 29);
        InsertArchive(db, actorId: 6000, kingdomId: 7,
            status: "none", everNoble: 0, lineageId: -1);

        List<long> result = CivilServiceExamCandidatePoolQuery.LoadLocal(
            db, "Membership", "Archive", "Officer", "Career",
            "Institution", "Candidate", "Session", pKingdomId: 7,
            pSocialOrigin: "commoner", pYear: 30,
            pTributeMode: true, pLimit: 96);

        Equal(1, result.Count,
            "same-year students are filtered before the bounded source window");
        Equal(6000L, result[0],
            "an older educated student remains visible behind same-year rows");
    }

    private static void HostGraduatesCannotConsumeTheForeignCandidateLimit()
    {
        using var db = OpenDatabase();
        InsertMembership(db, actorId: 1, startYear: 1);
        InsertMembership(db, actorId: 2, startYear: 2);
        InsertArchive(db, actorId: 1, kingdomId: 9,
            status: "none", everNoble: 0, lineageId: -1);
        InsertArchive(db, actorId: 2, kingdomId: 9,
            status: "none", everNoble: 0, lineageId: -1);
        Execute(db, "INSERT INTO Affiliation VALUES " +
            "(1,10,'Resident',-1,9),(2,10,'Resident',-1,9)");
        Execute(db, "INSERT INTO Session VALUES (100,7,20,'completed')");
        Execute(db, "INSERT INTO Candidate VALUES (1,7,100,'gongshi')");

        List<long> result = CivilServiceExamCandidatePoolQuery.
            LoadForeignResidents(db, "Membership", "Archive",
                "Affiliation", "Officer", "Candidate", "Session",
                new[] { 10L }, pHostKingdomId: 7, pYear: 30,
                pResidentState: "Resident", pTributeMode: true, pLimit: 1);

        Equal(1, result.Count,
            "one unqualified foreign resident remains visible");
        Equal(2L, result[0],
            "a prior host graduate is filtered before the SQL limit");
    }

    private static void ArchivedSlavesCannotStarveNewLocalCandidates()
    {
        using var db = OpenDatabase();
        for (int index = 0; index < 96; index++)
        {
            long actorId = 3000 + index;
            InsertMembership(db, actorId, startYear: 1);
            InsertArchive(db, actorId, kingdomId: 7,
                status: "slave_lineage", everNoble: 0, lineageId: -1);
        }
        InsertMembership(db, actorId: 4000, startYear: 30);
        InsertArchive(db, actorId: 4000, kingdomId: 7,
            status: "none", everNoble: 0, lineageId: -1);

        List<long> result = CivilServiceExamCandidatePoolQuery.LoadLocal(
            db, "Membership", "Archive", "Officer", "Career",
            "Institution", "Candidate", "Session", pKingdomId: 7,
            pSocialOrigin: "commoner", pYear: 31,
            pTributeMode: true, pLimit: 96);

        Equal(1, result.Count,
            "archived slaves are excluded before the bounded source window");
        Equal(4000L, result[0],
            "a later eligible student survives a century of slave records");
    }

    private static void SeniorCentralOfficialsCannotConsumeTheCandidateWindow()
    {
        using var db = OpenDatabase();
        for (int index = 0; index < 96; index++)
        {
            long actorId = 9000 + index;
            InsertMembership(db, actorId, startYear: 1);
            InsertArchive(db, actorId, kingdomId: 7,
                status: "none", everNoble: 0, lineageId: -1);
            Execute(db, "INSERT INTO Officer VALUES (" + actorId +
                ",7,1,'central','chancellor')");
        }
        InsertMembership(db, actorId: 10000, startYear: 30);
        InsertArchive(db, actorId: 10000, kingdomId: 7,
            status: "none", everNoble: 0, lineageId: -1);

        List<long> result = CivilServiceExamCandidatePoolQuery.LoadLocal(
            db, "Membership", "Archive", "Officer", "Career",
            "Institution", "Candidate", "Session", pKingdomId: 7,
            pSocialOrigin: "commoner", pYear: 31,
            pTributeMode: true, pLimit: 96);

        Equal(1, result.Count,
            "ineligible senior central officials are filtered before the bounded source window");
        Equal(10000L, result[0],
            "a valid young candidate remains visible behind senior central officials");
    }

    private static void WomenCannotEnterLocalOrForeignCandidatePools()
    {
        using var db = OpenDatabase();
        InsertMembership(db, actorId: 7000, startYear: 1);
        InsertArchive(db, actorId: 7000, kingdomId: 7,
            status: "none", everNoble: 0, lineageId: -1, sex: 1);
        InsertMembership(db, actorId: 7001, startYear: 1);
        InsertArchive(db, actorId: 7001, kingdomId: 7,
            status: "none", everNoble: 0, lineageId: -1, sex: 0);

        List<long> local = CivilServiceExamCandidatePoolQuery.LoadLocal(
            db, "Membership", "Archive", "Officer", "Career",
            "Institution", "Candidate", "Session", pKingdomId: 7,
            pSocialOrigin: "commoner", pYear: 30,
            pTributeMode: true, pLimit: 1);

        Equal(1, local.Count, "the local male candidate remains visible");
        Equal(7001L, local[0],
            "a woman cannot consume the bounded local examination window");

        InsertMembership(db, actorId: 8000, startYear: 1);
        InsertArchive(db, actorId: 8000, kingdomId: 9,
            status: "none", everNoble: 0, lineageId: -1, sex: 1);
        InsertMembership(db, actorId: 8001, startYear: 1);
        InsertArchive(db, actorId: 8001, kingdomId: 9,
            status: "none", everNoble: 0, lineageId: -1, sex: 0);
        Execute(db, "INSERT INTO Affiliation VALUES " +
            "(8000,10,'Resident',-1,9),(8001,10,'Resident',-1,9)");

        List<long> foreign = CivilServiceExamCandidatePoolQuery.
            LoadForeignResidents(db, "Membership", "Archive",
                "Affiliation", "Officer", "Candidate", "Session",
                new[] { 10L }, pHostKingdomId: 7, pYear: 30,
                pResidentState: "Resident", pTributeMode: true, pLimit: 1);

        Equal(1, foreign.Count,
            "the foreign male candidate remains visible");
        Equal(8001L, foreign[0],
            "a woman cannot consume the bounded foreign examination window");
    }

    private static SQLiteConnection OpenDatabase()
    {
        var db = new SQLiteConnection(
            "Data Source=:memory:;Version=3;New=True;");
        db.Open();
        Execute(db, "CREATE TABLE Membership (ACTOR_ID INTEGER," +
            "SCHOOL_ID TEXT,ACTIVE INTEGER,START_YEAR INTEGER," +
            "REPUTATION REAL)");
        Execute(db, "CREATE TABLE Archive (ID INTEGER,IS_ALIVE INTEGER," +
            "SEX INTEGER,KINGDOM_ID INTEGER,STATUS TEXT," +
            "EVER_NOBLE_BLOOD INTEGER," +
            "LINEAGE_ID INTEGER)");
        Execute(db, "CREATE TABLE Officer (ACTOR_ID INTEGER," +
            "KINGDOM_ID INTEGER,ACTIVE INTEGER,LAYER TEXT,OFFICE_ID TEXT)");
        Execute(db, "CREATE TABLE Career (ACTOR_ID INTEGER," +
            "KINGDOM_ID INTEGER)");
        Execute(db, "CREATE TABLE Institution (SCHOOL_ID TEXT," +
            "INSTITUTION_ID INTEGER,ACTIVE INTEGER)");
        Execute(db, "CREATE TABLE Candidate (ACTOR_ID INTEGER," +
            "KINGDOM_ID INTEGER,SESSION_ID INTEGER,QUALIFICATION TEXT)");
        Execute(db, "CREATE TABLE Session (ID INTEGER,KINGDOM_ID INTEGER," +
            "CYCLE_YEAR INTEGER,STATUS TEXT)");
        Execute(db, "CREATE TABLE Affiliation (ACTOR_ID INTEGER," +
            "RESIDENCE_CITY_ID INTEGER,LIFECYCLE_STATE TEXT," +
            "SERVICE_KINGDOM_ID INTEGER,HOME_KINGDOM_ID INTEGER)");
        return db;
    }

    private static void InsertMembership(SQLiteConnection pDb,
        long actorId, int startYear)
    {
        Execute(pDb, "INSERT INTO Membership VALUES (" + actorId +
            ",'ru',1," + startYear + ",10)");
    }

    private static void InsertArchive(SQLiteConnection pDb, long actorId,
        long kingdomId, string status, int everNoble, long lineageId,
        int sex = 0)
    {
        Execute(pDb, "INSERT INTO Archive VALUES (" + actorId +
            ",1," + sex + "," + kingdomId + ",'" + status + "'," + everNoble +
            "," + lineageId + ")");
    }

    private static void Execute(SQLiteConnection pDb, string pSql)
    {
        using var command = new SQLiteCommand(pSql, pDb);
        command.ExecuteNonQuery();
    }

    private static void True(bool pValue, string pMessage)
    {
        if (!pValue)
            throw new InvalidOperationException(pMessage + ": expected true");
    }

    private static void Equal<T>(T pExpected, T pActual, string pMessage)
    {
        if (EqualityComparer<T>.Default.Equals(pExpected, pActual)) return;
        throw new InvalidOperationException(pMessage + ": expected " +
            pExpected + ", actual " + pActual);
    }
}
