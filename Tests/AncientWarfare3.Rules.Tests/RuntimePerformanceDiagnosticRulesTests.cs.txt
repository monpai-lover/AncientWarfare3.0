using AncientWarfare3.core.policy;

internal static class RuntimePerformanceDiagnosticRulesTests
{
    public static void Run()
    {
        Equal(false, RuntimePerformanceDiagnosticRules.ShouldEmitTextLog(
                diagnosticsEnabled: false, benchmarkEnabled: false),
            "disabled diagnostics and benchmark mode emit no text log");
        Equal(true, RuntimePerformanceDiagnosticRules.ShouldEmitTextLog(
                diagnosticsEnabled: true, benchmarkEnabled: false),
            "diagnostic mode emits the runtime diagnostic text log");
        Equal(true, RuntimePerformanceDiagnosticRules.ShouldEmitTextLog(
                diagnosticsEnabled: false, benchmarkEnabled: true),
            "benchmark mode emits the runtime diagnostic text log");
        Equal(true, RuntimePerformanceDiagnosticRules.ShouldEmitTextLog(
                diagnosticsEnabled: true, benchmarkEnabled: true),
            "combined diagnostic and benchmark mode emits one text log");
        PerformanceIntervalMetricsUseRealElapsedTime();
        PerformanceIntervalLogUsesO1NativeCounts();
        BenchmarkFallbackCannotBeReintercepted();
        GeneratedFallbackInvalidatesPendingCompletion();
        BenchmarkTimeoutRulesBoundPendingWait();
        BenchmarkTimeoutPollingIsPendingOnlyAndCleansUp();
        BenchmarkTimeoutCoversExceptionalMapUpdate();
        BenchmarkAutoLoadWaitsForTheRequestedWorldAndSelectsX20();
    }

    private static void PerformanceIntervalMetricsUseRealElapsedTime()
    {
        Equal(true, RuntimePerformanceDiagnosticRules.
                ShouldStartIntervalBaseline(
                    intervalEnabled: false, diagnosticsEnabled: true),
            "enabling diagnostics starts a new reportable interval");
        Equal(false, RuntimePerformanceDiagnosticRules.
                ShouldStartIntervalBaseline(
                    intervalEnabled: true, diagnosticsEnabled: true),
            "an active interval keeps its existing baseline");
        Equal(false, RuntimePerformanceDiagnosticRules.
                ShouldStartIntervalBaseline(
                    intervalEnabled: true, diagnosticsEnabled: false),
            "disabling diagnostics does not start an interval");
        Equal(120L, RuntimePerformanceDiagnosticRules.IntervalFrameCount(
                intervalStartFrame: 0L, currentFrame: 120L),
            "first sample reports every frame since the interval baseline");
        Equal(120L, RuntimePerformanceDiagnosticRules.IntervalFrameCount(
                intervalStartFrame: 120L, currentFrame: 240L),
            "later samples report frames since the previous sample");
        Equal(0L, RuntimePerformanceDiagnosticRules.IntervalFrameCount(
                intervalStartFrame: 240L, currentFrame: 120L),
            "a regressed frame counter cannot produce a negative interval");
        Equal(60d, RuntimePerformanceDiagnosticRules.AverageFramesPerSecond(
                intervalFrames: 120L, intervalTicks: 2000L,
                timestampFrequency: 1000L),
            "average FPS uses real elapsed time rather than game time scale");
        Equal(0d, RuntimePerformanceDiagnosticRules.AverageFramesPerSecond(
                intervalFrames: 120L, intervalTicks: 0L,
                timestampFrequency: 1000L),
            "a zero-duration interval has no average FPS");
        Equal(0d, RuntimePerformanceDiagnosticRules.AverageFramesPerSecond(
                intervalFrames: 120L, intervalTicks: 2000L,
                timestampFrequency: 0L),
            "an unavailable timestamp frequency has no average FPS");
    }

    private static void PerformanceIntervalLogUsesO1NativeCounts()
    {
        string source = Read("Code", "core", "policy",
            "RuntimePerformanceDiagnostic.cs");
        Contains(source, " interval_real_ms=",
            "PERF log exposes real sampling interval duration");
        Contains(source, " interval_frames=",
            "PERF log exposes frames completed during the interval");
        Contains(source, " avg_fps=",
            "PERF log exposes average real frames per second");
        Contains(source, " live_population=",
            "PERF log exposes the current live population");
        Contains(source, " army_count=",
            "PERF log exposes the current army count");

        int emitGate = source.IndexOf("ShouldEmitTextLog(",
            StringComparison.Ordinal);
        int countRead = source.IndexOf("ReadLiveWorldCounts(", emitGate,
            StringComparison.Ordinal);
        True(emitGate >= 0 && countRead > emitGate,
            "world counts are read only inside the low-frequency log sample");

        int enableState = source.IndexOf("bool diagnosticsEnabled =",
            StringComparison.Ordinal);
        int baselineDecision = source.IndexOf(
            "ShouldStartIntervalBaseline(", enableState,
            StringComparison.Ordinal);
        int baselineClock = source.IndexOf(
            "_intervalStarted = Stopwatch.GetTimestamp();",
            baselineDecision, StringComparison.Ordinal);
        True(enableState >= 0 && baselineDecision > enableState &&
             baselineClock > baselineDecision,
            "the interval baseline starts only after diagnostics become enabled");

        int helper = source.IndexOf(
            "private static void ReadLiveWorldCounts(",
            StringComparison.Ordinal);
        int helperEnd = IndexAfter(source, "        private static ",
            helper + 1);
        True(helper >= 0 && helperEnd > helper,
            "PERF diagnostics expose a bounded world-count helper");
        string countSource = source.Substring(helper, helperEnd - helper);
        Contains(countSource, "units_only_alive?.Count",
            "live population reuses the native alive-unit list count");
        Contains(countSource, "armies?.Count",
            "army count reuses the native manager count");
        Equal(false, countSource.Contains("foreach") ||
                     countSource.Contains("for (") ||
                     countSource.Contains("getSimpleArray") ||
                     countSource.Contains("ToList("),
            "world-count sampling performs no world scan or materialization");
    }

    private static void BenchmarkFallbackCannotBeReintercepted()
    {
        string source = Read("Code", "patch",
            "AW_RuntimeBenchmarkAutoLoadPatch.cs");
        int prefix = source.IndexOf(
            "private static void StartTheGame_Prefix(ref bool pForceGenerate)",
            StringComparison.Ordinal);
        int dispatchedGate = source.IndexOf("if (_dispatched) return;", prefix,
            StringComparison.Ordinal);
        int forceOverride = source.IndexOf("pForceGenerate =", prefix,
            StringComparison.Ordinal);

        True(prefix >= 0 && dispatchedGate > prefix &&
             forceOverride > dispatchedGate,
            "a dispatched benchmark preserves stock forced-generation fallback");
    }

    private static void GeneratedFallbackInvalidatesPendingCompletion()
    {
        string benchmark = Read("Code", "patch",
            "AW_RuntimeBenchmarkAutoLoadPatch.cs");
        string savePatch = Read("Code", "patch", "AW_SavePatch.cs");
        int generatedPostfix = savePatch.IndexOf(
            "public static void GenerateNewMap_Postfix()",
            StringComparison.Ordinal);
        int clear = savePatch.IndexOf(
            "AW3SaveDirectoryRegistry.ClearForNewWorld();", generatedPostfix,
            StringComparison.Ordinal);
        True(generatedPostfix >= 0 && clear > generatedPostfix,
            "generated fallback clears the pre-observed benchmark directory");

        int completion = benchmark.IndexOf(
            "private static void OnBenchmarkWorldLoaded()",
            StringComparison.Ordinal);
        int missingPath = benchmark.IndexOf(
            "if (!AW3SaveDirectoryRegistry.TryGet(out string loadedPath))",
            completion, StringComparison.Ordinal);
        int cleanup = benchmark.IndexOf("StopWaitingForBenchmarkWorld();",
            missingPath, StringComparison.Ordinal);
        int invalid = benchmark.IndexOf(
            "AW3 benchmark auto-load invalid: completed world has no ",
            cleanup, StringComparison.Ordinal);
        int invalidReturn = benchmark.IndexOf("return;", invalid,
            StringComparison.Ordinal);
        int ready = benchmark.IndexOf("AW3 benchmark auto-load ready:",
            completion, StringComparison.Ordinal);
        int familyTree = benchmark.IndexOf("OpenValidationFamilyTree();",
            completion, StringComparison.Ordinal);
        True(completion >= 0 && missingPath > completion &&
             cleanup > missingPath && invalid > cleanup &&
             invalidReturn > invalid && ready > invalidReturn &&
             familyTree > ready,
            "generated fallback is invalid, cleans up, and cannot validate");
    }

    private static void BenchmarkTimeoutRulesBoundPendingWait()
    {
        System.Reflection.MethodInfo timeoutRule =
            typeof(RuntimePerformanceDiagnosticRules).GetMethod(
                "HasBenchmarkAutoLoadTimedOut",
                System.Reflection.BindingFlags.Public |
                System.Reflection.BindingFlags.Static);
        True(timeoutRule != null,
            "benchmark auto-load exposes a testable timeout rule");

        bool Invoke(bool pending, long nowTicks, long deadlineTicks)
        {
            return (bool)timeoutRule.Invoke(null,
                new object[] { pending, nowTicks, deadlineTicks });
        }

        Equal(false, Invoke(false, 200L, 100L),
            "non-pending benchmark work never times out");
        Equal(false, Invoke(true, 99L, 100L),
            "pending benchmark remains live before its deadline");
        Equal(true, Invoke(true, 100L, 100L),
            "pending benchmark expires exactly at its deadline");
        Equal(false, Invoke(true, 200L, 0L),
            "an unset deadline cannot expire benchmark work");
    }

    private static void BenchmarkTimeoutPollingIsPendingOnlyAndCleansUp()
    {
        string source = Read("Code", "patch",
            "AW_RuntimeBenchmarkAutoLoadPatch.cs");
        int timeoutPoll = source.IndexOf(
            "private static void CheckPendingBenchmarkAutoLoadTimeout()",
            StringComparison.Ordinal);
        int pendingGate = IndexAfter(source,
            "if (!_waitingForBenchmarkWorld) return;", timeoutPoll);
        int clockRead = IndexAfter(source, "DateTime.UtcNow.Ticks",
            pendingGate);
        int timeoutRule = IndexAfter(source,
            "HasBenchmarkAutoLoadTimedOut(", clockRead);
        int timeoutInvalid = IndexAfter(source,
            "FailBenchmarkAutoLoad(\"timed out\"", timeoutRule);
        int stop = source.IndexOf(
            "private static void StopWaitingForBenchmarkWorld()",
            StringComparison.Ordinal);
        int clearPending = source.IndexOf(
            "_waitingForBenchmarkWorld = false;", stop,
            StringComparison.Ordinal);
        int unsubscribe = source.IndexOf(
            "MapBox.on_world_loaded -= OnBenchmarkWorldLoaded;", stop,
            StringComparison.Ordinal);

        True(timeoutPoll >= 0 && pendingGate > timeoutPoll &&
             clockRead > pendingGate && timeoutRule > clockRead &&
             timeoutInvalid > timeoutRule,
            "timeout polling is a pending-only fast branch with explicit invalid");
        True(stop >= 0 && clearPending > stop && unsubscribe > stop,
            "timeout failure clears pending state and releases the callback");
    }

    private static void BenchmarkTimeoutCoversExceptionalMapUpdate()
    {
        string source = Read("Code", "patch",
            "AW_RuntimeBenchmarkAutoLoadPatch.cs");
        const string helperCall =
            "CheckPendingBenchmarkAutoLoadTimeout();";
        int postfix = source.IndexOf(
            "private static void CheckBenchmarkAutoLoadTimeout_Postfix()",
            StringComparison.Ordinal);
        int postfixCall = IndexAfter(source, helperCall, postfix);
        int finalizerAttribute = IndexAfter(source, "[HarmonyFinalizer]",
            postfix);
        int finalizer = IndexAfter(source,
            "private static Exception " +
            "CheckBenchmarkAutoLoadTimeout_Finalizer(Exception __exception)",
            finalizerAttribute);
        int finalizerCall = IndexAfter(source, helperCall, finalizer);
        int returnException = IndexAfter(source, "return __exception;",
            finalizerCall);
        int helper = source.IndexOf(
            "private static void CheckPendingBenchmarkAutoLoadTimeout()",
            StringComparison.Ordinal);

        True(postfix >= 0 && postfixCall > postfix &&
             finalizerAttribute > postfixCall && finalizer > finalizerAttribute &&
             finalizerCall > finalizer && returnException > finalizerCall,
            "normal and exceptional MapBox.Update paths share timeout cleanup");
        True(helper >= 0 && helper != postfix && helper != finalizer,
            "timeout state logic has one shared implementation");
    }

    private static void BenchmarkAutoLoadWaitsForTheRequestedWorldAndSelectsX20()
    {
        string source = Read("Code", "patch",
            "AW_RuntimeBenchmarkAutoLoadPatch.cs");

        Contains(source, "private static void OnBenchmarkWorldLoaded()",
            "benchmark auto-load has a dedicated completion callback");
        int subscribe = source.IndexOf(
            "MapBox.on_world_loaded += OnBenchmarkWorldLoaded;",
            StringComparison.Ordinal);
        int dispatch = source.IndexOf("loadWorld(path, false)",
            StringComparison.Ordinal);
        True(subscribe >= 0 && dispatch > subscribe,
            "completion is subscribed before the benchmark load dispatch");
        Contains(source, "AW3SaveDirectoryRegistry.TryGet(out string loadedPath)",
            "completion observes the directory actually loaded by WorldBox");
        Contains(source, "IsSameBenchmarkPath(_pendingPath, loadedPath)",
            "an unrelated world-loaded event cannot make the benchmark ready");
        Contains(source, "MapBox.on_world_loaded -= OnBenchmarkWorldLoaded;",
            "completion and failure paths can release the one-shot subscription");
        Contains(source, "Config.paused = false;",
            "the completed benchmark is explicitly unpaused");
        Contains(source, "Config.setWorldSpeed(\"x20\");",
            "the completed benchmark requests x20 speed");
        Contains(source, "Config.time_scale_asset.id == \"x20\"",
            "the benchmark verifies the active speed asset");
        Contains(source, "speed=x20 multiplier=20",
            "the ready log exposes the required speed and multiplier");
        Contains(source, "benchmark auto-load invalid",
            "failed completion emits a machine-detectable warning");
        Equal(false, source.Contains("Task.Run(") ||
                     source.Contains("new Thread("),
            "benchmark world-load handling remains on the game thread");
    }

    private static void Contains(string value, string token, string message)
    {
        True(value.Contains(token, StringComparison.Ordinal), message);
    }

    private static string Read(params string[] parts)
    {
        string[] pathParts = new string[parts.Length + 1];
        pathParts[0] = Directory.GetCurrentDirectory();
        Array.Copy(parts, 0, pathParts, 1, parts.Length);
        return File.ReadAllText(Path.Combine(pathParts));
    }

    private static int IndexAfter(string value, string token, int startIndex)
    {
        return startIndex < 0
            ? -1
            : value.IndexOf(token, startIndex, StringComparison.Ordinal);
    }

    private static void True(bool value, string message)
    {
        if (!value) throw new InvalidOperationException(message);
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(message +
                ": expected=" + expected + ", actual=" + actual);
    }
}
