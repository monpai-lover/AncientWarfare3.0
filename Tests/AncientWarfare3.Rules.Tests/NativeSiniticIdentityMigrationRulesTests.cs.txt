using System;
using AncientWarfare3.core.lineage;

internal static class NativeSiniticIdentityMigrationRulesTests
{
    public static void Run()
    {
        Equal(NativeSiniticIdentityMigrationAction.Skip,
            NativeSiniticIdentityMigrationRules.Decide(
                targetProfile: false, protectedName: false,
                completeIdentity: false, legacyWesternBranch: false,
                branchFamilyMismatch: false),
            "unrelated species are skipped");
        Equal(NativeSiniticIdentityMigrationAction.Skip,
            NativeSiniticIdentityMigrationRules.Decide(
                targetProfile: true, protectedName: true,
                completeIdentity: false, legacyWesternBranch: true,
                branchFamilyMismatch: true),
            "player and authored names stay protected");
        Equal(NativeSiniticIdentityMigrationAction.Reuse,
            NativeSiniticIdentityMigrationRules.Decide(
                targetProfile: true, protectedName: false,
                completeIdentity: true, legacyWesternBranch: false,
                branchFamilyMismatch: false),
            "a complete matching identity is idempotently reused");
        Equal(NativeSiniticIdentityMigrationAction.Repair,
            NativeSiniticIdentityMigrationRules.Decide(
                targetProfile: true, protectedName: false,
                completeIdentity: false, legacyWesternBranch: false,
                branchFamilyMismatch: false),
            "an incomplete identity is repaired lazily");
        Equal(NativeSiniticIdentityMigrationAction.Repair,
            NativeSiniticIdentityMigrationRules.Decide(
                targetProfile: true, protectedName: false,
                completeIdentity: true, legacyWesternBranch: true,
                branchFamilyMismatch: false),
            "a complete legacy Western branch still requires conversion");
        Equal(NativeSiniticIdentityMigrationAction.Repair,
            NativeSiniticIdentityMigrationRules.Decide(
                targetProfile: true, protectedName: false,
                completeIdentity: true, legacyWesternBranch: false,
                branchFamilyMismatch: true),
            "a later member adopts the already converted branch surname");
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(
                $"{message}: expected {expected}, got {actual}");
    }
}
