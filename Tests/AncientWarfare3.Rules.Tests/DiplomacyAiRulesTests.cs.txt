using AncientWarfare3.core.lineage;
using AncientWarfare3.core.performance;

internal static class DiplomacyAiRulesTests
{
    public static void Run()
    {
        if (!DiplomacyAiRules.ShouldRun(true))
            throw new InvalidOperationException(
                "enabled diplomacy AI should run");
        if (DiplomacyAiRules.ShouldRun(false))
            throw new InvalidOperationException(
                "disabled diplomacy AI should not run");

        AWPerformanceSettings.EnableAiAllianceActions = false;
        if (DiplomacyAiRules.AllowsAiInitiation(
                DiplomacyProposalType.Alliance))
            throw new InvalidOperationException(
                "disabled alliance AI should not initiate alliances");
        if (!DiplomacyAiRules.AllowsAiInitiation(
                DiplomacyProposalType.NonAggression))
            throw new InvalidOperationException(
                "alliance toggle must not disable non-aggression proposals");

        AWPerformanceSettings.EnableAiVassalActions = false;
        if (DiplomacyAiRules.AllowsAiInitiation(
                DiplomacyProposalType.Vassalize) ||
            DiplomacyAiRules.AllowsAiInitiation(
                DiplomacyProposalType.Tributary))
            throw new InvalidOperationException(
                "disabled vassal AI should not initiate subject actions");
        if (!DiplomacyAiRules.AllowsAiInitiation(
                DiplomacyProposalType.EndVassal))
            throw new InvalidOperationException(
                "vassal toggle must not disable ending existing relations");

        AWPerformanceSettings.EnableAiAllianceActions = true;
        AWPerformanceSettings.EnableAiVassalActions = true;
    }
}
