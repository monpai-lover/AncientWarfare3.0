using System;
using System.Linq;
using AncientWarfare3.core.lineage;

internal static class PeasantRebelBanditIslandRulesTests
{
    public static void Run()
    {
        True(PeasantRebelBanditIslandRules.ShouldStartEvacuation(
                true, true, 50, 100, 8, 1),
            "a two-cycle weak bandit under attack may evacuate");
        False(PeasantRebelBanditIslandRules.ShouldStartEvacuation(
                true, true, 61, 100, 8, 1),
            "a bandit at or above sixty percent strength does not evacuate");
        True(PeasantRebelBanditIslandRules.ShouldStartEvacuation(
                true, false, 100, 100, 3, 1),
            "a nearly empty stronghold may evacuate as a population fallback");
        False(PeasantRebelBanditIslandRules.ShouldStartEvacuation(
                true, true, 50, 100, 8, 0),
            "one threat cycle is not enough to evacuate");
        True(PeasantRebelBanditIslandRules.IsEligibleIsland(
                false, false, 12, true, false),
            "an empty buildable island with a coast is eligible");
        False(PeasantRebelBanditIslandRules.IsEligibleIsland(
                true, false, 12, true, false),
            "an island with a city is not eligible");
        True(PeasantRebelBanditIslandRules.IsEligiblePiracyTarget(
                true, true, false, false, 10),
            "a reachable coastal city with food is a piracy target");
        False(PeasantRebelBanditIslandRules.IsEligiblePiracyTarget(
                false, true, false, false, 10),
            "an inland city is not an island piracy target");
        True(PeasantRebelBanditIslandRules.NextThreatCycles(true, 0) == 1,
            "the first persistent threat records one cycle");
        True(PeasantRebelBanditIslandRules.NextThreatCycles(true, 1) == 2,
            "the threat counter saturates at two cycles");
        True(PeasantRebelBanditIslandRules.NextThreatCycles(false, 2) == 0,
            "a cleared threat resets the counter");
        StateDefaultsAndTransitionsAreSafe();
        IslandCandidatesRankBySafetyAndCapacity();
    }

    private static void IslandCandidatesRankBySafetyAndCapacity()
    {
        var facts = new[]
        {
            new BanditIslandCandidateFact(5, true, 80, 10, 20),
            new BanditIslandCandidateFact(3, true, 80, 10, 40),
            new BanditIslandCandidateFact(8, true, 90, 30, 100),
            new BanditIslandCandidateFact(1, false, 100, 1, 999)
        };
        long[] ranked = PeasantRebelBanditIslandRules.RankIslands(facts)
            .Select(fact => fact.IslandId).ToArray();
        True(ranked.SequenceEqual(new long[] { 8, 3, 5 }),
            "eligible islands rank by safety, area, distance, then stable ID");
    }

    private static void StateDefaultsAndTransitionsAreSafe()
    {
        var state = new PeasantRebelBanditStrongholdState();
        True(state.StrongholdKind == BanditStrongholdKind.Land,
            "legacy state defaults to a land stronghold");
        True(state.Migration.Stage == BanditMigrationStage.None,
            "legacy state defaults to no migration");
        True(PeasantRebelBanditIslandRules.CanTransition(
                BanditMigrationStage.None,
                BanditMigrationStage.Evaluating, manifestCount: 0),
            "evaluation can begin without a manifest");
        False(PeasantRebelBanditIslandRules.CanTransition(
                BanditMigrationStage.Evaluating,
                BanditMigrationStage.Boarding, manifestCount: 0),
            "boarding requires a non-empty manifest");
        True(PeasantRebelBanditIslandRules.CanTransition(
                BanditMigrationStage.Evaluating,
                BanditMigrationStage.Boarding, manifestCount: 2),
            "a locked manifest may begin boarding");
        False(PeasantRebelBanditIslandRules.CanTransition(
                BanditMigrationStage.Boarding,
                BanditMigrationStage.Founding, manifestCount: 2),
            "migration cannot skip the voyage stage");
    }

    private static void True(bool value, string message)
    {
        if (!value) throw new InvalidOperationException(message);
    }

    private static void False(bool value, string message)
    {
        if (value) throw new InvalidOperationException(message);
    }
}
