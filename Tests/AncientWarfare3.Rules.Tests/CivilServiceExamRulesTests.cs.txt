    private static void ExamModeAndCadenceAreStable()
    {
        Equal(CivilServiceExamMode.Imperial,
            CivilServiceExamRules.ResolveMode(hasMandate: true,
                hasEmpireTitle: false),
            "mandate realm uses imperial examinations");
        Equal(CivilServiceExamMode.Imperial,
            CivilServiceExamRules.ResolveMode(hasMandate: false,
                hasEmpireTitle: true),
            "empire title uses imperial examinations");
        Equal(CivilServiceExamMode.Tribute,
            CivilServiceExamRules.ResolveMode(hasMandate: false,
                hasEmpireTitle: false),
            "lesser realm uses tribute examinations");
        Equal(121, CivilServiceExamRules.FirstOpeningYear(120),
            "first sitting waits for a complete year");
        True(CivilServiceExamRules.IsCycleYear(121, 121),
            "anchor year opens a sitting");
        True(CivilServiceExamRules.IsCycleYear(124, 121),
            "sittings repeat every three years");
        False(CivilServiceExamRules.IsCycleYear(123, 121),
            "intermediate year has no sitting");
    }

    private static void EmptyCandidateRollWaitsForRecovery()
    {
        False(CivilServiceExamRules.ShouldOpenCandidateRoll(0),
            "an empty transient candidate index cannot commit an empty sitting");
        False(CivilServiceExamRules.ShouldOpenCandidateRoll(-1),
            "invalid candidate counts remain scheduled for recovery");
        True(CivilServiceExamRules.ShouldOpenCandidateRoll(1),
            "the sitting opens as soon as one eligible candidate is available");
        Equal(30, CivilServiceExamRules.EmptyCandidateRetryDays,
            "empty candidate rolls retry monthly instead of every authority tick");
    }

    private static void AiRankingBalancesMeritRulerAndDominantFaction()
    {
        var finalists = new[]
        {
            new CivilServiceAiRankingFacts(10, 100, 80, "ru"),
            new CivilServiceAiRankingFacts(11, 101, 78, "fa")
        };
        Equal("11,10", string.Join(",",
                CivilServiceExamRules.BuildAiRanking(finalists, "fa", 20)),
            "a weak ruler can let the dominant faction decide a close contest");
        Equal("10,11", string.Join(",",
                CivilServiceExamRules.BuildAiRanking(finalists, "fa", 80)),
            "an able ruler preserves a clear academic lead");

        var separated = new[]
        {
            new CivilServiceAiRankingFacts(20, 200, 85, "ru"),
            new CivilServiceAiRankingFacts(21, 201, 78, "fa")
        };
        Equal("20,21", string.Join(",",
                CivilServiceExamRules.BuildAiRanking(separated, "fa", 20)),
            "a raw score lead greater than six cannot be reversed");
        for (int ability = 0; ability <= 100; ability += 10)
        {
            int adjustment = CivilServiceExamRules.AiRankingAdjustment(
                rawScore: 80, candidateSchool: "fa",
                dominantSchool: "fa", rulerAbility: ability);
            True(adjustment >= -3 && adjustment <= 3,
                "AI ranking adjustment stays within the institutional bound");
        }
    }

    private static void CandidateSourcesInterleaveWithinTheBoundedScanBudget()
    {
        IReadOnlyList<long>[] sources =
        {
            Enumerable.Range(1, 140).Select(p => (long)p).ToArray(),
            new long[] { 1001L },
            new long[] { 2001L }
        };

        IReadOnlyList<long> result = CivilServiceExamRules.
            InterleaveCandidateSources(sources,
                CivilServiceExamRules.CandidateSourceLimit);

        True(result.Contains(121L),
            "a valid row after the old ninety-six-row window remains visible");
        True(result.Contains(1001L) && result.Contains(2001L),
            "round-robin scanning cannot starve the other social origins");
        True(result.Count <= CivilServiceExamRules.CandidateSourceLimit,
            "interleaved source work remains strictly bounded");
    }

    private static void ScoringIsDeterministicAndClassNeutral()
    {
        int first = CivilServiceExamRules.DeterministicJitter(41L, 77L,
            CivilServiceExamStage.Metropolitan);
        int second = CivilServiceExamRules.DeterministicJitter(41L, 77L,
            CivilServiceExamStage.Metropolitan);
        Equal(first, second, "stage jitter survives save-load retries");
        True(first >= -5 && first <= 5,
            "stage jitter remains in the designed band");
        Equal(CivilServiceExamRules.Score(70, 50, 12, first),
            CivilServiceExamRules.Score(70, 50, 12, first),
            "identical learned facts produce identical scores");
        Equal(100, CivilServiceExamRules.Score(100, 100, 100, 5),
            "exam score is clamped to one hundred");
        Equal(0, CivilServiceExamRules.Score(0, 0, 0, -5),
            "exam score is clamped to zero");
    }

                True(GovernorRotationRuntimeScope.IsActive,

        }

                "test exception crosses the rotation scope");

            "exceptional exit cannot leave career patches suppressed");
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException($"{name}: expected {expected}, got {actual}");
    }

    private static void True(bool value, string name)
    {
        if (!value) throw new InvalidOperationException(name + ": expected true");
    }

    private static void False(bool value, string name)
    {
        if (value) throw new InvalidOperationException(name + ": expected false");
    }
