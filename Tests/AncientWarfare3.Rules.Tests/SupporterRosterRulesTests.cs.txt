using System;
using AncientWarfare3.content.supporters;

internal static class SupporterRosterRulesTests
{
    public static void Run()
    {
        var entries = SupporterRosterData.Parse(new[]
        {
            "rank,name,amount,date,description",
            "2, Alice ,50,2026-08-16,",
            "bad,\"Bob, Jr.\",,,\"portrait, building\"",
            "3,alice,100,2026-08-27,second record",
            "4,   ,20,2026-08-28,missing name",
            "# ignored"
        });

        Equal(3, entries.Count, "blank names are skipped");
        Equal(1, entries[0].Rank, "ranks are normalized after sorting");
        Equal("Alice", entries[0].Name, "names are trimmed");
        Equal("Bob, Jr.", entries[1].Name, "quoted commas are parsed");
        Equal("portrait, building", entries[1].Description,
            "quoted descriptions are parsed");
        Equal("alice", entries[2].Name,
            "duplicate-name rows remain available for aggregation");
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(message + ": expected=" +
                                                expected + ", actual=" + actual);
    }
}
