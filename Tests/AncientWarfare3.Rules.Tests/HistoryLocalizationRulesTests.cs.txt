using AncientWarfare3.core.lineage;

internal static class HistoryLocalizationRulesTests
{
    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(
                $"{name}: expected {expected}, got {actual}");
    }

    public static void Run()
    {
        Equal("世子", HistoryLocalizationRules.Text("aw_heir_shizi", "cz"),
            "heir designation history resolves Shizi instead of leaking a key");
        Equal("太子", HistoryLocalizationRules.Text("aw_heir_taizi", "cz"),
            "heir designation history resolves Taizi");
        Equal("留后", HistoryLocalizationRules.Text("aw_heir_liuhou", "cz"),
            "military-acclaim history resolves Liuhou");
        Equal("嗣君", HistoryLocalizationRules.Text("aw_heir_sijun", "cz"),
            "civil-acclaim history resolves Sijun");
        Equal("无学派",
            HistoryLocalizationRules.Text("aw_court_school_none", "cz"),
            "empty court school uses the explicit no-school label");
        Equal("师从",
            HistoryLocalizationRules.Text("aw_hist_school_studied_under", "cz"),
            "disciple biography uses a localized Chinese relation");
        Equal(" studied under ",
            HistoryLocalizationRules.Text("aw_hist_school_studied_under", "en"),
            "disciple biography keeps the English relation in English only");
        Equal("師從",
            HistoryLocalizationRules.Text("aw_hist_school_studied_under", "ch"),
            "disciple biography uses a localized traditional Chinese relation");
        Equal(" is first granted ",
            HistoryLocalizationRules.Text("aw_hist_official_rank_grant_mid", "en"),
            "first official rank grant history resolves its middle fragment");
        Equal(" and is entered upon the official register.",
            HistoryLocalizationRules.Text("aw_hist_official_rank_grant_suffix", "en"),
            "first official rank grant history resolves its suffix");
        Equal("\u5b97\u5ba4\u5a5a\u76df",
            WarDisplayLabelRules.EventLabel("royal_marriage", "cz"),
            "royal marriage history resolves a localized event label");
        Equal("\u7532\u4e0e\u4e59\u7f14\u7ed3\u5a5a\u76df",
            WarDisplayLabelRules.NormalizeHistoryContent(
                "royal_marriage", "\u7532\u4e0e\u4e59", "cz"),
            "legacy royal marriage history gains its missing suffix");
        Equal("\u7532\u4e0e\u4e59\u7f14\u7ed3\u5a5a\u76df",
            WarDisplayLabelRules.NormalizeHistoryContent(
                "royal_marriage", "\u7532\u4e0e\u4e59\u7f14\u7ed3\u5a5a\u76df", "cz"),
            "complete royal marriage history is not suffixed twice");
        Equal("A married B",
            WarDisplayLabelRules.NormalizeHistoryContent(
                "royal_marriage", "A married B", "en"),
            "English royal marriage history remains a complete sentence");

        foreach ((string key, string cz, string en, string ch) in new[]
                 {
                     ("aw_hist_rebel_route_founding",
                         "\u9009\u62e9\u5efa\u56fd\u8def\u7ebf",
                         " chose the founding route",
                         "\u9078\u64c7\u5efa\u570b\u8def\u7dda"),
                     ("aw_hist_rebel_route_bandit",
                         "\u843d\u8349\u4e3a\u5bc7\u5e76\u7b51\u5be8",
                         " became bandits and fortified a hideout",
                         "\u843d\u8349\u70ba\u5bc7\u4e26\u7bc9\u5be8"),
                     ("aw_hist_bandit_suppression_started",
                         "\u53d1\u5175\u527f\u532a",
                         " began a suppression campaign",
                         "\u767c\u5175\u527f\u532a"),
                     ("aw_hist_bandit_converted",
                         "\u6539\u65d7\u5efa\u53f7\uff0c\u8f6c\u4e3a\u5efa\u56fd\u4e49\u519b",
                         " raised a state banner and became founding rebels",
                         "\u6539\u65d7\u5efa\u865f\uff0c\u8f49\u70ba\u5efa\u570b\u7fa9\u8ecd"),
                     ("aw_hist_bandit_destroyed",
                         "\u5c71\u5be8\u88ab\u653b\u7834\uff0c\u8d3c\u56fd\u8986\u706d",
                         " lost its hideout and was destroyed",
                         "\u5c71\u5be8\u88ab\u653b\u7834\uff0c\u8cca\u570b\u8986\u6ec5"),
                     ("aw_hist_bandit_pressure_annexed",
                         "\u56e0\u532a\u60a3\u6df1\u91cd\u5f52\u9644\u5c71\u5be8",
                         " joined the bandit stronghold after prolonged unrest",
                         "\u56e0\u532a\u60a3\u6df1\u91cd\u6b78\u9644\u5c71\u5be8"),
                     ("aw_hist_bandit_revolution_started",
                         "\u7fbd\u7ffc\u5df2\u4e30\uff0c\u6539\u65d7\u8d77\u4e49",
                         " grew strong enough to raise the rebel banner",
                         "\u7fbd\u7ffc\u5df2\u8c50\uff0c\u6539\u65d7\u8d77\u7fa9")
                 })
        {
            Equal(cz, HistoryLocalizationRules.Text(key, "cz"),
                "bandit lifecycle history localizes simplified " + key);
            Equal(en, HistoryLocalizationRules.Text(key, "en"),
                "bandit lifecycle history localizes English " + key);
            Equal(ch, HistoryLocalizationRules.Text(key, "ch"),
                "bandit lifecycle history localizes traditional " + key);
        }

        foreach ((string key, string expected) in new[]
                 {
                     ("aw_hist_school_authored", "著成"),
                     ("aw_hist_school_preserved", "传于此城"),
                     ("aw_hist_school_converted_from", "由"),
                     ("aw_hist_school_converted_to", "转入"),
                     ("aw_hist_school_rediscovered", "重得"),
                     ("aw_hist_school_revived", "复兴"),
                     ("aw_hist_school_debated_with", "与"),
                     ("aw_hist_school_debated_topic", "论辩"),
                     ("aw_hist_school_master_died", "逝世"),
                     ("aw_hist_school_descended_at", "降临于"),
                     ("aw_hist_school_travelled_to", "游学至"),
                     ("aw_hist_school_arrived_to_study", "来此游学"),
                     ("aw_hist_school_served_as", "出任"),
                     ("aw_hist_school_served_court", "入朝为客卿")
                 })
            Equal(expected, HistoryLocalizationRules.Text(key, "cz"),
                "school chronicle fragment localizes " + key);

        string schoolSources = string.Join("\n", new[]
        {
            "HistoricalSchoolActionService.cs",
            "HistoricalSchoolDebateService.cs",
            "SchoolGuestOfficeService.cs"
        }.Select(file => File.ReadAllText(Path.Combine(
            Directory.GetCurrentDirectory(), "Code", "core", "schools", file))));
        foreach (string literal in new[]
                 {
                     " authored ", " preserved ", "Converted from ",
                     "Rediscovered ", " revived ", " debated ",
                     " served as ", " served the court"
                 })
            Equal(false, schoolSources.Contains(literal),
                "school chronicle source contains no hard-coded English: " + literal);

        foreach ((string id, string expected) in new[]
                 {
                     ("livelihood", "民生"), ("famine", "饥荒"),
                     ("war", "战争"), ("defense", "守御"),
                     ("aggression", "征伐"), ("peace", "议和"),
                     ("diplomacy", "外交"), ("order", "治序"),
                     ("commerce", "商贸"), ("technology", "技艺"),
                     ("institutions", "制度"), ("medicine", "医道"),
                     ("epidemic", "疫病")
                 })
            Equal(expected, HistoryLocalizationRules.Text(
                    "aw_hist_school_topic_" + id, "cz"),
                "school debate topic localizes " + id);
        foreach ((string id, string expected) in new[]
                 {
                     ("draw", "未分胜负"),
                     ("narrow_first", "前者小胜"),
                     ("decisive_first", "前者大胜"),
                     ("narrow_second", "后者小胜"),
                     ("decisive_second", "后者大胜")
                 })
            Equal(expected, HistoryLocalizationRules.Text(
                    "aw_hist_school_outcome_" + id, "cz"),
                "school debate outcome localizes " + id);
    }
}
