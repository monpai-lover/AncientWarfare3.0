using AncientWarfare3.core.naming;

internal static class ActorManualNameRulesTests
{
    public static void Run()
    {
        ActorManualNameDraft xia = ActorManualNameRules.CreateDraft(
            ActorManualNameMode.Xia, "Ji", "Fa");
        Equal("JiFa", xia.DisplayName,
            "Xia family or shi precedes given name");

        ActorManualNameDraft nonXia = ActorManualNameRules.CreateDraft(
            ActorManualNameMode.NonXia, "Louis", "de Lyon");
        Equal("Louis de Lyon", nonXia.DisplayName,
            "non-Xia given name precedes family name");

        Equal(false, ActorManualNameRules.CreateDraft(
            ActorManualNameMode.Xia, "Ji", " ").IsValid,
            "given name is required");

        Equal("Ji Fa", ActorManualNameRules.CreateDisplayName(
            ActorManualNameMode.NonXia, "Ji", "Fa"),
            "non-Xia display uses a separator");

        Equal(ActorManualNameEditorState.Editing,
            ActorManualNameEditorRules.Resolve(
                ActorManualNameEditorState.Display,
                ActorManualNameEditorEvent.NameSelected,
                anyEditorFieldFocused: true),
            "selecting the displayed name expands the structured editor");
        Equal(ActorManualNameEditorState.Editing,
            ActorManualNameEditorRules.Resolve(
                ActorManualNameEditorState.Editing,
                ActorManualNameEditorEvent.FocusChanged,
                anyEditorFieldFocused: true),
            "moving between structured fields keeps the editor expanded");
        Equal(ActorManualNameEditorState.Display,
            ActorManualNameEditorRules.Resolve(
                ActorManualNameEditorState.Editing,
                ActorManualNameEditorEvent.FocusChanged,
                anyEditorFieldFocused: false),
            "leaving both structured fields restores the complete name");
        Equal(ActorManualNameEditorState.Display,
            ActorManualNameEditorRules.Resolve(
                ActorManualNameEditorState.Editing,
                ActorManualNameEditorEvent.WindowClosed,
                anyEditorFieldFocused: true),
            "closing the actor window restores display state");
    }

    private static void Equal<T>(T expected, T actual, string name)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual))
            throw new InvalidOperationException(
                $"{name}: expected {expected}, got {actual}");
    }
}
