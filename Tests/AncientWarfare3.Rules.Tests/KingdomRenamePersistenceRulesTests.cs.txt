using AncientWarfare3.core.lineage;
using AncientWarfare3.core.naming;

internal static class KingdomRenamePersistenceRulesTests
{
    public static void Run()
    {
        AWLocalizedNameEditDecision chineseEdit =
            AWLocalizedKingdomRenameRules.ResolveEdit(
                "ch", "New Chinese", "Native Realm", "Old Chinese");
        Equal("Native Realm", chineseEdit.NativeName,
            "a Chinese edit preserves the native identity slot");
        Equal("New Chinese", chineseEdit.ChineseName,
            "a Chinese edit updates only the Chinese identity slot");

        AWLocalizedNameEditDecision nativeEdit =
            AWLocalizedKingdomRenameRules.ResolveEdit(
                "en", "New Native", "Old Native", "Chinese Realm");
        Equal("New Native", nativeEdit.NativeName,
            "a non-Chinese edit updates only the native identity slot");
        Equal("Chinese Realm", nativeEdit.ChineseName,
            "a non-Chinese edit preserves the Chinese identity slot");

        AWLocalizedNameEditDecision canonicalStateName =
            AWLocalizedKingdomRenameRules.ResolveCanonicalStateName(
                " 周 ", "House", "房");
        Equal("周", canonicalStateName.NativeName,
            "a formal state name replaces the stale native slot across languages");
        Equal("周", canonicalStateName.ChineseName,
            "a formal state name replaces the stale Chinese slot before save");

        AWLocalizedNameEditDecision partnerChineseEdit =
            AWLocalizedKingdomRenameRules.ResolveEdit(
                "ch", "Shared Chinese", "Partner Native",
                "Partner Chinese");
        Equal("Partner Native", partnerChineseEdit.NativeName,
            "a split partner preserves its own inactive native slot");
        Equal("Shared Chinese", partnerChineseEdit.ChineseName,
            "a split partner receives only the edited Chinese base");
        AWLocalizedNameEditDecision originalChineseEdit =
            AWLocalizedKingdomRenameRules.ResolveEdit(
                "ch", "Shared Chinese", "Original Native",
                "Original Chinese");
        string sharedEnglish =
            AWLocalizedKingdomRenameRules.ResolveSharedProjection(
                "en", originalChineseEdit.NativeName,
                originalChineseEdit.ChineseName, "Legacy Database Name");
        Equal("Original Native", sharedEnglish,
            "language switching projects the shared authority native slot");
        Equal("Eastern Original Native",
            SuccessionDisputeDisplayRules.BuildQualifiedName(
                sharedEnglish, "east", active: true, language: "en"),
            "the original nameplate adds direction to the shared native base");
        Equal("Western Original Native",
            SuccessionDisputeDisplayRules.BuildQualifiedName(
                sharedEnglish, "west", active: true, language: "en"),
            "the rival nameplate adds direction to the same native base");
        Equal("Original Native",
            AWLocalizedKingdomRenameRules.ResolveSettlementBase(
                sharedEnglish, "Rival Native", "Shared Chinese"),
            "settlement keeps the current bilingual projection, not DB text");
        Equal("Rival Native",
            AWLocalizedKingdomRenameRules.ResolveSettlementBase(
                string.Empty, "Rival Native", "Shared Chinese"),
            "settlement uses a surviving rival before legacy DB text");
        Equal(true, AWLocalizedKingdomRenameRules.ShouldCommitManualEdit(
                originalAccepted: true, hasManualEditContext: true),
            "an accepted kingdom UI edit commits localized identity");
        Equal(false, AWLocalizedKingdomRenameRules.ShouldCommitManualEdit(
                originalAccepted: true, hasManualEditContext: false),
            "automatic generated names never enter the manual edit path");

        AWLocalizedNameIdentitySnapshot authorityIdentity = new(
            "Authority Native", "Authority Chinese", "Authority Given",
            "Authority Family", "authority_generator", 91L,
            AWLocalizedNameMigrationRules.CurrentSchemaVersion);
        AWLocalizedNameIdentitySnapshot rivalOldIdentity = new(
            "Rival Native", "Rival Chinese", "Rival Given",
            "Rival Family", "rival_generator", 37L, 0);
        AWLocalizedNameIdentitySnapshot selectedSurvivor =
            AWLocalizedKingdomIdentitySyncAdapter.SelectAuthority(
                original: null, rival: rivalOldIdentity);
        Equal("Rival Native", selectedSurvivor.NativeName,
            "an original-missing settlement selects the rival identity");
        Equal("Rival Chinese", selectedSurvivor.ChineseName,
            "the surviving rival retains both localized slots");

        long synchronizedId = -1L;
        AWLocalizedNameIdentitySnapshot synchronizedIdentity = null;
        var createRivalQueue = new AWBoundedLocalizedNameWriteQueue(2);
        int synchronizedWrites =
            AWLocalizedKingdomIdentitySyncAdapter.Synchronize(
                authorityIdentity, new[] { 84L }, (id, identity) =>
                {
                    synchronizedId = id;
                    synchronizedIdentity = identity;
                    createRivalQueue.Enqueue("Kingdom", id, identity);
                });
        Equal(1, synchronizedWrites,
            "CreateRival schedules exactly one rival identity write");
        Equal(84L, synchronizedId,
            "CreateRival writes the new rival kingdom identity");
        Equal("Authority Native", synchronizedIdentity.NativeName,
            "CreateRival copies the complete native identity slot");
        Equal("Authority Chinese", synchronizedIdentity.ChineseName,
            "CreateRival copies the complete Chinese identity slot");
        Equal("Authority Given", synchronizedIdentity.GivenName,
            "CreateRival copies structured given-name identity");
        Equal("Authority Family", synchronizedIdentity.FamilyComponent,
            "CreateRival copies structured family identity");
        Equal("authority_generator", synchronizedIdentity.GeneratorId,
            "CreateRival copies generator provenance");
        Equal(91L, synchronizedIdentity.CultureId,
            "CreateRival copies culture provenance");
        Equal(AWLocalizedNameMigrationRules.CurrentSchemaVersion,
            synchronizedIdentity.SchemaVersion,
            "CreateRival copies the current identity schema");
        Equal(true, createRivalQueue.TryDequeue(
                out AWLocalizedNamePendingWrite createRivalWrite),
            "CreateRival enqueues the synchronized rival identity");
        Equal(84L, createRivalWrite.ObjectId,
            "the async writer receives the rival kingdom ID");
        Equal("Authority Chinese", createRivalWrite.Snapshot.ChineseName,
            "the async writer receives the synchronized bilingual snapshot");

        var pending = new AWBoundedLocalizedNameWriteQueue(2);
        pending.Enqueue("Kingdom", 42L, Snapshot("Old Native", "Old Chinese"));
        pending.Enqueue("Kingdom", 42L, Snapshot("New Native", "New Chinese"));
        Equal(1, pending.Count,
            "an exact kingdom identity replaces its stale pending snapshot");
        Equal(true, pending.TryDequeue(out AWLocalizedNamePendingWrite write),
            "the replacement identity remains queued");
        Equal("New Native", write.Snapshot.NativeName,
            "the queued snapshot contains the latest native name");
        Equal("New Chinese", write.Snapshot.ChineseName,
            "the queued snapshot contains the latest Chinese name");

        AWLocalizedNameIdentitySnapshot saved =
            Snapshot("Saved Native", "Saved Chinese");
        AWLocalizedNameIdentitySnapshot staleDatabase =
            Snapshot("Database Native", "Database Chinese");
        AWLocalizedNameIdentitySnapshot restored =
            AWLocalizedNameRestoreRules.Merge(saved, staleDatabase,
                AWLocalizedNameMigrationRules.CurrentSchemaVersion);
        Equal("Saved Native", restored.NativeName,
            "current-schema save data is authoritative for native names");
        Equal("Saved Chinese", restored.ChineseName,
            "current-schema save data is authoritative for Chinese names");

        AWLocalizedNameIdentitySnapshot partialSaved = new(
            "Saved Native", string.Empty, "Saved Given", string.Empty,
            "saved_generator", 7L,
            AWLocalizedNameMigrationRules.CurrentSchemaVersion);
        AWLocalizedNameIdentitySnapshot completed =
            AWLocalizedNameRestoreRules.Merge(partialSaved, staleDatabase,
                AWLocalizedNameMigrationRules.CurrentSchemaVersion);
        Equal("Saved Native", completed.NativeName,
            "SQLite cannot overwrite a present current-schema slot");
        Equal("Database Chinese", completed.ChineseName,
            "SQLite fills a missing current-schema slot");

        Equal(true, AWLocalizedNameProjectionChangeRules.ShouldInvalidate(
                "Old Realm", "New Realm"),
            "a changed visible name invalidates name projections");
        Equal(false, AWLocalizedNameProjectionChangeRules.ShouldInvalidate(
                "New Realm", "New Realm"),
            "an unchanged projection does not invalidate caches twice");
        var invalidatedKingdoms = new HashSet<long>();
        Equal(true, AWLocalizedNameProjectionChangeRules.
                TryMarkInvalidated(invalidatedKingdoms, 42L,
                    "Old Realm", "New Realm"),
            "the first visible change schedules one kingdom invalidation");
        Equal(false, AWLocalizedNameProjectionChangeRules.
                TryMarkInvalidated(invalidatedKingdoms, 42L,
                    "Old Realm", "New Realm"),
            "the same kingdom cannot invalidate twice in one projection");
        Equal(false, AWLocalizedNameProjectionChangeRules.
                TryMarkInvalidated(invalidatedKingdoms, 43L,
                    "New Realm", "New Realm"),
            "an unchanged kingdom never enters the invalidation set");
        var prefixInvalidations = new HashSet<long>();
        Equal(true, AWLocalizedNameProjectionChangeRules.
                TryMarkInvalidated(prefixInvalidations, 10L,
                    "Zhou", "Eastern Zhou"),
            "adding the original direction invalidates once");
        Equal(true, AWLocalizedNameProjectionChangeRules.
                TryMarkInvalidated(prefixInvalidations, 20L,
                    "Rival", "Western Zhou"),
            "creating the rival direction invalidates that kingdom once");
        Equal(false, AWLocalizedNameProjectionChangeRules.
                TryMarkInvalidated(prefixInvalidations, 10L,
                    "Zhou", "Eastern Zhou"),
            "a batched prefix refresh cannot invalidate original twice");
        Equal(2, prefixInvalidations.Count,
            "one split presentation batch refreshes each changed kingdom once");
        int createRivalRefreshes = 0;
        AWLocalizedNameProjectionRefreshScope.Suppress(() =>
        {
            if (AWLocalizedNameProjectionRefreshScope.
                    ShouldRefreshAutomatically(isEditing: false))
                createRivalRefreshes++;
        });
        if (AWLocalizedNameProjectionChangeRules.TryMarkInvalidated(
                new HashSet<long>(), 20L, "Rival", "Western Zhou"))
            createRivalRefreshes++;
        Equal(1, createRivalRefreshes,
            "CreateRival suppresses its automatic rename refresh and batches one final refresh");
        Equal(true, AWLocalizedNameProjectionRefreshScope.
                ShouldRefreshAutomatically(isEditing: false),
            "CreateRival restores automatic refresh after its creation scope");
        try
        {
            AWLocalizedNameProjectionRefreshScope.Suppress(() =>
                throw new InvalidOperationException("creation failed"));
        }
        catch (InvalidOperationException)
        {
        }
        Equal(true, AWLocalizedNameProjectionRefreshScope.
                ShouldRefreshAutomatically(isEditing: false),
            "CreateRival restores automatic refresh when kingdom creation throws");

        var semanticRefreshes = new List<string>();
        KingdomRenameProjectionRefreshContract.Apply(
            () => semanticRefreshes.Add("mandate"),
            () => semanticRefreshes.Add("family-tree-cache"),
            () => semanticRefreshes.Add("family-tree-revision"));
        Equal("mandate,family-tree-cache,family-tree-revision",
            string.Join(",", semanticRefreshes),
            "one kingdom refresh applies every name-dependent semantic projection exactly once");
        Equal(true, SuccessionDisputeDisplayRules.
                HasPersistedClosingQualifier(
                    SuccessionDisputeStatus.Active, rivalKingdomId: 20L),
            "Close remembers an active prefix after cities are returned");
        Equal(true, SuccessionDisputeDisplayRules.
                HasPersistedClosingQualifier(
                    SuccessionDisputeStatus.PermanentSplit,
                    rivalKingdomId: 20L),
            "reunification remembers permanent-split prefixes");
        Equal(false, SuccessionDisputeDisplayRules.
                HasPersistedClosingQualifier(
                    SuccessionDisputeStatus.Prepared, rivalKingdomId: -1L),
            "a prepared dispute never invents closing prefixes");

        string uiBaseName = "New Zhou";
        Equal("New Zhou Kingdom", KingdomNameplateSuffixRules.ProjectName(
                uiBaseName, " Kingdom", pShouldDisplaySuffix: true),
            "the nameplate decorates the same base name shown by kingdom UI");

        (string east, string west) =
            SuccessionDisputeDisplayRules.BuildDistinctPair(
                uiBaseName, "east", "west", active: true, language: "en");
        Equal("Eastern New Zhou", east,
            "the original split keeps its direction around the updated base");
        Equal("Western New Zhou", west,
            "the rival split shares the updated base with its own direction");

        string root = Directory.GetCurrentDirectory();
        string editPatch = File.ReadAllText(Path.Combine(root, "Code", "patch",
            "naming", "AW_KingdomNameEditPatch.cs"));
        Equal(true, editPatch.Contains("typeof(KingdomWindow), \"onNameChange\"") &&
                    editPatch.Contains("__result"),
            "localized identity commits only after a successful kingdom UI edit");

        string projectionService = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "KingdomRenameProjectionService.cs"));
        Equal(true, projectionService.Contains("nameplate_manager?.clearCaches") &&
                    projectionService.Contains(
                        "HierarchicalVassalMapModeService.MarkKingdomDirty"),
            "rename projection invalidates vanilla and hierarchical nameplates");
        Equal(true, projectionService.Contains(
                        "KingdomRenameProjectionRefreshContract.Apply") &&
                    projectionService.Contains(
                        "MandateService.RefreshKingdomNameProjection") &&
                    projectionService.Contains(
                        "RulerAppellationService.InvalidateFamilyTreeProjectionCaches") &&
                    projectionService.Contains(
                        "FamilyTreeProjectionChange.DynastyOrStateName"),
            "rename projection binds all required semantic refresh callbacks");
        Equal(false, projectionService.Contains("DirtyMapIfActive"),
            "rename projection does not trigger unrelated map-mode redraws");
        Equal(false, projectionService.Contains("ModeGeometry") ||
                     projectionService.Contains("Texture"),
            "rename projection does not rebuild map geometry or textures");

        string stateNameService = File.ReadAllText(Path.Combine(root, "Code",
            "core", "lineage", "StateNameService.cs"));
        Equal(true, stateNameService.Contains(
                "AWLocalizedKingdomNameService.CommitCanonicalStateName"),
            "formal state-name projection enters the localized persistence boundary");
        int committedProjectionStart = stateNameService.IndexOf(
            "private static void ApplyCommittedProjection",
            StringComparison.Ordinal);
        int nextMethodStart = stateNameService.IndexOf(
            "private static bool TryReadBranchSeed", committedProjectionStart,
            StringComparison.Ordinal);
        string committedProjection = stateNameService.Substring(
            committedProjectionStart, nextMethodStart - committedProjectionStart);
        Equal(false, committedProjection.Contains("setName("),
            "formal state-name projection cannot bypass localized persistence with setName");

        string migrationService = File.ReadAllText(Path.Combine(root, "Code",
            "core", "naming", "AWLocalizedNameMigrationService.cs"));
        int migrateStart = migrationService.IndexOf(
            "private static void Migrate", StringComparison.Ordinal);
        int captureStart = migrationService.IndexOf(
            "AWLocalizedNamePersistence.Capture", migrateStart,
            StringComparison.Ordinal);
        int reconcileStart = migrationService.IndexOf(
            "StateNameService.ReconcileLocalizedIdentityBeforeRestore",
            migrateStart, StringComparison.Ordinal);
        Equal(true, reconcileStart >= migrateStart &&
                    reconcileStart < captureStart,
            "restore repairs an already projected formal state name before stale localized slots are captured");
        Equal(true, stateNameService.Contains(
                "string.Equals(pKingdom.name, committed"),
            "old-save repair only trusts a branch state name already projected into the live save");
    }

    private static AWLocalizedNameIdentitySnapshot Snapshot(
        string nativeName, string chineseName)
    {
        return new AWLocalizedNameIdentitySnapshot(nativeName, chineseName,
            "Given", "Family", "generator", 7L,
            AWLocalizedNameMigrationRules.CurrentSchemaVersion);
    }

    private static void Equal<T>(T expected, T actual, string message)
    {
        if (!Equals(expected, actual))
            throw new InvalidOperationException(
                $"{message}: expected {expected}, got {actual}");
    }
}
