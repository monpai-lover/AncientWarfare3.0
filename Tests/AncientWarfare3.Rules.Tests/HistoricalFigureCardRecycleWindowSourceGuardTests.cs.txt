using System;
using System.IO;

internal static class HistoricalFigureCardRecycleWindowSourceGuardTests
{
    public static void Run()
    {
        string root = Path.Combine(Directory.GetCurrentDirectory(), "Code",
            "ui", "windows");
        string draw = File.ReadAllText(Path.Combine(root,
            "HistoricalFigureDrawWindow.cs"));
        string recycle = File.ReadAllText(Path.Combine(root,
            "HistoricalFigureRecycleWindow.cs"));
        string activeDraw = StripBlockComments(draw);

        Contains(recycle, "HistoricalFigureCardRecycleSelectionRules.FilterVisible",
            "recycle window owns quality filtering");
        Contains(recycle, "ScrollRect",
            "recycle window has a real scroll list");
        Contains(recycle, "TryRecycle",
            "recycle window owns the collection transaction");
        Contains(recycle, "GetRecycleSourceCounts",
            "recycle window uses persisted source metadata");
        Contains(recycle, "SelectWeightedCrate",
            "recycle window uses source weighting");
        Contains(recycle, "HistoricalFigureCardCatalog.GetCards(crateId)",
            "recycle output uses the selected source crate");
        Contains(recycle, "GetSelectedCount(pCard.CardId)",
            "recycle list displays inventory after selected cards are reserved");
        Contains(recycle, "titleText.text = AW_L10n.Text",
            "recycle window localizes the native window title");
        Contains(recycle, "aw_historical_figure_cards_recycle_title",
            "recycle title uses the plural locale key registered by the runtime");
        Contains(recycle, "const int SlotColumns = 3",
            "recycle selection uses a three-column grid");
        Contains(recycle, "const int CardColumns = 3",
            "recycle inventory uses a three-column card grid");
        Contains(recycle, "CardHeight = 64f",
            "recycle inventory shows three compact rows in the default window");
        Contains(recycle, "i < 10",
            "recycle selection keeps the ten-card contract");
        Contains(recycle, "Position(_submit.GetComponent<RectTransform>()",
            "recycle submit is positioned from the selection panel layout");
        Contains(recycle, "OpenCardDetails(_lastOutput, _lastOutputCrateId, true)",
            "recycle result details retain a return-to-recycle target");
        Contains(recycle, "RenderTradeUpResult",
            "recycle success renders the same detailed result presentation");

        Contains(draw, "HistoricalFigureRecycleWindow.Open",
            "draw window exposes the dedicated recycle entry");
        Contains(draw, "const int InventoryPageSize = 20",
            "inventory uses a bounded twenty-card page");
        Contains(draw, "Skip(_inventoryPage * InventoryPageSize)",
            "inventory binds only the active page");
        Contains(draw, "_inventoryPreviousPage",
            "inventory exposes a previous-page control");
        Contains(draw, "_inventoryNextPage",
            "inventory exposes a next-page control");
        Contains(draw, "_inventoryPageLabel",
            "inventory exposes a page indicator");
        Contains(recycle, "const int PageSize = 20",
            "recycle uses a bounded twenty-card page");
        Contains(recycle, "Skip(_page * PageSize)",
            "recycle binds only the active page");
        Contains(recycle, "_previousPage",
            "recycle exposes a previous-page control");
        Contains(recycle, "_nextPage",
            "recycle exposes a next-page control");
        Contains(recycle, "_pageLabel",
            "recycle exposes a page indicator");
        Contains(draw, "aw_historical_figure_cards_recycle_back_to_recycle",
            "details localizes the return-to-recycle label");
        False(activeDraw.Contains("TryRecycle"),
            "draw window does not commit recycle transactions");
        False(activeDraw.Contains("ToggleRecycleCard"),
            "draw window does not own recycle selection");
        False(activeDraw.Contains("BuildRecycleInputIds"),
            "draw window does not expand recycle inputs");
        False(activeDraw.Contains("SelectedRecycleRarity"),
            "draw window does not track recycle rarity");
    }

    private static void Contains(string source, string value, string message)
    {
        if (!source.Contains(value))
            throw new InvalidOperationException(message);
    }

    private static void False(bool value, string message)
    {
        if (value) throw new InvalidOperationException(message);
    }

    private static string StripBlockComments(string source)
    {
        int start;
        while ((start = source.IndexOf("/*", StringComparison.Ordinal)) >= 0)
        {
            int end = source.IndexOf("*/", start + 2,
                StringComparison.Ordinal);
            if (end < 0) return source.Substring(0, start);
            source = source.Remove(start, end + 2 - start);
        }
        return source;
    }
}
